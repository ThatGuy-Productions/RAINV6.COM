    65→  const setMacros = useSessionStore((s) => s.setMacros)
    66→  const setMacroSource = useSessionStore((s) => s.setMacroSource)
    67→  const [showAssistant, setShowAssistant] = useState(true)

    69→  const [showCompare, setShowCompare] = useState(false)
    70→  const [showBlindTest, setShowBlindTest] = useState(false)

    73→  // P4-AI-DISCLOSURE: honest per-field AI disclosure state (EU AI Act Article 50)

    75→    vocals: 'none',

    81→  const renderStartTime = useRef<number>(0)
    82→  const progressIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null)
    83→  // BETA-ANALYTICS: server-persisted Session id for the currently loaded
    84→  // file. Lazily created on first render so anonymous/logged-out demo use
    85→  // never touches the DB — see ensureBackendSession() below.
    86→  const backendSessionId = useRef<string | null>(null)

    98→
    99→  // Cleanup progress interval on unmount
   100→  useEffect(() => {

   103→        clearInterval(progressIntervalRef.current)

   110→

   117→  // P2-4: when Simple Mode is toggled ON, immediately apply the deterministic

   127→    [simpleIntensity, genre, setMacros, setMacroSource, setSimpleMode],

   129→
   130→  // BETA-ANALYTICS: creates (once per loaded file) a real Session row via
   131→  // POST /api/rain/session, so activation/retention/funnel math has
   132→  // something real to read. Silently no-ops if the caller isn't logged in
   133→  // or the request fails — analytics must never block the mastering flow.
   134→  const ensureBackendSession = useCallback(async () => {
   135→    if (backendSessionId.current && backendSessionFileName.current === fileName) {
   136→      return backendSessionId.current
   137→    }
   138→    try {
   139→      const res = await fetch('/api/rain/session', {
   140→        method: 'POST',
   141→        headers: { 'Content-Type': 'application/json' },
   142→        body: JSON.stringify({ fileName, anonId: getAnonId() }),
   143→      })
   144→      if (!res.ok) return null
   145→      const data = (await res.json()) as { sessionId: string | null }
   146→      backendSessionId.current = data.sessionId
   147→      backendSessionFileName.current = fileName
   148→      return data.sessionId
   149→    } catch (e) {
   150→      console.warn('[analytics] ensureBackendSession failed:', e)
   151→      return null
   152→    }
   153→  }, [fileName])
   154→
   155→  // BETA-ANALYTICS: fires a render_completed or export_completed ping.
   156→  // Best-effort — failures are logged, never surfaced to the user.
   157→  const reportRenderEvent = useCallback(
   158→    async (
   159→      kind: 'render' | 'export',
   160→      details?: {
   161→        format?: string
   162→        outputFileHash?: string
   163→        loudnessLufs?: number
   164→        truePeakDbfs?: number
   165→        renderTimeMs?: number
   166→      },
   167→    ) => {
   168→      try {
   169→        await fetch('/api/rain/render', {
   170→          method: 'POST',
   171→          headers: { 'Content-Type': 'application/json' },
   172→          body: JSON.stringify({ kind, sessionId: backendSessionId.current, anonId: getAnonId(), ...details }),
   173→        })
   174→      } catch (e) {
   175→        console.warn(`[analytics] reportRenderEvent(${kind}) failed:`, e)
   176→      }
   177→    },
   178→    [],
   179→  )
   180→
   181→  const handleRender = async () => {
   182→    if (!fileName || isProcessing) return
   183→    setIsProcessing(true)
   184→    renderStartTime.current = Date.now()
   185→    // BETA-ANALYTICS: fire-and-forget — don't gate the render on this.
   186→    void ensureBackendSession()
   187→
   188→    // P3-BSROFORMER: clear any stale stem results from a previous render
   189→    // or manual Stems-tab run. Fresh results are populated from Stage 7
   190→    // of the render pipeline via the onStemsReady callback (below).
   191→    setStemResults(null)
   192→
   193→    // AUDIT-C5 FIX: create an AbortController for this render and store it so
   194→    // the Cancel button (ProcessingProgressPanel → cancelProcessing) can abort
   195→    // the in-flight render. Previously Cancel only flipped a flag the engine
   196→    // never checked, so the render kept running while the UI returned to idle.
   197→    const abortController = new AbortController()
   198→    setRenderAbortController(abortController)
   199→
   200→    // AUDIT-M9 FIX: previously used wall-clock time to fake a climbing progress
   201→    // bar within each stage (`stageProgress = (elapsed / 300) * 100`). That was
   202→    // fabricated — it filled the bar based on time, not actual DSP work.
   203→    // Now the within-stage indicator climbs gently toward 90% and resets to 0
   204→    // whenever the engine reports a real stage boundary (onProgress). The
   205→    // overall progress bar reflects real stage completion (stage/16 * 100).
   206→    const stageStartRef = { current: Date.now() }
   207→    progressIntervalRef.current = setInterval(() => {
   208→      const elapsedSinceStage = Date.now() - stageStartRef.current
   209→      // Climb toward 90% over ~2 seconds, then hold. Never reaches 100%
   210→      // until the engine confirms the next stage — honest "working" state.
   211→      const stageProgress = Math.min(90, (elapsedSinceStage / 2000) * 90)
   212→      setProcessingStageProgress(stageProgress)
   213→    }, 100)
   214→
   215→    try {
   216→      const result = await audioEngine.render(macros, genre, platform, (stage, total, name) => {
   217→        setProgress(stage, total, name)
   218→        // Reset the within-stage indicator — real stage boundary reached.
   219→        stageStartRef.current = Date.now()
   220→        setProcessingStageProgress(0)
   221→      }, abortController.signal, (stems) => {
   222→        // P3-BSROFORMER: Stage 7 emitted BS-RoFormer 4-pass stem results.
   223→        // Populate the session store so the Stems tab shows them immediately
   224→        // (no separate "Run Separation" click required).
   225→        setStemResults(stems)
   226→      }, simpleMode)
   227→
   228→      // Clear the progress interval
   229→      if (progressIntervalRef.current) {
   230→        clearInterval(progressIntervalRef.current)
   231→        progressIntervalRef.current = null
   232→      }
   233→      setProcessingStageProgress(100)
   234→
   235→      // Generate provenance certificate
   236→      const inputWavBlob = audioEngine.exportOriginalWav(24)
   237→      const outputWavBlob = audioEngine.exportWav(24)
   238→      const inputBuffer = await inputWavBlob.arrayBuffer()
   239→      const outputBuffer = await outputWavBlob.arrayBuffer()
   240→      const cert = await generateProvenance({
   241→        inputBuffer,
   242→        outputBuffer,
   243→        params: result.params as unknown as Record<string, unknown>,
   244→        analysis: {
   245→          lufs: result.analysis.lufs,
   246→          truePeak: result.analysis.truePeak,
   247→          rms: result.analysis.rms,
   248→          dynamicRange: result.analysis.dynamicRange,
   249→          bpm: result.analysis.bpm,
   250→          key: result.analysis.key,
   251→        },
   252→        // AUDIT-M8 FIX: pass the real output audio channels so a Chromaprint
   253→        // fingerprint hash is computed and embedded in the manifest.
   254→        outputChannels: audioEngine.getProcessedChannels() ?? undefined,
   255→        // P3-TPDF-MP3: pass the FLOAT32 input/output channels so the RAIN-CERT
   256→        // SHA-256 input/output hashes are computed over the deterministic DSP
   257→        // float buffer — NOT over the integer WAV bytes (which now carry TPDF
   258→        // dither noise at Stage 15 and would otherwise change every render).
   259→        inputChannels: audioEngine.getInputChannels() ?? undefined,
   260→        sampleRate: result.analysis.sampleRate,
   261→      })
   262→      setCert(cert)
   263→      setResult(result.analysis, result.score, result.params)
   264→      setInputAnalysis(audioEngine.currentAnalysis ?? useSessionStore.getState().inputAnalysis!)
   265→
   266→      // Log this render to session history
   267→      // P3-ANALYTICS: `duration` is the AUDIO file duration in SECONDS
   268→      // (used by computeAnalytics for storage-size + minutes-processed math),
   269→      // NOT the render wall-clock time. Render time is recorded separately
   270→      // via recordRenderTelemetry → IndexedDB (both cumulative engineStats
   271→      // counters AND a per-render telemetry row for the Analytics tab).
   272→      const renderDuration = Date.now() - renderStartTime.current
   273→      const audioDurationSec = useSessionStore.getState().fileDuration || 0
   274→      const renderTimestamp = Date.now()
   275→      addRenderLog({
   276→        timestamp: renderTimestamp,
   277→        fileName: fileName,
   278→        genre,
   279→        platform,
   280→        macroValues: { ...macros },
   281→        inputLufs: inputAnalysis?.lufs ?? 0,
   282→        outputLufs: result.analysis.lufs,
   283→        rainScore: result.score.overall,
   284→        duration: audioDurationSec,
   285→      })
   286→
   287→      // P2-ANALYTICS: persist a full per-render telemetry record (real
   288→      // per-stage DSP timings, format, bit depth, macro values, score, etc.)
   289→      // to IndexedDB. recordRenderTelemetry internally bumps the cumulative
   290→      // EngineStats counters (totalRenders, totalDspTimeMs, firstRenderAt,
   291→      // lastRenderAt) — so we no longer call recordRenderStat separately.
   292→      // The returned id links the QC snapshot below to this render.
   293→      const fileState = useSessionStore.getState()
   294→      const telemetryRecord = await recordRenderTelemetry({
   295→        timestamp: renderTimestamp,
   296→        renderDurationMs: renderDuration,
   297→        audioDurationSec,
   298→        genre,
   299→        platform,
   300→        format: 'WAV-24',
   301→        bitDepth: fileState.fileBitDepth || 24,
   302→        sampleRate: fileState.fileSampleRate || 48000,
   303→        channels: fileState.fileChannels || 2,
   304→        fileName: fileName ?? '(unknown)',
   305→        rainScore: result.score.overall,
   306→        inputLufs: inputAnalysis?.lufs ?? 0,
   307→        outputLufs: result.analysis.lufs,
   308→        outputTruePeak: result.analysis.truePeak,
   309→        macroValues: { ...macros },
   310→        stageTimings: { ...(result.stageTimings ?? {}) },
   311→      }).catch((e) => {
   312→        console.warn('[analytics] recordRenderTelemetry failed:', e)
   313→        return null
   314→      })
   315→
   316→      // P2-ANALYTICS: persist a QC snapshot for this render. Same
   317→      // computeQCResults() that QCTab uses for live display, summarized
   318→      // into the persisted QCRecord shape (pass/warn/fail counts + per-
   319→      // category tally + per-check status). The Analytics tab reads this
   320→      // from IndexedDB to compute pass/fail rates and common failure
   321→      // categories across the render history.
   322→      try {
   323→        const platformTarget = PLATFORM_TARGETS.find((p) => p.slug === platform) ?? PLATFORM_TARGETS[0]
   324→        const qcChecks = computeQCResults(result.analysis, platformTarget, cert)
   325→        const qcBody = summarizeQCResults(
   326→          qcChecks,
   327→          platform,
   328→          telemetryRecord?.id ?? null,
   329→          renderTimestamp,
   330→        )
   331→        void recordQCResult(qcBody).catch((e) =>
   332→          console.warn('[analytics] recordQCResult failed:', e),
   333→        )
   334→      } catch (e) {
   335→        console.warn('[analytics] QC snapshot computation failed:', e)
   336→      }
   337→
   338→      notifySuccess('Master complete', `RAIN Score: ${result.score.overall}`)
   339→
   340→      // BETA-ANALYTICS: mark this master as produced. No format/hash yet —
   341→      // those are attached per-export in handleExport below.
   342→      void reportRenderEvent('render', {
   343→        loudnessLufs: result.analysis.lufs,
   344→        truePeakDbfs: result.analysis.truePeak,
   345→        renderTimeMs: renderDuration,
   346→      })
   347→
   348→      // Trigger completion celebration
   349→      triggerCompletionCelebration()
   350→    } catch (e) {
   351→      // Clear the progress interval on error
   352→      if (progressIntervalRef.current) {
   353→        clearInterval(progressIntervalRef.current)
   354→        progressIntervalRef.current = null
   355→      }
   356→      // AUDIT-C5 FIX: user-initiated cancel is not an error — exit silently.
   357→      if (e instanceof Error && (e.name === 'CancelledError' || e.message === 'Render cancelled by user')) {
   358→        notifyError('Render cancelled', 'Processing was stopped by the user')
   359→        return
   360→      }
   361→      console.error('[RAIN render] error:', e)
   362→      setError(e instanceof Error ? e.message : 'Render failed')
   363→      notifyError('Render failed', e instanceof Error ? e.message : 'Unknown error during 16-stage processing')
   364→    } finally {
   365→      // Always clear the stored controller so a stale aborted signal can't
   366→      // leak into the next render.
   367→      setRenderAbortController(null)
   368→    }
   369→    // BUG FIX: removed `finally { setIsProcessing(false) }` — it was overwriting
   370→    // the `status:'complete'` (set by setResult) and `status:'failed'` (set by
   371→    // setError) back to `'idle'`, so the footer always showed "Ready" after
   372→    // every render. setResult and setError both set isProcessing:false already.
   373→  }
   374→
   375→  const handleAiSuggest = async () => {
   376→    setAiLoading(true)
   377→    const trigger = (window as unknown as { __rainAiSuggest?: () => void }).__rainAiSuggest
   378→    if (trigger) trigger()
   379→    // visual feedback only; actual completion handled in panel
   380→    aiSuggestTimeoutRef.current = setTimeout(() => setAiLoading(false), 1500)
   381→  }
   382→
   383→  const handleExport = async () => {
   384→    // P3-ANALYTICS: measure real export wall-clock time + actual byte count
   385→    // and persist them via recordExportDetails (which writes a row to the
   386→    // `exports` IndexedDB store AND bumps the cumulative exportCount /
   387→    // totalExportBytes / totalExportTimeMs counters in engineStats).
   388→    const exportStart = Date.now()
   389→    try {
   390→      const blob = audioEngine.exportWav(24)
   391→      const url = URL.createObjectURL(blob)
   392→      const a = document.createElement('a')
   393→      a.href = url
   394→      a.download = `${(fileName ?? 'rain-master').replace(/\.[^.]+$/, '')}_RAIN_mastered.wav`
   395→      document.body.appendChild(a)
   396→      a.click()
   397→      document.body.removeChild(a)
   398→      URL.revokeObjectURL(url)
   399→      const exportMs = Date.now() - exportStart
   400→      const fileState = useSessionStore.getState()
   401→      void recordExportDetails({
   402→        timestamp: Date.now(),
   403→        format: 'WAV-24',
   404→        bytes: blob.size,
   405→        durationMs: exportMs,
   406→        sampleRate: fileState.fileSampleRate || 48000,
   407→        bitDepth: 24,
   408→        channels: fileState.fileChannels || 2,
   409→        fileName: fileName ?? '(unknown)',
   410→      }).catch((e) =>
   411→        console.warn('[analytics] recordExportDetails failed:', e),
   412→      )
   413→      notifySuccess('WAV exported successfully', '24-bit PCM · RAIN mastered')
   414→
   415→      // BETA-ANALYTICS: this is the event that counts as "activation" —
   416→      // signup -> completed export. rainCert is set in the store by
   417→      // handleRender above, so outputHash is the real RAIN-CERT hash.
   418→      void reportRenderEvent('export', {
   419→        format: 'wav24',
   420→        outputFileHash: rainCert?.outputHash,
   421→      })
   422→    } catch (e) {
   423→      console.error('[RAIN export] error:', e)
   424→      notifyError('Export failed', e instanceof Error ? e.message : 'Could not export WAV file')
   425→    }
   426→  }
   427→
   428→  return (
   429→    <div className="grid lg:grid-cols-12 gap-4">
   430→      {/* Left column — main workflow */}
   431→      <div className="lg:col-span-8 space-y-4">
   432→        <UploadZone />
   433→
   434→        {fileName && (
   435→          <>
   436→            {/* Genre + Platform selectors */}
   437→            <div className="rain-panel rounded-lg p-4 grid sm:grid-cols-2 gap-3">
   438→              <div>
   439→                <label className="text-[10px] font-mono uppercase tracking-wider text-muted-foreground block mb-1">
   440→                  Genre
   441→                </label>
   442→                <select
   443→                  value={genre}
   444→                  onChange={(e) => setGenre(e.target.value)}
   445→                  className="w-full bg-rain-surface-2 border border-rain-border rounded-md px-3 py-2 text-sm focus:outline-none focus:border-rain-accent/50"
   446→                >
   447→                  {GENRES.map((g) => (
   448→                    <option key={g} value={g}>{g.charAt(0).toUpperCase() + g.slice(1)}</option>
   449→                  ))}
   450→                </select>
   451→              </div>
   452→              <div>
   453→                <label className="text-[10px] font-mono uppercase tracking-wider text-muted-foreground block mb-1">
   454→                  Platform Target ({PLATFORM_TARGETS.length})
   455→                </label>
   456→                <select
   457→                  value={platform}
   458→                  onChange={(e) => setPlatform(e.target.value)}
   459→                  className="w-full bg-rain-surface-2 border border-rain-border rounded-md px-3 py-2 text-sm focus:outline-none focus:border-rain-accent/50"
   460→                >
   461→                  {PLATFORM_TARGETS.map((p) => (
   462→                    <option key={p.slug} value={p.slug}>
   463→                      {p.label} · {p.targetLufs} LUFS · {p.codec}
   464→                    </option>
   465→                  ))}
   466→                </select>
   467→              </div>
   468→            </div>
   469→
   470→            {/* Genre Presets */}
   471→            <GenrePresets />
   472→
   473→            {/* P2-4: Simple / Pro mode toggle */}
   474→            <div className="rain-panel rounded-lg p-3 flex items-center justify-between">
   475→              <div className="flex items-center gap-3">
   476→                <Switch
   477→                  checked={simpleMode}
   478→                  onCheckedChange={handleSimpleModeToggle}
   479→                  aria-label="Toggle Simple Mode"
   480→                />
   481→                <div>
   482→                  <div className="text-[10px] font-mono uppercase tracking-wider text-muted-foreground">
   483→                    Mode
   484→                  </div>
   485→                  <div className="text-sm font-semibold">
