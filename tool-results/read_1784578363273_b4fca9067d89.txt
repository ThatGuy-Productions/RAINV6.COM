    81→function formatTrendPct(pct: number | null): string {
    82→  if (pct === null || !Number.isFinite(pct)) return '—'
    83→  const sign = pct > 0 ? '+' : ''
    84→  return `${sign}${pct.toFixed(0)}%`
    85→}
    86→
    87→function formatTrendScore(delta: number | null): string {
    88→  if (delta === null || !Number.isFinite(delta)) return '—'
    89→  const sign = delta > 0 ? '+' : ''
    90→  return `${sign}${delta.toFixed(1)}`
    91→}

    93→/** Human-readable label for an ActivityRecord.type. */
    94→function activityTypeLabel(type: string): string {
    95→  switch (type) {
    96→    case 'undo': return 'Undo'
    97→    case 'redo': return 'Redo'
    98→    case 'repair': return 'Repair run'
    99→    case 'ai-query': return 'AI Co-Master query'
   100→    case 'render': return 'Render'
   101→    case 'export': return 'Export'
   102→    case 'preset-apply': return 'Preset applied'
   103→    default: return type.charAt(0).toUpperCase() + type.slice(1).replace(/-/g, ' ')
   104→  }
   105→}

   613→// Score history chart — real per-render score line over time
   614→// ---------------------------------------------------------------------------
   615→
   616→function ScoreHistoryChart({
   617→  entries,
   618→}: {
   619→  entries: Array<{ timestamp: number; score: number; platform: string }>
   620→}) {
   621→  const W = 300
   622→  const H = 120
   623→  const PAD = 8
   624→  const minScore = 50
   625→  const maxScore = 100
   626→
   627→  if (entries.length < 2) return null
   628→
   629→  const xs = entries.map((_, i) =>
   630→    PAD + (i / (entries.length - 1)) * (W - 2 * PAD),
   631→  )
   632→  const ys = entries.map((e) => {
   633→    const clamped = Math.max(minScore, Math.min(maxScore, e.score))
   634→    return H - PAD - ((clamped - minScore) / (maxScore - minScore)) * (H - 2 * PAD)
   635→  })
   636→
   637→  const points = xs.map((x, i) => `${x.toFixed(2)},${ys[i].toFixed(2)}`).join(' ')
   638→  const areaPoints = `${xs[0].toFixed(2)},${H - PAD} ${points} ${xs[xs.length - 1].toFixed(2)},${H - PAD}`
   639→
   640→  const refLine = (score: number) => {
   641→    const clamped = Math.max(minScore, Math.min(maxScore, score))
   642→    return H - PAD - ((clamped - minScore) / (maxScore - minScore)) * (H - 2 * PAD)
   643→  }
   644→
   645→  return (
   646→    <svg viewBox={`0 0 ${W} ${H}`} className="w-full h-32" preserveAspectRatio="none">
   647→      <defs>
   648→        <linearGradient id="scoreFillReal" x1="0" y1="0" x2="0" y2="1">
   649→          <stop offset="0%" stopColor="#AAFF00" stopOpacity="0.3" />
   650→          <stop offset="100%" stopColor="#AAFF00" stopOpacity="0" />
   651→        </linearGradient>
   652→      </defs>
   653→      <line x1={PAD} x2={W - PAD} y1={refLine(90)} y2={refLine(90)}
   654→        stroke="#AAFF00" strokeOpacity="0.2" strokeDasharray="2 3" />
   655→      <line x1={PAD} x2={W - PAD} y1={refLine(75)} y2={refLine(75)}
   656→        stroke="#F97316" strokeOpacity="0.2" strokeDasharray="2 3" />
   657→      <polygon points={areaPoints} fill="url(#scoreFillReal)" />
   658→      <polyline points={points} fill="none" stroke="#AAFF00" strokeWidth="2" />
   659→      {xs.map((x, i) => (
   660→        <circle key={i} cx={x} cy={ys[i]} r="2" fill="#AAFF00" />
   661→      ))}
   662→    </svg>
   663→  )
   664→}
