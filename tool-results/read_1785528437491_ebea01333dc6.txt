   138→   138→async function sha256Bytes(data: Uint8Array): Promise<string> {
   139→   139→  if (typeof crypto === 'undefined' || !crypto.subtle) {
   140→   140→    throw new Error('WebCrypto not available — SHA-256 requires crypto.subtle')
   141→   141→  }
   142→   142→  // crypto.subtle.digest requires ArrayBufferView backed by ArrayBuffer.
   143→   143→  // Some TS typings reject Uint8Array<ArrayBufferLike>; copy into a fresh
   144→   144→  // ArrayBuffer-backed Uint8Array to be safe across TS lib variants.
   145→   145→  const buf = new ArrayBuffer(data.byteLength)
   146→   146→  new Uint8Array(buf).set(data)
   147→   147→  const digest = await crypto.subtle.digest('SHA-256', buf)
   148→   148→  const view = new Uint8Array(digest)
   149→   149→  let hex = ''
   150→   150→  for (let i = 0; i < view.length; i++) hex += view[i].toString(16).padStart(2, '0')
   151→   151→  return hex
   152→   152→}

   154→   154→// ---------------------------------------------------------------------------
   155→   155→// CRC-32 (PKZIP polynomial 0xEDB88320, table-driven)
   156→   156→// ---------------------------------------------------------------------------
   157→   157→
   158→   158→let CRC_TABLE: Uint32Array | null = null
   159→   159→
   160→   160→function crc32Table(): Uint32Array {
   161→   161→  if (CRC_TABLE) return CRC_TABLE
   162→   162→  const table = new Uint32Array(256)
   163→   163→  for (let i = 0; i < 256; i++) {
   164→   164→    let c = i
   165→   165→    for (let k = 0; k < 8; k++) {
   166→   166→      c = (c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1)
   167→   167→    }
   168→   168→    table[i] = c >>> 0
   169→   169→  }
   170→   170→  CRC_TABLE = table
   171→   171→  return table
   172→   172→}
   173→   173→
   174→   174→function crc32(data: Uint8Array): number {
   175→   175→  const table = crc32Table()
   176→   176→  let crc = 0xFFFFFFFF
   177→   177→  for (let i = 0; i < data.length; i++) {
   178→   178→    crc = (table[(crc ^ data[i]) & 0xFF] ^ (crc >>> 8)) >>> 0
   179→   179→  }
   180→   180→  return (crc ^ 0xFFFFFFFF) >>> 0
   181→   181→}

   183→   183→// ---------------------------------------------------------------------------
   184→   184→// Minimal ZIP writer (PKZIP 2.0, store-only / method 0)
   185→   185→// ---------------------------------------------------------------------------
   186→   186→//
   187→   187→// We deliberately do not compress. The dominant asset is the WAV master
   188→   188→// (already large and entropy-rich thanks to TPDF dither); compressing it
   189→   189→// client-side would burn CPU for ~5-10% size savings and force us to either
   190→   190→// pull in a dep (fflate) or implement DEFLATE inline. Store-only keeps the
   191→   191→// implementation dependency-free and well under 100 LOC.
   192→   192→
   193→   193→interface ZipEntry {
   194→   194→  filename: string
   195→   195→  data: Uint8Array
   196→   196→  crc: number
   197→   197→}
   198→   198→
   199→   199→function buildZip(entries: ZipEntry[]): Uint8Array {
   200→   200→  const enc = new TextEncoder()
   201→   201→  const localParts: Uint8Array[] = []
   202→   202→  const centralParts: Uint8Array[] = []
   203→   203→  let offset = 0
   204→   204→  const now = new Date()
   205→   205→  // DOS date/time (matches PKZIP convention; both fields 0 is also valid but
   206→   206→  // we use the real date so unzippers show a sensible timestamp).
   207→   207→  const dosTime = ((now.getHours() & 0x1F) << 11) | ((now.getMinutes() & 0x3F) << 5) | ((now.getSeconds() / 2) & 0x1F)
   208→   208→  const dosDate = ((((now.getFullYear() - 1980) & 0x7F) << 9) | (((now.getMonth() + 1) & 0x0F) << 5) | (now.getDate() & 0x1F))
   209→   209→
   210→   210→  for (const entry of entries) {
   211→   211→    const nameBytes = enc.encode(entry.filename)
   212→   212→    const crc = entry.crc
   213→   213→    const size = entry.data.length
   214→   214→
   215→   215→    // Local file header (30 bytes) + filename + data.
   216→   216→    const local = new Uint8Array(30 + nameBytes.length + size)
   217→   217→    const lv = new DataView(local.buffer)
   218→   218→    lv.setUint32(0, 0x04034b50, true)        // local file header signature
   219→   219→    lv.setUint16(4, 20, true)                // version needed to extract (2.0)
   220→   220→    lv.setUint16(6, 0, true)                 // general purpose bit flag
   221→   221→    lv.setUint16(8, 0, true)                 // compression method (0 = store)
   222→   222→    lv.setUint16(10, dosTime, true)          // last mod file time
   223→   223→    lv.setUint16(12, dosDate, true)          // last mod file date
   224→   224→    lv.setUint32(14, crc, true)              // CRC-32
   225→   225→    lv.setUint32(18, size, true)             // compressed size (== uncompressed for store)
   226→   226→    lv.setUint32(22, size, true)             // uncompressed size
   227→   227→    lv.setUint16(26, nameBytes.length, true) // filename length
   228→   228→    lv.setUint16(28, 0, true)                // extra field length
   229→   229→    local.set(nameBytes, 30)
   230→   230→    local.set(entry.data, 30 + nameBytes.length)
   231→   231→    localParts.push(local)
   232→   232→
   233→   233→    // Central directory header (46 bytes) + filename.
   234→   234→    const central = new Uint8Array(46 + nameBytes.length)
   235→   235→    const cv = new DataView(central.buffer)
   236→   236→    cv.setUint32(0, 0x02014b50, true)        // central file header signature
   237→   237→    cv.setUint16(4, 20, true)                // version made by
   238→   238→    cv.setUint16(6, 20, true)                // version needed to extract
   239→   239→    cv.setUint16(8, 0, true)                 // general purpose bit flag
   240→   240→    cv.setUint16(10, 0, true)                // compression method
   241→   241→    cv.setUint16(12, dosTime, true)          // last mod file time
   242→   242→    cv.setUint16(14, dosDate, true)          // last mod file date
   243→   243→    cv.setUint32(16, crc, true)              // CRC-32
   244→   244→    cv.setUint32(20, size, true)             // compressed size
   245→   245→    cv.setUint32(24, size, true)             // uncompressed size
   246→   246→    cv.setUint16(28, nameBytes.length, true) // filename length
   247→   247→    cv.setUint16(30, 0, true)                // extra field length
   248→   248→    cv.setUint16(32, 0, true)                // file comment length
   249→   249→    cv.setUint16(34, 0, true)                // disk number start
   250→   250→    cv.setUint16(36, 0, true)                // internal file attributes
   251→   251→    cv.setUint32(38, 0, true)                // external file attributes
   252→   252→    cv.setUint32(42, offset, true)           // relative offset of local header
   253→   253→    central.set(nameBytes, 46)
   254→   254→    centralParts.push(central)
   255→   255→
   256→   256→    offset += local.length
   257→   257→  }
   258→   258→
   259→   259→  const centralSize = centralParts.reduce((s, p) => s + p.length, 0)
   260→   260→  const centralOffset = offset
   261→   261→
   262→   262→  // End of central directory record (22 bytes).
   263→   263→  const eocd = new Uint8Array(22)
   264→   264→  const ev = new DataView(eocd.buffer)
   265→   265→  ev.setUint32(0, 0x06054b50, true)              // EOCD signature
   266→   266→  ev.setUint16(4, 0, true)                       // number of this disk
   267→   267→  ev.setUint16(6, 0, true)                       // disk where central directory starts
   268→   268→  ev.setUint16(8, entries.length, true)          // entries on this disk
   269→   269→  ev.setUint16(10, entries.length, true)         // total entries
   270→   270→  ev.setUint32(12, centralSize, true)            // central directory size
   271→   271→  ev.setUint32(16, centralOffset, true)          // offset of central directory
   272→   272→  ev.setUint16(20, 0, true)                      // comment length
   273→   273→
   274→   274→  // Concatenate everything.
   275→   275→  const totalLen = localParts.reduce((s, p) => s + p.length, 0)
   276→   276→    + centralSize
   277→   277→    + 22
   278→   278→  const out = new Uint8Array(totalLen)
   279→   279→  let pos = 0
   280→   280→  for (const p of localParts) { out.set(p, pos); pos += p.length }
   281→   281→  for (const p of centralParts) { out.set(p, pos); pos += p.length }
   282→   282→  out.set(eocd, pos)
   283→   283→  return out
   284→   284→}

   348→   348→export function escapeXml(s: string): string {
   349→   349→  return s
   350→   350→    .replace(/&/g, '&amp;')
   351→   351→    .replace(/</g, '&lt;')
   352→   352→    .replace(/>/g, '&gt;')
   353→   353→    .replace(/"/g, '&quot;')
   354→   354→    .replace(/'/g, '&apos;')
   355→   355→}

   357→   357→function isoDurationFromSeconds(totalSeconds?: number): string {
   358→   358→  if (!totalSeconds || totalSeconds <= 0) return 'PT0S'
   359→   359→  const s = Math.max(0, Math.floor(totalSeconds))
   360→   360→  const m = Math.floor(s / 60)
   361→   361→  const r = s - m * 60
   362→   362→  return `PT${m}M${r}S`
   363→   363→}

   422→   422→          const ipi = c.ipi ? `\n        <IPI>${escapeXml(c.ipi)}</IPI>` : ''
   423→   423→          const isni = c.isni ? `\n        <ISNI>${escapeXml(c.isni)}</ISNI>` : ''
   424→   424→          const share = typeof c.share === 'number' && !Number.isNaN(c.share)
   425→   425→            ? `\n        <Share>${c.share}</Share>`
   426→   426→            : ''
   427→   427→          return `      <ResourceContributor>\n        <PartyName><FullName>${escapeXml(c.name || '')}</FullName></PartyName>\n        <Role>${escapeXml(c.role)}</Role>${ipi}${isni}${share}\n      </ResourceContributor>`
   428→   428→        })
   429→   429→        .join('\n')
   430→   430→    : ''
   431→   431→
   432→   432→  const disclosureFields = ['vocals', 'instrumentation', 'composition', 'mixing', 'mastering']
   433→   433→  const disclosureBlock = disclosureFields
   434→   434→    .map((f) => `      <${f}>${meta.aiDisclosure?.[f] ?? 'none'}</${f}>`)
   435→   435→    .join('\n')
   436→   436→
   437→   437→  const dsps = meta.targetDsps && meta.targetDsps.length > 0
   438→   438→    ? meta.targetDsps
   439→   439→    : ['spotify', 'apple_music', 'youtube', 'tidal']
   440→   440→  const dealBlock = dsps
   441→   441→    .map((slug) => {
   442→   442→      const label = meta.dspLabels?.[slug]
   443→   443→        ? escapeXml(meta.dspLabels[slug])
   444→   444→        : escapeXml(slug)
   445→   445→      return `    <Deal>\n      <DSPName>${label}</DSPName>\n      <CommercialModel>PayAsYouGo</CommercialModel>\n    </Deal>`
   446→   446→    })
   447→   447→    .join('\n')
   448→   448→
   449→   449→  // Build optional blocks as a single string for clean insertion.
   450→   450→  const optionalReleaseBlocks = [
   451→   451→    releaseTypeBlock,
   452→   452→    pLineBlock,
   453→   453→    cLineBlock,
   454→   454→    labelBlock,
   455→   455→    distributorBlock,
   456→   456→    languageBlock,
   457→   457→    explicitBlock,
   458→   458→    publisherBlock,
   459→   459→    masterOwnerBlock,
   460→   460→    territoriesBlock,
   461→   461→    originalReleaseDate ? `    <OriginalReleaseDate>${originalReleaseDate}</OriginalReleaseDate>` : '',
   462→   462→    genreSub ? `    <SubGenre>${genreSub}</SubGenre>` : '',
   463→   463→  ].filter(Boolean).join('\n')
   464→   464→
   465→   465→  return `<?xml version="1.0" encoding="UTF-8"?>
   466→   466→<ern:NewReleaseMessage xmlns:ern="http://ddex.net/xml/ern/43" LanguageAndScriptCode="en">
   467→   467→  <MessageHeader>
   468→   468→    <MessageId>${messageId}</MessageId>
   469→   469→    <MessageCreatedDateTime>${created}</MessageCreatedDateTime>
   470→   470→    <MessageSender>
   471→   471→      <PartyName><FullName>RAIN V6 Studio</FullName></PartyName>
   472→   472→    </MessageSender>
   473→   473→  </MessageHeader>
   474→   474→  <ResourceList>
   475→   475→    <SoundRecording>
   476→   476→      <ResourceReference>A1</ResourceReference>
   477→   477→      <ISRC>${meta.isrc}</ISRC>
   478→   478→${iswcBlock}
   479→   479→      <ReferenceTitle>${title}</ReferenceTitle>
   480→   480→      <Duration>${duration}</Duration>
   481→   481→${contributorsBlock ? `      <ResourceContributorList>\n${contributorsBlock}\n      </ResourceContributorList>` : ''}
   482→   482→    </SoundRecording>
   483→   483→  </ResourceList>
   484→   484→  <Release>
   485→   485→    <ReleaseId>
   486→   486→      <ISRC>${meta.isrc}</ISRC>
   487→   487→      <UPC>${meta.upc}</UPC>
   488→   488→    </ReleaseId>
   489→   489→    <ReferenceTitle>${title}</ReferenceTitle>
   490→   490→    <DisplayArtist>
   491→   491→      <PartyName>
   492→   492→        <FullName>${artist}</FullName>
   493→   493→      </PartyName>
   494→   494→    </DisplayArtist>
   495→   495→${album ? `    <ReleaseDetails><DisplayTitleText>${album}</DisplayTitleText></ReleaseDetails>\n` : ''}    <Genre>${genre}</Genre>
   496→   496→    <YearOfOriginalRelease>${year}</YearOfOriginalRelease>
   497→   497→    <ReleaseDate>${releaseDate}</ReleaseDate>
   498→   498→${optionalReleaseBlocks ? optionalReleaseBlocks + '\n' : ''}    <AIInvolvement>
   499→   499→${disclosureBlock}
   500→   500→    </AIInvolvement>
   501→   501→  </Release>
   502→   502→  <DealList>
   503→   503→${dealBlock}
   504→   504→  </DealList>
   505→   505→</ern:NewReleaseMessage>`
   506→   506→}

   508→   508→// ---------------------------------------------------------------------------
   509→   509→// DDEX validator — well-formedness + required fields + ISRC format + UPC check
   510→   510→// ---------------------------------------------------------------------------
   511→   511→
   512→   512→export interface DdexValidationResult {
   513→   513→  ok: boolean
   514→   514→  errors: string[]
   515→   515→}
   516→   516→
   517→   517→/** ISO 3901 ISRC: CC-XXX-YY-NNNNN (12 alphanumeric, no dashes in raw form). */
   518→   518→const ISRC_RE = /^[A-Z]{2}[A-Z0-9]{3}[0-9]{7}$/
   519→   519→
   520→   520→/**
   521→   521→ * Validate a 12-digit UPC (EAN-13 minus the leading 0 — same check digit math).
   522→   522→ * Returns true if the embedded check digit matches the mod-10 checksum.
   523→   523→ */
   524→   524→export function validateUpcCheckDigit(upc: string): boolean {
   525→   525→  if (!/^\d{12}$/.test(upc)) return false
   526→   526→  let sum = 0
   527→   527→  for (let i = 0; i < 11; i++) {
   528→   528→    sum += parseInt(upc[i], 10) * (i % 2 === 0 ? 3 : 1)
   529→   529→  }
   530→   530→  const check = (10 - (sum % 10)) % 10
   531→   531→  return check === parseInt(upc[11], 10)
   532→   532→}
   533→   533→
   534→   534→export function validateIsrcFormat(isrc: string): boolean {
   535→   535→  return ISRC_RE.test(isrc.toUpperCase())
   536→   536→}
   537→   537→
   538→   538→export function validateDdex(xml: string): DdexValidationResult {
   539→   539→  const errors: string[] = []
   540→   540→  if (typeof DOMParser === 'undefined') {
   541→   541→    // Server-side path is not used for validation (the browser builds the
   542→   542→    // package). If we ever need server-side validation, we can parse via a
   543→   543→    // regex sanity pass; for now, surface the limitation honestly.
   544→   544→    errors.push('DOMParser not available in this environment')
   545→   545→    return { ok: false, errors }
   546→   546→  }
   547→   547→  const doc = new DOMParser().parseFromString(xml, 'application/xml')
   548→   548→  const parseError = doc.getElementsByTagName('parsererror')[0]
   549→   549→  if (parseError) {
   550→   550→    errors.push(`XML not well-formed: ${parseError.textContent?.slice(0, 200) ?? 'parse error'}`)
   551→   551→    return { ok: false, errors }
   552→   552→  }
   553→   553→  // Required elements per ERN 4.3.2 NewReleaseMessage.
   554→   554→  const required = [
   555→   555→    { tag: 'MessageId', parent: 'MessageHeader' },
   556→   556→    { tag: 'MessageCreatedDateTime', parent: 'MessageHeader' },
   557→   557→    { tag: 'ResourceReference', parent: 'SoundRecording' },
   558→   558→    { tag: 'ISRC', parent: 'SoundRecording' },
   559→   559→    { tag: 'ReferenceTitle', parent: 'SoundRecording' },
   560→   560→    { tag: 'Duration', parent: 'SoundRecording' },
   561→   561→    { tag: 'UPC', parent: 'ReleaseId' },
   562→   562→    { tag: 'DisplayArtist', parent: 'Release' },
   563→   563→    { tag: 'AIInvolvement', parent: 'Release' },
   564→   564→  ]
   565→   565→  for (const { tag } of required) {
   566→   566→    if (!doc.getElementsByTagName(tag).length) {
   567→   567→      errors.push(`Missing required element: <${tag}>`)
   568→   568→    }
   569→   569→  }
   570→   570→  // ISRC format check.
   571→   571→  const isrcEl = doc.getElementsByTagName('ISRC')[0]
   572→   572→  if (isrcEl) {
   573→   573→    const isrc = (isrcEl.textContent ?? '').trim()
   574→   574→    if (!validateIsrcFormat(isrc)) {
   575→   575→      errors.push(`ISRC '${isrc}' does not match ISO 3901 (CC-XXX-YY-NNNNN)`)
   576→   576→    }
   577→   577→  }
   578→   578→  // UPC check digit.
   579→   579→  const upcEl = doc.getElementsByTagName('UPC')[0]
   580→   580→  if (upcEl) {
   581→   581→    const upc = (upcEl.textContent ?? '').trim()
   582→   582→    if (!validateUpcCheckDigit(upc)) {
   583→   583→      errors.push(`UPC '${upc}' check digit is invalid`)
   584→   584→    }
   585→   585→  }
   586→   586→  // Root element must be ern:NewReleaseMessage.
   587→   587→  const root = doc.documentElement
   588→   588→  if (!root || root.localName !== 'NewReleaseMessage') {
   589→   589→    errors.push('Root element must be NewReleaseMessage (ern namespace)')
   590→   590→  }
   591→   591→  return { ok: errors.length === 0, errors }
   592→   592→}
   593→   593→
   594→   594→// ---------------------------------------------------------------------------
   595→   595→// Artwork validation (Spotify / Apple spec)
   596→   596→// ---------------------------------------------------------------------------
   597→   597→
   598→   598→export interface ArtworkValidationOk {
   599→   599→  ok: true
   600→   600→  dimensions: [number, number]
   601→   601→  format: 'image/jpeg' | 'image/png'
   602→   602→  sizeBytes: number
   603→   603→}
   604→   604→export interface ArtworkValidationFail {
   605→   605→  ok: false
   606→   606→  error: string
   607→   607→}
   608→   608→export type ArtworkValidation = ArtworkValidationOk | ArtworkValidationFail
   609→   609→
   610→   610→const ARTWORK_MIN = 1400
   611→   611→const ARTWORK_MAX = 3000
   612→   612→const ARTWORK_MAX_BYTES = 25 * 1024 * 1024 // 25 MB
   613→   613→
   614→   614→/**
   615→   615→ * Validate a cover-art File against DSP delivery specs:
   616→   616→ *   • JPEG or PNG
   617→   617→ *   • Square aspect ratio (1:1)
   618→   618→ *   • Dimensions between 1400×1400 and 3000×3000
   619→   619→ *   • File size ≤ 25 MB
   620→   620→ *
   621→   621→ * Uses `createImageBitmap` to decode the image and inspect pixel dimensions
   622→   622→ * (no DOM rendering required). Returns the validated dimensions + format on
   623→   623→ * success.
   624→   624→ */
   625→   625→export async function validateArtwork(
   626→   626→  file: File,
   627→   627→): Promise<ArtworkValidation> {
   628→   628→  if (file.size > ARTWORK_MAX_BYTES) {
   629→   629→    return { ok: false, error: `Artwork file size ${file.size} bytes exceeds 25 MB limit` }
   630→   630→  }
   631→   631→  const format = file.type === 'image/jpeg' || file.type === 'image/png'
   632→   632→    ? (file.type as 'image/jpeg' | 'image/png')
   633→   633→    : null
   634→   634→  if (!format) {
   635→   635→    return { ok: false, error: `Artwork must be JPEG or PNG (got ${file.type || 'unknown'})` }
   636→   636→  }
   637→   637→  if (typeof createImageBitmap === 'undefined') {
   638→   638→    return { ok: false, error: 'createImageBitmap not available in this environment' }
   639→   639→  }
   640→   640→  let bitmap: ImageBitmap
   641→   641→  try {
   642→   642→    bitmap = await createImageBitmap(file)
   643→   643→  } catch (e) {
   644→   644→    return { ok: false, error: `Failed to decode image: ${e instanceof Error ? e.message : 'unknown'}` }
   645→   645→  }
   646→   646→  const { width, height } = bitmap
   647→   647→  bitmap.close?.()
   648→   648→  if (width < ARTWORK_MIN || height < ARTWORK_MIN) {
   649→   649→    return { ok: false, error: `Artwork dimensions ${width}×${height} below minimum ${ARTWORK_MIN}×${ARTWORK_MIN}` }
   650→   650→  }
   651→   651→  if (width > ARTWORK_MAX || height > ARTWORK_MAX) {
   652→   652→    return { ok: false, error: `Artwork dimensions ${width}×${height} above maximum ${ARTWORK_MAX}×${ARTWORK_MAX}` }
   653→   653→  }
   654→   654→  if (width !== height) {
   655→   655→    return { ok: false, error: `Artwork must be square (1:1) — got ${width}×${height}` }
   656→   656→  }
   657→   657→  return { ok: true, dimensions: [width, height], format, sizeBytes: file.size }
   658→   658→}
   659→   659→
   660→   660→// ---------------------------------------------------------------------------
   661→   661→// Package assembly
   662→   662→// ---------------------------------------------------------------------------
   663→   663→
   664→   664→/**
   665→   665→ * Build a complete distribution package from the rendered audio + metadata.
   666→   666→ *
   667→   667→ * Layout inside the ZIP:
   668→   668→ *   ern.xml                — DDEX ERN 4.3.2 message
   669→   669→ *   manifest.json          — ReleaseManifest (assets, hashes, metadata)
   670→   670→ *   checksums.txt          — `<sha256>  <path>` lines for every asset
   671→   671→ *   assets/<title>.wav     — 24-bit WAV master (TPDF-dithered)
   672→   672→ *   assets/<title>.mp3     — 320 kbps MP3 (LAME, TPDF-dithered Int16 input)
   673→   673→ *   artwork/cover.<ext>    — cover art (if supplied)
   674→   674→ *
   675→   675→ * The function:
   676→   676→ *   1. Renders the AudioBuffer to WAV + MP3 via the real audio-engine encoders.
   677→   677→ *   2. Validates the DDEX XML (well-formedness + required fields + ISRC/UPC).
   678→   678→ *   3. Computes SHA-256 over every asset, the manifest, the checksums file.
   679→   679→ *   4. Packs everything into a real ZIP Blob (store-only, with CRC-32).
   680→   680→ *   5. Computes the package SHA-256 (the ZIP bytes themselves).
   681→   681→ *
   682→   682→ * Throws if DDEX validation fails OR if the WAV/MP3 encoders throw.
   683→   683→ */
   684→   684→export async function buildDistributionPackage(
   685→   685→  audioBuffer: AudioBuffer,
   686→   686→  metadata: {
   687→   687→    title: string
   688→   688→    artist: string
   689→   689→    album?: string
   690→   690→    genre?: string
   691→   691→    /** DDEX genre:subgenre. TASK B enrichment. */
   692→   692→    genreSubgenre?: string
   693→   693→    year?: string
   694→   694→    isrc: string
   695→   695→    upc: string
   696→   696→    /** ISWC for the underlying composition. TASK B enrichment. */
   697→   697→    iswc?: string
   698→   698→    releaseDate?: string
   699→   699→    /** Original release date for re-releases. TASK B enrichment. */
   700→   700→    originalReleaseDate?: string
   701→   701→    /** Release type — single / ep / album / compilation. TASK B enrichment. */
   702→   702→    releaseType?: 'single' | 'ep' | 'album' | 'compilation'
   703→   703→    /** Label name. TASK B enrichment. */
   704→   704→    label?: string
   705→   705→    /** Distributor name. TASK B enrichment. */
   706→   706→    distributor?: string
   707→   707→    /** P-line holder. TASK B enrichment. */
   708→   708→    pLine?: string
   709→   709→    /** C-line. TASK B enrichment. */
   710→   710→    cLine?: string
   711→   711→    /** Publisher name. TASK B enrichment. */
   712→   712→    publisher?: string
   713→   713→    /** PRO / collecting society. TASK B enrichment. */
   714→   714→    pro?: string
   715→   715→    /** Master rights owner. TASK B enrichment. */
   716→   716→    masterOwner?: string
   717→   717→    /** ISO 639-2 lyrical language code. TASK B enrichment. */
   718→   718→    language?: string
   719→   719→    /** Explicit-lyrics rating. TASK B enrichment. */
   720→   720→    explicitLyrics?: 'none' | 'explicit' | 'clean'
   721→   721→    /** Parental Advisory flag. TASK B enrichment. */
   722→   722→    parentalAdvisory?: boolean
   723→   723→    /** Territories (ISO 3166 codes or ['WORLDWIDE']). TASK B enrichment. */
   724→   724→    territories?: string[]
   725→   725→    /** Track contributors. TASK B enrichment. */
   726→   726→    contributors?: Array<{
   727→   727→      name: string
   728→   728→      role: string
   729→   729→      ipi?: string
   730→   730→      isni?: string
   731→   731→      share?: number
   732→   732→    }>
   733→   733→    durationSeconds?: number
   734→   734→    aiDisclosure?: Record<string, 'none' | 'assisted' | 'generated'>
   735→   735→    targetDsps?: string[]
   736→   736→    dspLabels?: Record<string, string>
   737→   737→  },
   738→   738→  artworkFile?: File,
   739→   739→): Promise<{
   740→   740→  manifest: ReleaseManifest
   741→   741→  packageBlob: Blob
   742→   742→  packageSha256: string
   743→   743→  packageBytes: Uint8Array
   744→   744→  validation: DdexValidationResult
   745→   745→}> {
   746→   746→  // 1) Build + validate the DDEX XML first — fail fast if metadata is bad.
   747→   747→  const ddexXml = buildDdexErnXml({
   748→   748→    title: metadata.title,
   749→   749→    artist: metadata.artist,
   750→   750→    album: metadata.album,
   751→   751→    genre: metadata.genre,
   752→   752→    genreSubgenre: metadata.genreSubgenre,
   753→   753→    year: metadata.year,
   754→   754→    isrc: metadata.isrc,
   755→   755→    upc: metadata.upc,
   756→   756→    iswc: metadata.iswc,
   757→   757→    releaseDate: metadata.releaseDate,
   758→   758→    originalReleaseDate: metadata.originalReleaseDate,
   759→   759→    releaseType: metadata.releaseType,
   760→   760→    label: metadata.label,
   761→   761→    distributor: metadata.distributor,
   762→   762→    pLine: metadata.pLine,
   763→   763→    cLine: metadata.cLine,
   764→   764→    publisher: metadata.publisher,
   765→   765→    pro: metadata.pro,
   766→   766→    masterOwner: metadata.masterOwner,
   767→   767→    language: metadata.language,
   768→   768→    explicitLyrics: metadata.explicitLyrics,
   769→   769→    parentalAdvisory: metadata.parentalAdvisory,
   770→   770→    territories: metadata.territories,
   771→   771→    contributors: metadata.contributors,
   772→   772→    durationSeconds: metadata.durationSeconds ?? audioBuffer.duration,
   773→   773→    aiDisclosure: metadata.aiDisclosure,
   774→   774→    targetDsps: metadata.targetDsps,
   775→   775→    dspLabels: metadata.dspLabels,
   776→   776→  })
   777→   777→  const validation = validateDdex(ddexXml)
   778→   778→  if (!validation.ok) {
   779→   779→    throw new Error(`DDEX validation failed: ${validation.errors.join('; ')}`)
   780→   780→  }
   781→   781→
   782→   782→  // 2) Render WAV + MP3. These call the real encoders in audio-engine.ts
   783→   783→  //    (audioBufferToWav appends a RIFF INFO chunk + TPDF dither; audioBufferToMp3
   784→   784→  //    runs real LAME + ID3v2.3 tag). We pass null provenance here because the
   785→   785→  //    distribution package attests to its own SHA-256s via the manifest; the
   786→   786→  //    RAIN-CERT provenance flow lives on the Provenance tab and is orthogonal.
   787→   787→  const wavBlob = audioBufferToWav(audioBuffer, 24, null)
   788→   788→  const mp3Blob = audioBufferToMp3(audioBuffer, 320, null)
   789→   789→
   790→   790→  // 3) Convert Blobs to Uint8Array (one copy each — fine for the typical
   791→   791→  //    30-100 MB master sizes we deal with).
   792→   792→  const wavBytes = new Uint8Array(await wavBlob.arrayBuffer())
   793→   793→  const mp3Bytes = new Uint8Array(await mp3Blob.arrayBuffer())
   794→   794→  const ernBytes = new TextEncoder().encode(ddexXml)
   795→   795→
   796→   796→  const safeTitle = (metadata.title || 'rain-release')
   797→   797→    .replace(/[^a-zA-Z0-9-_]+/g, '_')
   798→   798→    .slice(0, 64) || 'rain-release'
   799→   799→
   800→   800→  // 4) Compute SHA-256 for every asset.
   801→   801→  const assets: DistributionAsset[] = []
   802→   802→  const wavSha = await sha256Bytes(wavBytes)
   803→   803→  assets.push({
   804→   804→    type: 'audio',
   805→   805→    filename: `assets/${safeTitle}.wav`,
   806→   806→    mimeType: 'audio/wav',
   807→   807→    sizeBytes: wavBytes.byteLength,
   808→   808→    sha256: wavSha,
   809→   809→    data: wavBytes,
   810→   810→  })
   811→   811→  const mp3Sha = await sha256Bytes(mp3Bytes)
   812→   812→  assets.push({
   813→   813→    type: 'audio',
   814→   814→    filename: `assets/${safeTitle}.mp3`,
   815→   815→    mimeType: 'audio/mpeg',
   816→   816→    sizeBytes: mp3Bytes.byteLength,
   817→   817→    sha256: mp3Sha,
   818→   818→    data: mp3Bytes,
   819→   819→  })
   820→   820→
   821→   821→  // 5) Optional artwork.
   822→   822→  let artworkMeta: ReleaseManifest['artwork']
   823→   823→  if (artworkFile) {
   824→   824→    const art = await validateArtwork(artworkFile)
   825→   825→    if (!art.ok) {
   826→   826→      throw new Error(`Artwork validation failed: ${art.error}`)
   827→   827→    }
   828→   828→    const artBytes = new Uint8Array(await artworkFile.arrayBuffer())
   829→   829→    const artSha = await sha256Bytes(artBytes)
   830→   830→    const ext = art.format === 'image/jpeg' ? 'jpg' : 'png'
   831→   831→    assets.push({
   832→   832→      type: 'artwork',
   833→   833→      filename: `artwork/cover.${ext}`,
   834→   834→      mimeType: art.format,
   835→   835→      sizeBytes: artBytes.byteLength,
   836→   836→      sha256: artSha,
   837→   837→      data: artBytes,
   838→   838→    })
   839→   839→    artworkMeta = { width: art.dimensions[0], height: art.dimensions[1], format: art.format }
   840→   840→  }
   841→   841→
   842→   842→  // 6) Build the manifest (without `data` for each asset — manifest.json
   843→   843→  //    never embeds bytes, only the hash + metadata).
   844→   844→  const releaseId = `RAIN-RLS-${metadata.isrc}-${Date.now().toString(36)}`
   845→   845→  const manifest: ReleaseManifest = {
   846→   846→    releaseId,
   847→   847→    isrc: metadata.isrc,
   848→   848→    upc: metadata.upc,
   849→   849→    title: metadata.title,
   850→   850→    artist: metadata.artist,
   851→   851→    album: metadata.album ?? '',
   852→   852→    genre: metadata.genre ?? '',
   853→   853→    genreSubgenre: metadata.genreSubgenre,
   854→   854→    releaseDate: metadata.releaseDate ?? new Date().toISOString().slice(0, 10),
   855→   855→    originalReleaseDate: metadata.originalReleaseDate,
   856→   856→    releaseType: metadata.releaseType,
   857→   857→    label: metadata.label,
   858→   858→    distributor: metadata.distributor,
   859→   859→    pLine: metadata.pLine,
   860→   860→    cLine: metadata.cLine,
   861→   861→    publisher: metadata.publisher,
   862→   862→    pro: metadata.pro,
   863→   863→    masterOwner: metadata.masterOwner,
   864→   864→    iswc: metadata.iswc,
   865→   865→    language: metadata.language,
   866→   866→    explicitLyrics: metadata.explicitLyrics,
   867→   867→    parentalAdvisory: metadata.parentalAdvisory,
   868→   868→    territories: metadata.territories,
   869→   869→    contributors: metadata.contributors,
   870→   870→    ddexVersion: '4.3.2',
   871→   871→    generatedAt: new Date().toISOString(),
   872→   872→    assets: assets.map((a) => ({
   873→   873→      type: a.type,
   874→   874→      filename: a.filename,
   875→   875→      mimeType: a.mimeType,
   876→   876→      sizeBytes: a.sizeBytes,
   877→   877→      sha256: a.sha256,
   878→   878→    })),
   879→   879→    artwork: artworkMeta,
   880→   880→  }
   881→   881→  const manifestJson = JSON.stringify(manifest, null, 2)
   882→   882→  const manifestBytes = new TextEncoder().encode(manifestJson)
   883→   883→  const manifestSha = await sha256Bytes(manifestBytes)
   884→   884→
   885→   885→  // 7) checksums.txt — one line per asset + the manifest itself.
   886→   886→  const checksumsLines = assets.map((a) => `${a.sha256}  ${a.filename}`)
   887→   887→  checksumsLines.push(`${manifestSha}  manifest.json`)
   888→   888→  const checksumsTxt = checksumsLines.join('\n') + '\n'
   889→   889→  const checksumsBytes = new TextEncoder().encode(checksumsTxt)
   890→   890→  const checksumsSha = await sha256Bytes(checksumsBytes)
   891→   891→
   892→   892→  // 8) Push the metadata assets (ern.xml, manifest.json, checksums.txt).
   893→   893→  assets.push({
   894→   894→    type: 'metadata',
   895→   895→    filename: 'ern.xml',
   896→   896→    mimeType: 'application/xml',
   897→   897→    sizeBytes: ernBytes.byteLength,
   898→   898→    sha256: await sha256Bytes(ernBytes),
   899→   899→    data: ernBytes,
   900→   900→  })
   901→   901→  assets.push({
   902→   902→    type: 'metadata',
   903→   903→    filename: 'manifest.json',
   904→   904→    mimeType: 'application/json',
   905→   905→    sizeBytes: manifestBytes.byteLength,
   906→   906→    sha256: manifestSha,
   907→   907→    data: manifestBytes,
   908→   908→  })
   909→   909→  assets.push({
   910→   910→    type: 'metadata',
   911→   911→    filename: 'checksums.txt',
   912→   912→    mimeType: 'text/plain',
   913→   913→    sizeBytes: checksumsBytes.byteLength,
   914→   914→    sha256: checksumsSha,
   915→   915→    data: checksumsBytes,
   916→   916→  })
   917→   917→
   918→   918→  // 9) Build the ZIP. Order matters for human-inspection: ern.xml first, then
   919→   919→  //    manifest, then checksums, then assets/, then artwork/.
   920→   920→  const zipOrder = ['ern.xml', 'manifest.json', 'checksums.txt']
   921→   921→  const orderedAssets = [...assets].sort((a, b) => {
   922→   922→    const ia = zipOrder.indexOf(a.filename)
   923→   923→    const ib = zipOrder.indexOf(b.filename)
   924→   924→    if (ia !== -1 && ib !== -1) return ia - ib
   925→   925→    if (ia !== -1) return -1
   926→   926→    if (ib !== -1) return 1
   927→   927→    return a.filename.localeCompare(b.filename)
   928→   928→  })
   929→   929→  const entries: ZipEntry[] = orderedAssets.map((a) => ({
   930→   930→    filename: a.filename,
   931→   931→    data: a.data,
   932→   932→    crc: crc32(a.data),
   933→   933→  }))
   934→   934→  const zipBytes = buildZip(entries)
   935→   935→
   936→   936→  // 10) Compute the package SHA-256 (over the ZIP bytes themselves).
   937→   937→  const packageSha256 = await sha256Bytes(zipBytes)
   938→   938→  const packageBlob = new Blob([zipBytes.buffer as ArrayBuffer], { type: 'application/zip' })
   939→   939→
   940→   940→  return {
   941→   941→    manifest,
   942→   942→    packageBlob,
   943→   943→    packageSha256,
   944→   944→    packageBytes: zipBytes,
   945→   945→    validation,
   946→   946→  }
   947→   947→}

   950→   950→// IndexedDB delivery queue
   951→   951→// ---------------------------------------------------------------------------
   952→   952→
   953→   953→const DB_NAME = 'rain-distribution'
   954→   954→const DB_VERSION = 1
   955→   955→const JOB_STORE = 'delivery-jobs'
   956→   956→const PKG_STORE = 'delivery-packages'
   957→   957→
   958→   958→function openDistributionDb(): Promise<IDBDatabase> {
   959→   959→  return new Promise((resolve, reject) => {
   960→   960→    if (typeof indexedDB === 'undefined') {
   961→   961→      reject(new Error('IndexedDB not available'))
   962→   962→      return
   963→   963→    }
   964→   964→    const req = indexedDB.open(DB_NAME, DB_VERSION)
   965→   965→    req.onupgradeneeded = () => {
   966→   966→      const db = req.result
   967→   967→      if (!db.objectStoreNames.contains(JOB_STORE)) {
   968→   968→        db.createObjectStore(JOB_STORE, { keyPath: 'id' })
   969→   969→      }
   970→   970→      if (!db.objectStoreNames.contains(PKG_STORE)) {
   971→   971→        db.createObjectStore(PKG_STORE)
   972→   972→      }
   973→   973→    }
   974→   974→    req.onsuccess = () => resolve(req.result)
   975→   975→    req.onerror = () => reject(req.error)
   976→   976→  })
   977→   977→}
   978→   978→
   979→   979→async function withDb<T>(fn: (db: IDBDatabase) => Promise<T> | T): Promise<T> {
   980→   980→  const db = await openDistributionDb()
   981→   981→  try {
   982→   982→    return await fn(db)
   983→   983→  } finally {
   984→   984→    db.close()
   985→   985→  }
   986→   986→}

   988→   988→/**
   989→   989→ * Persist a delivery job (metadata only) to IndexedDB. If the job's status is
   990→   990→ * `packaged`, the package bytes are stored separately in `delivery-packages`
   991→   991→ * (keyed by job.id) so the queue list view can load all jobs without
   992→   992→ * deserialising multi-MB blobs.
   993→   993→ */
   994→   994→export async function persistDeliveryJob(
   995→   995→  job: DeliveryJob,
   996→   996→  packageBytes?: Uint8Array,
   997→   997→): Promise<void> {
   998→   998→  if (typeof indexedDB === 'undefined') return
   999→   999→  await withDb((db) => new Promise<void>((resolve, reject) => {
  1000→  1000→    const tx = db.transaction([JOB_STORE, PKG_STORE], 'readwrite')
  1001→  1001→    const jobStore = tx.objectStore(JOB_STORE)
  1002→  1002→    jobStore.put(job)
  1003→  1003→    if (packageBytes) {
  1004→  1004→      const pkgStore = tx.objectStore(PKG_STORE)
  1005→  1005→      // IndexedDB structured-clones its inputs; Uint8Array is supported
  1006→  1006→      // natively. Wrap in a fresh ArrayBuffer-backed array to satisfy the
  1007→  1007→      // structured-clone algorithm across TS lib variants.
  1008→  1008→      const clone = new Uint8Array(packageBytes.byteLength)
  1009→  1009→      clone.set(packageBytes)
  1010→  1010→      pkgStore.put(clone, job.id)
  1011→  1011→    }
  1012→  1012→    tx.oncomplete = () => resolve()
  1013→  1013→    tx.onerror = () => reject(tx.error)
  1014→  1014→    tx.onabort = () => reject(tx.error)
  1015→  1015→  }))
  1016→  1016→}

  1018→  1018→/** Load all delivery jobs, newest first. */
  1019→  1019→export async function loadDeliveryJobs(): Promise<DeliveryJob[]> {
  1020→  1020→  if (typeof indexedDB === 'undefined') return []
  1021→  1021→  try {
  1022→  1022→    return await withDb((db) => new Promise<DeliveryJob[]>((resolve, reject) => {
  1023→  1023→      const tx = db.transaction(JOB_STORE, 'readonly')
  1024→  1024→      const store = tx.objectStore(JOB_STORE)
  1025→  1025→      const req = store.getAll()
  1026→  1026→      req.onsuccess = () => {
  1027→  1027→        const jobs = (req.result as DeliveryJob[] | undefined) ?? []
  1028→  1028→        jobs.sort((a, b) => b.createdAt - a.createdAt)
  1029→  1029→        resolve(jobs)
  1030→  1030→      }
  1031→  1031→      req.onerror = () => reject(req.error)
  1032→  1032→    }))
  1033→  1033→  } catch {
  1034→  1034→    return []
  1035→  1035→  }
  1036→  1036→}

  1072→  1072→  id: string,

  1117→  1117→export interface SubmitResultOk {
  1118→  1118→  ok: true
  1119→  1119→  providerResponse: string

  1122→  1122→  ok: false
  1123→  1123→  error: string

  1126→  1126→export type SubmitResult = SubmitResultOk | SubmitResultFail
  1127→  1127→
  1128→  1128→/**
  1129→  1129→ * Submit a packaged delivery job to LabelGrid (or whichever DSP aggregator is
  1130→  1130→ * configured via env vars).
  1131→  1131→ *
  1132→  1132→ * Flow:
  1133→  1133→ *   1. Loads the persisted package bytes from IndexedDB.
  1134→  1134→ *   2. POSTs a multipart/form-data request to `/api/rain/distribute` with:
  1135→  1135→ *        - `manifest`: JSON string of ReleaseManifest
  1136→  1136→ *        - `package`: the ZIP Blob (application/zip)
  1137→  1137→ *        - `jobId`: the DeliveryJob id
  1138→  1138→ *   3. The server route checks `process.env.LABELGRID_API_KEY`:
  1139→  1139→ *        - If missing → 409 with `{ requiresCredentials: true, error: '...' }`.
  1140→  1140→ *          The client surfaces this honestly — the package IS built and IS
  1141→  1141→ *          ready, but no delivery has been attempted.
  1142→  1142→ *        - If present → the server makes a real `fetch()` to
  1143→  1143→ *          `LABELGRID_API_URL` (default `https://api.labelgrid.com/v1/deliveries`)
  1144→  1144→ *          with `Authorization: Bearer <key>` and forwards the multipart body.
  1145→  1145→ *          2xx → `{ ok: true, providerResponse }`. Non-2xx →
  1146→  1146→ *          `{ ok: false, error: <provider body>, requiresCredentials: false }`.
  1147→  1147→ *   4. The client updates the DeliveryJob status accordingly.
  1148→  1148→ *
  1149→  1149→ * Honest disclosure: the LABELGRID_API_KEY env var is NOT set in the default
  1150→  1150→ * sandbox `.env` file. The function will return `requiresCredentials: true`
  1151→  1151→ * until the operator sets the env var and restarts the dev server. This is
  1152→  1152→ * intentional and matches the user directive: "build the integration layer
  1153→  1153→ * so the workflow is complete once credentials are supplied".
  1154→  1154→ */
  1155→  1155→export async function submitToLabelGrid(job: DeliveryJob): Promise<SubmitResult> {
  1156→  1156→  const pkg = await loadDeliveryPackage(job.id)
  1157→  1157→  if (!pkg) {
  1158→  1158→    return {
  1159→  1159→      ok: false,
  1160→  1160→      error: 'Package bytes not found in IndexedDB — rebuild the package before submitting.',
  1161→  1161→      requiresCredentials: false,
  1162→  1162→    }
  1163→  1163→  }
  1164→  1164→  // Mark the job as submitting so the UI can render an in-flight state.
  1165→  1165→  await updateDeliveryJob(job.id, {
  1166→  1166→    status: 'submitting',
  1167→  1167→    submittedAt: Date.now(),
  1168→  1168→    error: undefined,
  1169→  1169→  })
  1170→  1170→
  1171→  1171→  try {
  1172→  1172→    const fd = new FormData()
  1173→  1173→    fd.append('jobId', job.id)
  1174→  1174→    fd.append('manifest', JSON.stringify(job.manifest))
  1175→  1175→    fd.append('package', new Blob([pkg.buffer as ArrayBuffer], { type: 'application/zip' }), `${job.manifest.releaseId}.zip`)
  1176→  1176→
  1177→  1177→    const resp = await fetch('/api/rain/distribute', {
  1178→  1178→      method: 'POST',
  1179→  1179→      body: fd,
  1180→  1180→    })
  1181→  1181→    const body = await resp.json().catch(() => ({ error: 'Non-JSON provider response' })) as {
  1182→  1182→      ok?: boolean
  1183→  1183→      providerResponse?: string
  1184→  1184→      error?: string
  1185→  1185→      requiresCredentials?: boolean
  1186→  1186→    }
  1187→  1187→
  1188→  1188→    if (resp.ok && body.ok) {
  1189→  1189→      await updateDeliveryJob(job.id, {
  1190→  1190→        status: 'delivered',
  1191→  1191→        deliveredAt: Date.now(),
  1192→  1192→        providerResponse: body.providerResponse ?? 'Delivered',
  1193→  1193→      })
  1194→  1194→      return { ok: true, providerResponse: body.providerResponse ?? 'Delivered' }
  1195→  1195→    }
  1196→  1196→
  1197→  1197→    const requiresCredentials = body.requiresCredentials === true
  1198→  1198→    const errMsg = body.error ?? `Provider returned HTTP ${resp.status}`
  1199→  1199→    await updateDeliveryJob(job.id, {
  1200→  1200→      status: 'failed',
  1201→  1201→      error: errMsg,
  1202→  1202→      providerResponse: body.providerResponse,
  1203→  1203→    })
  1204→  1204→    return { ok: false, error: errMsg, requiresCredentials }
  1205→  1205→  } catch (e) {
  1206→  1206→    const errMsg = e instanceof Error ? e.message : 'Network error during submission'
  1207→  1207→    await updateDeliveryJob(job.id, {
  1208→  1208→      status: 'failed',
  1209→  1209→      error: errMsg,
  1210→  1210→    })
  1211→  1211→    return { ok: false, error: errMsg, requiresCredentials: false }
  1212→  1212→  }
  1213→  1213→}
