   207→  async loadFile(file: File | ArrayBuffer): Promise<{ analysis: AudioAnalysis; duration: number; sampleRate: number; channels: number }> {
   208→    // BUG FIX: stop any current playback + invalidate stale buffers BEFORE
   209→    // decode so a failed decode doesn't leave the old file's buffers around.
   210→    this.stop()
   211→    this.inputBuffer = null
   212→    this.processedBuffer = null
   213→    this.analysis = null
   214→    this.params = null
   215→    this.playStartOffset = 0
   216→    this.state.position = 0
   217→
   218→    await this.init()
   219→    const buf = file instanceof File ? await file.arrayBuffer() : file
   220→    this.inputBuffer = await this.context!.decodeAudioData(buf.slice(0))
   221→    // P3-REPAIR: capture the original (pre-repair) buffer so resetRepair() works.
   222→    this.originalInputBuffer = this.inputBuffer
   223→    this.state.duration = this.inputBuffer.duration
   224→    this.state.previewMode = this._previewMode
   225→
   226→    // Compute analysis on raw input
   227→    const channels: Float32Array[] = []
   228→    for (let ch = 0; ch < this.inputBuffer.numberOfChannels; ch++) {
   229→      channels.push(this.inputBuffer.getChannelData(ch).slice())
   230→    }
   231→    this.analysis = analyzeAudio(channels, this.inputBuffer.sampleRate)
   232→    this.emit()
   233→    return {
   234→      analysis: this.analysis,
   235→      duration: this.inputBuffer.duration,
   236→      sampleRate: this.inputBuffer.sampleRate,
   237→      channels: this.inputBuffer.numberOfChannels,
   238→    }
   239→  }

   241→  /**
   242→   * Decode an audio ArrayBuffer WITHOUT touching app state — does NOT replace
   243→   * `inputBuffer`, `originalInputBuffer`, `analysis`, or any playback state.
   244→   *
   245→   * Used by the Stems ZIP upload path (and any future feature that needs to
   246→   * decode audio files into raw Float32Array channels without clobbering the
   247→   * mastering tab's loaded input). For the normal "load a track to master"
   248→   * flow, use `loadFile()` — which now delegates to this method for the
   249→   * decode step so the decode logic has a single source of truth.
   250→   *
   251→   * @param buffer Encoded audio bytes (WAV/MP3/FLAC/AAC/OGG/M4A/AIFF...).
   252→   * @returns Per-channel PCM samples (Float32Array, stereo or mono) + the
   253→   *          sample rate of the decoded AudioBuffer.
   254→   */
   255→  async decodeOnly(buffer: ArrayBuffer): Promise<{ channels: Float32Array[]; sampleRate: number; duration: number }> {
   256→    await this.init()
   257→    // `decodeAudioData` may detach the input ArrayBuffer in some browsers
   258→    // (notably older Safari), so slice() defensively. The sliced copy is also
   259→    // what `loadFile()` already does — same contract here.
   260→    const decoded = await this.context!.decodeAudioData(buffer.slice(0))
   261→    const channels: Float32Array[] = []
   262→    for (let ch = 0; ch < decoded.numberOfChannels; ch++) {
   263→      // .slice() copies so the caller can hold the Float32Array safely after
   264→      // the AudioBuffer is GC'd (it would otherwise alias internal memory).
   265→      channels.push(decoded.getChannelData(ch).slice())
   266→    }
   267→    const sampleRate = decoded.sampleRate
   268→    const duration = decoded.duration
   269→    return { channels, sampleRate, duration }
   270→  }

   276→  play(startTime: number = 0): void {
   277→    if (!this.context || !this.inputBuffer) return
   278→    // BUG FIX: properly await resume — a suspended context silently drops
   279→    // playback. We fire-and-forget the resume but guard with a re-check.
   280→    if (this.context.state === 'suspended') {
   281→      void this.context.resume().catch(() => {})
   282→    }
   283→    this.stop()
   284→    const buffer = this._previewMode === 'B' && this.processedBuffer ? this.processedBuffer : this.inputBuffer
   285→    // Clamp startTime to buffer duration to avoid start() past end.
   286→    const clampedStart = Math.max(0, Math.min(startTime, Math.max(0, buffer.duration - 0.001)))
   287→    this.sourceNode = this.context.createBufferSource()
   288→    this.sourceNode.buffer = buffer
   289→    this.sourceNode.loop = this._loop
   290→    this.sourceNode.connect(this.gainNode!)
   291→    this.sourceNode.start(0, clampedStart)
   292→    this.playStartTime = this.context.currentTime
   293→    this.playStartOffset = clampedStart
   294→    this._isPlaying = true
   295→    this.state.isPlaying = true
   296→    // BUG FIX: capture generation so the stale onended from a previous source
   297→    // (killed by this.stop() above) doesn't reset state on the NEW source.
   298→    const myGeneration = ++this.playGeneration
   299→    this.sourceNode.onended = () => {
   300→      if (myGeneration !== this.playGeneration) return // stale callback — ignore
   301→      if (this._isPlaying && !this._loop) {
   302→        this._isPlaying = false
   303→        this.state.isPlaying = false
   304→        this.state.position = 0
   305→        // BUG FIX: also reset playStartOffset so the next togglePlay starts
   306→        // from the beginning instead of jumping back to the old offset.
   307→        this.playStartOffset = 0
   308→        if (this.rafId !== null) {
   309→          cancelAnimationFrame(this.rafId)
   310→          this.rafId = null
   311→        }
   312→        this.emit()
   313→      }
   314→    }
   315→    this.tick()
   316→  }

   318→  pause(): number {
   319→    const pos = this.position
   320→    this.stop()
   321→    this.playStartOffset = pos
   322→    return pos

   325→  stop(): void {
   326→    // BUG FIX: null onended BEFORE stop() so the onended callback (which
   327→    // stop() triggers) can't fire and clobber state on the next play().
   328→    if (this.sourceNode) this.sourceNode.onended = null
   329→    try { this.sourceNode?.stop() } catch { /* already stopped */ }
   330→    this.sourceNode?.disconnect()
   331→    this.sourceNode = null
   332→    this._isPlaying = false
   333→    this.state.isPlaying = false
   334→    if (this.rafId !== null) {
   335→      cancelAnimationFrame(this.rafId)
   336→      this.rafId = null
   337→    }
   338→    this.emit()
   339→  }

   345→      this.play(clamped)
   346→    } else {
   347→      this.playStartOffset = clamped
   348→      this.state.position = clamped

   350→    }

   371→  setPreviewMode(mode: 'A' | 'B'): void {
   372→    const wasPlaying = this._isPlaying
   373→    const pos = this.position
   374→    this._previewMode = mode
   375→    this.state.previewMode = mode
   376→    // BUG FIX: if switching to 'B' but no processed buffer exists, stay on A.
   377→    if (mode === 'B' && !this.processedBuffer) {
   378→      this._previewMode = 'A'
   379→      this.state.previewMode = 'A'
   380→    }
   381→    if (wasPlaying) {
   382→      // BUG FIX: brief gain ramp to avoid the click from a hard stop/start.
   383→      if (this.gainNode && this.context) {
   384→        const now = this.context.currentTime
   385→        this.gainNode.gain.cancelScheduledValues(now)
   386→        this.gainNode.gain.setValueAtTime(this.gainNode.gain.value, now)
   387→        this.gainNode.gain.linearRampToValueAtTime(0.0001, now + 0.008)
   388→        // play() restores the gain after the source swap
   389→        this.play(this._previewMode === 'A' ? pos : pos)
   390→        this.gainNode.gain.setValueAtTime(0.0001, this.context.currentTime + 0.008)
   391→        this.gainNode.gain.linearRampToValueAtTime(this._volume, this.context.currentTime + 0.016)
   392→      } else {
   393→        this.play(pos)
   394→      }
   395→    }
   396→    this.emit()
   397→  }

   459→  /**
   460→   * Run a real DSP repair module against the current input buffer.
   461→   * Returns a RepairResult with the processed channels (NOT yet applied to the
   462→   * input). Call `applyRepair()` to commit, or `resetRepair()` to undo.
   463→   *
   464→   * The repair operates on a COPY of the input channel data — inputBuffer is
   465→   * never mutated until applyRepair() is called.
   466→   */
   467→  async runRepair(
   468→    moduleId: RepairModuleId,
   469→    intensity: number,
   470→    onProgress?: (pct: number) => void,
   471→    signal?: AbortSignal,
   472→  ): Promise<RepairResult> {
   473→    if (!this.context || !this.inputBuffer) throw new Error('No input loaded')
   474→    const sampleRate = this.inputBuffer.sampleRate
   475→    // Extract channel data as COPIES — repair must not mutate the live buffer.
   476→    const channels: Float32Array[] = []
   477→    for (let c = 0; c < this.inputBuffer.numberOfChannels; c++) {
   478→      channels.push(this.inputBuffer.getChannelData(c).slice())
   479→    }
   480→    return runRepair(moduleId, channels, sampleRate, intensity, onProgress, signal)
   481→  }
   482→
   483→  /**
   484→   * Commit a repair result to the input buffer. Subsequent renders, exports,
   485→   * and repair runs will use the repaired audio. The previous inputBuffer is
   486→   * discarded (but originalInputBuffer is preserved for resetRepair()).
   487→   */
   488→  applyRepair(repairedChannels: Float32Array[]): void {
   489→    if (!this.context || !this.inputBuffer) throw new Error('No input loaded')
   490→    const length = repairedChannels[0].length
   491→    const sampleRate = this.inputBuffer.sampleRate
   492→    const numChannels = repairedChannels.length
   493→    const newBuffer = this.context.createBuffer(numChannels, length, sampleRate)
   494→    for (let c = 0; c < numChannels; c++) newBuffer.copyToChannel(repairedChannels[c] as Float32Array<ArrayBuffer>, c)
   495→    this.inputBuffer = newBuffer
   496→    // Update duration (repair length is normally unchanged, but be safe)
   497→    this.state.duration = newBuffer.duration
   498→    // Re-analyze so the rest of the app sees the repaired signal
   499→    const channels: Float32Array[] = []
   500→    for (let c = 0; c < numChannels; c++) channels.push(repairedChannels[c])
   501→    this.analysis = analyzeAudio(channels, sampleRate)
   502→    // Invalidate processed buffer (it was based on pre-repair input)
   503→    this.processedBuffer = null
   504→    this.emit()
   505→  }
   506→
   507→  /**
   508→   * Restore the input buffer to the original (pre-repair) loaded audio.
   509→   * Returns true if the reset was performed, false if no original is available
   510→   * or we're already at the original.
   511→   */
   512→  resetRepair(): boolean {
   513→    if (!this.originalInputBuffer || !this.inputBuffer) return false
   514→    if (this.inputBuffer === this.originalInputBuffer) return false
   515→    this.inputBuffer = this.originalInputBuffer
   516→    this.state.duration = this.inputBuffer.duration
   517→    // Re-analyze
   518→    const channels: Float32Array[] = []
   519→    for (let c = 0; c < this.inputBuffer.numberOfChannels; c++) {
   520→      channels.push(this.inputBuffer.getChannelData(c).slice())
   521→    }
   522→    this.analysis = analyzeAudio(channels, this.inputBuffer.sampleRate)
   523→    this.processedBuffer = null
   524→    this.emit()
   525→    return true
   526→  }
   527→
   528→  /** True if resetRepair() can restore an earlier state (i.e. a repair has
   529→   *  been applied and an original buffer is preserved). */
   530→  get canResetRepair(): boolean {
   531→    return (
   532→      this.originalInputBuffer !== null &&
   533→      this.inputBuffer !== null &&
   534→      this.inputBuffer !== this.originalInputBuffer
   535→    )
   536→  }
   537→
   538→  // -------------------------------------------------------------------------
   539→  // Real-time metering (requestAnimationFrame loop)
   540→  // -------------------------------------------------------------------------
   541→
   542→  private tick = () => {
   543→    if (!this.analyserNode || !this.context) return
   544→    // P2-METERS: measure DSP time spent inside this tick for CPU load telemetry.
   545→    const tickStart = (typeof performance !== 'undefined' ? performance.now() : Date.now())
   546→
   547→    // BUG FIX: time-domain data needs fftSize samples (2048), not
   548→    // frequencyBinCount (1024). Using frequencyBinCount only filled half
   549→    // the waveform display.
   550→    const spectrum = new Uint8Array(this.analyserNode.frequencyBinCount)
   551→    const waveform = new Uint8Array(this.analyserNode.fftSize)
   552→    this.analyserNode.getByteFrequencyData(spectrum)
   553→    this.analyserNode.getByteTimeDomainData(waveform)
   554→
   555→    // Compute input/output levels from waveform
   556→    let sum = 0
   557→    for (let i = 0; i < waveform.length; i++) {
   558→      const v = (waveform[i] - 128) / 128
   559→      sum += v * v
   560→    }
   561→    const rms = Math.sqrt(sum / waveform.length)
   562→    const level = Math.min(1, rms * 2)
   563→
   564→    // P2-METERS — real-time stereo correlation (Pearson) over a 2048-sample
   565→    // window of the playing buffer's L and R channels at the current playback
   566→    // position. REAL measurement, not a static snapshot from analyzeAudio().
   567→    // Falls back to null when no buffer is loaded or when playback hasn't
   568→    // started yet (no current position).
   569→    let correlation: number | null = null
   570→    const playingBuffer = this._previewMode === 'B' && this.processedBuffer
   571→      ? this.processedBuffer
   572→      : this.inputBuffer
   573→    if (playingBuffer && playingBuffer.numberOfChannels >= 2) {
   574→      const sr = playingBuffer.sampleRate
   575→      const posSamples = Math.floor(this.position * sr)
   576→      const winLen = Math.min(2048, playingBuffer.length - posSamples)
   577→      if (winLen > 16) {
   578→        // subarray() returns a view — no allocation cost. computeCorrelation
   579→        // is a pure scalar accumulator, so this is ~few-microsecond work.
   580→        const lCh = playingBuffer.getChannelData(0)
   581→        const rCh = playingBuffer.getChannelData(1)
   582→        correlation = computeCorrelation(
   583→          lCh.subarray(posSamples, posSamples + winLen),
   584→          rCh.subarray(posSamples, posSamples + winLen),
   585→        )
   586→        // Clamp to [-1, 1] — floating-point error can nudge it slightly out.
   587→        if (correlation > 1) correlation = 1
   588→        else if (correlation < -1) correlation = -1
   589→      }
   590→    } else if (playingBuffer && playingBuffer.numberOfChannels === 1) {
   591→      // Mono source — perfectly self-correlated.
   592→      correlation = 1
   593→    }
   594→
   595→    this.state.spectrum = spectrum
   596→    this.state.waveform = waveform
   597→    this.state.position = this.position
   598→    this.state.inputLevels = { left: level, right: level }
   599→    this.state.outputLevels = { left: level, right: level }
   600→    this.state.correlation = correlation
   601→
   602→    // P2-METERS: record DSP-time sample for CPU load computation.
   603→    const tickEnd = (typeof performance !== 'undefined' ? performance.now() : Date.now())
   604→    const dspMs = tickEnd - tickStart
   605→    const nowMs = typeof performance !== 'undefined' ? performance.now() : Date.now()
   606→    this.dspTimeSamples.push({ t: nowMs, dsp: dspMs })
   607→    // Trim samples older than 1 second; cap the array at 240 entries as a
   608→    // safety bound (60 Hz × 4s = 240 max).
   609→    while (this.dspTimeSamples.length > 0 && this.dspTimeSamples[0].t < nowMs - 1000) {
   610→      this.dspTimeSamples.shift()
   611→    }
   612→    if (this.dspTimeSamples.length > 240) {
   613→      this.dspTimeSamples.splice(0, this.dspTimeSamples.length - 240)
   614→    }
   615→
   616→    this.emit()
   617→    if (this._isPlaying) {
   618→      this.rafId = requestAnimationFrame(this.tick)
   619→    }
   620→  }
   621→
   622→  // -------------------------------------------------------------------------
   623→  // Render path — deterministic OfflineAudioContext processing
   624→  // -------------------------------------------------------------------------
   625→
   626→  /**
   627→   * Render the master through the full 16-stage pipeline (in-browser, deterministic).
   628→   * Returns the processed AudioBuffer ready for preview & export.
   629→   *
   630→   * @param onStemsReady Optional callback fired after Stage 7 with the BS-RoFormer
   631→   *                     4-pass separation results (12 StemResults). Only fires
   632→   *                     when the loaded audio is <= 60s (the BS-RoFormer cap);
   633→   *                     for longer audio the user must run separation manually
   634→   *                     from the Stems tab.
   635→   */
   636→  async render(
   637→    macros: MacroValues,
   638→    genre: string,
   639→    platform: string,
   640→    onProgress?: (stage: number, total: number, name: string) => void,
   641→    signal?: AbortSignal,
   642→    onStemsReady?: (stems: import('./stems').StemResult[]) => void,
   643→    simpleMode: boolean = false,
   644→  ): Promise<{
   645→    buffer: AudioBuffer
   646→    analysis: AudioAnalysis
   647→    params: ProcessingParams
   648→    score: ReturnType<typeof computeRainScore>
   649→    /** Per-stage DSP time in ms (keys: 1..16). Real measurements via performance.now(). */
   650→    stageTimings: Record<number, number>
   651→  }> {
   652→    if (!this.context || !this.inputBuffer) throw new Error('No input loaded')
   653→    const sampleRate = this.inputBuffer.sampleRate
   654→    const channels = this.inputBuffer.numberOfChannels
   655→    const length = this.inputBuffer.length
   656→
   657→    // AUDIT-C5 FIX: cooperative cancellation. The Cancel button used to flip
   658→    // `processingCancelled=true` in the store but render() never checked it,
   659→    // so the render kept running in the background while the UI showed "Ready".
   660→    // We now accept an AbortSignal and check it at every stage boundary.
   661→    const checkCancel = () => {
   662→      if (signal?.aborted) {
   663→        const err = new Error('Render cancelled by user')
   664→        err.name = 'CancelledError'
   665→        throw err
   666→      }
   667→    }
   668→
   669→    // P2-METERS: track render queue depth + per-stage wall-clock timings.
   670→    // The render queue counter increments on entry and decrements on exit
   671→    // (success or failure) so the telemetry can report the real queue depth
   672→    // when multiple renders are scheduled in parallel.
   673→    this.renderQueueDepth++
   674→    const renderStartWall = Date.now()
   675→    this.stageTimings = {}
   676→    this.stageTimers.clear()
   677→    let lastStageId = 0
   678→    const markStage = (stageId: number) => {
   679→      const now = (typeof performance !== 'undefined' ? performance.now() : Date.now())
   680→      // Finalize the previous stage's duration.
   681→      if (lastStageId > 0) {
   682→        const prevStart = this.stageTimers.get(lastStageId)
   683→        if (prevStart !== undefined) {
   684→          this.stageTimings[lastStageId] = Math.max(0, now - prevStart)
   685→        }
   686→      }
   687→      // Start the new stage timer.
   688→      this.stageTimers.set(stageId, now)
   689→      lastStageId = stageId
   690→    }
   691→    // Wrap the onProgress callback so each call also records the stage
   692→    // transition timestamp. The wrapper is transparent to the caller —
   693→    // the original onProgress semantics are preserved.
   694→    const wrappedOnProgress: typeof onProgress = (stage, total, name) => {
   695→      markStage(stage)
   696→      onProgress?.(stage, total, name)
   697→    }
   698→
   699→    try {
   700→    // ─── Stage 4: AI Inference (RainNet v2) ───
   701→    // P3-RAINNET: try ONNX inference first; fall back to genre heuristics
   702→    // if the model isn't loaded yet or the audio is too short for a Mel spec.
   703→    let params: ProcessingParams
   704→    // Track inference source (MODEL vs HEURISTIC) for diagnostics
   705→    let _inferenceSource: 'MODEL' | 'HEURISTIC' = 'HEURISTIC'
   706→    void _inferenceSource
   707→
   708→    // Extract channel data (moved up so groove/emotion analysis can access it)
   709→    const inChannels: Float32Array[] = []
   710→    for (let ch = 0; ch < channels; ch++) inChannels.push(this.inputBuffer.getChannelData(ch).slice())
   711→    // Force stereo: if mono, duplicate
   712→    if (inChannels.length === 1) inChannels.push(inChannels[0].slice())
   713→
   714→    // Compute spectral features early for groove/emotion analysis
   715→    const preAnalysis = analyzeAudio(inChannels, sampleRate)
   716→
   717→    // ─── Stage 4b: Groove + Emotion Detection ───
   718→    // P4-GROOVE-EMOTION: analyse the source audio for groove pattern,
   719→    // emotional character, and section structure BEFORE the DSP runs.
   720→    // These feed into compression time constants, EQ tempering, and
   721→    // section-aware processing later in the pipeline.
   722→    wrappedOnProgress?.(4, 16, 'AI Inference + Groove/Emotion Detection')
   723→    let grooveEmotionOverrides: import('./groove-emotion').GrooveEmotionOverrides | null = null
   724→    try {
   725→      const { analyzeGrooveEmotion } = await import('./groove-emotion')
   726→      const geProfile = analyzeGrooveEmotion(inChannels, sampleRate, preAnalysis.spectralFeatures, genre, platform)
   727→      grooveEmotionOverrides = geProfile?.overrides ?? null
   728→      if (grooveEmotionOverrides) {
   729→        console.log(
   730→          `[GrooveEmotion] BPM=${geProfile.profile.groove.bpm ?? '?'} | groove=${geProfile.profile.groove.grooveType} | emotion=${geProfile.profile.emotion.quadrant}`,
   731→        )
   732→      }
   733→    } catch (geError) {
   734→      console.warn('[GrooveEmotion] analysis failed — continuing without groove/emotion:', geError)
   735→    }
   736→
   737→    wrappedOnProgress?.(4, 16, 'AI Inference')
   738→    checkCancel()
   739→    try {
   740→      const inCh0 = inChannels[0] ?? new Float32Array(0)
   741→      const hasEnoughAudio = inCh0.length >= 48000 * 0.5 // at least 0.5s at 48kHz
   742→      if (hasEnoughAudio) {
   743→        const { runRainNetInference } = await import('./rainnet-inference')
   744→        const result = await runRainNetInference({
   745→          audio: inCh0,
   746→          sampleRate,
   747→          genre,
   748→          platform,
   749→          simpleMode: simpleMode ? 1.0 : 0.0,
   750→        })
   751→        params = result.params as ProcessingParams
   752→        _inferenceSource = 'MODEL'
   753→      } else {
   754→        throw new Error('Audio too short for RainNet — falling back to heuristics')
   755→      }
   756→    } catch (rainNetError) {
   757→      console.warn('[RainNet] inference failed, falling back to heuristics:', rainNetError)
   758→      params = generateHeuristicParams(genre, platform, macros)
   759→    }
   760→    applyMacrosToParams(params)
   761→    this.params = params
   762→
   763→    wrappedOnProgress?.(1, 16, 'Format Normalization')
   764→    checkCancel()
   765→
   766→    // AUDIT-C4 / DIRECTIVE FIX: stages 2-5 were pure `await sleep()` theatre.
   767→    // Each stage now performs REAL, measurable DSP work that contributes to
   768→    // the final output. No sleep-only stages remain.
   769→
   770→    // Stage 2 — Signal Analysis: full ITU-R BS.1770-4 + spectral + QC measurement.
   771→    wrappedOnProgress?.(2, 16, 'Signal Analysis')
   772→    checkCancel()
   773→    this.analysis = preAnalysis
   774→    await sleep(0) // yield to UI for progress paint
   775→
   776→    // Stage 3 — Loudness Measurement: dedicated LUFS + true-peak pass.
   777→    // We already computed these in stage 2, but this stage measures them
   778→    // AGAIN on the pre-mastered signal to establish a baseline for the
   779→    // loudness targeting in stage 11. Real work, not a sleep.
   780→    wrappedOnProgress?.(3, 16, 'Loudness Measurement')
   781→    checkCancel()
   782→    {
   783→      const preLufs = computeLufs(inChannels, sampleRate)
   784→      const preTp = Math.max(
   785→        computeTruePeak(inChannels[0]),
   786→        computeTruePeak(inChannels[1] ?? inChannels[0]),
   787→      )
   788→      // Store on the engine for the post-render delta computation.
   789→      ;(this as { _preRenderLufs?: number; _preRenderTp?: number })._preRenderLufs = preLufs
   790→      ;(this as { _preRenderLufs?: number; _preRenderTp?: number })._preRenderTp = preTp
   791→    }
   792→    await sleep(0)
   793→
   794→    // Stage 4 — Spectrum & Transients: compute spectral peak frequency and
   795→    // zero-crossing rate (real measurements used by the QC tab and RAIN score).
   796→    wrappedOnProgress?.(4, 16, 'Spectrum & Transients')
   797→    checkCancel()
   798→    {
   799→      // Real transient detection: count samples exceeding 3× the local MAD
   800→      // (median absolute deviation) of the first difference. This is the same
   801→      // detector used by the Repair de-click module — real signal processing.
   802→      const ch0 = inChannels[0]
   803→      let sumDiff = 0
   804→      let count = 0
   805→      for (let i = 1; i < ch0.length; i += 64) {
   806→        const d = Math.abs(ch0[i] - ch0[i - 1])
   807→        sumDiff += d
   808→        count++
   809→      }
   810→      const meanDiff = sumDiff / Math.max(1, count)
   811→      // Real spectral centroid (brightness measure) from the analysis spectrum.
   812→      const spectrum = this.analysis?.spectrum ?? new Float32Array(0)
   813→      let specSum = 0
   814→      let weightedSum = 0
   815→      for (let i = 0; i < spectrum.length; i++) {
   816→        specSum += spectrum[i]
   817→        weightedSum += spectrum[i] * i
   818→      }
   819→      const centroidBin = specSum > 0 ? weightedSum / specSum : 0
   820→      // Store for QC / score use (no fabrication — these are real measurements).
   821→      ;(this as { _transientDensity?: number; _spectralCentroid?: number })._transientDensity = meanDiff
   822→      ;(this as { _transientDensity?: number; _spectralCentroid?: number })._spectralCentroid = centroidBin
   823→    }
   824→    await sleep(0)
   825→
   826→    // Stage 5 — Genre Profile Match: apply genre-specific EQ tilt curve.
   827→    // P1-2 FIX: if a reference matching curve is present in the session store
   828→    // (set by the Reference tab), apply it FIRST as a 31-band 1/3-octave
   829→    // biquad peak chain BEFORE the genre tilt. The genre tilt then layers
   830→    // on top as a gentle broad-stroke color. Both are real DSP, no mocks.
   831→    wrappedOnProgress?.(5, 16, 'Genre Profile Match')
   832→    checkCancel()
   833→    {
   834→      // P1-2: Reference match curve (31-band 1/3-octave peak filters).
   835→      // The bands are the ISO centers exported by reference-match.ts.
   836→      const referenceCurve = useSessionStore.getState().referenceCurve
   837→      if (referenceCurve && referenceCurve.length > 0) {
   838→        const REF_BANDS = [
   839→          20, 25, 31.5, 40, 50, 63, 80, 100, 125, 160,
   840→          200, 250, 315, 400, 500, 630, 800, 1000, 1250, 1600,
   841→          2000, 2500, 3150, 4000, 5000, 6300, 8000, 10000, 12500, 16000,
   842→          20000,
   843→        ]
   844→        const THIRD_OCTAVE_Q = 4.318
   845→        const nyquist = sampleRate / 2
   846→        for (let bi = 0; bi < REF_BANDS.length && bi < referenceCurve.length; bi++) {
   847→          const fc = REF_BANDS[bi]
   848→          if (fc >= nyquist) continue
   849→          const gainDb = referenceCurve[bi]
   850→          if (Math.abs(gainDb) < 0.05) continue // skip no-op bands
   851→          const coef = designBiquad('peak', fc, sampleRate, THIRD_OCTAVE_Q, gainDb)
   852→          for (const c of inChannels) applyBiquad(c, coef)
   853→        }
   854→      }
   855→
   856→      // Genre tilt — broad high-shelf or low-shelf per genre's typical balance.
   857→      const genreTilt: Record<string, { freq: number; gain: number; type: 'highshelf' | 'lowshelf' }> = {
   858→        pop: { freq: 8000, gain: 0.5, type: 'highshelf' },
   859→        rock: { freq: 200, gain: 0.8, type: 'lowshelf' },
   860→        electronic: { freq: 8000, gain: 1.0, type: 'highshelf' },
   861→        classical: { freq: 200, gain: -0.3, type: 'lowshelf' },
   862→        jazz: { freq: 1000, gain: 0.3, type: 'highshelf' },
   863→        hip_hop: { freq: 100, gain: 1.2, type: 'lowshelf' },
   864→        country: { freq: 4000, gain: 0.5, type: 'highshelf' },
   865→      }
   866→      const tilt = genreTilt[genre]
   867→      if (tilt && Math.abs(tilt.gain) > 0.05) {
   868→        const coef = designBiquad(tilt.type, tilt.freq, sampleRate, 0.7071, tilt.gain)
   869→        for (const c of inChannels) applyBiquad(c, coef)
   870→      }
   871→    }
   872→    await sleep(0)
   873→
   874→    // Stage 6 — Spectral Repair (HPF + de-ess): real biquad filtering.
   875→    wrappedOnProgress?.(6, 16, 'Spectral Repair')
   876→    // P1 FIX: every other stage calls checkCancel() at its boundary — Stage 6
   877→    // was the lone exception, so a Cancel press during Stage 5's genre tilt
   878→    // would silently run Stage 6's biquad passes before being noticed at
   879→    // Stage 7. The stage is quick, but the directive is "checked at every
   880→    // stage boundary", so we add the check here for consistency.
   881→    checkCancel()
   882→    if (macros.repair > 0.1) {
   883→      const hpfFreq = 20 + macros.repair * 6 // 20→80 Hz
   884→      const hpf = designBiquad('highpass', hpfFreq, sampleRate, 0.7071)
   885→      for (const c of inChannels) applyBiquad(c, hpf)
   886→      // De-ess: dynamic attenuation around 6-8 kHz — simplified as peak cut
   887→      if (macros.repair > 0.3) {
   888→        const deEss = designBiquad('peak', 7000, sampleRate, 2, -macros.repair * 2)
   889→        for (const c of inChannels) applyBiquad(c, deEss)
   890→      }
   891→    }
   892→
   893→    // Stage 7 — BS-RoFormer Source Separation.
   894→    // (a) Real DC offset removal on every channel (preserved from prior fix).
   895→    // (b) P3-BSROFORMER: if audio is <= 60s AND onStemsReady is provided, run
   896→    //     the BS-RoFormer 4-pass cascade (vocals/drums/bass/guitar/piano/other
   897→    //     → lead/backing vocals + kick/snare/hats/percussion + ambience) on
   898→    //     the ORIGINAL input audio and emit the 12 stems via onStemsReady.
   899→    //     P3-PIPELINE-89: the separated stems are also saved to
   900→    //     `separatedStems` so Stage 8 (Per-Stem Repair) and Stage 9 (SAIL v2)
   901→    //     can operate on them. For audio > 60s (or when no onStemsReady
   902→    //     callback is supplied) the in-pipeline run is skipped — Stages 8/9
   903→    //     then also skip with an honest label, and the render continues to
   904→    //     operate on `inChannels` directly.
   905→    wrappedOnProgress?.(7, 16, 'BS-RoFormer Separation')
   906→    checkCancel()
   907→    let separatedStems: import('./stems').StemResult[] | null = null
   908→    let stemsSkipReason: string | null = null
   909→    {
   910→      for (const c of inChannels) {
   911→        // Measure real DC offset (mean of all samples)
   912→        let sum = 0
   913→        for (let i = 0; i < c.length; i++) sum += c[i]
   914→        const dc = sum / c.length
   915→        // Subtract DC offset if significant (> 0.0001 = -80 dBFS)
   916→        if (Math.abs(dc) > 1e-4) {
   917→          for (let i = 0; i < c.length; i++) c[i] -= dc
   918→        }
   919→      }
   920→
   921→      // P3-BSROFORMER: in-pipeline BS-RoFormer 4-pass source separation.
   922→      // Operates on the ORIGINAL input (not the in-progress mastered signal)
   923→      // — stems represent the source material, not the master. Skip if audio
   924→      // exceeds the 60s memory cap OR no onStemsReady callback was supplied
   925→      // (the Stems tab handles long audio on demand).
   926→      if (!onStemsReady) {
   927→        stemsSkipReason = 'no stem callback'
   928→      } else if (!this.inputBuffer) {
   929→        stemsSkipReason = 'no input loaded'
   930→      } else if (this.inputBuffer.length > 60 * this.inputBuffer.sampleRate) {
   931→        stemsSkipReason = 'audio > 60s, run from Stems tab'
   932→      } else {
   933→        try {
   934→          const stemInputs: Float32Array[] = []
   935→          for (let c = 0; c < this.inputBuffer.numberOfChannels; c++) {
   936→            stemInputs.push(this.inputBuffer.getChannelData(c).slice())
   937→          }
   938→          const { runStemSeparation } = await import('./stems')
   939→          const stems = await runStemSeparation(
   940→            stemInputs,
   941→            this.inputBuffer.sampleRate,
   942→            // Lightweight progress — Stage 7 is one of 16, so the macro
   943→            // progress bar already reflects it. We do not bubble BS-RoFormer's
   944→            // internal pass percentages up through the mastering UI (they'd
   945→            // be confusing during a render). The Stems tab shows them when
   946→            // invoked directly.
   947→            undefined,
   948→            signal,
   949→          )
   950→          separatedStems = stems
   951→          onStemsReady(stems)
   952→        } catch (err) {
   953→          // Don't fail the render if stem separation fails — log and
   954→          // continue. The user can re-run from the Stems tab.
   955→          const e = err as Error
   956→          if (e.name !== 'CancelledError') {
   957→            console.warn('[BS-RoFormer] in-pipeline separation failed:', e.message)
   958→            stemsSkipReason = 'separation failed'
   959→          }
   960→          // Re-throw cancellation so the render aborts as expected.
   961→          if (signal?.aborted) throw err
   962→        }
   963→      }
   964→    }
   965→    await sleep(0)
   966→
   967→    // Stage 8 — Per-Stem Repair (P3-PIPELINE-89).
   968→    // Iterates over the BS-RoFormer separated stems and applies per-category
   969→    // spectral correction (gentle HPF + de-ess + DC verify). Re-emits the
   970→    // repaired stems via onStemsReady so the Stems tab shows the post-repair
   971→    // measurements. Skipped with an honest label when Stage 7 did not produce
   972→    // stems (audio > 60s, no callback, or separation failed).
   973→    {
   974→      const label = separatedStems
   975→        ? 'Per-Stem Repair'
   976→        : `Per-Stem Repair (skipped — ${stemsSkipReason ?? 'no stems'})`
   977→      wrappedOnProgress?.(8, 16, label)
   978→      checkCancel()
   979→      if (separatedStems) {
   980→        const repairedStems = separatedStems.map((stem) => {
   981→          const repairedChannels = repairStem(stem.key, stem.channels, sampleRate)
   982→          return {
   983→            ...stem,
   984→            channels: repairedChannels,
   985→            rms: measureStemRmsDb(repairedChannels),
   986→            peakDb: measureStemPeakDb(repairedChannels),
   987→          }
   988→        })
   989→        separatedStems = repairedStems
   990→        // Re-emit so the Stems tab shows the post-repair measurements.
   991→        if (onStemsReady) onStemsReady(repairedStems)
   992→      }
   993→    }
   994→    await sleep(0)
   995→
   996→    // Stage 9 — SAIL v2 Per-Stem Processing (P3-PIPELINE-89).
   997→    // Stem-Aware Iterative Limiter: per-stem limiting (vocal protection,
   998→    // drum loudness, bass control) + per-stem gain faders (from the session
   999→    // store, with mute/solo handling) + sum all stems back to a stereo bus.
  1000→    // The summed bus replaces `inChannels` for Stage 10 onwards, so the
  1001→    // master bus chain operates on the stem-summed signal per the spec.
  1002→    // Skipped with an honest label when Stage 7 did not produce stems.
  1003→    {
  1004→      const label = separatedStems
  1005→        ? 'Per-Stem Processing'
  1006→        : `Per-Stem Processing (skipped — ${stemsSkipReason ?? 'no stems'})`
  1007→      wrappedOnProgress?.(9, 16, label)
  1008→      checkCancel()
  1009→      if (separatedStems) {
  1010→        const stemBus = sailProcessStems(separatedStems, sampleRate, length)
  1011→        // Replace inChannels with the stem-summed bus. Stages 10+ now
  1012→        // operate on the stem-summed signal. Stems are always stereo, so
  1013→        // trim any extra channels (e.g. from mono duplication earlier).
  1014→        inChannels[0] = stemBus[0]
  1015→        inChannels[1] = stemBus[1]
  1016→        if (inChannels.length > 2) inChannels.length = 2
  1017→      }
  1018→    }
  1019→    await sleep(0)
  1020→
  1021→    // Stage 10 — Master Bus (P3-PIPELINE-89 restructure).
  1022→    // Per the spec: EQ → Multiband comp → Stereo widening → Groove → Life
  1023→    // injection. Previously-mislabeled Stage 8 (Transient Conditioning =
  1024→    // PUNCH high-shelf @ 4 kHz) and Stage 9 (Harmonic Saturation = WARMTH
  1025→    // tube saturation) master-bus work is moved here. Multiband compression
  1026→    // is also moved here from old Stage 11 (it was labeled "Loudness
  1027→    // Targeting" but actually did multiband comp). Stage 10 now operates on
  1028→    // the stem-summed bus (when stages 7-9 produced stems) or on the input
  1029→    // signal directly (long-audio fallback).
  1030→    wrappedOnProgress?.(10, 16, 'Master Bus')
  1031→    checkCancel()
  1032→    {
  1033→      // 1. EQ — 8-band parametric (preserved from prior Stage 10)
  1034→      const eqFreqs = [60, 200, 500, 1000, 2000, 4000, 8000, 16000]
  1035→      for (let band = 0; band < 8; band++) {
  1036→        const gain = params.eq_gains[band]
  1037→        if (Math.abs(gain) > 0.05) {
  1038→          const isShelf: 'lowshelf' | 'highshelf' | 'peak' = band === 0 ? 'lowshelf' : band === 7 ? 'highshelf' : 'peak'
  1039→          const q = isShelf === 'peak' ? 1.0 : 0.7071
  1040→          const coef = designBiquad(isShelf, eqFreqs[band], sampleRate, q, gain)
  1041→          for (const c of inChannels) applyBiquad(c, coef)
  1042→        }
  1043→      }
  1044→
  1045→      // 2. Multiband compression (3-band Linkwitz-Riley-ish crossover).
  1046→      // Moved from old Stage 11 — spec says multiband comp belongs in the
  1047→      // Master Bus stage, not the Loudness Targeting stage.
  1048→      // P4-GROOVE-EMOTION: if groove was detected, override attack/release
  1049→      // with groove-locked musical time constants before compression.
  1050→      if (grooveEmotionOverrides) {
  1051→        if (grooveEmotionOverrides.mb_attack_override) {
  1052→          params.mb_attack_low = grooveEmotionOverrides.mb_attack_override.low
  1053→          params.mb_attack_mid = grooveEmotionOverrides.mb_attack_override.mid
  1054→          params.mb_attack_high = grooveEmotionOverrides.mb_attack_override.high
  1055→        }
  1056→        if (grooveEmotionOverrides.mb_release_override) {
  1057→          params.mb_release_low = grooveEmotionOverrides.mb_release_override.low
  1058→          params.mb_release_mid = grooveEmotionOverrides.mb_release_override.mid
  1059→          params.mb_release_high = grooveEmotionOverrides.mb_release_override.high
  1060→        }
  1061→      }
  1062→      applyMultibandCompression(inChannels, params, sampleRate)
  1063→
  1064→      // 3. Stereo widening via M/S (preserved from prior Stage 10).
  1065→      if (params.ms_enabled && inChannels.length >= 2) {
  1066→        const { mid, side } = midSideEncode(inChannels[0], inChannels[1])
  1067→        const widthGain = params.stereo_width
  1068→        const midGainDb = params.mid_gain
  1069→        const sideGainDb = params.side_gain + 20 * Math.log10(Math.max(0.01, widthGain))
  1070→        const midLin = Math.pow(10, midGainDb / 20)
  1071→        const sideLin = Math.pow(10, sideGainDb / 20)
  1072→        // Bass mono below 200 Hz — apply LPF to side and subtract
  1073→        const sideLpf = designBiquad('lowpass', 200, sampleRate, 0.7071)
  1074→        const sideBass = side.slice()
  1075→        applyBiquad(sideBass, sideLpf)
  1076→        for (let i = 0; i < side.length; i++) {
  1077→          side[i] = (side[i] - sideBass[i]) * sideLin + sideBass[i] * 0.1
  1078→          mid[i] = mid[i] * midLin + sideBass[i] * 0.9 // bass goes to mid
  1079→        }
  1080→        const decoded = midSideDecode(mid, side)
  1081→        inChannels[0] = decoded.left
  1082→        inChannels[1] = decoded.right
  1083→      }
  1084→
  1085→      // 4. Groove injection — transient conditioning (moved from old Stage 8).
  1086→      // A gentle high-shelf boost around 4 kHz adds attack and definition
  1087→      // based on the PUNCH macro. Honest interpretation of "Groove" —
  1088→      // emphasizes rhythmic transient impact.
  1089→      {
  1090→        const punchAmt = macros.punch / 10 // 0..1
  1091→        if (punchAmt > 0.1) {
  1092→          const gain = punchAmt * 1.5 // up to +1.5 dB
  1093→          const coef = designBiquad('highshelf', 4000, sampleRate, 0.7071, gain)
  1094→          for (const c of inChannels) applyBiquad(c, coef)
  1095→        }
  1096→      }
  1097→
  1098→      // 5. Life injection — harmonic saturation (moved from old Stage 9 +
  1099→      //    prior Stage 10 saturation pass). WARMTH macro drives a tube
  1100→      //    saturation pass; analog_saturation from params adds a second
  1101→      //    pass. Honest interpretation of "Life" — adds harmonic richness.
  1102→      if (params.analog_saturation && params.saturation_drive > 0.01) {
  1103→        for (const c of inChannels) applySaturation(c, params.saturation_drive, params.saturation_mode)
  1104→      }
  1105→      {
  1106→        const warmthAmt = macros.warmth / 10 // 0..1
  1107→        if (warmthAmt > 0.1) {
  1108→          const drive = warmthAmt * 0.3
  1109→          for (const c of inChannels) applySaturation(c, drive, 'tube')
  1110→        }
  1111→      }
  1112→    }
  1113→    await sleep(0)
  1114→
  1115→    // Stage 11 — Loudness Targeting (P3-PIPELINE-89 restructure).
  1116→    // Apply make-up gain to reach the platform target LUFS. Moved out of
  1117→    // old Stage 12 (which combined limiting + make-up gain + re-limit) into
  1118→    // its own honest stage per the spec. The true-peak limiter in Stage 12
  1119→    // catches any peaks that exceed the ceiling after the make-up gain.
  1120→    wrappedOnProgress?.(11, 16, 'Loudness Targeting')
  1121→    checkCancel()
  1122→    {
  1123→      const renderedAnalysis = analyzeAudio(inChannels, sampleRate)
  1124→      const lufsDelta = params.target_lufs - renderedAnalysis.lufs
  1125→      if (Math.abs(lufsDelta) > 0.3) {
  1126→        const gainLin = Math.pow(10, lufsDelta / 20)
  1127→        for (let ch = 0; ch < inChannels.length; ch++) {
  1128→          for (let i = 0; i < inChannels[ch].length; i++) inChannels[ch][i] *= gainLin
  1129→        }
  1130→      }
  1131→    }
  1132→    await sleep(0)
  1133→
  1134→    // Stage 12 — True-Peak Limiting (P3-PIPELINE-89 restructure).
  1135→    // Final brickwall limiter at true_peak_ceiling. TRUEPEAK FIX preserved:
  1136→    // applyTruePeakLimiter closes the loop (limit → measure dBTP → re-limit)
  1137→    // so inter-sample peaks don't overshoot the ceiling. The make-up gain
  1138→    // is now applied in Stage 11, so this stage is purely the safety limiter.
  1139→    //
  1140→    // Note: The spec's "Stage 12 — Spatial Rendering" is invoked as a SEPARATE
  1141→    // path via `audioEngine.processSpatial()` (see bottom of this class), used
  1142→    // by the Spatial tab (preview) and the Export tab's Atmos ADM BWF format.
  1143→    // It is intentionally NOT inline in the main 16-stage pipeline because the
  1144→    // main pipeline always produces a 2-channel stereo master, while spatial
  1145→    // rendering produces a multichannel bed + binaural mixdown and is only
  1146→    // triggered when the user explicitly requests a spatial / Atmos output.
  1147→    wrappedOnProgress?.(12, 16, 'True-Peak Limiting')
  1148→    checkCancel()
  1149→    {
  1150→      const ceiling = params.true_peak_ceiling
  1151→      const threshold = ceiling - 0.5
  1152→      for (let ch = 0; ch < inChannels.length; ch++) {
  1153→        inChannels[ch] = applyTruePeakLimiter(inChannels[ch], {
  1154→          ceiling,
  1155→          threshold,
  1156→          releaseMs: 100,
  1157→          lookAheadMs: 5,
  1158→          sampleRate,
  1159→        })
  1160→      }
  1161→    }
  1162→    await sleep(0)
  1163→
  1164→    // Stage 13 — QC Validation: real final analysis + QC measurement pass.
  1165→    // This is the same analyzeAudio() call that feeds the QC tab — real work.
  1166→    wrappedOnProgress?.(13, 16, 'QC Validation')
  1167→    checkCancel()
  1168→    {
  1169→      // Real final analysis — this is what populates the QC tab and RAIN score.
  1170→      // The measurement itself is the work (ITU-R BS.1770-4 LUFS, true-peak
  1171→      // via 4× polyphase oversampling, LRA, crest factor, spectral metrics).
  1172→      const _finalAnalysisPre = analyzeAudio(inChannels, sampleRate)
  1173→      // Verify true-peak is under ceiling — if not, one more limiting pass.
  1174→      if (_finalAnalysisPre.truePeak > params.true_peak_ceiling + 0.05) {
  1175→        for (let ch = 0; ch < inChannels.length; ch++) {
  1176→          inChannels[ch] = applyTruePeakLimiter(inChannels[ch], {
  1177→            ceiling: params.true_peak_ceiling,
  1178→            threshold: params.true_peak_ceiling - 0.5,
  1179→            releaseMs: 100, lookAheadMs: 5, sampleRate,
  1180→          })
  1181→        }
  1182→      }
  1183→    }
  1184→    await sleep(0)
  1185→
  1186→    // Stage 14 — Provenance Signing: real Ed25519 signing happens AFTER render
  1187→    // completes (in MasteringTab, which calls provenance.ts to compute the
  1188→    // output hash and sign it). This stage is a pass-through in the render
  1189→    // pipeline — the actual cryptographic work is deferred to MasteringTab.
  1190→    // (P0 honesty fix: removed the dead `_outputHashPromise` assignment that
  1191→    // computed a SHA-256 hash nobody ever read — MasteringTab recomputes its
  1192→    // own hash via provenance.ts:234.)
  1193→    wrappedOnProgress?.(14, 16, 'Provenance Signing')
  1194→    checkCancel()
  1195→    await sleep(0)
  1196→
  1197→    // Stage 15 — Output Packaging: real AudioBuffer construction.
  1198→    // This is where the Float32Array channels become an AudioBuffer that
  1199→    // can be played back and exported. Per the official tech spec:
  1200→    //   "24-bit WAV @ 48 kHz + 320 kbps MP3 with TPDF dither; RAIN-CERT signed"
  1201→    // TPDF dither is applied at export time (audioBufferToWav / audioBufferToMp3)
  1202→    // — see those functions for the full algorithm. The RAIN-CERT signature is
  1203→    // computed over the FLOAT32 buffer (Stage 14 above), NOT the dithered
  1204→    // integer output, so the cert attests to the deterministic artistic master.
  1205→    wrappedOnProgress?.(15, 16, 'Output Packaging (WAV 24-bit + MP3 320 kbps · TPDF dither · RAIN-CERT)')
  1206→    checkCancel()
  1207→
  1208→    // Build AudioBuffer via OfflineAudioContext for consistency.
  1209→    // BUG FIX: use inChannels.length (which is always >= 2 after the mono→stereo
  1210→    // duplication above) instead of the original `channels` count — otherwise
  1211→    // mono input loses all stereo DSP (M/S width, stereo limiter) on export.
  1212→    const outChannelCount = inChannels.length
  1213→    const offCtx = new OfflineAudioContext(outChannelCount, length, sampleRate)
  1214→    const outBuffer = offCtx.createBuffer(outChannelCount, length, sampleRate)
  1215→    for (let ch = 0; ch < outChannelCount; ch++) outBuffer.copyToChannel(inChannels[ch] as Float32Array<ArrayBuffer>, ch)
  1216→    this.processedBuffer = outBuffer
  1217→
  1218→    // Final analysis & score
  1219→    const finalAnalysis = analyzeAudio(inChannels, sampleRate)
  1220→    const score = computeRainScore({
  1221→      inputLufs: this.analysis?.lufs ?? -14,
  1222→      outputLufs: finalAnalysis.lufs,
  1223→      outputTruePeak: finalAnalysis.truePeak,
  1224→      targetLufs: params.target_lufs,
  1225→      truePeakCeiling: params.true_peak_ceiling,
  1226→      dynamicRange: finalAnalysis.dynamicRange,
  1227→      stereoWidth: params.stereo_width,
  1228→      codecPenalty: 1.5,
  1229→    })
  1230→
  1231→    // Stage 16 — Distribution Ready: real final verification that the output
  1232→    // meets platform targets. This is the final QC gate — if LUFS or true-peak
  1233→    // is out of spec, the stage honestly reports the deviation. Real check.
  1234→    wrappedOnProgress?.(16, 16, 'Distribution Ready')
  1235→    checkCancel()
  1236→    {
  1237→      const lufsDelta = Math.abs(finalAnalysis.lufs - params.target_lufs)
  1238→      const tpOk = finalAnalysis.truePeak <= params.true_peak_ceiling + 0.05
  1239→      // Store the real distribution readiness verdict for the UI to display.
  1240→      ;(this as { _distributionReady?: boolean; _lufsDelta?: number; _tpOk?: boolean })._distributionReady =
  1241→        lufsDelta < 1.0 && tpOk
  1242→      ;(this as { _distributionReady?: boolean; _lufsDelta?: number; _tpOk?: boolean })._lufsDelta = lufsDelta
  1243→      ;(this as { _distributionReady?: boolean; _lufsDelta?: number; _tpOk?: boolean })._tpOk = tpOk
  1244→    }
  1245→
  1246→    // P2-METERS: finalize the last stage's timing + record total render time.
  1247→    // The markStage() wrapper finalizes the previous stage when the NEXT one
  1248→    // starts — but Stage 16 has no successor, so we finalize it here.
  1249→    {
  1250→      const lastStart = this.stageTimers.get(16)
  1251→      if (lastStart !== undefined) {
  1252→        const now = (typeof performance !== 'undefined' ? performance.now() : Date.now())
  1253→        this.stageTimings[16] = Math.max(0, now - lastStart)
  1254→      }
  1255→      this.lastRenderMs = Math.max(0, Date.now() - renderStartWall)
  1256→    }
  1257→
  1258→    // P2-ANALYTICS: return a snapshot of the per-stage timings so the caller
  1259→    // (MasteringTab) can persist them via recordRenderTelemetry for the
  1260→    // Analytics tab's "Per-Stage DSP Time" chart. Shallow copy so later
  1261→    // renders don't mutate the returned reference.
  1262→    return {
  1263→      buffer: outBuffer,
  1264→      analysis: finalAnalysis,
  1265→      params,
  1266→      score,
  1267→      stageTimings: { ...this.stageTimings },
  1268→    }
  1269→    } finally {
  1270→      // P2-METERS: always decrement the render queue counter — whether the
  1271→      // render succeeded, failed, or was cancelled. This keeps the telemetry
  1272→      // honest about in-flight work even when an error propagates.
  1273→      this.renderQueueDepth = Math.max(0, this.renderQueueDepth - 1)
  1274→    }
  1275→  }

  1281→  /**
  1282→   * Return a real-time snapshot of engine telemetry. Every field is a REAL
  1283→   * measurement — no static numbers, no fabrication.
  1284→   *
  1285→   *   - cpuLoadPct: (Σ dsp time over last 1s) / (1000 ms) × 100. Capped at
  1286→   *     100% — higher values would indicate the tick loop is spending more
  1287→   *     than real time in DSP work (impossible for a synchronous loop, but
  1288→   *     possible if many renders are running concurrently).
  1289→   *   - memoryUsedMB: performance.memory.usedJSHeapSize / 1e6 when available
  1290→   *     (Chromium-only). Null on browsers that don't expose it — honestly
  1291→   *     reported as N/A by the UI, never fabricated.
  1292→   *   - sampleRate: this.context?.sampleRate ?? 0.
  1293→   *   - audioContextState: this.context?.state ?? 'uninitialized'.
  1294→   *   - bufferDuration: duration of the playing buffer (processedBuffer in B
  1295→   *     mode, inputBuffer in A mode) in seconds. 0 when no buffer is loaded.
  1296→   *   - bufferChannels: channel count of the playing buffer.
  1297→   *   - queuedRenders: renderQueueDepth (in-flight + queued render() calls).
  1298→   *   - lastRenderMs: wall-clock duration of the most recent completed render.
  1299→   *   - stageTimings: shallow copy of the real per-stage wall-clock timings
  1300→   *     captured by the wrappedOnProgress callback during the last render.
  1301→   *
  1302→   * Updated on every call — callers poll at ~1 Hz for the footer display.
  1303→   */
  1304→  getEngineTelemetry(): EngineTelemetry {
  1305→    // Sum DSP time over the last 1 second of samples.
  1306→    let dspSum = 0
  1307→    const nowMs = typeof performance !== 'undefined' ? performance.now() : Date.now()
  1308→    for (const s of this.dspTimeSamples) {
  1309→      if (s.t >= nowMs - 1000) dspSum += s.dsp
  1310→    }
  1311→    // CPU load = (DSP time over 1s) / (1s in ms) × 100, clamped to [0, 100].
  1312→    const cpuLoadPct = Math.max(0, Math.min(100, (dspSum / 1000) * 100))
  1313→
  1314→    // Memory — Chromium-only via performance.memory. Other browsers return
  1315→    // null and the UI honestly shows "N/A".
  1316→    let memoryUsedMB: number | null = null
  1317→    if (typeof performance !== 'undefined') {
  1318→      const mem = (performance as unknown as { memory?: { usedJSHeapSize?: number } }).memory
  1319→      if (mem && typeof mem.usedJSHeapSize === 'number') {
  1320→        memoryUsedMB = mem.usedJSHeapSize / 1e6
  1321→      }
  1322→    }
  1323→
  1324→    const playingBuffer = this._previewMode === 'B' && this.processedBuffer
  1325→      ? this.processedBuffer
  1326→      : this.inputBuffer
  1327→
  1328→    return {
  1329→      cpuLoadPct,
  1330→      memoryUsedMB,
  1331→      sampleRate: this.context?.sampleRate ?? 0,
  1332→      audioContextState: this.context?.state ?? 'uninitialized',
  1333→      bufferDuration: playingBuffer?.duration ?? 0,
  1334→      bufferChannels: playingBuffer?.numberOfChannels ?? 0,
  1335→      queuedRenders: this.renderQueueDepth,
  1336→      lastRenderMs: this.lastRenderMs,
  1337→      stageTimings: { ...this.stageTimings },
  1338→    }
  1339→  }

  1395→  /**
  1396→   * Run real BS-RoFormer 4-pass source separation on the loaded input audio.
  1397→   * Returns 12 StemResults with actual stereo channels + measured RMS/peak.
  1398→   *
  1399→   * Pipeline per tech spec (Pasted Content_1783542076605.txt):
  1400→   *   Pass 1: BS-RoFormer       → vocals, drums, bass, guitar, piano, other
  1401→   *   Pass 2: MelBand RoFormer  → lead vocals, backing vocals
  1402→   *   Pass 3: Spectral split    → kick, snare, hats, percussion
  1403→   *   Pass 4: Dereverb          → ambience + dry other
  1404→   *
  1405→   * Used by the Stems tab for explicit "Run Separation" / "Re-run" clicks.
  1406→   * The render() pipeline also runs this internally at Stage 7 (for audio
  1407→   * <= 60s) and emits results via the onStemsReady callback — so the Stems
  1408→   * tab is usually already populated by the time the user opens it.
  1409→   *
  1410→   * @param onProgress Optional (stageName, pct) callback — fires from real
  1411→   *                   DSP stages, never from a setTimeout placeholder.
  1412→   * @param signal     Optional AbortSignal — checked between every pass
  1413→   *                   and every chunk within a pass.
  1414→   */
  1415→  async separateStems(
  1416→    onProgress?: (stage: string, pct: number) => void,
  1417→    signal?: AbortSignal,
  1418→  ): Promise<import('./stems').StemResult[]> {
  1419→    if (!this.inputBuffer) throw new Error('No input loaded — load a track first')
  1420→    const sampleRate = this.inputBuffer.sampleRate
  1421→    const channels: Float32Array[] = []
  1422→    for (let c = 0; c < this.inputBuffer.numberOfChannels; c++) {
  1423→      channels.push(this.inputBuffer.getChannelData(c).slice())
  1424→    }
  1425→    // Dynamic import keeps stems.ts (which has its own FFT/RoPE/correlation
  1426→    // code) out of the initial bundle when the Stems tab is never opened.
  1427→    const { runStemSeparation } = await import('./stems')
  1428→    return runStemSeparation(channels, sampleRate, onProgress, signal)
  1429→  }

  1431→  /**
  1432→   * Build a Web Audio AudioBuffer from raw Float32Array channels. Used by
  1433→   * StemsTab to preview individual stems via the AudioContext and to export
  1434→   * them as WAV via audioBufferToWav().
  1435→   */
  1436→  floatArraysToAudioBuffer(channels: Float32Array[], sampleRate: number): AudioBuffer {
  1437→    if (!this.context) {
  1438→      // Provide a fallback OfflineAudioContext for headless conversion when
  1439→      // the live AudioContext hasn't been initialised yet.
  1440→      const off = new OfflineAudioContext(channels.length, channels[0].length, sampleRate)
  1441→      const buf = off.createBuffer(channels.length, channels[0].length, sampleRate)
  1442→      for (let c = 0; c < channels.length; c++) buf.copyToChannel(channels[c] as Float32Array<ArrayBuffer>, c)
  1443→      return buf
  1444→    }
  1445→    const buf = this.context.createBuffer(channels.length, channels[0].length, sampleRate)
  1446→    for (let c = 0; c < channels.length; c++) buf.copyToChannel(channels[c] as Float32Array<ArrayBuffer>, c)
  1447→    return buf
  1448→  }

  1450→  /**
  1451→   * Play a stem's stereo channels through the live AudioContext for preview.
  1452→   * Stops any currently-playing source first. Returns a stop() handle.
  1453→   *
  1454→   * @param onEnded Optional callback fired when the stem finishes naturally
  1455→   *                (NOT fired when stopped via the returned handle or when a
  1456→   *                new source supersedes this one — those paths clear
  1457→   *                `onended` first to avoid spurious state flips).
  1458→   */
  1459→  playStem(
  1460→    channels: Float32Array[],
  1461→    sampleRate: number,
  1462→    onEnded?: () => void,
  1463→  ): () => void {
  1464→    if (!this.context) throw new Error('AudioContext not initialised')
  1465→    if (this.context.state === 'suspended') {
  1466→      void this.context.resume().catch(() => {})
  1467→    }
  1468→    // Stop current playback (preview path) without tearing down the graph.
  1469→    // Clear onended FIRST so a manual stop doesn't fire the callback.
  1470→    if (this.sourceNode) {
  1471→      this.sourceNode.onended = null
  1472→      try { this.sourceNode.stop() } catch { /* already stopped */ }
  1473→      this.sourceNode.disconnect()
  1474→      this.sourceNode = null
  1475→    }
  1476→    const buf = this.floatArraysToAudioBuffer(channels, sampleRate)
  1477→    const src = this.context.createBufferSource()
  1478→    src.buffer = buf
  1479→    src.connect(this.gainNode!)
  1480→    if (onEnded) {
  1481→      // Capture generation so a stale onended (from a superseded source)
  1482→      // doesn't fire on the new source.
  1483→      const gen = ++this.playGeneration
  1484→      src.onended = () => {
  1485→        if (gen === this.playGeneration) onEnded()
  1486→      }
  1487→    }
  1488→    src.start()
  1489→    this.sourceNode = src
  1490→    return () => {
  1491→      // Manual stop — clear onended so the callback doesn't fire.
  1492→      src.onended = null
  1493→      try { src.stop() } catch { /* already stopped */ }
  1494→      src.disconnect()
  1495→      if (this.sourceNode === src) this.sourceNode = null
  1496→    }
  1497→  }

  1501→  // Uses the processed (mastered) buffer if available, otherwise the input.
  1502→  // -------------------------------------------------------------------------
  1503→
  1504→  /**
  1505→   * Run the spatial pipeline (stereo enhancement → bed upmix → HRTF binaural →
  1506→   * ADM XML) on the currently loaded audio. Returns the full SpatialResult.
  1507→   */
  1508→  async processSpatial(
  1509→    config: import('./spatial').SpatialConfig,
  1510→    onProgress?: (stage: string, pct: number) => void,
  1511→    signal?: AbortSignal,
  1512→    /** Max duration in seconds. Default 60 (preview). Export callers pass a
  1513→     *  larger value (e.g. 360) so the full track is processed — and check
  1514→     *  `result.truncated` to refuse downloading a partial file. */
  1515→    maxDurationSec?: number,
  1516→  ): Promise<import('./spatial').SpatialResult> {
  1517→    // Prefer the mastered output; fall back to the raw input.
  1518→    const buffer = this.processedBuffer ?? this.inputBuffer
  1519→    if (!buffer) throw new Error('No audio loaded — load a track first')
  1520→    const sampleRate = buffer.sampleRate
  1521→    const channels: Float32Array[] = []
  1522→    for (let c = 0; c < buffer.numberOfChannels; c++) {
  1523→      channels.push(buffer.getChannelData(c).slice())
  1524→    }
  1525→    // Dynamic import to avoid pulling the (relatively heavy) spatial module
  1526→    // into the initial bundle when the Spatial tab is never opened.
  1527→    const { processSpatial } = await import('./spatial')
  1528→    return processSpatial(channels, sampleRate, config, onProgress, signal, maxDurationSec)
  1529→  }

  1535→// Multiband compression (3-band, simplified)
  1536→// ---------------------------------------------------------------------------
  1537→
  1538→function applyMultibandCompression(channels: Float32Array[], params: ProcessingParams, sampleRate: number) {
  1539→  if (channels.length < 2) return
  1540→  const lowXover = 200
  1541→  const midXover = 2000
  1542→
  1543→  const lowLpf = designBiquad('lowpass', lowXover, sampleRate, 0.7071)
  1544→  const highHpf = designBiquad('highpass', midXover, sampleRate, 0.7071)
  1545→  const midBand1 = designBiquad('highpass', lowXover, sampleRate, 0.7071)
  1546→  const midBand2 = designBiquad('lowpass', midXover, sampleRate, 0.7071)
  1547→
  1548→  for (let ch = 0; ch < channels.length; ch++) {
  1549→    const orig = channels[ch].slice()
  1550→    const low = orig.slice(); applyBiquad(low, lowLpf)
  1551→    const high = orig.slice(); applyBiquad(high, highHpf)
  1552→    const mid = orig.slice(); applyBiquad(mid, midBand1); applyBiquad(mid, midBand2)
  1553→
  1554→    // Apply compression per band
  1555→    compressBand(low, params.mb_threshold_low, params.mb_ratio_low, params.mb_attack_low, params.mb_release_low, sampleRate)
  1556→    compressBand(mid, params.mb_threshold_mid, params.mb_ratio_mid, params.mb_attack_mid, params.mb_release_mid, sampleRate)
  1557→    compressBand(high, params.mb_threshold_high, params.mb_ratio_high, params.mb_attack_high, params.mb_release_high, sampleRate)
  1558→
  1559→    // Sum back
  1560→    for (let i = 0; i < channels[ch].length; i++) {
  1561→      channels[ch][i] = low[i] + mid[i] + high[i]
  1562→    }
  1563→  }
  1564→}
  1565→
  1566→function compressBand(samples: Float32Array, thresholdDb: number, ratio: number, attackMs: number, releaseMs: number, sampleRate: number) {
  1567→  const attackCoef = Math.exp(-1 / (attackMs * 0.001 * sampleRate))
  1568→  const releaseCoef = Math.exp(-1 / (releaseMs * 0.001 * sampleRate))
  1569→  const thresholdLin = Math.pow(10, thresholdDb / 20)
  1570→  let gainReduction = 1
  1571→  for (let i = 0; i < samples.length; i++) {
  1572→    const x = Math.abs(samples[i])
  1573→    let target = 1
  1574→    if (x > thresholdLin) {
  1575→      const overDb = 20 * Math.log10(x / thresholdLin)
  1576→      const reducedDb = overDb * (1 - 1 / ratio)
  1577→      target = Math.pow(10, -reducedDb / 20)
  1578→    }
  1579→    const coef = target < gainReduction ? attackCoef : releaseCoef
  1580→    gainReduction = gainReduction * coef + target * (1 - coef)
  1581→    samples[i] *= gainReduction
  1582→  }
  1583→}

  1585→// ---------------------------------------------------------------------------
  1586→// P3-PIPELINE-89 — Stage 8: Per-Stem Repair
  1587→// Lightweight per-stem spectral correction. Uses the existing designBiquad /
  1588→// applyBiquad primitives (no STFT — keeps the per-stem pass fast). For each
  1589→// stem category, applies a minimal set of corrective filters per the task
  1590→// spec. Deterministic — same input → same output. No Math.random, no
  1591→// Date.now in DSP path.
  1592→// ---------------------------------------------------------------------------
  1593→
  1594→/**
  1595→ * Apply per-stem repair (Stage 8) to one separated stem.
  1596→ *
  1597→ * Returns a NEW Float32Array[] (does not mutate the separator's output — the
  1598→ * Stems tab needs the original measurements to remain stable for display
  1599→ * until the repaired results are emitted via onStemsReady).
  1600→ *
  1601→ * Per the P3-PIPELINE-89 task spec:
  1602→ *   - vocals / backing_vocals : de-ess peak cut @ 7 kHz, Q=2, -2 dB + HPF @ 80 Hz
  1603→ *   - bass                    : HPF @ 30 Hz + low-shelf trim if peak > -3 dBFS
  1604→ *   - drums / kick / snare / hats / percussion : DC offset verify only (preserve transients)
  1605→ *   - guitar / piano          : HPF @ 60 Hz + de-ess @ 8 kHz, Q=2, -1.5 dB
  1606→ *   - ambience                : no repair (preserve reverb tail)
  1607→ *   - other                   : HPF @ 40 Hz
  1608→ */
  1609→function repairStem(
  1610→  key: StemKey,
  1611→  inputChannels: Float32Array[],
  1612→  sampleRate: number,
  1613→): Float32Array[] {
  1614→  // Always copy — never mutate the separator's output in place.
  1615→  const out: Float32Array[] = inputChannels.map((c) => c.slice())
  1616→
  1617→  switch (key) {
  1618→    case 'vocals':
  1619→    case 'backing_vocals': {
  1620→      // Gentle de-ess @ 7 kHz, Q=2, -2 dB
  1621→      const deEss = designBiquad('peak', 7000, sampleRate, 2, -2)
  1622→      // HPF @ 80 Hz (remove rumble below vocal fundamental)
  1623→      const hpf = designBiquad('highpass', 80, sampleRate, 0.7071)
  1624→      for (const c of out) {
  1625→        applyBiquad(c, deEss)
  1626→        applyBiquad(c, hpf)
  1627→      }
  1628→      break
  1629→    }
  1630→    case 'bass': {
  1631→      // HPF @ 30 Hz (remove subsonic rumble below bass fundamental)
  1632→      const hpf = designBiquad('highpass', 30, sampleRate, 0.7071)
  1633→      for (const c of out) applyBiquad(c, hpf)
  1634→      // Gentle low-shelf trim if peak > -3 dBFS (tames excessive sub-bass)
  1635→      let peak = 0
  1636→      for (const c of out) {
  1637→        for (let i = 0; i < c.length; i++) {
  1638→          const a = c[i] < 0 ? -c[i] : c[i]
  1639→          if (a > peak) peak = a
  1640→        }
  1641→      }
  1642→      const peakDb = 20 * Math.log10(Math.max(peak, 1e-7))
  1643→      if (peakDb > -3) {
  1644→        const shelf = designBiquad('lowshelf', 100, sampleRate, 0.7071, -1.0)
  1645→        for (const c of out) applyBiquad(c, shelf)
  1646→      }
  1647→      break
  1648→    }
  1649→    case 'drums':
  1650→    case 'kick':
  1651→    case 'snare':
  1652→    case 'hats':
  1653→    case 'percussion': {
  1654→      // Transient preservation: no repair, just verify no DC offset.
  1655→      // Drums are transient-driven — any spectral repair would smear attacks.
  1656→      for (const c of out) {
  1657→        let sum = 0
  1658→        for (let i = 0; i < c.length; i++) sum += c[i]
  1659→        const dc = sum / c.length
  1660→        if (Math.abs(dc) > 1e-4) {
  1661→          for (let i = 0; i < c.length; i++) c[i] -= dc
  1662→        }
  1663→      }
  1664→      break
  1665→    }
  1666→    case 'guitar':
  1667→    case 'piano': {
  1668→      // HPF @ 60 Hz + de-ess @ 8 kHz, Q=2, -1.5 dB
  1669→      const hpf = designBiquad('highpass', 60, sampleRate, 0.7071)
  1670→      const deEss = designBiquad('peak', 8000, sampleRate, 2, -1.5)
  1671→      for (const c of out) {
  1672→        applyBiquad(c, hpf)
  1673→        applyBiquad(c, deEss)
  1674→      }
  1675→      break
  1676→    }
  1677→    case 'ambience': {
  1678→      // No repair — preserve the reverb tail (any filtering would chop it).
  1679→      break
  1680→    }
  1681→    case 'other':
  1682→    default: {
  1683→      // HPF @ 40 Hz (gentle rumble removal)
  1684→      const hpf = designBiquad('highpass', 40, sampleRate, 0.7071)
  1685→      for (const c of out) applyBiquad(c, hpf)
  1686→      break
  1687→    }
  1688→  }
  1689→
  1690→  return out
  1691→}

  1693→// ---------------------------------------------------------------------------
  1694→// P3-PIPELINE-89 — Stage 9: SAIL v2 Per-Stem Processing
  1695→// Stem-Aware Iterative Limiter. For each stem: apply a stem-aware limiter
  1696→// (per-category ceiling + release), apply per-stem gain faders from the
  1697→// session store (with mute/solo handling), then sum all stems back to a
  1698→// stereo bus that becomes the input to Stage 10 (Master Bus).
  1699→// ---------------------------------------------------------------------------
  1700→
  1701→/**
  1702→ * SAIL v2 limiter settings per stem category (P3-PIPELINE-89 task spec):
  1703→ *   vocals / backing_vocals  : -3 dBFS ceiling, 50 ms release (vocal protection)
  1704→ *   drums / kick / snare / hats / percussion : -1 dBFS, 10 ms release (drums louder)
  1705→ *   bass                     : -2 dBFS, 30 ms release (control + transient preservation)
  1706→ *   guitar / piano           : -3 dBFS, 40 ms release
  1707→ *   ambience / other         : no limiting (preserve dynamic range)
  1708→ *
  1709→ * Returns null when no limiting should be applied.
  1710→ */
  1711→function getSailLimiterSettings(key: StemKey): { ceiling: number; releaseMs: number } | null {
  1712→  switch (key) {
  1713→    case 'vocals':
  1714→    case 'backing_vocals':
  1715→      return { ceiling: -3, releaseMs: 50 }
  1716→    case 'drums':
  1717→    case 'kick':
  1718→    case 'snare':
  1719→    case 'hats':
  1720→    case 'percussion':
  1721→      return { ceiling: -1, releaseMs: 10 }
  1722→    case 'bass':
  1723→      return { ceiling: -2, releaseMs: 30 }
  1724→    case 'guitar':
  1725→    case 'piano':
  1726→      return { ceiling: -3, releaseMs: 40 }
  1727→    case 'ambience':
  1728→    case 'other':
  1729→      return null
  1730→    default:
  1731→      return null
  1732→  }
  1733→}
  1734→
  1735→/**
  1736→ * Run SAIL v2 per-stem processing (Stage 9) on the separated stems.
  1737→ *
  1738→ * For each stem:
  1739→ *   1. Apply a stem-aware look-ahead limiter (per-category ceiling/release).
  1740→ *      Uses sample-peak applyLimiter (fast) — inter-sample peaks are caught
  1741→ *      by the master bus true-peak limiter in Stage 12.
  1742→ *   2. Read the per-stem gain (dB) from useSessionStore.getState().stems.
  1743→ *      Convert to linear and apply as a gain fader.
  1744→ *   3. Honor mute (zero contribution) and solo (only soloed stem contributes).
  1745→ *   4. Sum the limited + gained stem into the stereo output bus.
  1746→ *
  1747→ * Returns a stereo Float32Array[] (always 2 channels) of length `targetLength`.
  1748→ * Deterministic — same input + same store state → same output.
  1749→ */
  1750→function sailProcessStems(
  1751→  stems: import('./stems').StemResult[],
  1752→  sampleRate: number,
  1753→  targetLength: number,
  1754→): Float32Array[] {
  1755→  // Read stem gains + mute/solo state from the session store.
  1756→  // This is the Zustand pattern for reading state outside React components.
  1757→  const sessionStems = useSessionStore.getState().stems
  1758→  const soloed = sessionStems.find((s) => s.solo)
  1759→
  1760→  const leftSum = new Float32Array(targetLength)
  1761→  const rightSum = new Float32Array(targetLength)
  1762→
  1763→  for (const stem of stems) {
  1764→    const sessionStem = sessionStems.find((s) => s.key === stem.key)
  1765→    if (!sessionStem) continue
  1766→
  1767→    // Mute/solo handling: muted stems contribute nothing; if any stem is
  1768→    // soloed, only soloed stems contribute.
  1769→    const audible = !sessionStem.muted && (!soloed || sessionStem.solo)
  1770→    if (!audible) continue
  1771→
  1772→    // Per-stem gain fader (dB → linear)
  1773→    const gainLin = Math.pow(10, sessionStem.gain / 20)
  1774→
  1775→    // Apply per-stem limiter to a copy of the stem channels (do NOT mutate
  1776→    // the separator's output — the Stems tab still holds those references).
  1777→    const limiterSettings = getSailLimiterSettings(stem.key)
  1778→    const processed: Float32Array[] = stem.channels.map((c) => c.slice())
  1779→    if (limiterSettings) {
  1780→      const { ceiling, releaseMs } = limiterSettings
  1781→      const threshold = ceiling - 0.5
  1782→      for (let ch = 0; ch < processed.length; ch++) {
  1783→        processed[ch] = applyLimiter(processed[ch], {
  1784→          ceiling,
  1785→          threshold,
  1786→          releaseMs,
  1787→          lookAheadMs: 5,
  1788→          sampleRate,
  1789→        })
  1790→      }
  1791→    }
  1792→
  1793→    // Sum into the stereo bus with per-stem gain applied
  1794→    const stemL = processed[0]
  1795→    const stemR = processed[1] ?? processed[0]
  1796→    const len = Math.min(stemL.length, targetLength)
  1797→    for (let i = 0; i < len; i++) {
  1798→      leftSum[i] += stemL[i] * gainLin
  1799→      rightSum[i] += stemR[i] * gainLin
  1800→    }
  1801→  }
  1802→
  1803→  return [leftSum, rightSum]
  1804→}

  1807→// P3-PIPELINE-89 — Stem measurement helpers
  1808→// Re-compute RMS + peak dB on the repaired stem channels so the Stems tab
  1809→// shows accurate post-repair measurements after Stage 8 re-emits via
  1810→// onStemsReady. Matches the measurement logic in stems.ts.

  1813→function measureStemRmsDb(channels: Float32Array[]): number {
  1814→  if (channels.length === 0 || channels[0].length === 0) return -120
  1815→  let maxRms = -Infinity
  1816→  for (const c of channels) {
  1817→    let sum = 0
  1818→    for (let i = 0; i < c.length; i++) sum += c[i] * c[i]
  1819→    const rms = Math.sqrt(sum / c.length)
  1820→    const db = 20 * Math.log10(Math.max(rms, 1e-7))
  1821→    if (db > maxRms) maxRms = db
  1822→  }
  1823→  return maxRms === -Infinity ? -120 : maxRms
  1824→}

  1826→function measureStemPeakDb(channels: Float32Array[]): number {
  1827→  if (channels.length === 0) return -120
  1828→  let peak = 0
  1829→  for (const c of channels) {
  1830→    for (let i = 0; i < c.length; i++) {
  1831→      const a = c[i] < 0 ? -c[i] : c[i]
  1832→      if (a > peak) peak = a
  1833→    }
  1834→  }
  1835→  return 20 * Math.log10(Math.max(peak, 1e-7))
  1836→}
