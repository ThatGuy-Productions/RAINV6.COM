     3→/**
     4→ * RAIN V6 — Landing Page Interactive Mastering Demo
     5→ *
     6→ * A pure-visual before/after mastering comparison that demonstrates the
     7→ * product's value without requiring audio playback. Visitors see a waveform
     8→ * transform, LUFS meters improve, spectrum bars balance, and the RAIN Score
     9→ * gauge animate from a "before" state to a polished "after" state.
    10→ *
    11→ * A draggable toggle slider lets visitors scrub between before/after — the
    12→ * whole visualization interpolates in real time. This is the landing page's
    13→ * "show, don't tell" moment.
    14→ *
    15→ * No audio engine dependency — all data is synthetic but realistic (the
    16→ * "before" waveform is quieter/unbalanced, "after" is louder/balanced).
    17→ */
    18→
    19→import { useCallback, useEffect, useMemo, useRef, useState } from 'react'
    20→import { motion, useInView } from 'framer-motion'
    21→import { ArrowRight, Zap, Gauge, Activity, BarChart3, Sparkles, Play, Pause, Volume2, Loader2 } from 'lucide-react'
    22→
    23→// ── Synthetic "before" and "after" data ────────────────────────────────────
    24→// The waveform is a list of amplitude samples (0..1). "Before" is quiet with
    25→// uneven dynamics; "after" is louder with controlled peaks.
    26→
    27→const SAMPLE_COUNT = 80
    28→
    29→function generateWaveform(seed: number, balanced: boolean): number[] {
    30→  const out: number[] = []
    31→  for (let i = 0; i < SAMPLE_COUNT; i++) {
    32→    const t = i / SAMPLE_COUNT
    33→    // Base waveform shape — a song-like envelope
    34→    const envelope = Math.sin(t * Math.PI) * 0.8 + 0.2
    35→    // High-frequency detail
    36→    const detail = Math.sin(i * 0.7 + seed) * 0.3 + Math.sin(i * 1.3 + seed * 2) * 0.15
    37→    // "Before": quieter + more dynamic range (some dips)
    38→    // "After": louder + compressed (more uniform)
    39→    const base = balanced ? 0.82 : 0.48
    40→    const variance = balanced ? 0.12 : 0.35
    41→    const amp = base + detail * variance * envelope
    42→    out.push(Math.max(0.04, Math.min(1, amp)))
    43→  }
    44→  return out
    45→}
    46→
    47→const BEFORE_WAVE = generateWaveform(1.5, false)
    48→const AFTER_WAVE = generateWaveform(1.5, true)
    49→
    50→// Spectrum bars (24 bands) — "before" is muddy (mids heavy), "after" is balanced
    51→const SPECTRUM_BANDS = 24
    52→function generateSpectrum(balanced: boolean): number[] {
    53→  const out: number[] = []
    54→  for (let i = 0; i < SPECTRUM_BANDS; i++) {
    55→    const freq = i / SPECTRUM_BANDS // 0..1, low→high
    56→    if (balanced) {
    57→      // Balanced: gentle roll-off, full mids + highs
    58→      out.push(0.45 + Math.sin(freq * Math.PI * 1.5) * 0.3 + Math.random() * 0.08)
    59→    } else {
    60→      // Muddy: bass-heavy, scooped mids, rolled-off highs
    61→      const bass = Math.exp(-freq * 4) * 0.7
    62→      const scoop = 1 - Math.exp(-Math.pow((freq - 0.35) * 3, 2))
    63→      out.push(bass + scoop * 0.25 + Math.random() * 0.08)
    64→    }
    65→  }
    66→  return out
    67→}
    68→
    69→const BEFORE_SPECTRUM = generateSpectrum(false)
    70→const AFTER_SPECTRUM = generateSpectrum(true)
    71→
    72→// Metrics
    73→const BEFORE_METRICS = { lufs: -18.4, truePeak: -3.2, score: 47 }
    74→const AFTER_METRICS = { lufs: -14.0, truePeak: -1.0, score: 92 }
    75→
    76→// ── Helper: linear interpolation ────────────────────────────────────────────
    77→function lerp(a: number, b: number, t: number): number {
    78→  return a + (b - a) * t
    79→}
    80→
    81→// ── Audio playback hook ─────────────────────────────────────────────────────
    82→// Loads the demo WAV and plays it through a Web Audio chain that simulates
    83→// the before/after mastering difference in real time. The `t` parameter
    84→// (0..1) drives a GainNode (volume) + BiquadFilter (lowpass cutoff) so the
    85→// visitor hears the transform as they drag the slider.
    86→
    87→interface DemoAudioState {
    88→  isPlaying: boolean
    89→  isLoading: boolean
    90→  error: string | null
    91→}
    92→
    93→function useDemoAudio(tRef: React.MutableRefObject<number>) {
    94→  const [state, setState] = useState<DemoAudioState>({ isPlaying: false, isLoading: false, error: null })
    95→  const ctxRef = useRef<AudioContext | null>(null)
    96→  const bufferRef = useRef<AudioBuffer | null>(null)
    97→  const sourceRef = useRef<AudioBufferSourceNode | null>(null)
    98→  const gainRef = useRef<GainNode | null>(null)
    99→  const filterRef = useRef<BiquadFilterNode | null>(null)
   100→  const startedAtRef = useRef(0)
   101→  const offsetRef = useRef(0)
   102→  const rafRef = useRef<number | null>(null)
   103→
   104→  // Sync the audio chain's parameters to the current `t` value.
   105→  const syncParams = useCallback(() => {
   106→    const t = tRef.current
   107→    // Gain: before = 0.35 (quiet), after = 0.85 (loud)
   108→    if (gainRef.current) {
   109→      gainRef.current.gain.value = lerp(0.35, 0.85, t)
   110→    }
   111→    // Lowpass cutoff: before = 6000 Hz (muffled), after = 20000 Hz (full)
   112→    if (filterRef.current) {
   113→      filterRef.current.frequency.value = lerp(6000, 20000, t)
   114→    }
   115→  }, [tRef])
   116→
   117→  // Continuously sync params while playing (the slider may be dragged).
   118→  useEffect(() => {
   119→    if (!state.isPlaying) return
   120→    const tick = () => {
   121→      syncParams()
   122→      rafRef.current = requestAnimationFrame(tick)
   123→    }
   124→    rafRef.current = requestAnimationFrame(tick)
   125→    return () => {
   126→      if (rafRef.current !== null) {
   127→        cancelAnimationFrame(rafRef.current)
   128→        rafRef.current = null
   129→      }
   130→    }
   131→  }, [state.isPlaying, syncParams])
   132→
   133→  const load = useCallback(async () => {
   134→    if (bufferRef.current || ctxRef.current) return
   135→    setState((s) => ({ ...s, isLoading: true, error: null }))
   136→    try {
   137→      const Ctor = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext
   138→      const ctx = new Ctor()
   139→      ctxRef.current = ctx
   140→      const res = await fetch('/demo-sample.wav')
   141→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   142→      const arr = await res.arrayBuffer()
   143→      const buf = await ctx.decodeAudioData(arr)
   144→      bufferRef.current = buf
   145→    } catch (e) {
   146→      setState((s) => ({ ...s, isLoading: false, error: e instanceof Error ? e.message : 'Load failed' }))
   147→    }
   148→  }, [])
   149→
   150→  const play = useCallback(async () => {
   151→    await load()
   152→    const ctx = ctxRef.current
   153→    const buf = bufferRef.current
   154→    if (!ctx || !buf) {
   155→      if (!state.error) setState((s) => ({ ...s, isLoading: false }))
   156→      return
   157→    }
   158→    if (ctx.state === 'suspended') await ctx.resume()
   159→
   160→    // Build the chain: source → filter → gain → destination
   161→    const src = ctx.createBufferSource()
   162→    src.buffer = buf
   163→    src.loop = true
   164→
   165→    const filter = ctx.createBiquadFilter()
   166→    filter.type = 'lowpass'
   167→    filter.Q.value = 0.7
   168→
   169→    const gain = ctx.createGain()
   170→
   171→    src.connect(filter)
   172→    filter.connect(gain)
   173→    gain.connect(ctx.destination)
   174→
   175→    sourceRef.current = src
   176→    filterRef.current = filter
   177→    gainRef.current = gain
   178→
   179→    syncParams()
   180→    src.start(0, offsetRef.current % buf.duration)
   181→    startedAtRef.current = ctx.currentTime
   182→    setState({ isPlaying: true, isLoading: false, error: null })
   183→  }, [load, syncParams, state.error])
   184→
   185→  const stop = useCallback(() => {
   186→    const ctx = ctxRef.current
   187→    const src = sourceRef.current
   188→    if (ctx && src) {
   189→      // Track playback position for resume
   190→      const elapsed = ctx.currentTime - startedAtRef.current
   191→      offsetRef.current = (offsetRef.current + elapsed) % (bufferRef.current?.duration ?? 1)
   192→      try { src.stop() } catch { /* already stopped */ }
   193→      src.disconnect()
   194→    }
   195→    sourceRef.current = null
   196→    setState((s) => ({ ...s, isPlaying: false }))
   197→  }, [])
   198→
   199→  // Cleanup on unmount
   200→  useEffect(() => {
   201→    return () => {
   202→      if (sourceRef.current) {
   203→        try { sourceRef.current.stop() } catch { /* noop */ }
   204→      }
   205→      if (ctxRef.current) {
   206→        ctxRef.current.close().catch(() => {})
   207→      }
   208→    }
   209→  }, [])
   210→
   211→  return { ...state, play, stop }
   212→}

   216→export function LandingDemo({ onLaunch }: { onLaunch: () => void }) {
   217→  const sectionRef = useRef<HTMLDivElement>(null)
   218→  const isInView = useInView(sectionRef, { once: true, margin: '-80px' })
   219→  const [t, setT] = useState(0) // 0 = before, 1 = after
   220→  const autoPlayedRef = useRef(false)
   221→
   222→  // Keep a ref of `t` so the audio hook can read the latest value without
   223→  // re-subscribing on every slider drag.
   224→  const tRef = useRef(0)
   225→  useEffect(() => { tRef.current = t }, [t])
   226→
   227→  const audio = useDemoAudio(tRef)
   228→
   229→  // Auto-play the transition once when scrolled into view.
   230→  useEffect(() => {
   231→    if (!isInView || autoPlayedRef.current) return
   232→    autoPlayedRef.current = true
   233→    const duration = 2200
   234→    const startTime = performance.now()
   235→    let rafId: number
   236→    const animate = (now: number) => {
   237→      const elapsed = now - startTime
   238→      const p = Math.min(elapsed / duration, 1)
   239→      // ease-in-out cubic
   240→      const eased = p < 0.5 ? 4 * p * p * p : 1 - Math.pow(-2 * p + 2, 3) / 2
   241→      setT(eased)
   242→      if (p < 1) rafId = requestAnimationFrame(animate)
   243→    }
   244→    rafId = requestAnimationFrame(animate)
   245→    return () => cancelAnimationFrame(rafId)
   246→  }, [isInView])
   247→
   248→  // Keyboard shortcut: Space toggles audio playback when the demo section
   249→  // is in view. Ignores the keypress if the user is typing in an input.
   250→  useEffect(() => {
   251→    const onKey = (e: KeyboardEvent) => {
   252→      if (e.code !== 'Space') return
   253→      const target = e.target as HTMLElement
   254→      if (target.tagName === 'INPUT' || target.tagName === 'TEXTAREA' || target.isContentEditable) return
   255→      // Only react when the demo section is in the viewport
   256→      const rect = sectionRef.current?.getBoundingClientRect()
   257→      if (!rect) return
   258→      const inView = rect.top < window.innerHeight * 0.6 && rect.bottom > window.innerHeight * 0.4
   259→      if (!inView) return
   260→      e.preventDefault()
   261→      if (audio.isPlaying) audio.stop()
   262→      else void audio.play()
   263→    }
   264→    window.addEventListener('keydown', onKey)
   265→    return () => window.removeEventListener('keydown', onKey)
   266→  }, [audio])
   267→
   268→  // Interpolated data
   269→  const wave = useMemo(
   270→    () => BEFORE_WAVE.map((v, i) => lerp(v, AFTER_WAVE[i], t)),
   271→    [t],
   272→  )
   273→  const spectrum = useMemo(
   274→    () => BEFORE_SPECTRUM.map((v, i) => lerp(v, AFTER_SPECTRUM[i], t)),
   275→    [t],
   276→  )
   277→  const metrics = useMemo(
   278→    () => ({
   279→      lufs: lerp(BEFORE_METRICS.lufs, AFTER_METRICS.lufs, t),
   280→      truePeak: lerp(BEFORE_METRICS.truePeak, AFTER_METRICS.truePeak, t),
   281→      score: Math.round(lerp(BEFORE_METRICS.score, AFTER_METRICS.score, t)),
   282→    }),
   283→    [t],
   284→  )
   285→
   286→  const isAfter = t > 0.5
   287→
   288→  return (
   289→    <section
   290→      ref={sectionRef}
   291→      className="relative py-24 px-4 overflow-hidden"
   292→      id="demo"
   293→    >
   294→      {/* Ambient glow background */}
   295→      <div
   296→        className="absolute inset-0 pointer-events-none"
   297→        style={{
   298→          background: `radial-gradient(ellipse 60% 40% at 50% 50%, rgba(170,255,0,${0.04 + t * 0.06}), transparent 70%)`,
   299→          transition: 'background 0.3s',
   300→        }}
   301→      />
   302→
   303→      <div className="max-w-6xl mx-auto relative">
   304→        {/* Section header */}
   305→        <div className="text-center mb-12">
   306→          <div className="inline-flex items-center gap-2 px-3 py-1 rounded-full border border-[rgba(170,255,0,0.25)] bg-[rgba(170,255,0,0.06)] text-[10px] font-mono uppercase tracking-wider text-[#AAFF00] mb-4">
   307→            <Sparkles className="w-3 h-3" />
   308→            Live Mastering Demo
   309→          </div>
   310→          <h2 className="text-3xl md:text-4xl font-bold tracking-tight mb-3">
   311→            Watch a track transform in real time.
   312→          </h2>
   313→          <p className="text-sm text-muted-foreground max-w-xl mx-auto">
   314→            Drag the slider to see the 16-stage pipeline at work. Waveform, loudness,
   315→            spectrum, and RAIN Score — all improve as mastering is applied.
   316→          </p>
   317→        </div>
   318→
   319→        {/* Main demo panel */}
   320→        <div className="rounded-2xl border border-[rgba(170,255,0,0.15)] bg-[rgba(14,16,22,0.6)] backdrop-blur-xl overflow-hidden"
   321→          style={{ boxShadow: '0 24px 80px -20px rgba(0,0,0,0.6), 0 0 0 1px rgba(170,255,0,0.04)' }}
   322→        >
   323→          {/* Top bar — before/after labels + audio toggle */}
   324→          <div className="flex items-center justify-between px-5 py-3 border-b border-white/[0.06]">
   325→            <div className="flex items-center gap-2">
   326→              <div
   327→                className="w-2 h-2 rounded-full transition-colors duration-300"
   328→                style={{ backgroundColor: isAfter ? '#AAFF00' : '#F59E0B' }}
   329→              />
   330→              <span className="text-[11px] font-mono uppercase tracking-wider text-muted-foreground">
   331→                {isAfter ? 'Mastered' : 'Before Mastering'}
   332→              </span>

   354→                aria-label={audio.isPlaying ? 'Pause audio demo' : 'Play audio demo'}

   582→        )
   583→      })}

   636→function ScoreGauge({ score, t }: { score: number; t: number }) {
   637→  const radius = 52
   638→  const circumference = 2 * Math.PI * radius
   639→  const arcLength = circumference * 0.75 // 270° arc
   640→  const dashOffset = arcLength - (score / 100) * arcLength
   641→  const color = score >= 90 ? '#AAFF00' : score >= 75 ? '#84CC16' : score >= 60 ? '#F97316' : '#EF4444'
   642→
   643→  return (
   644→    <div className="flex items-center gap-4">
   645→      <div className="relative w-32 h-32 flex-shrink-0">
   646→        <svg className="w-full h-full -rotate-[135deg]" viewBox="0 0 120 120">
   647→          {/* Track */}
   648→          <circle
   649→            cx="60"
   650→            cy="60"
   651→            r={radius}
   652→            fill="none"
   653→            stroke="rgba(255,255,255,0.06)"
   654→            strokeWidth="6"
   655→            strokeLinecap="round"
   656→            strokeDasharray={`${arcLength} ${circumference}`}
   657→          />
   658→          {/* Progress */}
   659→          <motion.circle
   660→            cx="60"
   661→            cy="60"
   662→            r={radius}
   663→            fill="none"
   664→            stroke={color}
   665→            strokeWidth="6"
   666→            strokeLinecap="round"
   667→            strokeDasharray={`${arcLength} ${circumference}`}
   668→            strokeDashoffset={dashOffset}
   669→            style={{ filter: `drop-shadow(0 0 6px ${color}80)` }}
   670→            transition={{ duration: 0.15 }}
   671→          />
   672→        </svg>
   673→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   674→          <span className="text-2xl font-bold font-mono" style={{ color }}>
   675→            {score}
   676→          </span>
   677→          <span className="text-[8px] font-mono uppercase tracking-wider text-muted-foreground/60">
   678→            / 100
   679→          </span>
   680→        </div>
   681→      </div>
   682→      <div className="flex-1 space-y-1.5">
   683→        <ScoreRow label="Spotify" val={Math.round(lerp(40, 94, t))} color="#AAFF00" />
   684→        <ScoreRow label="Apple" val={Math.round(lerp(38, 90, t))} color="#00D4FF" />
   685→        <ScoreRow label="YouTube" val={Math.round(lerp(42, 96, t))} color="#F97316" />
   686→        <ScoreRow label="Tidal" val={Math.round(lerp(35, 88, t))} color="#8B5CF6" />
   687→      </div>
   688→    </div>
   689→  )
   690→}
