     3→/**
     4→ * RAIN V6 — Ditto-standard release metadata form.
     5→ *
     6→ * Single source of truth for everything the DDEX ERN 4.3.2 manifest, the
     7→ * RAIN-CERT provenance cert, and the ID3/RIFF embedded tags need to know
     8→ * about a release. Reads + writes the Zustand `metadata` slice directly
     9→ * (auto-saves on every keystroke); the "Save to release" button is just a
    10→ * visual confirmation toast.
    11→ *
    12→ * Sections (accordion, all expanded by default on desktop):
    13→ *   1. Release Information
    14→ *   2. Track Information
    15→ *   3. Copyright & Publishing
    16→ *   4. Contributors / Credits  (table with add/remove rows)
    17→ *   5. AI Disclosure           (5-row × 3-column grid)
    18→ *   6. Release Notes / Comments
    19→ *
    20→ * Sticky top toolbar: live validation badge, "Generate ISRC + UPC",
    21→ * "Save to release" (toast), "Reset" (confirm dialog).
    22→ *
    23→ * Visual language: dark rain-panel/rain-surface-2, lime accent, font-mono
    24→ * uppercase labels, subtle Framer Motion entrance.
    25→ */

   110→const DISCLOSURE_FIELDS: { key: keyof AiDisclosure; label: string; tooltip: string }[] = [
   111→  { key: 'vocals', label: 'Vocals', tooltip: 'Lead + backing vocal performances. Generated = AI voice clone or synth voice.' },
   112→  { key: 'instrumentation', label: 'Instrumentation', tooltip: 'Instrumental performances. Generated = MIDI rendered by an AI instrument model.' },
   113→  { key: 'composition', label: 'Composition', tooltip: 'The underlying musical work (melody + harmony). Generated = AI-composed.' },
   114→  { key: 'mixing', label: 'Mixing', tooltip: 'Mix balance + automation. Assisted = AI tools used as one input among many.' },
   115→  { key: 'mastering', label: 'Mastering', tooltip: 'Final loudness + colour. RAIN V6 mastering is AI-assisted by default.' },
   116→]
   117→
   118→function issueForField(issues: { field: string; message: string }[], field: string): string | undefined {
   119→  return issues.find((i) => i.field === field || i.field.startsWith(field + '.'))?.message
   120→}

   126→export function MetadataForm() {
   127→  const metadata = useSessionStore((s) => s.metadata)
   128→  const setMetadata = useSessionStore((s) => s.setMetadata)
   129→  const reset = useSessionStore((s) => s.reset)
   130→
   131→  const [resetOpen, setResetOpen] = React.useState(false)
   132→  const [saving, setSaving] = React.useState(false)
   133→
   134→  // ---- Live validation ----
   135→  const issues = React.useMemo(() => validateMetadata(metadata), [metadata])
   136→  const issueCount = issues.length
   137→  const ready = issueCount === 0
   138→
   139→  // ---- Update helpers (each writes through the Zustand shallow-merge) ----
   140→  const update = React.useCallback(
   141→    (patch: Partial<TrackMetadata>) => setMetadata(patch),
   142→    [setMetadata],
   143→  )
   144→
   145→  const updateDisclosure = React.useCallback(
   146→    (key: keyof AiDisclosure, value: 'none' | 'assisted' | 'generated') => {
   147→      setMetadata({
   148→        aiDisclosure: { ...(metadata.aiDisclosure as AiDisclosure), [key]: value },
   149→      })
   150→    },
   151→    [metadata.aiDisclosure, setMetadata],
   152→  )
   153→
   154→  // ---- Featured-artists parsing: comma-separated string → contributors[] ----
   155→  // We reconcile the contributors array: featured-artist rows that were auto-
   156→  // generated from the input are removed and re-added whenever the string
   157→  // changes, preserving user-added rows for other roles.
   158→  const featuredString = React.useMemo(
   159→    () => (metadata.contributors ?? [])
   160→      .filter((c) => c.role === 'featured')
   161→      .map((c) => c.name)
   162→      .join(', '),
   163→    [metadata.contributors],
   164→  )
   165→
   166→  const setFeaturedArtists = React.useCallback(
   167→    (raw: string) => {
   168→      const names = raw.split(',').map((s) => s.trim()).filter(Boolean)
   169→      const nonFeatured = (metadata.contributors ?? []).filter((c) => c.role !== 'featured')
   170→      const featured: Contributor[] = names.map((name) => ({ name, role: 'featured' }))
   171→      setMetadata({ contributors: [...nonFeatured, ...featured] })
   172→    },
   173→    [metadata.contributors, setMetadata],
   174→  )
   175→
   176→  // ---- Auto-populate: when primary artist changes and there's no performer row, add one ----
   177→  const ensurePerformer = React.useCallback(
   178→    (artist: string) => {
   179→      const existing = (metadata.contributors ?? []).find((c) => c.role === 'performer')
   180→      if (existing) {
   181→        // Update the existing performer's name (don't add a duplicate).
   182→        const next = (metadata.contributors ?? []).map((c) =>
   183→          c.role === 'performer' ? { ...c, name: artist } : c,
   184→        )
   185→        setMetadata({ contributors: next })
   186→      } else if (artist.trim()) {
   187→        setMetadata({
   188→          contributors: [{ name: artist, role: 'performer' }, ...(metadata.contributors ?? [])],
   189→        })
   190→      }
   191→    },
   192→    [metadata.contributors, setMetadata],
   193→  )
   194→
   195→  // ---- Contributor row operations ----
   196→  const addContributor = React.useCallback(() => {
   197→    const next: Contributor[] = [
   198→      ...(metadata.contributors ?? []),
   199→      { name: '', role: 'songwriter', share: undefined },
   200→    ]
   201→    setMetadata({ contributors: next })
   202→  }, [metadata.contributors, setMetadata])
   203→
   204→  const updateContributor = React.useCallback(
   205→    (index: number, patch: Partial<Contributor>) => {
   206→      const next = (metadata.contributors ?? []).map((c, i) =>
   207→        i === index ? { ...c, ...patch } : c,
   208→      )
   209→      setMetadata({ contributors: next })
   210→    },
   211→    [metadata.contributors, setMetadata],
   212→  )
   213→
   214→  const removeContributor = React.useCallback(
   215→    (index: number) => {
   216→      const next = (metadata.contributors ?? []).filter((_, i) => i !== index)
   217→      setMetadata({ contributors: next })
   218→    },
   219→    [metadata.contributors, setMetadata],
   220→  )
   221→
   222→  // ---- Writer share total ----
   223→  const writerShareTotal = React.useMemo(() => {
   224→    return (metadata.contributors ?? [])
   225→      .filter((c) => WRITER_ROLES.includes(c.role))
   226→      .reduce((sum, c) => sum + (typeof c.share === 'number' && !Number.isNaN(c.share) ? c.share : 0), 0)
   227→  }, [metadata.contributors])
   228→
   229→  // ---- Re-release toggle ----
   230→  const [isRerelease, setIsRerelease] = React.useState(Boolean(metadata.originalReleaseDate))
   231→
   232→  // ---- Genre:Subgenre parsing (combined "Genre:Subgenre" string in metadata.genreSubgenre) ----
   233→  const [selectedGenre, selectedSubgenre] = React.useMemo(() => {
   234→    const v = metadata.genreSubgenre ?? ''
   235→    const [g, s] = v.split(':')
   236→    return [g ?? '', s ?? '']
   237→  }, [metadata.genreSubgenre])
   238→
   239→  const setGenreSubgenre = React.useCallback(
   240→    (genre: string, subgenre: string) => {
   241→      update({
   242→        genreSubgenre: subgenre ? `${genre}:${subgenre}` : genre,
   243→        // Also keep the legacy `genre` field in sync (top-level only).
   244→        genre: genre.toLowerCase(),
   245→      })
   246→    },
   247→    [update],
   248→  )
   249→
   250→  // ---- Territories multi-select ----
   251→  const toggleTerritory = React.useCallback(
   252→    (code: string) => {
   253→      const current = metadata.territories ?? []
   254→      // If 'WORLDWIDE' is in the list and the user picks a specific country,
   255→      // remove 'WORLDWIDE' (mutually exclusive intent).
   256→      let next: string[]
   257→      if (current.includes(code)) {
   258→        next = current.filter((c) => c !== code)
   259→      } else {
   260→        next = code === 'WORLDWIDE'
   261→          ? ['WORLDWIDE']
   262→          : [...current.filter((c) => c !== 'WORLDWIDE'), code]
   263→      }
   264→      if (next.length === 0) next = ['WORLDWIDE'] // never empty
   265→      update({ territories: next })
   266→    },
   267→    [metadata.territories, update],
   268→  )
   269→
   270→  // ---- Toolbar actions ----
   271→  const handleGenerateIds = React.useCallback(() => {
   272→    update({ isrc: generateIsrc(), upc: generateUpc() })
   273→    notifyInfo('Identifiers generated', 'ISRC + UPC populated with valid check digits.')
   274→  }, [update])
   275→
   276→  const handleSave = React.useCallback(() => {
   277→    setSaving(true)
   278→    // The store is already up to date (auto-save). This is just a
   279→    // visual confirmation + a brief saving indicator for UX.
   280→    setTimeout(() => {
   281→      setSaving(false)
   282→      if (ready) {
   283→        notifySuccess('Metadata saved', 'Release metadata is valid and ready for distribution.')
   284→      } else {
   285→        notifyWarning(
   286→          'Metadata saved with issues',
   287→          `${issueCount} validation issue${issueCount === 1 ? '' : 's'} remain — review before distributing.`,
   288→        )
   289→      }
   290→    }, 350)
   291→  }, [ready, issueCount])
   292→
   293→  const handleReset = React.useCallback(() => {
   294→    reset()
   295→    setResetOpen(false)
   296→    notifyInfo('Metadata reset', 'All fields restored to defaults.')
   297→  }, [reset])
   298→
   299→  // ---- Auto-suggest P-line holder from artist + copyrightYear ----
   300→  const suggestPline = React.useCallback(() => {
   301→    const yr = metadata.copyrightYear || String(new Date().getFullYear())
   302→    const who = metadata.artist || metadata.copyrightHolder || ''
   303→    if (who) update({ copyrightHolder: `${yr} ${who}` })
   304→  }, [metadata.artist, metadata.copyrightHolder, metadata.copyrightYear, update])
   305→
   306→  // ---------------------------------------------------------------------
   307→  // Render
   308→  // ---------------------------------------------------------------------
   309→
   310→  const disclosureHasGenerated = React.useMemo(() => {
   311→    const d = metadata.aiDisclosure
   312→    if (!d) return false
   313→    return (Object.values(d) as string[]).some((v) => v === 'generated')
   314→  }, [metadata.aiDisclosure])
   315→
   316→  // Accordion value state — all expanded by default.
   317→  const [accordionValue, setAccordionValue] = React.useState<string[]>([
   318→    'release', 'track', 'copyright', 'contributors', 'ai', 'notes',
   319→  ])
   320→
   321→  return (
   322→    <div className="space-y-3">
   323→      {/* ----------------------------------------------------------------- */}
   324→      {/* Sticky toolbar                                                    */}
   325→      {/* ----------------------------------------------------------------- */}
   326→      <div
   327→        className="sticky top-0 z-20 -mx-4 px-4 py-2.5 mb-3 backdrop-blur-xl bg-[rgba(10,12,18,0.85)] border-b border-rain-border"
   328→        style={{ boxShadow: '0 4px 16px -8px rgba(0,0,0,0.5)' }}
   329→      >
   330→        <div className="flex flex-wrap items-center gap-2">
   331→          <div className="flex items-center gap-2 mr-auto">
   332→            <ListChecks className="w-4 h-4 text-rain-accent" />
   333→            <span className="text-[10px] font-mono uppercase tracking-wider text-muted-foreground">
   334→              Validation
   335→            </span>
   336→            <Badge
   337→              variant="outline"
   338→              className={
   339→                ready
   340→                  ? 'border-rain-accent/50 text-rain-accent bg-rain-accent/10'
   341→                  : 'border-amber-400/50 text-amber-400 bg-amber-400/10'
   342→              }
   343→            >
   344→              {ready ? (
   345→                <>
   346→                  <CheckCircle2 className="w-3 h-3" />
   347→                  Ready
   348→                </>
   349→              ) : (
   350→                <>
   351→                  <AlertTriangle className="w-3 h-3" />
   352→                  {issueCount} issue{issueCount === 1 ? '' : 's'}
   353→                </>
   354→              )}
   355→            </Badge>
   356→          </div>
   357→
   358→          <Button
   359→            type="button"
   360→            size="sm"
   361→            variant="outline"
   362→            onClick={handleGenerateIds}
   363→            className="bg-rain-surface-2 border-rain-border hover:border-rain-accent/50 hover:text-rain-accent font-mono text-[11px]"
   364→          >
   365→            <Wand2 className="w-3.5 h-3.5" />
   366→            Generate ISRC + UPC
   367→          </Button>
   368→
   369→          <Button
   370→            type="button"
   371→            size="sm"
   372→            onClick={handleSave}
   373→            disabled={saving}
   374→            className="bg-rain-accent text-black hover:bg-rain-accent/90 font-mono text-[11px]"
   375→          >
   376→            {saving ? (
   377→              <Loader2 className="w-3.5 h-3.5 animate-spin" />
   378→            ) : (
   379→              <Save className="w-3.5 h-3.5" />
   380→            )}
   381→            Save to release
   382→          </Button>
   383→
   384→          <Button
   385→            type="button"
   386→            size="sm"
   387→            variant="outline"
   388→            onClick={() => setResetOpen(true)}
   389→            className="bg-rain-surface-2 border-rain-border hover:border-red-400/50 hover:text-red-400 font-mono text-[11px]"
   390→          >
   391→            <RotateCcw className="w-3.5 h-3.5" />
   392→            Reset
   393→          </Button>
   394→        </div>
   395→      </div>
   396→
   397→      {/* ----------------------------------------------------------------- */}
   398→      {/* Accordion form                                                    */}
   399→      {/* ----------------------------------------------------------------- */}
   400→      <motion.div
   401→        initial={{ opacity: 0, y: 6 }}
   402→        animate={{ opacity: 1, y: 0 }}
   403→        transition={{ duration: 0.25, ease: 'easeOut' }}
   404→      >
   405→        <Accordion
   406→          type="multiple"
   407→          value={accordionValue}
   408→          onValueChange={setAccordionValue}
   409→          className="rain-panel rounded-lg px-4"
   410→        >
   411→          {/* ----------------------------------------------------------- */}
   412→          {/* 1. Release Information                                       */}
   413→          {/* ----------------------------------------------------------- */}
   414→          <AccordionItem value="release" className="border-rain-border">
   415→            <AccordionTrigger className="text-sm font-mono uppercase tracking-wider text-rain-accent hover:no-underline">
   416→              <span className="flex items-center gap-2">
   417→                <span className="text-[9px] text-muted-foreground">01</span>
   418→                Release Information
   419→              </span>
   420→            </AccordionTrigger>
   421→            <AccordionContent>
   422→              <div className="grid lg:grid-cols-2 gap-3 pb-2">
   423→                <Field
   424→                  label="Release Title"
   425→                  value={metadata.title}
   426→                  onValueChange={(v) => update({ title: v })}
   427→                  required
   428→                  placeholder="e.g. Midnight Protocol"
   429→                  error={issueForField(issues, 'title')}
   430→                  warning={!metadata.title ? 'Required' : undefined}
   431→                />
   432→                <Field
   433→                  label="Primary Artist"
   434→                  value={metadata.artist}
   435→                  onValueChange={(v) => {
   436→                    update({ artist: v })
   437→                    ensurePerformer(v)
   438→                  }}
   439→                  required
   440→                  placeholder="e.g. GLM Collective"
   441→                  error={issueForField(issues, 'artist')}
   442→                  warning={!metadata.artist ? 'Required' : undefined}
   443→                />
   444→
   445→                <RadioField
   446→                  label="Release Type"
   447→                  value={metadata.releaseType ?? 'single'}
   448→                  onValueChange={(v) => update({ releaseType: v as TrackMetadata['releaseType'] })}
   449→                  layout="grid"
   450→                  options={[
   451→                    { value: 'single', label: 'Single' },
   452→                    { value: 'ep', label: 'EP' },
   453→                    { value: 'album', label: 'Album' },
   454→                    { value: 'compilation', label: 'Compilation' },
   455→                  ]}
   456→                  hint="Drives DDEX <ReleaseType> + DSP categorisation."
   457→                />
   458→
   459→                <Field
   460→                  label="Featured Artists"
   461→                  value={featuredString}
   462→                  onValueChange={setFeaturedArtists}
   463→                  placeholder="Comma-separated — added as contributors with role 'featured'"
   464→                  hint="Parsed into the Contributors table with role 'featured'."
   465→                  containerClassName="lg:col-span-2"
   466→                />
   467→
   468→                <Field
   469→                  label="Label"
   470→                  value={metadata.label ?? ''}
   471→                  onValueChange={(v) => update({ label: v })}
   472→                  placeholder="Record label name (optional)"
   473→                />
   474→                <Field
   475→                  label="Distributor"
   476→                  value={metadata.distributor ?? ''}
   477→                  onValueChange={(v) => update({ distributor: v })}
   478→                  placeholder="RAIN V6"
   479→                  hint="Default 'RAIN V6' — change if delivering through a different aggregator."
   480→                />
   481→
   482→                <Field
   483→                  label="Release Date"
   484→                  type="date"
   485→                  value={metadata.releaseDate ?? ''}
   486→                  onValueChange={(v) => update({ releaseDate: v })}
   487→                  error={issueForField(issues, 'releaseDate')}
   488→                  hint="Official release date — drives the DDEX <ReleaseDate> element."
   489→                />
   490→                <div className="flex flex-col gap-1">
   491→                  <div className="flex items-center justify-between">
   492→                    <label className="text-[10px] font-mono uppercase tracking-wider text-muted-foreground">
   493→                      Re-release?
   494→                    </label>
   495→                    <Switch
   496→                      checked={isRerelease}
   497→                      onCheckedChange={(checked) => {
   498→                        setIsRerelease(checked)
   499→                        if (!checked) update({ originalReleaseDate: '' })
   500→                      }}
   501→                    />
   502→                  </div>
   503→                  {isRerelease && (
   504→                    <Field
   505→                      label="Original Release Date"
   506→                      type="date"
   507→                      value={metadata.originalReleaseDate ?? ''}
   508→                      onValueChange={(v) => update({ originalReleaseDate: v })}
   509→                      error={issueForField(issues, 'originalReleaseDate')}
   510→                      containerClassName="mt-1"
   511→                    />
   512→                  )}
   513→                  {!isRerelease && (
   514→                    <div className="text-[9px] font-mono text-muted-foreground/70">
   515→                      Toggle on if this is a re-release of previously-released material.
   516→                    </div>
   517→                  )}
   518→                </div>
   519→
   520→                {/* Territories multi-select */}
   521→                <div className="flex flex-col gap-1 lg:col-span-2">
   522→                  <label className="text-[10px] font-mono uppercase tracking-wider text-muted-foreground">
   523→                    Territories
   524→                  </label>
   525→                  <Popover>
   526→                    <PopoverTrigger asChild>
   527→                      <button
   528→                        type="button"
   529→                        className="w-full bg-rain-surface-2 border border-rain-border rounded-md px-2 py-1.5 text-xs font-mono text-left flex items-center justify-between hover:border-rain-accent/40 transition-colors"
   530→                      >
   531→                        <span className="truncate">
   532→                          {(metadata.territories ?? []).length === 0
   533→                            ? '— none —'
   534→                            : (metadata.territories ?? ['WORLDWIDE']).join(', ')}
   535→                        </span>
   536→                        <span className="text-[9px] text-muted-foreground">
   537→                          {(metadata.territories ?? []).length} selected
   538→                        </span>
   539→                      </button>
   540→                    </PopoverTrigger>
   541→                    <PopoverContent
   542→                      className="w-80 bg-rain-surface-2 border-rain-border p-2 max-h-80 overflow-y-auto rain-scrollbar"
   543→                      align="start"
   544→                    >
   545→                      <div className="grid grid-cols-1 gap-1">
   546→                        {TERRITORY_OPTIONS.map((t) => (
   547→                          <label
   548→                            key={t.value}
   549→                            className="flex items-center gap-2 px-2 py-1 rounded hover:bg-rain-accent/10 cursor-pointer text-xs font-mono"
   550→                          >
   551→                            <Checkbox
   552→                              checked={(metadata.territories ?? []).includes(t.value)}
   553→                              onCheckedChange={() => toggleTerritory(t.value)}
   554→                            />
   555→                            <span className="text-rain-accent/80 w-10">{t.value === 'WORLDWIDE' ? 'WW' : t.value}</span>
   556→                            <span className="text-foreground/90">{t.label}</span>
   557→                          </label>
   558→                        ))}
   559→                      </div>
   560→                    </PopoverContent>
   561→                  </Popover>
   562→                  <div className="text-[9px] font-mono text-muted-foreground/70">
   563→                    Default Worldwide. Picking specific countries removes Worldwide.
   564→                  </div>
   565→                </div>
   566→              </div>
   567→            </AccordionContent>
