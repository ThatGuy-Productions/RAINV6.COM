    24→    24→async function reportBetaExportEvent(format: string, outputFileHash: string | undefined) {
    25→    25→  try {
    26→    26→    await fetch('/api/rain/render', {
    27→    27→      method: 'POST',
    28→    28→      headers: { 'Content-Type': 'application/json' },
    29→    29→      body: JSON.stringify({ kind: 'export', format, outputFileHash }),
    30→    30→    })
    31→    31→  } catch (e) {
    32→    32→    console.warn('[analytics] reportBetaExportEvent failed:', e)
    33→    33→  }
    34→    34→}

    58→    58→/** Extract the Chromaprint hash from a RAIN-CERT manifest (if present). */
    59→    59→function fingerprintFromCert(cert: ProvenanceCertificate | null): string | undefined {
    60→    60→  if (!cert) return undefined
    61→    61→  const a = cert.manifest.assertions.find((x) => x.label === 'org.rain.fingerprint')
    62→    62→  const h = a?.data?.hash
    63→    63→  return typeof h === 'string' ? h : undefined
    64→    64→}

    66→    66→export function ExportTab() {
    67→    67→  const fileName = useSessionStore((s) => s.fileName)
    68→    68→  const hasProcessed = useSessionStore((s) => s.hasProcessed)
    69→    69→  const metadata = useSessionStore((s) => s.metadata)
    70→    70→  const setMetadata = useSessionStore((s) => s.setMetadata)
    71→    71→  // P2-EXPORT: read the real cert from the store so the toggles can embed it.
    72→    72→  const rainCert = useSessionStore((s) => s.rainCert)
    73→    73→
    74→    74→  const [selectedFormat, setSelectedFormat] = useState('wav24')
    75→    75→  // P2-EXPORT directive: five real toggles. Each one is honored byte-for-byte
    76→    76→  // by the encoder AND re-verified after export via verifyExported*().
    77→    77→  const [embedProvenance, setEmbedProvenance] = useState(true)
    78→    78→  const [embedSignature, setEmbedSignature] = useState(true)
    79→    79→  const [embedFingerprint, setEmbedFingerprint] = useState(true)
    80→    80→  const [embedMetadata, setEmbedMetadata] = useState(true)
    81→    81→  const [attachCertificate, setAttachCertificate] = useState(false)
    82→    82→  const [isExporting, setIsExporting] = useState(false)
    83→    83→  const [verification, setVerification] = useState<ExportVerificationResult | null>(null)
    84→    84→
    85→    85→  const fingerprint = useMemo(() => fingerprintFromCert(rainCert), [rainCert])
    86→    86→
    87→    87→  // Build the ExportOptions object that gets passed to the encoder. This is
    88→    88→  // the SINGLE source of truth for what gets embedded — the encoder reads no
    89→    89→  // other state.
    90→    90→  const buildOptions = (): ExportOptions => ({
    91→    91→    embedProvenance,
    92→    92→    embedSignature,
    93→    93→    embedFingerprint,
    94→    94→    embedMetadata,
    95→    95→    attachCertificate,
    96→    96→    metadata: {
    97→    97→      title: metadata.title,
    98→    98→      artist: metadata.artist,
    99→    99→      album: metadata.album || undefined,
   100→   100→      year: metadata.year || undefined,
   101→   101→      isrc: metadata.isrc || undefined,
   102→   102→      comment: metadata.comment || undefined,
   103→   103→    },
   104→   104→    fingerprint,
   105→   105→  })
   106→   106→
   107→   107→  const isAtmos = selectedFormat === 'atmos'
   108→   108→  const isMp3 = selectedFormat === 'mp3_320'
   109→   109→  const isWav = selectedFormat === 'wav24' || selectedFormat === 'wav16'
   110→   110→
   111→   111→  // Toggles are only meaningful for WAV/MP3. Atmos package uses its own
   112→   112→  // embedding path (axml chunk + sidecar XML), so we show a notice rather
   113→   113→  // than fake-enabled toggles.
   114→   114→  const togglesRelevant = isWav || isMp3
   115→   115→
   116→   116→  const handleExport = async () => {
   117→   117→    if (!hasProcessed) return
   118→   118→    setIsExporting(true)
   119→   119→    setVerification(null)
   120→   120→    // P2-ANALYTICS: measure real export wall-clock + actual blob size for
   121→   121→    // the Analytics tab's "Export Format Distribution" chart. Each branch
   122→   122→    // (atmos / mp3 / wav) records its own format string + size + duration.
   123→   123→    const exportStart = Date.now()
   124→   124→    const fileState = useSessionStore.getState()
   125→   125→    const recordExport = (format: string, blob: Blob, bitDepthOverride?: number) => {
   126→   126→      void recordExportDetails({
   127→   127→        timestamp: Date.now(),
   128→   128→        format,
   129→   129→        bytes: blob.size,
   130→   130→        durationMs: Date.now() - exportStart,
   131→   131→        sampleRate: fileState.fileSampleRate || 48000,
   132→   132→        bitDepth: bitDepthOverride ?? fileState.fileBitDepth ?? 24,
   133→   133→        channels: fileState.fileChannels || 2,
   134→   134→        fileName: fileState.fileName ?? '(unknown)',
   135→   135→      }).catch((e) => console.warn('[analytics] recordExportDetails failed:', e))
   136→   136→      void reportBetaExportEvent(selectedFormat, rainCert?.outputHash)
   137→   137→    }
   138→   138→    try {
   139→   139→      const format = FORMATS.find((f) => f.id === selectedFormat)!
   140→   140→      const baseName = (metadata.title || fileName || 'rain-master').replace(/\s+/g, '_')
   141→   141→
   142→   142→      // --- Atmos path: separate code path, no ID3/LIST toggles apply ---
   143→   143→      if (format.id === 'atmos') {
   144→   144→        // EXPORT CAP: the spatial pipeline processes the full track (no silent
   145→   145→        // 60s preview truncation). 360s (6 min) is the memory-safe ceiling for
   146→   146→        // a 7.1.4 bed (12ch × 360s × 48k × 4B ≈ 830 MB persistent). Tracks
   147→   147→        // beyond this get an EXPLICIT error — never a silently truncated file.
   148→   148→        const ATMOS_EXPORT_MAX_SEC = 360
   149→   149→        const sourceBuffer = audioEngine.getProcessedBuffer() ?? null
   150→   150→        const sourceDuration = sourceBuffer?.duration ?? 0
   151→   151→        if (sourceDuration > ATMOS_EXPORT_MAX_SEC) {
   152→   152→          const mm = Math.floor(sourceDuration / 60)
   153→   153→          const ss = Math.floor(sourceDuration % 60)
   154→   154→          throw new Error(
   155→   155→            `Atmos export supports tracks up to ${Math.floor(ATMOS_EXPORT_MAX_SEC / 60)} minutes. ` +
   156→   156→            `Your track is ${mm}:${ss.toString().padStart(2, '0')}. ` +
   157→   157→            `Please trim the track or export as WAV / MP3 instead.`,
   158→   158→          )
   159→   159→        }
   160→   160→        const config: SpatialConfig = {
   161→   161→          bedFormat: '7.1.4',
   162→   162→          outputMode: 'MULTICHANNEL',
   163→   163→          hrtf: 'SPHERICAL',
   164→   164→          objects: 16,
   165→   165→          width: 100,
   166→   166→          centerFocus: 50,
   167→   167→        }
   168→   168→        // Pass the export cap so the full track (up to 6 min) is processed.
   169→   169→        // If the result is somehow still truncated (defensive), refuse to ship.
   170→   170→        const result = await audioEngine.processSpatial(config, undefined, undefined, ATMOS_EXPORT_MAX_SEC)
   171→   171→        if (result.truncated) {
   172→   172→          throw new Error(
   173→   173→            `Atmos export was truncated to ${result.processedSeconds.toFixed(0)}s ` +
   174→   174→            `(track is ${result.inputSeconds.toFixed(0)}s). Refusing to export a partial file. ` +
   175→   175→            `Please trim the track to ≤ ${ATMOS_EXPORT_MAX_SEC}s.`,
   176→   176→          )
   177→   177→        }
   178→   178→        // Full-source package: .atmos.wav + ADM XML sidecar + spatial.json +
   179→   179→        // README.txt + MANIFEST.json (real SHA-256 over every file).
   180→   180→        const blob = await exportAtmosPackage(result, { title: metadata.title, artist: metadata.artist }, config)
   181→   181→        triggerDownload(blob, `${baseName}_Atmos_7.1.4.zip`)
   182→   182→        recordExport('ATMOS-7.1.4', blob, 24)
   183→   183→        notifySuccess(
   184→   184→          'Atmos full-source package complete',
   185→   185→          `${result.processedSeconds.toFixed(1)}s · 5 files · ${(blob.size / 1024 / 1024).toFixed(1)} MB`,
   186→   186→        )
   187→   187→        setIsExporting(false)
   188→   188→        return
   189→   189→      }
   190→   190→
   191→   191→      const options = buildOptions()
   192→   192→
   193→   193→      // --- MP3 path ---
   194→   194→      if (format.id === 'mp3_320') {
   195→   195→        const blob = audioEngine.exportMp3(320, rainCert, options)
   196→   196→        // P2-EXPORT Step 6: verify the produced bytes actually contain what
   197→   197→        // the toggles promised. The verify function re-parses the ID3v2 tag
   198→   198→        // and reports per-toggle ✓/✗.
   199→   199→        const result = await verifyExportedMp3(blob, options)
   200→   200→        setVerification(result)
   201→   201→
   202→   202→        if (options.attachCertificate && rainCert) {
   203→   203→          // Sidecar path: ZIP the MP3 + the cert.json together. The cert.json
   204→   204→          // is the FULL cert (sig + manifest + fingerprint intact) — the
   205→   205→          // sidecar IS the authoritative cert.
   206→   206→          const audioBytes = new Uint8Array(await blob.arrayBuffer())
   207→   207→          const certJson = JSON.stringify(rainCert, null, 2)
   208→   208→          const zipBlob = buildSidecarZip(
   209→   209→            audioBytes,
   210→   210→            `${baseName}_RAIN.mp3`,
   211→   211→            certJson,
   212→   212→            `${baseName}.cert.json`,
   213→   213→          )
   214→   214→          triggerDownload(zipBlob, `${baseName}_RAIN.zip`)
   215→   215→          recordExport('MP3-320+SIDECAR', zipBlob, 16)
   216→   216→        } else {
   217→   217→          triggerDownload(blob, `${baseName}_RAIN.mp3`)
   218→   218→          recordExport('MP3-320', blob, 16)
   219→   219→        }
   220→   220→        reportExportResult(result, 'MP3')
   221→   221→        setIsExporting(false)
   222→   222→        return
   223→   223→      }
   224→   224→
   225→   225→      // --- WAV path (24-bit / 16-bit) ---
   226→   226→      const bitDepth = (format.bitDepth === 24 ? 24 : 16) as 16 | 24
   227→   227→      const blob = audioEngine.exportWav(bitDepth, rainCert, options)
   228→   228→      const result = await verifyExportedWav(blob, options)
   229→   229→      setVerification(result)
   230→   230→
   231→   231→      if (options.attachCertificate && rainCert) {
   232→   232→        const audioBytes = new Uint8Array(await blob.arrayBuffer())
   233→   233→        const certJson = JSON.stringify(rainCert, null, 2)
   234→   234→        const zipBlob = buildSidecarZip(
   235→   235→          audioBytes,
   236→   236→          `${baseName}_RAIN.wav`,
   237→   237→          certJson,
   238→   238→          `${baseName}.cert.json`,
   239→   239→        )
   240→   240→        triggerDownload(zipBlob, `${baseName}_RAIN.zip`)
   241→   241→        recordExport(bitDepth === 24 ? 'WAV-24+SIDECAR' : 'WAV-16+SIDECAR', zipBlob, bitDepth)
   242→   242→      } else {
   243→   243→        triggerDownload(blob, `${baseName}_RAIN.wav`)
   244→   244→        recordExport(bitDepth === 24 ? 'WAV-24' : 'WAV-16', blob, bitDepth)
   245→   245→      }
   246→   246→      reportExportResult(result, 'WAV')
   247→   247→    } catch (e) {
   248→   248→      console.error('[RAIN export] error:', e)
   249→   249→      notifyError('Export failed', e instanceof Error ? e.message : 'Unknown error')
   250→   250→    } finally {
   251→   251→      setIsExporting(false)
   252→   252→    }
   253→   253→  }
   254→   254→
   255→   255→  return (
   256→   256→    <div className="grid lg:grid-cols-3 gap-4">
   257→   257→      <div className="lg:col-span-2 space-y-4">
   258→   258→        <div className="rain-panel rounded-lg p-4">
   259→   259→          <div className="text-[10px] font-mono uppercase tracking-wider text-muted-foreground mb-1">
   260→   260→            Export Format
   261→   261→          </div>
   262→   262→          <div className="text-sm font-semibold mb-3">Choose authoritative master</div>
   263→   263→          <div className="grid sm:grid-cols-2 gap-2">
   264→   264→            {FORMATS.map((f) => (
   265→   265→              <button
   266→   266→                key={f.id}
   267→   267→                onClick={() => setSelectedFormat(f.id)}
   268→   268→                disabled={!hasProcessed}
   269→   269→                className={`text-left p-3 rounded-md border transition-all disabled:opacity-50 disabled:cursor-not-allowed ${
   270→   270→                  selectedFormat === f.id
   271→   271→                    ? 'border-rain-accent bg-rain-accent/10'
   272→   272→                    : 'border-rain-border bg-rain-surface-2/60 hover:border-rain-accent/50'
   273→   273→                }`}
   274→   274→              >
   275→   275→                <div className="flex items-start gap-2">
   276→   276→                  <div className={selectedFormat === f.id ? 'text-rain-accent' : 'text-muted-foreground'}>
   277→   277→                    {f.icon}
   278→   278→                  </div>
   279→   279→                  <div className="flex-1 min-w-0">
   280→   280→                    <div className="text-sm font-semibold mb-0.5">{f.label}</div>
   281→   281→                    <div className="text-[10px] text-muted-foreground leading-tight mb-1">{f.desc}</div>
   282→   282→                    <div className="text-[9px] font-mono text-muted-foreground">{f.size}</div>
   283→   283→                  </div>
   284→   284→                </div>
   285→   285→              </button>
   286→   286→            ))}
   287→   287→          </div>
   288→   288→          {/* Full-bandwidth guarantee banner — addresses the "clean cutoff
   289→   289→              above 16-18 kHz" issue. WAV is lossless PCM (Nyquist = 24 kHz
   290→   290→              @ 48 k). MP3 LAME's default lowpass was disabled (patched in
   291→   291→              node_modules) so 320 kbps exports preserve the full top octave. */}
   292→   292→          <div className="mt-3 flex items-start gap-2 p-2.5 rounded-md border border-rain-accent/30 bg-rain-accent/5">
   293→   293→            <CheckCircle2 className="w-4 h-4 text-rain-accent flex-shrink-0 mt-0.5" />
   294→   294→            <div className="text-[10px] leading-tight">
   295→   295→              <span className="font-semibold text-rain-accent">Full Bandwidth Guarantee</span>
   296→   296→              <span className="text-muted-foreground"> — WAV exports are lossless PCM (Nyquist = sample-rate / 2). MP3 320 kbps exports have LAME's default lowpass <em>disabled</em>, preserving the full 20–24 kHz top octave. No 16–18 kHz encoder cliff.</span>
   297→   297→            </div>
   298→   298→          </div>
   299→   299→        </div>
   300→   300→
   301→   301→        <div className="rain-panel rounded-lg p-4">
   302→   302→          <div className="text-[10px] font-mono uppercase tracking-wider text-muted-foreground mb-1">
   303→   303→            Metadata (ID3 / RIFF INFO — used by Metadata toggle)
   304→   304→          </div>
   305→   305→          <div className="text-sm font-semibold mb-3">Tag the master</div>
   306→   306→          <div className="grid sm:grid-cols-2 gap-3">
   307→   307→            <MetadataField label="Title" value={metadata.title} onChange={(v) => setMetadata({ title: v })} />
   308→   308→            <MetadataField label="Artist" value={metadata.artist} onChange={(v) => setMetadata({ artist: v })} />
   309→   309→            <MetadataField label="Album" value={metadata.album} onChange={(v) => setMetadata({ album: v })} />
   310→   310→            <MetadataField label="Track #" value={metadata.trackNumber} onChange={(v) => setMetadata({ trackNumber: v })} />
   311→   311→            <MetadataField label="Year" value={metadata.year} onChange={(v) => setMetadata({ year: v })} />
   312→   312→            <MetadataField label="ISRC" value={metadata.isrc} onChange={(v) => setMetadata({ isrc: v })} placeholder="US-XXX-YY-NNNNN" />
   313→   313→            <MetadataField label="Comment" value={metadata.comment} onChange={(v) => setMetadata({ comment: v })} />
   314→   314→          </div>
   315→   315→        </div>
   316→   316→
   317→   317→        {/* P2-EXPORT: 5 real toggles, grouped logically per the directive. */}
   318→   318→        <div className="rain-panel rounded-lg p-4">
   319→   319→          <div className="text-[10px] font-mono uppercase tracking-wider text-muted-foreground mb-1">
   320→   320→            Provenance
   321→   321→          </div>
   322→   322→          <div className="text-sm font-semibold mb-3">What gets embedded in the audio bytes</div>
   323→   323→          {!togglesRelevant && (
   324→   324→            <div className="text-[10px] text-muted-foreground/80 italic mb-2">
   325→   325→              Toggles below apply to WAV/MP3 only. The Atmos path embeds metadata

   327→   327→              it does not use these ID3/RIFF INFO toggles.
   328→   328→            </div>
   329→   329→          )}
   330→   330→          <div className={`space-y-2 ${!togglesRelevant ? 'opacity-50 pointer-events-none' : ''}`}>
   331→   331→            <ToggleRow
   332→   332→              label="RAIN-CERT certificate (Ed25519)"

   334→   334→              checked={embedProvenance}
   335→   335→              onChange={setEmbedProvenance}
   336→   336→              disabled={!rainCert}

   338→   338→            />
   339→   339→            <ToggleRow
   340→   340→              label="Ed25519 digital signature"
   341→   341→              desc={isWav ? 'WAV LIST/INFO ISIG field with sig hex (independent of cert JSON)' : 'MP3 ID3v2 TXXX "RAIN_SIGNATURE" frame with sig hex'}
   342→   342→              checked={embedSignature}
   343→   343→              onChange={setEmbedSignature}
   344→   344→              disabled={!rainCert}
   345→   345→              disabledNote={!rainCert ? 'No cert available — render the pipeline first' : undefined}

   506→   506→  if (result.ok) {
   507→   507→    notifySuccess(
   508→   508→      `${formatLabel} export verified`,
   509→   509→      `All toggles honored · ${result.sizeBytes.toLocaleString()} bytes`,
   510→   510→    )
   511→   511→  } else {
   512→   512→    const failed = Object.entries(result.checks)
   513→   513→      .filter(([, v]) => !v.ok)
   514→   514→      .map(([k]) => k)
   515→   515→    notifyError(
   516→   516→      `${formatLabel} export verification failed`,
   517→   517→      `Mismatch on: ${failed.join(', ')}. The encoder did not honor one or more toggles.`,
   518→   518→    )
   519→   519→  }

   525→   525→  const a = document.createElement('a')

   527→   527→  a.download = filename
   528→   528→  document.body.appendChild(a)
   529→   529→  a.click()
   530→   530→  document.body.removeChild(a)
   531→   531→  URL.revokeObjectURL(url)
   532→   532→}
