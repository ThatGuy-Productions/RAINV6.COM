    70→function platformLabel(slug: string): string {

    78→function formatTrendPct(pct: number | null): string {
    79→  if (pct === null || !Number.isFinite(pct)) return '—'
    80→  const sign = pct > 0 ? '+' : ''
    81→  return `${sign}${pct.toFixed(0)}%`
    82→}
    83→
    84→function formatTrendScore(delta: number | null): string {
    85→  if (delta === null || !Number.isFinite(delta)) return '—'
    86→  const sign = delta > 0 ? '+' : ''
    87→  return `${sign}${delta.toFixed(1)}`
    88→}

    90→/** Human-readable label for an ActivityRecord.type. */
    91→function activityTypeLabel(type: string): string {
    92→  switch (type) {
    93→    case 'undo': return 'Undo'
    94→    case 'redo': return 'Redo'
    95→    case 'repair': return 'Repair run'
    96→    case 'ai-query': return 'AI Co-Master query'
    97→    case 'render': return 'Render'
    98→    case 'export': return 'Export'
    99→    case 'preset-apply': return 'Preset applied'
   100→    default: return type.charAt(0).toUpperCase() + type.slice(1).replace(/-/g, ' ')
   101→  }
   102→}

   610→// Score history chart — real per-render score line over time
   611→// ---------------------------------------------------------------------------
   612→
   613→function ScoreHistoryChart({
   614→  entries,
   615→}: {
   616→  entries: Array<{ timestamp: number; score: number; platform: string }>
   617→}) {
   618→  const W = 300
   619→  const H = 120
   620→  const PAD = 8
   621→  const minScore = 50
   622→  const maxScore = 100
   623→
   624→  if (entries.length < 2) return null
   625→
   626→  const xs = entries.map((_, i) =>
   627→    PAD + (i / (entries.length - 1)) * (W - 2 * PAD),
   628→  )
   629→  const ys = entries.map((e) => {
   630→    const clamped = Math.max(minScore, Math.min(maxScore, e.score))
   631→    return H - PAD - ((clamped - minScore) / (maxScore - minScore)) * (H - 2 * PAD)
   632→  })
   633→
   634→  const points = xs.map((x, i) => `${x.toFixed(2)},${ys[i].toFixed(2)}`).join(' ')
   635→  const areaPoints = `${xs[0].toFixed(2)},${H - PAD} ${points} ${xs[xs.length - 1].toFixed(2)},${H - PAD}`
   636→
   637→  const refLine = (score: number) => {
   638→    const clamped = Math.max(minScore, Math.min(maxScore, score))
   639→    return H - PAD - ((clamped - minScore) / (maxScore - minScore)) * (H - 2 * PAD)
   640→  }
   641→
   642→  return (
   643→    <svg viewBox={`0 0 ${W} ${H}`} className="w-full h-32" preserveAspectRatio="none">
   644→      <defs>
   645→        <linearGradient id="scoreFillReal" x1="0" y1="0" x2="0" y2="1">
   646→          <stop offset="0%" stopColor="#AAFF00" stopOpacity="0.3" />
   647→          <stop offset="100%" stopColor="#AAFF00" stopOpacity="0" />
   648→        </linearGradient>
   649→      </defs>
   650→      <line x1={PAD} x2={W - PAD} y1={refLine(90)} y2={refLine(90)}
   651→        stroke="#AAFF00" strokeOpacity="0.2" strokeDasharray="2 3" />
   652→      <line x1={PAD} x2={W - PAD} y1={refLine(75)} y2={refLine(75)}
   653→        stroke="#F97316" strokeOpacity="0.2" strokeDasharray="2 3" />
   654→      <polygon points={areaPoints} fill="url(#scoreFillReal)" />
   655→      <polyline points={points} fill="none" stroke="#AAFF00" strokeWidth="2" />
   656→      {xs.map((x, i) => (
   657→        <circle key={i} cx={x} cy={ys[i]} r="2" fill="#AAFF00" />
   658→      ))}
   659→    </svg>
   660→  )
   661→}
