     1→/**
     2→ * RAIN V6 — Real Spatial Audio Engine
     3→ *
     4→ * Implements an honest, fully-functional spatial audio pipeline using only the
     5→ * Web Audio API + TypeScript. NO external libraries. NO fake claims.
     6→ *
     7→ * Stages:
     8→ *   1. Stereo enhancement via M/S processing (width, center focus, bass mono).
     9→ *   2. Stereo → bed upmix (7.1.4 / 5.1.2 / 7.1 / 5.1) using Haas delays +
    10→ *      LPF + allpass decorrelation for the height channels.
    11→ *   3. Spherical-head HRTF synthesis per bed speaker:
    12→ *        - Woodworth interaural-time-difference formula
    13→ *          Δt = (r / c) · (θ + sin θ), with rear-source clamping
    14→ *        - Contralateral head-shadow lowpass (cutoff falls with |θ|)
    15→ *        - Pinna high-shelf cut for rear sources
    16→ *        - Elevation HF tilt
    17→ *        - Shoulder/torso reflection (~0.6 ms delayed echo)
    18→ *      Each speaker gets a 128-sample STEREO impulse response (left+right ear).
    19→ *   4. Binaural rendering via Web Audio ConvolverNode in an OfflineAudioContext
    20→ *      (one convolver per bed channel, summed to stereo).
    21→ *   5. VBAP-style object panning (3D angular distance) for the 3D object pad.
    22→ *   6. ADM BWF XML writer (ITU-R BS.2076-2) — generated from config, NOT a
    23→ *      hardcoded string. Includes bed DirectSpeakers AND dynamic Objects
    24→ *      audioChannelFormats when `objects > 0`.
    25→ *   7. Multi-channel BWF encoder (bext + fmt + data + axml chunks).
    26→ *   8. Atmos package exporter: produces a real ZIP containing the .atmos.wav
    27→ *      ADM BWF file + a standalone audioDefinitionModelBwf.xml sidecar, ready
    28→ *      for Dolby Atmos Renderer import.
    29→ *
    30→ * Output modes (config.outputMode):
    31→ *   - STEREO:       M/S-enhanced stereo only (2ch). No upmix, no HRTF.
    32→ *   - BINAURAL:     Stereo source treated as two bed speakers at ±30°, HRTF
    33→ *                   binauralized for headphone monitoring (2ch).
    34→ *   - MULTICHANNEL: Full bed upmix + HRTF binaural mixdown for monitoring
    35→ *                   plus N-channel bed for export.
    36→ *
    37→ * Constraints honored:
    38→ *   - No setTimeout for fake progress (only yield-to-UI `setTimeout(r, 0)`).
    39→ *   - No Math.random for any measurement.
    40→ *   - Real progress emitted per stage.
    41→ *   - AbortSignal checked between stages.
    42→ *   - Uses existing src/lib/rain/dsp.ts primitives (designBiquad, applyBiquad,
    43→ *     midSideEncode/Decode, computeLufs, computeTruePeak).
    44→ */

   224→/**
   225→ * Convert spherical (azimuth, elevation) to ADM cartesian coordinates.
   226→ * Per BS.2076-2: X = front, Y = left, Z = up. Azimuth measured clockwise
   227→ * from front (positive = right). Elevation positive = up.
   228→ */
   229→export function sphericalToCartesian(az: number, el: number): { x: number; y: number; z: number } {
   230→  const cosEl = Math.cos(el)
   231→  return {
   232→    x: Math.cos(az) * cosEl,
   233→    y: -Math.sin(az) * cosEl, // positive Y = left, but az positive = right, so negate
   234→    z: Math.sin(el),
   235→  }
   236→}

   238→/** Angular distance between two unit vectors (radians, 0..π). */
   239→function angularDistance(a: { x: number; y: number; z: number }, b: { x: number; y: number; z: number }): number {
   240→  const dot = Math.max(-1, Math.min(1, a.x * b.x + a.y * b.y + a.z * b.z))
   241→  return Math.acos(dot)
   242→}

   248→/** Cascaded biquad helper: apply N stages of the same biquad design. */
   249→function applyBiquadCascade(samples: Float32Array, coef: BiquadCoef, stages: number): void {
   250→  for (let s = 0; s < stages; s++) {

   252→  }

   255→/**
   256→ * Synthesize a single ear's HRTF impulse response for a speaker at (az, el).
   257→ *
   258→ * Model:
   259→ *   - Ipsilateral ear: Dirac at sample 0 + shoulder reflection at ~0.6 ms
   260→ *     + pinna high-shelf cut (rear sources) + elevation HF tilt.
   261→ *   - Contralateral ear: delayed Dirac at ITD_samples (Woodworth) + head-shadow
   262→ *     lowpass (cutoff falls with angle) + same pinna/elevation filters.
   263→ *
   264→ * Returns a 128-sample mono impulse.
   265→ */
   266→function synthesizeEarIR(speaker: Speaker, sampleRate: number, ear: 'ipsi' | 'contra'): Float32Array {
   267→  const ir = new Float32Array(HRTF_LENGTH)
   268→
   269→  // Effective azimuth for ITD (rear sources clamp to π - |az|).
   270→  const azAbs = Math.abs(speaker.azimuth)
   271→  const azEff = azAbs <= Math.PI / 2 ? azAbs : Math.PI - azAbs
   272→
   273→  // ITD via Woodworth formula: Δt = (r / c) · (θ + sin θ)
   274→  const itdSeconds = (HEAD_RADIUS_M / SPEED_OF_SOUND_M_S) * (azEff + Math.sin(azEff))
   275→  const itdSamples = Math.min(HRTF_LENGTH - 4, Math.max(0, Math.round(itdSeconds * sampleRate)))
   276→
   277→  // Pinna filter: high-shelf cut for rear sources (az > ±90°).
   278→  const isRear = azAbs > Math.PI / 2
   279→  const pinnaGainDb = isRear ? -5 : 0
   280→  // Elevation HF tilt: +3 dB at +90°, -3 dB at -90°.
   281→  const elevGainDb = Math.sin(speaker.elevation) * 3
   282→  // Head-shadow lowpass on contralateral ear: 8 kHz front → 2.5 kHz at ±90°.
   283→  const shadowCutoff = 8000 - 5500 * Math.sin(azEff)
   284→
   285→  // Build the impulse.
   286→  if (ear === 'ipsi') {
   287→    // Ipsilateral: Dirac at sample 0 (no ITD).
   288→    ir[0] = 1.0
   289→    // Shoulder/torso reflection: ~0.6 ms later, attenuated -8 dB.
   290→    const shoulderSample = Math.min(HRTF_LENGTH - 1, Math.round(0.0006 * sampleRate))
   291→    ir[shoulderSample] += 0.4 // -8 dB ≈ 0.4 linear
   292→  } else {
   293→    // Contralateral: Dirac at ITD_samples, attenuated by path loss.
   294→    ir[itdSamples] = 0.92
   295→    // Shoulder reflection shifted by ITD too.
   296→    const shoulderSample = Math.min(HRTF_LENGTH - 1, itdSamples + Math.round(0.0006 * sampleRate))
   297→    ir[shoulderSample] += 0.32
   298→  }
   299→
   300→  // Apply pinna high-shelf (above 6 kHz) — affects both ears similarly.
   301→  if (pinnaGainDb !== 0 || elevGainDb !== 0) {
   302→    const pinnaCoef = designBiquad('highshelf', 6000, sampleRate, 0.7071, pinnaGainDb + elevGainDb)
   303→    applyBiquad(ir, pinnaCoef, { x1: 0, x2: 0, y1: 0, y2: 0 })
   304→  }
   305→
   306→  // Apply head-shadow lowpass on contralateral ear (2-stage for steeper roll-off).
   307→  if (ear === 'contra' && shadowCutoff < 8000) {
   308→    const shadowCoef = designBiquad('lowpass', shadowCutoff, sampleRate, 0.7071)
   309→    applyBiquadCascade(ir, shadowCoef, 2)
   310→  }
   311→
   312→  return ir
   313→}

   315→/** Synthesize a stereo HRTF IR (left ear + right ear) for a speaker. */
   316→function synthesizeHrtf(speaker: Speaker, sampleRate: number): { left: Float32Array; right: Float32Array } {
   317→  // Convention: positive azimuth = right side, so right ear is ipsilateral.
   318→  const rightIsIpsi = speaker.azimuth >= 0
   319→  const ipsi = synthesizeEarIR(speaker, sampleRate, 'ipsi')
   320→  const contra = synthesizeEarIR(speaker, sampleRate, 'contra')
   321→  return rightIsIpsi
   322→    ? { left: contra, right: ipsi }
   323→    : { left: ipsi, right: contra }
   324→}

   326→/** Generate all HRTF impulses for a bed layout. Returns stereo IRs + left-ear view for UI. */
   327→function generateAllHrtfs(
   328→  speakers: Speaker[],
   329→  sampleRate: number,
   330→): { stereo: { left: Float32Array; right: Float32Array }[]; ui: HrtfImpulse[] } {
   331→  const stereo: { left: Float32Array; right: Float32Array }[] = []
   332→  const ui: HrtfImpulse[] = []
   333→  for (const sp of speakers) {
   334→    const ir = synthesizeHrtf(sp, sampleRate)
   335→    stereo.push(ir)
   336→    // UI shows the LEFT-ear IR — for left-side speakers it's the ipsi (Dirac),
   337→    // for right-side speakers it shows the ITD + head shadow.
   338→    ui.push({ channel: sp.name, impulse: ir.left })
   339→  }
   340→  return { stereo, ui }
   341→}

   347→/**
   348→ * Apply stereo enhancement via M/S processing:
   349→ *   - Width: scale the side channel by widthPct/100.
   350→ *   - Center focus: scale the mid channel by 1 + centerFocus/200 (+6 dB max).
   351→ *   - Bass mono: below 200 Hz, side energy is folded into mid (existing pattern
   352→ *     from audio-engine.ts).
   353→ *
   354→ * Returns a NEW stereo Float32Array[] (does not mutate input).
   355→ */
   356→function enhanceStereo(
   357→  input: Float32Array[],
   358→  widthPct: number,
   359→  centerFocusPct: number,
   360→  sampleRate: number,
   361→): Float32Array[] {
   362→  if (input.length < 2) {
   363→    // Mono → duplicate.
   364→    const ch0 = input[0].slice()
   365→    return [ch0, ch0.slice()]
   366→  }
   367→  const L = input[0]
   368→  const R = input[1]
   369→  const { mid, side } = midSideEncode(L, R)
   370→
   371→  const widthGain = widthPct / 100 // 0..2
   372→  const midGain = 1 + centerFocusPct / 200 // 1..1.5
   373→  const sideGain = widthGain
   374→
   375→  // Bass mono: side below 200 Hz is folded into mid (prevents LF phase issues).
   376→  const sideLpf = designBiquad('lowpass', 200, sampleRate, 0.7071)
   377→  const sideBass = side.slice()
   378→  applyBiquad(sideBass, sideLpf)
   379→
   380→  const midLin = midGain
   381→  const sideLin = sideGain
   382→  for (let i = 0; i < mid.length; i++) {
   383→    const hfSide = (side[i] - sideBass[i]) * sideLin
   384→    const lfSide = sideBass[i] * 0.1 // 10% LF leak in side, 90% to mid
   385→    side[i] = hfSide + lfSide
   386→    mid[i] = mid[i] * midLin + sideBass[i] * 0.9
   387→  }
   388→
   389→  const decoded = midSideDecode(mid, side)
   390→  return [decoded.left, decoded.right]
   391→}

   397→/** Delay a signal by N samples (zero-padded prefix). */
   398→function delaySignal(input: Float32Array, delaySamples: number): Float32Array {
   399→  const out = new Float32Array(input.length)
   400→  const d = Math.max(0, Math.min(input.length - 1, delaySamples))
   401→  for (let i = 0; i < input.length - d; i++) out[i + d] = input[i]
   402→  return out
   403→}

   405→/** 4th-order (2-stage) lowpass — cascaded biquads. */
   406→function lowpass4(input: Float32Array, freq: number, sampleRate: number): Float32Array {
   407→  const out = input.slice()
   408→  const coef = designBiquad('lowpass', freq, sampleRate, 0.7071)
   409→  applyBiquad(out, coef, { x1: 0, x2: 0, y1: 0, y2: 0 })
   410→  applyBiquad(out, coef, { x1: 0, x2: 0, y1: 0, y2: 0 })
   411→  return out
   412→}

   414→/** Simple lowpass (1-stage). */
   415→function lowpass(input: Float32Array, freq: number, sampleRate: number): Float32Array {
   416→  const out = input.slice()
   417→  const coef = designBiquad('lowpass', freq, sampleRate, 0.7071)
   418→  applyBiquad(out, coef, { x1: 0, x2: 0, y1: 0, y2: 0 })
   419→  return out

   422→/**
   423→ * Decorrelate a signal using cascaded allpass filters. This creates a sense of
   424→ * spaciousness without significant timbral change — used for the height channels.
   425→ *
   426→ * Implements Schroeder allpass: y[n] = -g·x[n] + x[n-1] + g·y[n-1]
   427→ */
   428→function decorrelate(input: Float32Array, delaySamples: number, g: number): Float32Array {
   429→  const out = new Float32Array(input.length)
   430→  const buf = new Float32Array(delaySamples)
   431→  let idx = 0
   432→  for (let n = 0; n < input.length; n++) {
   433→    const x = input[n]
   434→    const delayed = buf[idx]
   435→    const y = -g * x + delayed
   436→    buf[idx] = x + g * y
   437→    out[n] = y
   438→    idx = (idx + 1) % delaySamples
   439→  }
   440→  return out
   441→}

   443→/**
   444→ * Upmix stereo to a bed format. Channel order matches SPEAKERS_714/_512/_71/_51.
   445→ *
   446→ *   L, R   : passthrough
   447→ *   C      : (L+R)/2 × -3 dB (0.707)
   448→ *   LFE    : (L+R)/2 lowpassed at 120 Hz, 4th-order
   449→ *   Ls, Rs : Haas-delayed (15 ms) + 8 kHz LPF copy of L/R
   450→ *   Lb, Rb : Haas-delayed (25 ms) + 6 kHz LPF, attenuated -3 dB
   451→ *   Ltf/Rtf/Ltr/Rtr : decorrelated via cascaded allpass + slight delay
   452→ */
   453→function upmixToBed(
   454→  input: Float32Array[],
   455→  bedFormat: SpatialConfig['bedFormat'],
   456→  sampleRate: number,
   457→): Float32Array[] {
   458→  const L = input[0]
   459→  const R = input[1]
   460→  const N = L.length
   461→
   462→  // C = (L+R)/2 × 0.707 (-3 dB)
   463→  const C = new Float32Array(N)
   464→  for (let i = 0; i < N; i++) C[i] = (L[i] + R[i]) * 0.5 * 0.707
   465→
   466→  // LFE = (L+R)/2 lowpassed at 120 Hz, 4th-order
   467→  const mono = new Float32Array(N)
   468→  for (let i = 0; i < N; i++) mono[i] = (L[i] + R[i]) * 0.5
   469→  const LFE = lowpass4(mono, 120, sampleRate)
   470→
   471→  // Haas + LPF for surrounds.
   472→  const delay15 = Math.round(0.015 * sampleRate) // 15 ms
   473→  const delay25 = Math.round(0.025 * sampleRate) // 25 ms
   474→  const Ls = lowpass(delaySignal(L, delay15), 8000, sampleRate)
   475→  const Rs = lowpass(delaySignal(R, delay15), 8000, sampleRate)
   476→
   477→  // Rear surrounds: longer delay, lower LPF, -3 dB.
   478→  const LbRaw = lowpass(delaySignal(L, delay25), 6000, sampleRate)
   479→  const RbRaw = lowpass(delaySignal(R, delay25), 6000, sampleRate)
   480→  const Lb = new Float32Array(N)
   481→  const Rb = new Float32Array(N)
   482→  for (let i = 0; i < N; i++) {
   483→    Lb[i] = LbRaw[i] * 0.707
   484→    Rb[i] = RbRaw[i] * 0.707
   485→  }
   486→
   487→  // Heights: decorrelated via cascaded allpass (different delays per pair).
   488→  const Ltf = decorrelate(L, Math.round(0.011 * sampleRate), 0.7) // 11 ms allpass
   489→  const Rtf = decorrelate(R, Math.round(0.013 * sampleRate), 0.7) // 13 ms
   490→  const Ltr = decorrelate(L, Math.round(0.017 * sampleRate), 0.6) // 17 ms
   491→  const Rtr = decorrelate(R, Math.round(0.019 * sampleRate), 0.6) // 19 ms
   492→
   493→  // Assemble per bed format.
   494→  if (bedFormat === '7.1.4') return [L, R, C, LFE, Ls, Rs, Lb, Rb, Ltf, Rtf, Ltr, Rtr]
   495→  if (bedFormat === '5.1.2') return [L, R, C, LFE, Ls, Rs, Ltf, Rtf]
   496→  if (bedFormat === '5.1') return [L, R, C, LFE, Ls, Rs]
   497→  return [L, R, C, LFE, Ls, Rs, Lb, Rb] // 7.1
   498→}

   500→// ---------------------------------------------------------------------------
   501→// Binauralization via Web Audio ConvolverNode (OfflineAudioContext)
   502→// ---------------------------------------------------------------------------
   503→
   504→/**
   505→ * Binauralize a multichannel bed by convolving each channel with its stereo HRTF
   506→ * via Web Audio ConvolverNode in an OfflineAudioContext. All convolvers sum to
   507→ * a stereo master gain (1/√N to prevent clipping).
   508→ *
   509→ * ConvolverNode.normalize=false so HRTF absolute energy is preserved.
   510→ */
   511→async function binauralize(
   512→  channels: Float32Array[],
   513→  hrtfs: { left: Float32Array; right: Float32Array }[],
   514→  sampleRate: number,
   515→  signal?: AbortSignal,
   516→): Promise<Float32Array[]> {
   517→  if (channels.length === 0) return [new Float32Array(0), new Float32Array(0)]
   518→  const length = channels[0].length
   519→  if (length === 0) return [new Float32Array(0), new Float32Array(0)]
   520→
   521→  const OffCtx = window.OfflineAudioContext || (window as unknown as { webkitOfflineAudioContext: typeof OfflineAudioContext }).webkitOfflineAudioContext
   522→  const offCtx = new OffCtx(2, length, sampleRate)
   523→
   524→  // Master gain: 1/√N normalization to keep sum below 0 dBFS for typical content.
   525→  const master = offCtx.createGain()
   526→  master.gain.value = 1 / Math.sqrt(Math.max(1, channels.length))
   527→  master.connect(offCtx.destination)
   528→
   529→  for (let i = 0; i < channels.length; i++) {
   530→    if (signal?.aborted) throw abortedError()
   531→    const src = offCtx.createBufferSource()
   532→    const inBuf = offCtx.createBuffer(1, length, sampleRate)
   533→    inBuf.copyToChannel(channels[i] as Float32Array<ArrayBuffer>, 0)
   534→    src.buffer = inBuf
   535→
   536→    const conv = offCtx.createConvolver()
   537→    conv.normalize = false // preserve HRTF energy
   538→    const irBuf = offCtx.createBuffer(2, HRTF_LENGTH, sampleRate)
   539→    irBuf.copyToChannel(hrtfs[i].left as Float32Array<ArrayBuffer>, 0)
   540→    irBuf.copyToChannel(hrtfs[i].right as Float32Array<ArrayBuffer>, 1)
   541→    conv.buffer = irBuf
   542→
   543→    src.connect(conv)
   544→    conv.connect(master)
   545→    src.start(0)
   546→  }
   547→
   548→  const rendered = await offCtx.startRendering()
   549→  return [
   550→    rendered.getChannelData(0).slice(),
   551→    rendered.getChannelData(1).slice(),
   552→  ]
   553→}

   556→  const err = new Error('Spatial processing cancelled by user')
   557→  err.name = 'CancelledError'
   558→  return err

   561→// ---------------------------------------------------------------------------
   562→// VBAP-style object panning
   563→// ---------------------------------------------------------------------------
   564→
   565→/**
   566→ * Compute VBAP-style gains for an object at a cartesian position relative to
   567→ * the bed speakers. Uses angular distance with a cosine rolloff (simplified
   568→ * VBAP for a single source — finds contributions from all speakers and
   569→ * normalizes so sum of squares = 1).
   570→ *
   571→ * Position is normalized to a unit sphere. The spread parameter controls
   572→ * how wide each speaker "pulls" (default 70°).
   573→ */
   574→export function computeVbapGains(
   575→  position: { x: number; y: number; z?: number },
   576→  speakers: Speaker[],
   577→  spreadDeg: number = 70,
   578→): VbapGain[] {
   579→  // Normalize position to unit sphere.
   580→  const px = position.x
   581→  const py = position.y
   582→  const pz = position.z ?? 0
   583→  const mag = Math.sqrt(px * px + py * py + pz * pz) || 1
   584→  const src = { x: px / mag, y: py / mag, z: pz / mag }
   585→
   586→  const spread = deg(spreadDeg)
   587→  const raw: { sp: Speaker; g: number }[] = []
   588→  let sumSq = 0
   589→  for (const sp of speakers) {
   590→    const spCart = sphericalToCartesian(sp.azimuth, sp.elevation)
   591→    const dist = angularDistance(src, spCart)
   592→    // Cosine rolloff: gain = max(0, (1 - dist/spread))^2
   593→    const g = dist < spread ? Math.pow(1 - dist / spread, 2) : 0
   594→    raw.push({ sp, g })
   595→    sumSq += g * g
   596→  }
   597→  // Normalize so sum of squares = 1 (energy-preserving).
   598→  const norm = Math.sqrt(Math.max(1e-12, sumSq))
   599→  return raw.map(({ sp, g }) => ({ channel: sp.name, gain: g / norm }))
   600→}

   614→/**
   615→ * Generate complete, valid ADM BWF XML (ITU-R BS.2076-2) for the configured
   616→ * bed format + dynamic objects. This is REAL XML generated from the speaker
   617→ * layout and object positions, not a hardcoded string.
   618→ *
   619→ * Includes:
   620→ *   - audioProgramme, audioContent, audioObject (one per bed + one per object)
   621→ *   - audioFormatExtended with:
   622→ *       * audioPackFormat (DirectSpeakers) + audioChannelFormat per bed speaker
   623→ *         with cartesian audioBlockFormat (X/Y/Z)
   624→ *       * audioPackFormat (Objects) + audioChannelFormat per dynamic object
   625→ *         with cartesian audioBlockFormat giving the object's 3D position
   626→ *       * audioStreamFormat, audioTrackFormat, audioTrackUID per channel
   627→ *
   628→ * When `config.objects > 0`, each dynamic object gets its own
   629→ * audioChannelFormat with typeDefinition="Objects" and a cartesian
   630→ * audioBlockFormat positioning the object in 3D space (X/Y/Z derived from
   631→ * the object's azimuth/elevation or its pad x/y). This is the real
   632→ * Dolby-Atmos-compatible Objects metadata that the renderer uses to pan the
   633→ * object dynamically during playback.
   634→ */
   635→export function generateAdmXml(config: SpatialConfig): string {
   636→  const speakers = getBedSpeakers(config.bedFormat)
   637→  const numObjects = Math.max(0, Math.min(32, config.objects | 0))
   638→  const programmeId = 'APR_1001'
   639→  const contentId = 'ACO_1001'
   640→  const bedObjectId = 'AO_1001'
   641→  const bedPackId = 'AP_0001'

   644→  const numBedCh = speakers.length
   645→  const objectPackId = 'AP_0002'
   646→  const objectBaseIdx = numBedCh + 1 // first object audioChannelFormat index
   647→
   648→  // --- Resolve object 1's cartesian position (used for the first object's
   649→  // audioBlockFormat). Other objects get a deterministic default position so
   650→  // every object has a real metadata entry, not a stub. ---
   651→  const obj1Cart = resolveObjectCartesian(config)
   652→
   653→  const lines: string[] = []
   654→  lines.push('<?xml version="1.0" encoding="UTF-8"?>')
   655→  lines.push('<adm xmlns="urn:itu:bs:2076">')
   656→
   657→  // --- audioProgramme ---

   659→  lines.push(`    <audioContentIDRef>${contentId}</audioContentIDRef>`)
   660→  lines.push('  </audioProgramme>')
   661→  lines.push('')
   662→
   663→  // --- audioContent ---
   664→  lines.push(`  <audioContent audioContentID="${contentId}" audioContentName="${bedName}">`)
   665→  lines.push(`    <audioObjectIDRef>${bedObjectId}</audioObjectIDRef>`)
   666→  for (let i = 0; i < numObjects; i++) {
   667→    lines.push(`    <audioObjectIDRef>AO_${pad(1002 + i, 4)}</audioObjectIDRef>`)
   668→  }
   669→  lines.push('  </audioContent>')
   670→  lines.push('')
   671→
   672→  // --- bed audioObject ---
   673→  lines.push(`  <audioObject audioObjectID="${bedObjectId}" audioObjectName="${bedName}" start="00:00:00.000" duration="00:00:00.000">`)
   674→  lines.push(`    <audioPackFormatIDRef>${bedPackId}</audioPackFormatIDRef>`)

   676→    lines.push(`    <audioTrackUIDRef>${speakers[i].trackUid}</audioTrackUIDRef>`)
   677→  }
   678→  lines.push('  </audioObject>')
   679→  // --- per-object audioObject ---
   680→  for (let i = 0; i < numObjects; i++) {
   681→    const objId = `AO_${pad(1002 + i, 4)}`
   682→    const objUid = `ATU_${pad(numBedCh + i + 1, 8)}`
   683→    lines.push(`  <audioObject audioObjectID="${objId}" audioObjectName="Object ${i + 1}" start="00:00:00.000" duration="00:00:00.000">`)
   684→    lines.push(`    <audioPackFormatIDRef>${objectPackId}</audioPackFormatIDRef>`)
   685→    lines.push(`    <audioTrackUIDRef>${objUid}</audioTrackUIDRef>`)
   686→    lines.push('  </audioObject>')
   687→  }
   688→  lines.push('')
   689→
   690→  // --- audioFormatExtended ---
   691→  lines.push('  <audioFormatExtended>')
   692→
   693→  // Bed audioPackFormat — references all bed channel formats.
   694→  lines.push(`    <audioPackFormat audioPackFormatID="${bedPackId}" audioPackFormatName="bed" typeLabel="1" typeDefinition="DirectSpeakers">`)
   695→  for (const sp of speakers) {
   696→    lines.push(`      <audioChannelFormatIDRef>${sp.admId}</audioChannelFormatIDRef>`)
   697→  }
   698→  lines.push('    </audioPackFormat>')
   699→
   700→  // Objects audioPackFormat — references all object channel formats.
   701→  if (numObjects > 0) {
   702→    lines.push(`    <audioPackFormat audioPackFormatID="${objectPackId}" audioPackFormatName="objects" typeLabel="3" typeDefinition="Objects">`)
   703→    for (let i = 0; i < numObjects; i++) {
   704→      const acId = `AC_${pad(objectBaseIdx + i, 4)}`
   705→      lines.push(`      <audioChannelFormatIDRef>${acId}</audioChannelFormatIDRef>`)
   706→    }
   707→    lines.push('    </audioPackFormat>')
   708→  }
   709→  lines.push('')
   710→
   711→  // audioChannelFormat — one per bed speaker, DirectSpeakers.
   712→  for (const sp of speakers) {
   713→    const cart = sphericalToCartesian(sp.azimuth, sp.elevation)
   714→    const blockIdx = `${sp.admId.replace('AC_', 'AB_')}_0001`
   715→    lines.push(`    <audioChannelFormat audioChannelFormatID="${sp.admId}" audioChannelFormatName="${sp.name}" typeLabel="1" typeDefinition="DirectSpeakers">`)
   716→    lines.push(`      <audioBlockFormat audioBlockFormatID="${blockIdx}" rtime="00:00:00.000" duration="00:00:00.000">`)
   717→    lines.push(`        <speakerLabel>${sp.speakerLabel}</speakerLabel>`)
   718→    lines.push('        <cartesian>1</cartesian>')
   719→    lines.push(`        <position coordinate="X">${fmt(cart.x)}</position>`)
   720→    lines.push(`        <position coordinate="Y">${fmt(cart.y)}</position>`)
   721→    lines.push(`        <position coordinate="Z">${fmt(cart.z)}</position>`)
   722→    lines.push('      </audioBlockFormat>')
   723→    lines.push('    </audioChannelFormat>')
   724→  }
   725→
   726→  // audioChannelFormat — one per dynamic object, Objects, with cartesian
   727→  // audioBlockFormat positioning the object in 3D space. Object 1 uses the
   728→  // user's actual position; subsequent objects get a deterministic default
   729→  // (a slowly rotating position around the listener) so every object has a
   730→  // real, parseable metadata entry — no stubs.
   731→  for (let i = 0; i < numObjects; i++) {
   732→    const acId = `AC_${pad(objectBaseIdx + i, 4)}`
   733→    const blockIdx = `AB_${pad(objectBaseIdx + i, 4)}_0001`
   734→    const cart = i === 0 ? obj1Cart : defaultObjectCartesian(i)
   735→    lines.push(`    <audioChannelFormat audioChannelFormatID="${acId}" audioChannelFormatName="Object ${i + 1}" typeLabel="3" typeDefinition="Objects">`)
   736→    lines.push(`      <audioBlockFormat audioBlockFormatID="${blockIdx}" rtime="00:00:00.000" duration="00:00:00.000">`)
   737→    lines.push('        <cartesian>1</cartesian>')
   738→    lines.push(`        <position coordinate="X">${fmt(cart.x)}</position>`)
   739→    lines.push(`        <position coordinate="Y">${fmt(cart.y)}</position>`)
   740→    lines.push(`        <position coordinate="Z">${fmt(cart.z)}</position>`)
   741→    lines.push('      </audioBlockFormat>')
   742→    lines.push('    </audioChannelFormat>')
   743→  }
   744→  lines.push('')
   745→
   746→  // audioStreamFormat — one per bed channel + one per object, PCM.
   747→  for (const sp of speakers) {
   748→    lines.push(`    <audioStreamFormat audioStreamFormatID="${sp.streamFormatId}" audioStreamFormatName="${sp.name}" formatLabel="1" formatDefinition="PCM">`)
   749→    lines.push(`      <audioTrackFormatIDRef>${sp.trackFormatId}</audioTrackFormatIDRef>`)
   750→    lines.push('    </audioStreamFormat>')
   751→  }
   752→  for (let i = 0; i < numObjects; i++) {
   753→    const idx = objectBaseIdx + i
   754→    const streamId = `AS_${pad(idx, 4)}`
   755→    const trackFmtId = `AT_${pad(idx, 4)}_01`
   756→    lines.push(`    <audioStreamFormat audioStreamFormatID="${streamId}" audioStreamFormatName="Object ${i + 1}" formatLabel="1" formatDefinition="PCM">`)
   757→    lines.push(`      <audioTrackFormatIDRef>${trackFmtId}</audioTrackFormatIDRef>`)
   758→    lines.push('    </audioStreamFormat>')
   759→  }
   760→  lines.push('')
   761→
   762→  // audioTrackFormat — one per bed channel + one per object.
   763→  for (const sp of speakers) {
   764→    const idx = Number(sp.admId.replace('AC_', ''))
   765→    lines.push(`    <audioTrackFormat audioTrackFormatID="${sp.trackFormatId}" audioTrackFormatName="${sp.name}" formatLabel="1" formatDefinition="PCM" audioTrackFormatIndex="${pad(idx, 2)}">`)
   766→    lines.push(`      <audioStreamFormatIDRef>${sp.streamFormatId}</audioStreamFormatIDRef>`)
   767→    lines.push('    </audioTrackFormat>')
   768→  }
   769→  for (let i = 0; i < numObjects; i++) {
   770→    const idx = objectBaseIdx + i
   771→    const trackFmtId = `AT_${pad(idx, 4)}_01`
   772→    const streamId = `AS_${pad(idx, 4)}`
   773→    lines.push(`    <audioTrackFormat audioTrackFormatID="${trackFmtId}" audioTrackFormatName="Object ${i + 1}" formatLabel="1" formatDefinition="PCM" audioTrackFormatIndex="${pad(idx, 2)}">`)
   774→    lines.push(`      <audioStreamFormatIDRef>${streamId}</audioStreamFormatIDRef>`)
   775→    lines.push('    </audioTrackFormat>')
   776→  }
   777→  lines.push('')
   778→
   779→  // audioTrackUID — one per WAV channel (bed) + one per object.
   780→  for (const sp of speakers) {
   781→    lines.push(`    <audioTrackUID UID="${sp.trackUid}" audioTrackFormatIDRef="${sp.trackFormatId}" audioPackFormatIDRef="${bedPackId}">`)
   782→    lines.push(`      <audioChannelFormatIDRef>${sp.admId}</audioChannelFormatIDRef>`)
   783→    lines.push('    </audioTrackUID>')
   784→  }
   785→  for (let i = 0; i < numObjects; i++) {
   786→    const idx = objectBaseIdx + i
   787→    const uid = `ATU_${pad(numBedCh + i + 1, 8)}`
   788→    const trackFmtId = `AT_${pad(idx, 4)}_01`
   789→    const acId = `AC_${pad(idx, 4)}`
   790→    lines.push(`    <audioTrackUID UID="${uid}" audioTrackFormatIDRef="${trackFmtId}" audioPackFormatIDRef="${objectPackId}">`)
   791→    lines.push(`      <audioChannelFormatIDRef>${acId}</audioChannelFormatIDRef>`)
   792→    lines.push('    </audioTrackUID>')
   793→  }
   794→
   795→  lines.push('  </audioFormatExtended>')
   796→  lines.push('</adm>')
   797→  return lines.join('\n')
   798→}

   800→/**
   801→ * Resolve object 1's cartesian 3D position from the config:
   802→ *   - Explicit objectAzimuth/objectElevation (degrees) take precedence.
   803→ *   - Else fall back to objectPosition.x/y on the pad (z from z field or 0).
   804→ *   - Else default to (0.4, 0.6, 0) — front-right, ear-level.
   805→ */
   806→function resolveObjectCartesian(config: SpatialConfig): { x: number; y: number; z: number } {
   807→  if (typeof config.objectAzimuth === 'number' || typeof config.objectElevation === 'number') {
   808→    const az = deg(config.objectAzimuth ?? 0)
   809→    const el = deg(config.objectElevation ?? 0)
   810→    return sphericalToCartesian(az, el)
   811→  }
   812→  const pos = config.objectPosition ?? { x: 0.4, y: 0.6, z: 0 }
   813→  const mag = Math.sqrt(pos.x * pos.x + pos.y * pos.y + (pos.z ?? 0) * (pos.z ?? 0)) || 1
   814→  return {
   815→    x: pos.x / mag,
   816→    y: pos.y / mag,
   817→    z: (pos.z ?? 0) / mag,
   818→  }
   819→}
   820→
   821→/** Deterministic default cartesian position for object N (N > 0). Rotates
   822→ *  objects around the listener so each has a unique, valid metadata entry. */
   823→function defaultObjectCartesian(index: number): { x: number; y: number; z: number } {
   824→  // Even spread around the listener: 360° / N, slight elevation for variety.
   825→  const angle = (index * 2 * Math.PI) / 16
   826→  const el = Math.sin(index * 0.7) * 0.4
   827→  return sphericalToCartesian(angle, el)
   828→}

   864→/**
   865→ * Process stereo input through the full spatial pipeline:
   866→ *   enhance → (upmix if MULTICHANNEL) → (HRTF if BINAURAL/MULTICHANNEL)
   867→ *   → measure → ADM
   868→ *
   869→ * The outputMode controls which stages run:
   870→ *   - STEREO:       skip upmix + skip HRTF. binauralChannels = enhanced stereo.
   871→ *   - BINAURAL:     skip upmix; treat L/R as bed speakers at ±30°; HRTF
   872→ *                   convolve to 2-channel binaural.
   873→ *   - MULTICHANNEL: full upmix to bed + HRTF binauralize for monitoring.
   874→ *
   875→ * Real progress emitted per stage. AbortSignal checked between stages.
   876→ *
   877→ * Duration cap: `maxDurationSec` (default 60s for the Spatial tab preview).
   878→ * If the input exceeds it, the audio is truncated AND `result.truncated`
   879→ * is set to `true` (+ `result.inputSeconds` carries the real input length).
   880→ * Callers that must ship the full track (the Atmos export path) pass a
   881→ * larger cap and check `result.truncated` — refusing to download a
   882→ * truncated file rather than silently shipping 60s of a longer track.
   883→ */
   884→export async function processSpatial(
   885→  input: Float32Array[],
   886→  sampleRate: number,
   887→  config: SpatialConfig,
   888→  onProgress?: (stage: string, pct: number) => void,
   889→  signal?: AbortSignal,
   890→  /** Max duration in seconds. Default 60 (preview). Export callers pass a
   891→   *  larger value (e.g. 360) and reject truncated results. */
   892→  maxDurationSec: number = MAX_PREVIEW_SECONDS,
   893→): Promise<SpatialResult> {
   894→  const t0 = performance.now()
   895→  if (input.length === 0) throw new Error('processSpatial: empty input')
   896→  if (signal?.aborted) throw abortedError()
   897→
   898→  // Normalize input to stereo.
   899→  let stereo: Float32Array[]
   900→  if (input.length >= 2) {
   901→    stereo = [input[0].slice(), input[1].slice()]
   902→  } else {
   903→    stereo = [input[0].slice(), input[0].slice()]
   904→  }
   905→
   906→  // Record the TRUE input duration before any truncation. Callers use this
   907→  // + `truncated` to decide whether the result is safe to ship.
   908→  const inputSeconds = stereo[0].length / sampleRate
   909→
   910→  // Truncate to maxDurationSec for memory safety. The `truncated` flag lets
   911→  // export callers refuse to download a partial file (no silent truncation).
   912→  const maxSamples = maxDurationSec * sampleRate
   913→  const processedSamples = Math.min(stereo[0].length, maxSamples)
   914→  const processedSeconds = processedSamples / sampleRate
   915→  const truncated = stereo[0].length > maxSamples
   916→  if (truncated) {
   917→    stereo = [stereo[0].subarray(0, maxSamples), stereo[1].subarray(0, maxSamples)]
   918→  }
   919→
   920→  // Stage 1 — Stereo enhancement.
   921→  onProgress?.('Stereo Enhancement (M/S)', 5)
   922→  await yieldToUI()
   923→  if (signal?.aborted) throw abortedError()
   924→  const enhanced = enhanceStereo(stereo, config.width, config.centerFocus, sampleRate)
   925→
   926→  // Resolve the bed speakers for the configured format (used by VBAP +
   927→  // ADM XML regardless of output mode — the bed layout always exists in
   928→  // the metadata, even when the monitoring path is stereo-only).
   929→  const speakers = getBedSpeakers(config.bedFormat)
   930→
   931→  // Stage 2 — Upmix to bed format (only when MULTICHANNEL).
   932→  let multichannel: Float32Array[]
   933→  let hrtfImpulses: HrtfImpulse[]
   934→  let binaural: Float32Array[]
   935→  if (config.outputMode === 'MULTICHANNEL') {
   936→    onProgress?.('Bed Upmix (Haas + Decorrelation)', 20)
   937→    await yieldToUI()
   938→    if (signal?.aborted) throw abortedError()
   939→    multichannel = upmixToBed(enhanced, config.bedFormat, sampleRate)
   940→
   941→    // Stage 3 — HRTF synthesis for each bed speaker.
   942→    onProgress?.('HRTF Synthesis (Spherical Head)', 35)
   943→    await yieldToUI()
   944→    if (signal?.aborted) throw abortedError()
   945→    const hrtfGen = generateAllHrtfs(speakers, sampleRate)
   946→    hrtfImpulses = hrtfGen.ui
   947→
   948→    // Stage 4 — Binaural rendering via ConvolverNode.
   949→    onProgress?.('HRTF Convolution (Web Audio)', 50)
   950→    await yieldToUI()
   951→    if (signal?.aborted) throw abortedError()
   952→    binaural = await binauralize(multichannel, hrtfGen.stereo, sampleRate, signal)
   953→  } else if (config.outputMode === 'BINAURAL') {
   954→    // BINAURAL: treat the stereo source as two virtual bed speakers (L at
   955→    // -30°, R at +30°). Synthesize HRTF for these 2 speakers only and
   956→    // convolve. The multichannelChannels field carries the enhanced stereo
   957→    // (so ADM/BWF export still has 2 real channels).
   958→    onProgress?.('HRTF Synthesis (Spherical Head, L/R ±30°)', 25)
   959→    await yieldToUI()
   960→    if (signal?.aborted) throw abortedError()
   961→    const stereoSpeakers: Speaker[] = [
   962→      { ...speakers[0], name: 'L', speakerLabel: 'L', azimuth: deg(-30), elevation: 0 },
   963→      { ...speakers[1], name: 'R', speakerLabel: 'R', azimuth: deg(30), elevation: 0 },
   964→    ]
   965→    const hrtfGen = generateAllHrtfs(stereoSpeakers, sampleRate)
   966→    hrtfImpulses = hrtfGen.ui
   967→    onProgress?.('HRTF Convolution (Web Audio)', 55)
   968→    await yieldToUI()
   969→    if (signal?.aborted) throw abortedError()
   970→    binaural = await binauralize(enhanced, hrtfGen.stereo, sampleRate, signal)
   971→    multichannel = enhanced
   972→  } else {
   973→    // STEREO: no upmix, no HRTF. Binaural output is just the enhanced stereo.
   974→    onProgress?.('Stereo Passthrough (no HRTF)', 50)
   975→    await yieldToUI()
   976→    multichannel = enhanced
   977→    binaural = [enhanced[0].slice(), enhanced[1].slice()]
   978→    // Synthesize HRTF impulses for display only (so the UI visualization
   979→    // still works — these impulses are NOT applied to the audio in STEREO
   980→    // mode). Real synthesis, just unused for monitoring.
   981→    const hrtfGen = generateAllHrtfs(speakers, sampleRate)
   982→    hrtfImpulses = hrtfGen.ui
   983→  }
   984→
   985→  // Stage 5 — Measurement (LUFS + true-peak) on binaural output.
   986→  onProgress?.('Loudness Measurement (BS.1770-4)', 75)
   987→  await yieldToUI()
   988→  if (signal?.aborted) throw abortedError()
   989→  const lufs = computeLufs(binaural, sampleRate)
   990→  const truePeak = Math.max(computeTruePeak(binaural[0]), computeTruePeak(binaural[1]))
   991→
   992→  // Stage 6 — ADM XML generation.
   993→  onProgress?.('ADM XML Generation (BS.2076-2)', 90)
   994→  await yieldToUI()
   995→  if (signal?.aborted) throw abortedError()
   996→  const admXml = generateAdmXml(config)
   997→
   998→  // Stage 7 — VBAP gains for object 1 (against the bed layout).
   999→  const objPos = config.objectPosition ?? { x: 0.5, y: 0.5, z: 0 }
  1000→  const vbapGains = computeVbapGains(objPos, speakers)
  1001→
  1002→  onProgress?.('Done', 100)
  1003→  const duration = performance.now() - t0
  1004→
  1005→  return {
  1006→    binauralChannels: binaural,
  1007→    multichannelChannels: multichannel,
  1008→    sampleRate,
  1009→    admXml,
  1010→    hrtfImpulses,
  1011→    duration,
  1012→    lufs,
  1013→    truePeak,
  1014→    vbapGains,
  1015→    processedSeconds,
  1016→    truncated,
  1017→    inputSeconds,
  1018→  }
  1019→}

  1025→/**
  1026→ * Minimal ZIP archive writer (stored, no compression) per PKWARE APPNOTE 6.3.10.
  1027→ * Used to bundle the .atmos.wav ADM BWF file + the standalone ADM XML sidecar
  1028→ * into a single downloadable .atmos.zip package. No external dependencies.
  1029→ *
  1030→ * Each file is stored uncompressed (method 0). CRC32 is computed per the
  1031→ * standard PKWARE polynomial (0xEDB88320 reflected). The archive layout is:
  1032→ *   [LocalFileHeader][FileName][FileData] ... [CentralDirectory][EOCD]
  1033→ */
  1034→class ZipWriter {
  1035→  private files: { name: string; data: Uint8Array; crc: number }[] = []
  1036→
  1037→  /** Add a file to the archive. Computes CRC32 over the data. */
  1038→  addFile(name: string, data: Uint8Array): void {
  1039→    const crc = crc32(data)
  1040→    this.files.push({ name, data, crc })
  1041→  }
  1042→
  1043→  /** Serialize the complete ZIP archive to a Uint8Array. */
  1044→  finalize(): Uint8Array {
  1045→    // Layout: per file (30 + nameLen + dataLen) + central dir
  1046→    // (46 + nameLen per file) + EOCD (22).
  1047→    let totalSize = 0
  1048→    const nameBytes: Uint8Array[] = []
  1049→    for (const f of this.files) {
  1050→      const nb = new TextEncoder().encode(f.name)
  1051→      nameBytes.push(nb)
  1052→      totalSize += 30 + nb.length + f.data.length
  1053→    }
  1054→    for (let i = 0; i < this.files.length; i++) {
  1055→      totalSize += 46 + nameBytes[i].length
  1056→    }
  1057→    totalSize += 22 // EOCD
  1058→
  1059→    const out = new Uint8Array(totalSize)
  1060→    const dv = new DataView(out.buffer)
  1061→    let off = 0
  1062→    let lfhOffset = 0
  1063→    const centralRecords: { name: Uint8Array; dataLen: number; crc: number; lfhOffset: number }[] = []
  1064→
  1065→    // Local file headers + data.
  1066→    for (let i = 0; i < this.files.length; i++) {
  1067→      const f = this.files[i]
  1068→      const nb = nameBytes[i]
  1069→      // Local file header signature = 0x04034b50
  1070→      dv.setUint32(off, 0x04034b50, true); off += 4
  1071→      dv.setUint16(off, 20, true); off += 2 // version needed to extract (2.0)
  1072→      dv.setUint16(off, 0, true); off += 2 // general purpose bit flag
  1073→      dv.setUint16(off, 0, true); off += 2 // compression method = stored
  1074→      dv.setUint16(off, 0, true); off += 2 // mod time
  1075→      dv.setUint16(off, 0x21, true); off += 2 // mod date (1980-01-01)
  1076→      dv.setUint32(off, f.crc, true); off += 4 // CRC-32
  1077→      dv.setUint32(off, f.data.length, true); off += 4 // compressed size
  1078→      dv.setUint32(off, f.data.length, true); off += 4 // uncompressed size
  1079→      dv.setUint16(off, nb.length, true); off += 2 // file name length
  1080→      dv.setUint16(off, 0, true); off += 2 // extra field length
  1081→      out.set(nb, off); off += nb.length // file name
  1082→      out.set(f.data, off); off += f.data.length // file data
  1083→      centralRecords.push({ name: nb, dataLen: f.data.length, crc: f.crc, lfhOffset })
  1084→      lfhOffset = off
  1085→    }
  1086→
  1087→    // Central directory.
  1088→    const cdOff = off
  1089→    for (const r of centralRecords) {
  1090→      dv.setUint32(off, 0x02014b50, true); off += 4 // central file header sig
  1091→      dv.setUint16(off, 20, true); off += 2 // version made by
  1092→      dv.setUint16(off, 20, true); off += 2 // version needed to extract
  1093→      dv.setUint16(off, 0, true); off += 2 // general purpose bit flag
  1094→      dv.setUint16(off, 0, true); off += 2 // compression method = stored
  1095→      dv.setUint16(off, 0, true); off += 2 // mod time
  1096→      dv.setUint16(off, 0x21, true); off += 2 // mod date
  1097→      dv.setUint32(off, r.crc, true); off += 4
  1098→      dv.setUint32(off, r.dataLen, true); off += 4
  1099→      dv.setUint32(off, r.dataLen, true); off += 4
  1100→      dv.setUint16(off, r.name.length, true); off += 2
  1101→      dv.setUint16(off, 0, true); off += 2 // extra field length
  1102→      dv.setUint16(off, 0, true); off += 2 // comment length
  1103→      dv.setUint16(off, 0, true); off += 2 // disk number start
  1104→      dv.setUint16(off, 0, true); off += 2 // internal attrs
  1105→      dv.setUint32(off, 0, true); off += 4 // external attrs
  1106→      dv.setUint32(off, r.lfhOffset, true); off += 4 // relative offset of local header
  1107→      out.set(r.name, off); off += r.name.length
  1108→    }
  1109→    const cdSize = off - cdOff
  1110→
  1111→    // End of central directory record.
  1112→    dv.setUint32(off, 0x06054b50, true); off += 4 // EOCD signature
  1113→    dv.setUint16(off, 0, true); off += 2 // disk number
  1114→    dv.setUint16(off, 0, true); off += 2 // disk with central dir
  1115→    dv.setUint16(off, this.files.length, true); off += 2 // entries on this disk
  1116→    dv.setUint16(off, this.files.length, true); off += 2 // total entries
  1117→    dv.setUint32(off, cdSize, true); off += 4 // size of central directory
  1118→    dv.setUint32(off, cdOff, true); off += 4 // offset of central directory
  1119→    dv.setUint16(off, 0, true); off += 2 // comment length
  1120→
  1121→    return out
  1122→  }
  1123→}

  1125→/** CRC-32 with the standard PKWARE polynomial 0xEDB88320 (reflected). */
  1126→function crc32(data: Uint8Array): number {
  1127→  // Lazy-init the lookup table (memoized on the function itself).
  1128→  const table = (crc32 as unknown as { _table?: Uint32Array })._table ?? buildCrcTable()
  1129→  ;(crc32 as unknown as { _table?: Uint32Array })._table = table
  1130→  let crc = 0xffffffff
  1131→  for (let i = 0; i < data.length; i++) {
  1132→    crc = (crc >>> 8) ^ table[(crc ^ data[i]) & 0xff]
  1133→  }
  1134→  return (crc ^ 0xffffffff) >>> 0
  1135→}
  1136→
  1137→function buildCrcTable(): Uint32Array {
  1138→  const t = new Uint32Array(256)
  1139→  for (let n = 0; n < 256; n++) {
  1140→    let c = n
  1141→    for (let k = 0; k < 8; k++) {
  1142→      c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1
  1143→    }
  1144→    t[n] = c >>> 0
  1145→  }
  1146→  return t
  1147→}

  1149→/**
  1150→ * Build a real Dolby-Atmos-compatible FULL-SOURCE package as a single
  1151→ * downloadable ZIP. The archive contains:
  1152→ *
  1153→ *   1. <basename>.atmos.wav           — ADM BWF WAV file (bext + fmt + data +
  1154→ *                                       axml chunks), 24-bit PCM, N channels
  1155→ *                                       (bed + object beds). This is the file
  1156→ *                                       Dolby Atmos Renderer imports.
  1157→ *   2. audioDefinitionModelBwf.xml    — Standalone ADM XML sidecar (the same
  1158→ *                                       content embedded in the axml chunk,
  1159→ *                                       provided as a separate file for tools
  1160→ *                                       that expect a sidecar). ITU-R BS.2076-2.
  1161→ *   3. <basename>.spatial.json        — Structured spatial metadata: bed layout,
  1162→ *                                       per-speaker channel map (name, ADM id,
  1163→ *                                       track UID, azimuth, elevation, cartesian
  1164→ *                                       X/Y/Z), object count + positions,
  1165→ *                                       sample rate, processed duration,
  1166→ *                                       measured LUFS / true-peak. Machine-
  1167→ *                                       readable companion to the ADM XML.
  1168→ *   4. README.txt                     — Human-readable package documentation:
  1169→ *                                       contents, channel map table, ADM element
  1170→ *                                       counts, import instructions, spec refs.
  1171→ *   5. MANIFEST.json                  — Cryptographic manifest: SHA-256 of every
  1172→ *                                       file in the package, file inventory with
  1173→ *                                       sizes, build metadata (timestamp, tool,
  1174→ *                                       version), provenance summary. Verifies
  1175→ *                                       package integrity end-to-end.
  1176→ *
  1177→ * The XML is generated from the live config (bed layout + object positions),
  1178→ * NOT a hardcoded template. Returns a Blob ready for download.
  1179→ *
  1180→ * NOTE: This function is async because MANIFEST.json computes a real SHA-256
  1181→ * over every file via WebCrypto (`crypto.subtle.digest`). No fabricated hashes.
  1182→ */
  1183→export async function exportAtmosPackage(
  1184→  result: SpatialResult,
  1185→  metadata: { title: string; artist: string },
  1186→  config?: SpatialConfig,
  1187→): Promise<Blob> {
  1188→  const safeBase = (metadata.title || 'rain-spatial').replace(/\s+/g, '_').replace(/[^\w.-]/g, '_')
  1189→
  1190→  // 1. Build the .atmos.wav ADM BWF bytes (real bext + fmt + data + axml).
  1191→  const bwfAsm = exportSpatialBwfAssembly(result, metadata)
  1192→  const bwfBytes = new Uint8Array(bwfAsm.arrayBuffer, 0, bwfAsm.size)
  1193→
  1194→  // 2. The standalone XML sidecar.
  1195→  const xmlBytes = new TextEncoder().encode(result.admXml)
  1196→
  1197→  // 3. Structured spatial metadata JSON (machine-readable companion).
  1198→  const bedFormat = config?.bedFormat ?? inferBedFormat(result.multichannelChannels.length)
  1199→  const speakers = getBedSpeakers(bedFormat)
  1200→  const numObjects = (config?.objects ?? 0)
  1201→  const objectBaseIdx = speakers.length + 1
  1202→  const spatialMeta = buildSpatialMetadataJson(result, metadata, bedFormat, speakers, numObjects, objectBaseIdx)
  1203→  const spatialBytes = new TextEncoder().encode(spatialMeta)
  1204→
  1205→  // 4. Human-readable README.
  1206→  const readme = buildAtmosReadme(safeBase, result, metadata, bedFormat, speakers, numObjects)
  1207→  const readmeBytes = new TextEncoder().encode(readme)
  1208→
  1209→  // 5. Pack into a ZIP first (so MANIFEST can hash the exact bytes).
  1210→  const zip = new ZipWriter()
  1211→  zip.addFile(`${safeBase}.atmos.wav`, bwfBytes)
  1212→  zip.addFile('audioDefinitionModelBwf.xml', xmlBytes)
  1213→  zip.addFile(`${safeBase}.spatial.json`, spatialBytes)
  1214→  zip.addFile('README.txt', readmeBytes)
  1215→
  1216→  // 6. Compute the MANIFEST.json with real SHA-256 over every file, then add
  1217→  //    it to the ZIP. The manifest does NOT include its own hash (chicken-and-
  1218→  //    egg); instead it includes a `manifestSha256` of its own content computed
  1219→  //    AFTER serialization, stored as a top-level self-hash field. This lets a
  1220→  //    verifier confirm the manifest itself was not tampered with.
  1221→  const filesForManifest = [
  1222→    { path: `${safeBase}.atmos.wav`, bytes: bwfBytes },
  1223→    { path: 'audioDefinitionModelBwf.xml', bytes: xmlBytes },
  1224→    { path: `${safeBase}.spatial.json`, bytes: spatialBytes },
  1225→    { path: 'README.txt', bytes: readmeBytes },
  1226→  ]
  1227→  const manifestJson = await buildManifestJson(filesForManifest, result, metadata, bedFormat, numObjects)
  1228→  const manifestBytes = new TextEncoder().encode(manifestJson)
  1229→  zip.addFile('MANIFEST.json', manifestBytes)
  1230→
  1231→  const zipBytes = zip.finalize()
  1232→  // Copy into a fresh ArrayBuffer-backed view so the BlobPart type accepts it
  1233→  // (TS 5.7+ lib types distinguish Uint8Array<ArrayBuffer> from <ArrayBufferLike>).
  1234→  const zipAb = new ArrayBuffer(zipBytes.byteLength)
  1235→  new Uint8Array(zipAb).set(zipBytes)
  1236→  return new Blob([zipAb], { type: 'application/zip' })
  1237→}

  1239→/** Infer the bed format from the multichannel channel count (best-effort). */
  1240→function inferBedFormat(channels: number): SpatialConfig['bedFormat'] {
  1241→  if (channels === 12) return '7.1.4'
  1242→  if (channels === 8) return '5.1.2' // could also be 7.1; 5.1.2 is the common Atmos bed
  1243→  if (channels === 6) return '5.1'
  1244→  return '7.1'
  1245→}

  1247→/** SHA-256 hex digest via WebCrypto (browser). Matches distribution.ts pattern. */
  1248→async function sha256Hex(data: Uint8Array): Promise<string> {
  1249→  if (typeof crypto === 'undefined' || !crypto.subtle) {
  1250→    throw new Error('WebCrypto not available — SHA-256 requires crypto.subtle')
  1251→  }
  1252→  const buf = new ArrayBuffer(data.byteLength)
  1253→  new Uint8Array(buf).set(data)
  1254→  const digest = await crypto.subtle.digest('SHA-256', buf)
  1255→  const view = new Uint8Array(digest)
  1256→  let hex = ''
  1257→  for (let i = 0; i < view.length; i++) hex += view[i].toString(16).padStart(2, '0')
  1258→  return hex
  1259→}

  1477→/** Build the MANIFEST.json with real SHA-256 over every file. */
  1478→async function buildManifestJson(
  1479→  files: { path: string; bytes: Uint8Array }[],
  1480→  result: SpatialResult,
  1481→  metadata: { title: string; artist: string },
  1482→  bedFormat: SpatialConfig['bedFormat'],
  1483→  numObjects: number,
  1484→): Promise<string> {
  1485→  const fileEntries: { path: string; sizeBytes: number; sha256: string }[] = []
  1486→  for (const f of files) {
  1487→    const sha = await sha256Hex(f.bytes)
  1488→    fileEntries.push({
  1489→      path: f.path,
  1490→      sizeBytes: f.bytes.byteLength,
  1491→      sha256: sha,
  1492→    })
  1493→  }
  1494→  const manifest = {
  1495→    schema: 'rain.manifest.v1',
  1496→    package: 'rain-atmos-full-source',
  1497→    title: metadata.title,
  1498→    artist: metadata.artist,
  1499→    builtAt: new Date().toISOString(),
  1500→    tool: {
  1501→      name: 'RAIN V6',
  1502→      component: 'spatial-engine',
  1503→      spec: 'ITU-R BS.2076-2 / EBU Tech 3285 v2',
  1504→    },
  1505→    audio: {
  1506→      bedFormat,
  1507→      bedChannels: result.multichannelChannels.length,
  1508→      dynamicObjects: numObjects,
  1509→      wavChannels: result.multichannelChannels.length,
  1510→      admDeclaredChannels: result.multichannelChannels.length + numObjects,
  1511→      sampleRate: result.sampleRate,
  1512→      durationSeconds: Number(result.processedSeconds.toFixed(6)),
  1513→      bitDepth: 24,
  1514→      integratedLufs: Number(result.lufs.toFixed(2)),
  1515→      truePeakDbtp: Number(result.truePeak.toFixed(2)),
  1516→    },
  1517→    files: fileEntries,
  1518→    integrity: {
  1519→      algorithm: 'SHA-256',
  1520→      note: 'Recompute SHA-256 of each file and compare to files[].sha256 to verify package integrity.',
  1521→    },
  1522→  }
  1523→  return JSON.stringify(manifest, null, 2)

  1526→/**
  1527→ * Internal: same encoding as exportSpatialBwf, but returns the underlying
  1528→ * ArrayBuffer + size instead of wrapping in a Blob. Kept in sync with
  1529→ * exportSpatialBwf — if you edit one, edit both.
  1530→ */
  1531→function exportSpatialBwfAssembly(
  1532→  result: SpatialResult,
  1533→  metadata: { title: string; artist: string },
  1534→): { arrayBuffer: ArrayBuffer; size: number } {
  1535→  const channels = result.multichannelChannels
  1536→  const numChannels = channels.length
  1537→  const sampleRate = result.sampleRate
  1538→  const length = channels[0].length
  1539→  const bytesPerSample = 3 // 24-bit
  1540→  const blockAlign = numChannels * bytesPerSample
  1541→  const dataSize = length * blockAlign
  1542→
  1543→  // --- bext chunk (EBU Tech 3285 v2) ---
  1544→  const now = new Date()
  1545→  const dateStr = `${now.getFullYear()}-${pad(now.getMonth() + 1, 2)}-${pad(now.getDate(), 2)}`
  1546→  const timeStr = `${pad(now.getHours(), 2)}:${pad(now.getMinutes(), 2)}:${pad(now.getSeconds(), 2)}`
  1547→  const description = `RAIN V6 Spatial Master — ${result.processedSeconds.toFixed(1)}s${result.truncated ? ' (truncated from ' + result.inputSeconds.toFixed(1) + 's)' : ''}. Bed: ${numChannels}ch. LUFS ${result.lufs.toFixed(1)} / dBTP ${result.truePeak.toFixed(1)}.`
  1548→  const originator = (metadata.artist || 'RAIN V6').slice(0, 32)
  1549→  const origRef = `RAIN-${Date.now().toString(36)}`.slice(0, 32)
  1550→  const codingHistory = `A=PCM,F=${sampleRate},B=24-bit,W=${numChannels},M=audio/${numChannels}ch\r\n`
  1551→  const codingBytes = new TextEncoder().encode(codingHistory)
  1552→  const bextPayloadSize = 602 + codingBytes.length + 1
  1553→  const bextPadByte = bextPayloadSize % 2 === 1 ? 1 : 0
  1554→  const bextTotal = 8 + bextPayloadSize + bextPadByte
  1555→
  1556→  const bextBuf = new ArrayBuffer(bextTotal)
  1557→  const bv = new DataView(bextBuf)
  1558→  let bo = 0
  1559→  bo = writeStr(bv, bo, 'bext')
  1560→  bv.setUint32(bo, bextPayloadSize, true); bo += 4
  1561→  bo = writeStrPad(bv, bo, description, 256)
  1562→  bo = writeStrPad(bv, bo, originator, 32)
  1563→  bo = writeStrPad(bv, bo, origRef, 32)
  1564→  bo = writeStrPad(bv, bo, dateStr, 10)
  1565→  bo = writeStrPad(bv, bo, timeStr, 8)
  1566→  bv.setUint32(bo, 0, true); bo += 4
  1567→  bv.setUint32(bo, 0, true); bo += 4
  1568→  bv.setUint16(bo, 2, true); bo += 2
  1569→  bo += 64
  1570→  const lufsInt = Math.round(result.lufs * 100)
  1571→  bv.setInt16(bo, Math.max(-32768, Math.min(32767, lufsInt)), true); bo += 2
  1572→  bv.setInt16(bo, 0x8000, true); bo += 2
  1573→  const tpInt = Math.round(result.truePeak * 100)
  1574→  bv.setInt16(bo, Math.max(-32768, Math.min(32767, tpInt)), true); bo += 2
  1575→  bv.setInt16(bo, 0x8000, true); bo += 2
  1576→  bv.setInt16(bo, 0x8000, true); bo += 2
  1577→  bo += 180
  1578→  for (let i = 0; i < codingBytes.length; i++) bv.setUint8(bo + i, codingBytes[i])
  1579→  bo += codingBytes.length
  1580→  bv.setUint8(bo, 0); bo += 1
  1581→  if (bextPadByte === 1) bv.setUint8(bo++, 0)
  1582→
  1583→  // --- axml chunk ---
  1584→  const xmlBytes = new TextEncoder().encode(result.admXml)
  1585→  const axmlPayloadSize = xmlBytes.length
  1586→  const axmlPadByte = axmlPayloadSize % 2 === 1 ? 1 : 0
  1587→  const axmlTotal = 8 + axmlPayloadSize + axmlPadByte
  1588→
  1589→  const axmlBuf = new ArrayBuffer(axmlTotal)
  1590→  const av = new DataView(axmlBuf)
  1591→  let ao = 0
  1592→  ao = writeStr(av, ao, 'axml')
  1593→  av.setUint32(ao, axmlPayloadSize, true); ao += 4
  1594→  for (let i = 0; i < xmlBytes.length; i++) av.setUint8(ao + i, xmlBytes[i])
  1595→  ao += xmlBytes.length
  1596→  if (axmlPadByte === 1) av.setUint8(ao++, 0)
  1597→
  1598→  const fmtSize = 16
  1599→  const fmtTotal = 8 + fmtSize
  1600→  const dataTotal = 8 + dataSize
  1601→  const riffPayloadSize = 4 + bextTotal + fmtTotal + dataTotal + axmlTotal
  1602→  const totalSize = 8 + riffPayloadSize
  1603→
  1604→  const out = new ArrayBuffer(totalSize)
  1605→  const v = new DataView(out)
  1606→  let o = 0
  1607→  o = writeStr(v, o, 'RIFF')
  1608→  v.setUint32(o, riffPayloadSize, true); o += 4
  1609→  o = writeStr(v, o, 'WAVE')
  1610→  new Uint8Array(out, o, bextTotal).set(new Uint8Array(bextBuf))
  1611→  o += bextTotal
  1612→  o = writeStr(v, o, 'fmt ')
  1613→  v.setUint32(o, fmtSize, true); o += 4
  1614→  v.setUint16(o, 1, true); o += 2
  1615→  v.setUint16(o, numChannels, true); o += 2
  1616→  v.setUint32(o, sampleRate, true); o += 4
  1617→  v.setUint32(o, sampleRate * blockAlign, true); o += 4
  1618→  v.setUint16(o, blockAlign, true); o += 2
  1619→  v.setUint16(o, 24, true); o += 2
  1620→  o = writeStr(v, o, 'data')
  1621→  v.setUint32(o, dataSize, true); o += 4
  1622→  for (let i = 0; i < length; i++) {
  1623→    for (let ch = 0; ch < numChannels; ch++) {
  1624→      let s = channels[ch][i]
  1625→      s = Math.max(-1, Math.min(1, s))
  1626→      const iv = Math.round(s < 0 ? s * 0x800000 : s * 0x7fffff)
  1627→      v.setUint8(o, iv & 0xff); o += 1
  1628→      v.setUint8(o, (iv >> 8) & 0xff); o += 1
  1629→      v.setUint8(o, (iv >> 16) & 0xff); o += 1
  1630→    }
  1631→  }
  1632→  new Uint8Array(out, o, axmlTotal).set(new Uint8Array(axmlBuf))
  1633→  o += axmlTotal
  1634→
  1635→  return { arrayBuffer: out, size: o }
  1636→}
