     4→ * RAIN V6 — Real Analytics Engine
     5→ *
     6→ * Derives EVERY displayed analytics metric from actual stored data:
     7→ *   - `renderHistory` (Zustand store, RenderLogEntry[]) — last 20 renders, in-memory
     8→ *   - `engineStats`   (IndexedDB, single-record) — cumulative counters
     9→ *   - `renders`       (IndexedDB) — per-render telemetry w/ 16-stage timings
    10→ *   - `qc`            (IndexedDB) — per-render QC pass/warn/fail breakdown
    11→ *   - `exports`       (IndexedDB) — per-export format + bytes + duration
    12→ *   - `activity`      (IndexedDB) — undo / redo / repair / AI-query events
    13→ *
    14→ * NO fabricated numbers, NO Math.random, NO hardcoded sample arrays.
    15→ * If there is no data, downstream components show "No data" / "—".
    16→ *
    17→ * The `RenderLogEntry.duration` field is the *audio* file duration in SECONDS
    18→ * (not render-wall-clock time — that is tracked separately via `recordRenderStat`).
    19→ * This makes the storage-size formula physically meaningful:
    20→ *   bytes = duration_sec * sampleRate * bitDepth * channels / 8
    21→ */

   235→function startOfLocalDay(d: Date): Date {
   236→  return new Date(d.getFullYear(), d.getMonth(), d.getDate(), 0, 0, 0, 0)
   237→}
   238→
   239→function mmdd(d: Date): string {
   240→  const m = String(d.getMonth() + 1).padStart(2, '0')
   241→  const day = String(d.getDate()).padStart(2, '0')
   242→  return `${m}-${day}`
   243→}
   244→
   245→function getPlatformTargetLufs(platform: string): number | null {
   246→  const target = PLATFORM_TARGETS.find((p) => p.slug === platform)
   247→  return target ? target.targetLufs : null
   248→}
   249→
   250→/** Audio file size estimate in MB given duration in seconds. */
   251→function audioSizeMb(durationSec: number): number {
   252→  return (durationSec * DEFAULT_SAMPLE_RATE * DEFAULT_BIT_DEPTH * DEFAULT_CHANNELS) / 8 / 1e6
   253→}
   254→
   255→function safeMemoryMb(): number | null {
   256→  if (typeof performance === 'undefined') return null
   257→  const mem = (performance as unknown as { memory?: { usedJSHeapSize?: number } }).memory
   258→  if (mem && typeof mem.usedJSHeapSize === 'number') {
   259→    return mem.usedJSHeapSize / 1e6
   260→  }
   261→  return null
   262→}

   268→export function computeAnalytics(renderHistory: RenderLogEntry[]): AnalyticsSummary {
   269→  const now = new Date()
   270→  const todayStart = startOfLocalDay(now)
   271→
   272→  // ---- Daily window (last 30 days, oldest → today) ----
   273→  const daily: DailyMetric[] = []
   274→  for (let i = WINDOW_DAYS - 1; i >= 0; i--) {
   275→    const dayStart = new Date(todayStart.getTime() - i * DAY_MS)
   276→    const dayStartMs = dayStart.getTime()
   277→    const dayEndMs = dayStartMs + DAY_MS
   278→
   279→    const dayRenders = renderHistory.filter(
   280→      (r) => r.timestamp >= dayStartMs && r.timestamp < dayEndMs,
   281→    )
   282→    const renders = dayRenders.length
   283→    // duration is in SECONDS (audio). Sum → seconds; /60 → minutes.
   284→    const minutes = dayRenders.reduce((acc, r) => acc + r.duration / 60, 0)
   285→    const score = renders > 0
   286→      ? dayRenders.reduce((acc, r) => acc + r.rainScore, 0) / renders
   287→      : 0
   288→    const dayStorageMb = dayRenders.reduce((acc, r) => acc + audioSizeMb(r.duration), 0)
   289→
   290→    daily.push({
   291→      date: mmdd(dayStart),
   292→      renders,
   293→      minutes,
   294→      score,
   295→      storageMb: dayStorageMb,
   296→    })
   297→  }
   298→
   299→  // Make storage cumulative across the 30-day window
   300→  let cumulative = 0
   301→  for (const d of daily) {
   302→    cumulative += d.storageMb
   303→    d.storageMb = cumulative
   304→  }
   305→
   306→  // ---- Totals across ALL history (not just 30 days) ----
   307→  const totalRenders = renderHistory.length
   308→  const avgScore = totalRenders > 0
   309→    ? renderHistory.reduce((a, b) => a + b.rainScore, 0) / totalRenders
   310→    : 0
   311→  const totalMinutes = renderHistory.reduce((a, b) => a + b.duration / 60, 0)
   312→  const totalStorageMb = renderHistory.reduce((a, b) => a + audioSizeMb(b.duration), 0)
   313→
   314→  // ---- Platform breakdown ----
   315→  const platformMap = new Map<string, RenderLogEntry[]>()
   316→  for (const r of renderHistory) {
   317→    const arr = platformMap.get(r.platform) ?? []
   318→    arr.push(r)
   319→    platformMap.set(r.platform, arr)
   320→  }
   321→  const platformBreakdown: PlatformBreakdownEntry[] = Array.from(platformMap.entries())
   322→    .map(([platform, entries]) => {
   323→      const count = entries.length
   324→      const entriesAvgScore = entries.reduce((a, b) => a + b.rainScore, 0) / count
   325→      const targetLufs = getPlatformTargetLufs(platform)
   326→      const avgLufsDelta = targetLufs !== null
   327→        ? entries.reduce((a, b) => a + (b.outputLufs - targetLufs), 0) / count
   328→        : 0
   329→      const pct = totalRenders > 0 ? (count / totalRenders) * 100 : 0
   330→      return { platform, count, avgScore: entriesAvgScore, avgLufsDelta, pct }
   331→    })
   332→    .sort((a, b) => b.count - a.count)
   333→
   334→  // ---- Score history (chronological) ----
   335→  const scoreHistory: ScoreHistoryEntry[] = renderHistory
   336→    .slice()
   337→    .sort((a, b) => a.timestamp - b.timestamp)
   338→    .map((r) => ({ timestamp: r.timestamp, score: r.rainScore, platform: r.platform }))
   339→
   340→  // ---- Trend: last 7 days vs previous 7 days ----
   341→  const last7Start = todayStart.getTime() - 7 * DAY_MS
   342→  const prev7Start = todayStart.getTime() - 14 * DAY_MS
   343→  const last7 = renderHistory.filter((r) => r.timestamp >= last7Start)
   344→  const prev7 = renderHistory.filter(
   345→    (r) => r.timestamp >= prev7Start && r.timestamp < last7Start,
   346→  )
   347→  const rendersDeltaPct = prev7.length > 0
   348→    ? ((last7.length - prev7.length) / prev7.length) * 100
   349→    : last7.length > 0 ? null : null // null = no baseline
   350→  const last7Avg = last7.length > 0 ? last7.reduce((a, b) => a + b.rainScore, 0) / last7.length : null
   351→  const prev7Avg = prev7.length > 0 ? prev7.reduce((a, b) => a + b.rainScore, 0) / prev7.length : null
   352→  const scoreDelta = last7Avg !== null && prev7Avg !== null ? last7Avg - prev7Avg : null
   353→
   354→  // ---- P2-METERS: avg LUFS over last 10 renders (real, not sample data) ----
   355→  const sortedByTime = renderHistory.slice().sort((a, b) => a.timestamp - b.timestamp)
   356→  const last10 = sortedByTime.slice(-10)
   357→  const avgLufsLast10 = last10.length > 0
   358→    ? last10.reduce((a, b) => a + b.outputLufs, 0) / last10.length
   359→    : null
   360→
   361→  // ---- P2-METERS: score trend direction (last 5 vs previous 5) ----
   362→  // Compares the mean RAIN score of the most recent 5 renders against the
   363→  // 5 renders before that. Direction is 'improving' if delta > +0.5,
   364→  // 'declining' if delta < -0.5, 'stable' if |delta| <= 0.5, and
   365→  // 'insufficient' if we don't have at least 5 renders in each window.
   366→  const last5 = sortedByTime.slice(-5)
   367→  const prev5 = sortedByTime.slice(-10, -5)
   368→  let scoreTrendDirection: 'improving' | 'declining' | 'stable' | 'insufficient' = 'insufficient'
   369→  let scoreTrendDelta: number | null = null
   370→  if (last5.length === 5 && prev5.length === 5) {
   371→    const last5Avg = last5.reduce((a, b) => a + b.rainScore, 0) / 5
   372→    const prev5Avg = prev5.reduce((a, b) => a + b.rainScore, 0) / 5
   373→    scoreTrendDelta = last5Avg - prev5Avg
   374→    if (scoreTrendDelta > 0.5) scoreTrendDirection = 'improving'
   375→    else if (scoreTrendDelta < -0.5) scoreTrendDirection = 'declining'
   376→    else scoreTrendDirection = 'stable'
   377→  }
   378→
   379→  // ---- P2-METERS: most-used macro values (mean + mode per macro key) ----
   380→  // Real aggregate over renderHistory. For each of the 7 macros, compute the
   381→  // arithmetic mean and the mode (most common rounded-to-integer value).
   382→  // Returns zero-filled stats when renderHistory is empty.
   383→  const MACRO_KEYS: Array<keyof MacroValues> = [
   384→    'brighten', 'glue', 'width', 'punch', 'warmth', 'space', 'repair',
   385→  ]
   386→  const macroUsage: MacroUsageStats = {
   387→    means: {},
   388→    modes: {},
   389→    sampleCount: renderHistory.length,
   390→  }
   391→  for (const key of MACRO_KEYS) {
   392→    if (renderHistory.length === 0) {
   393→      macroUsage.means[key] = 0
   394→      macroUsage.modes[key] = 0
   395→      continue
   396→    }
   397→    let sum = 0
   398→    const roundedCounts = new Map<number, number>()
   399→    for (const r of renderHistory) {
   400→      const v = r.macroValues[key]
   401→      sum += v
   402→      const rounded = Math.round(v)
   403→      roundedCounts.set(rounded, (roundedCounts.get(rounded) ?? 0) + 1)
   404→    }
   405→    macroUsage.means[key] = sum / renderHistory.length
   406→    // Mode = rounded value with the highest count. Ties broken by lower value.
   407→    let modeVal = 0
   408→    let modeCount = -1
   409→    for (const [val, cnt] of Array.from(roundedCounts.entries()).sort((a, b) => a[0] - b[0])) {
   410→      if (cnt > modeCount) {
   411→        modeCount = cnt
   412→        modeVal = val
   413→      }
   414→    }
   415→    macroUsage.modes[key] = modeVal
   416→  }
   417→
   418→  // ---- P2-METERS: genre distribution (real per-genre render counts) ----
   419→  const genreMap = new Map<string, number>()
   420→  for (const r of renderHistory) {
   421→    genreMap.set(r.genre, (genreMap.get(r.genre) ?? 0) + 1)
   422→  }
   423→  const genreDistribution: GenreDistributionEntry[] = Array.from(genreMap.entries())
   424→    .map(([genre, count]) => ({
   425→      genre,
   426→      count,
   427→      pct: totalRenders > 0 ? (count / totalRenders) * 100 : 0,
   428→    }))
   429→    .sort((a, b) => b.count - a.count)
   430→
   431→  return {
   432→    daily,
   433→    totalRenders,
   434→    avgScore,
   435→    totalMinutes,
   436→    totalStorageMb,
   437→    platformBreakdown,
   438→    scoreHistory,
   439→    engineStats: { ...EMPTY_ENGINE_STATS },
   440→    memoryMb: safeMemoryMb(),
   441→    trend: { rendersDeltaPct, scoreDelta },
   442→    avgLufsLast10,
   443→    scoreTrendDirection,
   444→    scoreTrendDelta,
   445→    macroUsage,
   446→    genreDistribution,
   447→  }
   448→}

   450→// ---------------------------------------------------------------------------
   451→// IndexedDB persistence for EngineStats + per-event records
   452→// Pattern mirrors provenance.ts `withDb` — connection is closed in finally.
   453→//
   454→// Schema v2 (this audit) adds 4 new object stores alongside the existing
   455→// `engineStats` single-record store:
   456→//   - `renders`   — one row per render (telemetry + 16-stage timings)
   457→//   - `qc`        — one row per render QC snapshot
   458→//   - `exports`   — one row per WAV/MP3/Atmos export
   459→//   - `activity`  — one row per user action (undo / redo / repair / ai-query)
   460→// All stores are append-only with autoincrement numeric keys; queries use
   461→// openCursor + collect. We don't bother with indexes — every analytics
   462→// query walks the full list once on AnalyticsTab mount, which is cheap for
   463→// the expected scale (hundreds of rows, not millions).
   464→// ---------------------------------------------------------------------------
   465→
   466→const ANALYTICS_DB = 'rain-analytics'
   467→const STATS_STORE = 'engineStats'
   468→const STATS_KEY = 'current'
   469→const RENDER_STORE = 'renders'
   470→const QC_STORE = 'qc'
   471→const EXPORT_STORE = 'exports'
   472→const ACTIVITY_STORE = 'activity'
   473→
   474→function openDb(): Promise<IDBDatabase> {
   475→  return new Promise((resolve, reject) => {
   476→    const req = indexedDB.open(ANALYTICS_DB, 2)
   477→    req.onupgradeneeded = () => {
   478→      const db = req.result
   479→      // v1 store — cumulative counters (single record, keyed 'current').
   480→      if (!db.objectStoreNames.contains(STATS_STORE)) {
   481→        db.createObjectStore(STATS_STORE)
   482→      }
   483→      // v2 stores — per-event append-only logs.
   484→      if (!db.objectStoreNames.contains(RENDER_STORE)) {
   485→        db.createObjectStore(RENDER_STORE, { keyPath: 'id' })
   486→      }
   487→      if (!db.objectStoreNames.contains(QC_STORE)) {
   488→        db.createObjectStore(QC_STORE, { keyPath: 'id' })
   489→      }
   490→      if (!db.objectStoreNames.contains(EXPORT_STORE)) {
   491→        db.createObjectStore(EXPORT_STORE, { keyPath: 'id' })
   492→      }
   493→      if (!db.objectStoreNames.contains(ACTIVITY_STORE)) {
   494→        db.createObjectStore(ACTIVITY_STORE, { keyPath: 'id' })
   495→      }
   496→    }
   497→    req.onsuccess = () => resolve(req.result)
   498→    req.onerror = () => reject(req.error)
   499→  })
   500→}
   501→
   502→async function withDb<T>(fn: (db: IDBDatabase) => Promise<T> | T): Promise<T> {
   503→  const db = await openDb()
   504→  try {
   505→    return await fn(db)
   506→  } finally {
   507→    db.close()
   508→  }
   509→}

   511→// Generic read-all helper for any of the v2 append-only stores. The generic
   512→// is constrained to records with a numeric `timestamp` so the sort below
   513→// type-checks — every record type we persist (renders / qc / exports /
   514→// activity) has a top-level `timestamp: number` field.
   515→function readAll<T extends { timestamp: number }>(storeName: string): Promise<T[]> {
   516→  return withDb((db) => new Promise<T[]>((resolve, reject) => {
   517→    if (!db.objectStoreNames.contains(storeName)) {
   518→      resolve([])
   519→      return
   520→    }
   521→    const tx = db.transaction(storeName, 'readonly')
   522→    const req = tx.objectStore(storeName).getAll()
   523→    req.onsuccess = () => {
   524→      const arr = (req.result ?? []) as T[]
   525→      // Sort oldest → newest by timestamp for predictable aggregation.
   526→      arr.sort((a, b) => (a.timestamp ?? 0) - (b.timestamp ?? 0))
   527→      resolve(arr)
   528→    }
   529→    req.onerror = () => reject(req.error)
   530→  }))
   531→}

   533→function putRecord<T extends { id: string }>(storeName: string, record: T): Promise<void> {
   534→  return withDb((db) => new Promise<void>((resolve, reject) => {
   535→    if (!db.objectStoreNames.contains(storeName)) {
   536→      // DB still on v1 schema (upgrade hasn't fired for this connection).
   537→      // Silently drop — analytics must never break the render flow.
   538→      resolve()
   539→      return
   540→    }
   541→    const tx = db.transaction(storeName, 'readwrite')
   542→    tx.objectStore(storeName).put(record)
   543→    tx.oncomplete = () => resolve()
   544→    tx.onerror = () => reject(tx.error)
   545→  }))
   546→}

   548→function clearStore(storeName: string): Promise<void> {
   549→  return withDb((db) => new Promise<void>((resolve, reject) => {
   550→    if (!db.objectStoreNames.contains(storeName)) {
   551→      resolve()
   552→      return
   553→    }
   554→    const tx = db.transaction(storeName, 'readwrite')
   555→    tx.objectStore(storeName).clear()
   556→    tx.oncomplete = () => resolve()
   557→    tx.onerror = () => reject(tx.error)
   558→  }))
   559→}

   561→function newId(prefix: string): string {
   562→  // crypto.randomUUID when available; deterministic fallback otherwise.
   563→  // No Math.random — keeps the "no random data" guarantee (this is identity,
   564→  // not analytics data, but the linter grep would flag Math.random anywhere).
   565→  if (typeof crypto !== 'undefined' && 'randomUUID' in crypto) {
   566→    return crypto.randomUUID()
   567→  }
   568→  return `${prefix}-${Date.now()}-${(typeof performance !== 'undefined' && performance.now
   569→    ? Math.floor(performance.now() * 1000)
   570→    : 0)}`
   571→}

   573→export async function loadEngineStats(): Promise<EngineStats> {
   574→  if (typeof indexedDB === 'undefined') return { ...EMPTY_ENGINE_STATS }
   575→  try {
   576→    return await withDb((db) => new Promise<EngineStats>((resolve, reject) => {
   577→      const tx = db.transaction(STATS_STORE, 'readonly')
   578→      const store = tx.objectStore(STATS_STORE)
   579→      const req = store.get(STATS_KEY)
   580→      req.onsuccess = () => {
   581→        const stored = req.result as EngineStats | undefined
   582→        // Merge with EMPTY_ENGINE_STATS so newly-added counters default to 0
   583→        // when reading a v1-shaped record from an older session.
   584→        resolve({
   585→          ...EMPTY_ENGINE_STATS,
   586→          ...(stored ?? {}),
   587→        })
   588→      }
   589→      req.onerror = () => reject(req.error)
   590→    }))
   591→  } catch {
   592→    return { ...EMPTY_ENGINE_STATS }
   593→  }
   594→}

   830→export interface QCAggregates {
   831→  total: number
   832→  passRate: number  // 0..1 across all checks of all renders
   833→  warnRate: number
   834→  failRate: number
   835→  /** Most-failed check IDs, sorted desc by fail count. */
   836→  topFailures: Array<{ id: string; fails: number; occurrences: number }>
   837→  /** Per-category pass/warn/fail tallies. */
   838→  categories: Record<string, { pass: number; warn: number; fail: number }>
   839→}
   840→
   841→export function computeQCAggregates(qc: QCRecord[]): QCAggregates {
   842→  if (qc.length === 0) {
   843→    return {
   844→      total: 0,
   845→      passRate: 0,
   846→      warnRate: 0,
   847→      failRate: 0,
   848→      topFailures: [],
   849→      categories: {},
   850→    }
   851→  }
   852→  const totals = { pass: 0, warn: 0, fail: 0 }
   853→  const categories: Record<string, { pass: number; warn: number; fail: number }> = {}
   854→  const failCounts = new Map<string, { fails: number; occurrences: number }>()
   855→
   856→  // QCRecord.categories is the per-category tally for a single render;
   857→  // QCRecord.checks is the per-check list. We use `checks` for the per-id
   858→  // failure ranking (more granular) and `categories` for the rollup.
   859→  for (const r of qc) {
   860→    for (const cat of Object.keys(r.categories)) {
   861→      const c = r.categories[cat]
   862→      const acc = categories[cat] ?? { pass: 0, warn: 0, fail: 0 }
   863→      acc.pass += c.pass
   864→      acc.warn += c.warn
   865→      acc.fail += c.fail
   866→      categories[cat] = acc
   867→      totals.pass += c.pass
   868→      totals.warn += c.warn
   869→      totals.fail += c.fail
   870→    }
   871→    for (const chk of r.checks) {
   872→      const acc = failCounts.get(chk.id) ?? { fails: 0, occurrences: 0 }
   873→      acc.occurrences += 1
   874→      if (chk.status === 'fail') acc.fails += 1
   875→      failCounts.set(chk.id, acc)
   876→    }
   877→  }
   878→  const total = totals.pass + totals.warn + totals.fail
   879→  const topFailures = Array.from(failCounts.entries())
   880→    .map(([id, v]) => ({ id, fails: v.fails, occurrences: v.occurrences }))
   881→    .sort((a, b) => b.fails - a.fails)
   882→    .slice(0, 6)
   883→  return {
   884→    total,
   885→    passRate: total > 0 ? totals.pass / total : 0,
   886→    warnRate: total > 0 ? totals.warn / total : 0,
   887→    failRate: total > 0 ? totals.fail / total : 0,
   888→    topFailures,
   889→    categories,
   890→  }
   891→}

   893→export interface ExportAggregates {
   894→  total: number
   895→  totalBytes: number
   896→  avgMs: number  // average export wall-clock
   897→  /** Per-format count + byte total. */
   898→  byFormat: Array<{ format: string; count: number; bytes: number; pct: number }>
   899→}
   900→
   901→export function computeExportAggregates(exports: ExportRecord[]): ExportAggregates {
   902→  if (exports.length === 0) {
   903→    return { total: 0, totalBytes: 0, avgMs: 0, byFormat: [] }
   904→  }
   905→  const total = exports.length
   906→  const totalBytes = exports.reduce((a, b) => a + Math.max(0, b.bytes), 0)
   907→  const totalMs = exports.reduce((a, b) => a + Math.max(0, b.durationMs), 0)
   908→  const byFormatMap = new Map<string, { count: number; bytes: number }>()
   909→  for (const e of exports) {
   910→    const acc = byFormatMap.get(e.format) ?? { count: 0, bytes: 0 }
   911→    acc.count += 1
   912→    acc.bytes += Math.max(0, e.bytes)
   913→    byFormatMap.set(e.format, acc)
   914→  }
   915→  const byFormat = Array.from(byFormatMap.entries())
   916→    .map(([format, v]) => ({
   917→      format,
   918→      count: v.count,
   919→      bytes: v.bytes,
   920→      pct: total > 0 ? (v.count / total) * 100 : 0,
   921→    }))
   922→    .sort((a, b) => b.count - a.count)
   923→  return { total, totalBytes, avgMs: total > 0 ? totalMs / total : 0, byFormat }
   924→}

   926→export interface ActivityAggregates {
   927→  total: number
   928→  /** Per-type count, sorted desc. */
   929→  byType: Array<{ type: string; count: number; lastAt: number | null }>
   930→  /** Last N activity records for the log panel (newest first). */
   931→  recent: ActivityRecord[]
   932→}
   933→
   934→export function computeActivityAggregates(
   935→  activity: ActivityRecord[],
   936→  recentLimit = 20,
   937→): ActivityAggregates {
   938→  if (activity.length === 0) {
   939→    return { total: 0, byType: [], recent: [] }
   940→  }
   941→  const byTypeMap = new Map<string, { count: number; lastAt: number | null }>()
   942→  for (const a of activity) {
   943→    const acc = byTypeMap.get(a.type) ?? { count: 0, lastAt: null }
   944→    acc.count += 1
   945→    if (a.timestamp > (acc.lastAt ?? 0)) acc.lastAt = a.timestamp
   946→    byTypeMap.set(a.type, acc)
   947→  }
   948→  const byType = Array.from(byTypeMap.entries())
   949→    .map(([type, v]) => ({ type, count: v.count, lastAt: v.lastAt }))
   950→    .sort((a, b) => b.count - a.count)
   951→  // Recent: newest first (input is oldest → newest).
   952→  const recent = activity.slice(-recentLimit).reverse()
   953→  return { total: activity.length, byType, recent }
   954→}

   956→export interface StageTimingAverages {
   957→  /** Per-stage average ms across all renders. */
   958→  averages: Array<{ id: number; name: string; avgMs: number; samples: number }>
   959→  /** Total render count used to compute the averages. */
   960→  sampleCount: number
   961→}
   962→
   963→export function computeStageTimingAverages(
   964→  renders: RenderTelemetryRecord[],
   965→): StageTimingAverages {
   966→  const sampleCount = renders.length
   967→  const sums = new Map<number, number>()
   968→  const counts = new Map<number, number>()
   969→  for (const r of renders) {
   970→    for (const [k, v] of Object.entries(r.stageTimings)) {
   971→      const id = Number(k)
   972→      if (!Number.isFinite(id)) continue
   973→      sums.set(id, (sums.get(id) ?? 0) + Math.max(0, v))
   974→      counts.set(id, (counts.get(id) ?? 0) + 1)
   975→    }
   976→  }
   977→  const averages = PIPELINE_STAGES.map((stage) => {
   978→    const samples = counts.get(stage.id) ?? 0
   979→    const sum = sums.get(stage.id) ?? 0
   980→    return {
   981→      id: stage.id,
   982→      name: stage.name,
   983→      avgMs: samples > 0 ? sum / samples : 0,
   984→      samples,
   985→    }
   986→  })
   987→  return { averages, sampleCount }
   988→}

  1014→export function formatMs(ms: number): string {
  1015→  if (!Number.isFinite(ms) || ms <= 0) return '—'
  1016→  if (ms < 1000) return `${Math.round(ms)} ms`
  1017→  return `${(ms / 1000).toFixed(ms < 10000 ? 2 : 1)} s`
  1018→}
  1019→
  1020→export function formatMb(mb: number): string {
  1021→  if (!Number.isFinite(mb) || mb <= 0) return '0 MB'
  1022→  if (mb < 1) return `${(mb * 1024).toFixed(1)} KB`
  1023→  if (mb < 1024) return `${mb.toFixed(1)} MB`
  1024→  return `${(mb / 1024).toFixed(2)} GB`
  1025→}
  1026→
  1027→/** Format a byte count as B / KB / MB / GB. */
  1028→export function formatBytes(bytes: number): string {
  1029→  if (!Number.isFinite(bytes) || bytes <= 0) return '0 B'
  1030→  if (bytes < 1024) return `${Math.round(bytes)} B`
  1031→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
  1032→  if (bytes < 1024 * 1024 * 1024) return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
  1033→  return `${(bytes / (1024 * 1024 * 1024)).toFixed(2)} GB`
  1034→}
