   207→   207→  async loadFile(file: File | ArrayBuffer): Promise<{ analysis: AudioAnalysis; duration: number; sampleRate: number; channels: number }> {
   208→   208→    // BUG FIX: stop any current playback + invalidate stale buffers BEFORE
   209→   209→    // decode so a failed decode doesn't leave the old file's buffers around.
   210→   210→    this.stop()
   211→   211→    this.inputBuffer = null
   212→   212→    this.processedBuffer = null
   213→   213→    this.analysis = null
   214→   214→    this.params = null
   215→   215→    this.playStartOffset = 0
   216→   216→    this.state.position = 0
   217→   217→
   218→   218→    await this.init()
   219→   219→    const buf = file instanceof File ? await file.arrayBuffer() : file
   220→   220→    this.inputBuffer = await this.context!.decodeAudioData(buf.slice(0))
   221→   221→    // P3-REPAIR: capture the original (pre-repair) buffer so resetRepair() works.
   222→   222→    this.originalInputBuffer = this.inputBuffer
   223→   223→    this.state.duration = this.inputBuffer.duration
   224→   224→    this.state.previewMode = this._previewMode
   225→   225→
   226→   226→    // Compute analysis on raw input
   227→   227→    const channels: Float32Array[] = []
   228→   228→    for (let ch = 0; ch < this.inputBuffer.numberOfChannels; ch++) {
   229→   229→      channels.push(this.inputBuffer.getChannelData(ch).slice())
   230→   230→    }
   231→   231→    this.analysis = analyzeAudio(channels, this.inputBuffer.sampleRate)
   232→   232→    this.emit()
   233→   233→    return {
   234→   234→      analysis: this.analysis,
   235→   235→      duration: this.inputBuffer.duration,
   236→   236→      sampleRate: this.inputBuffer.sampleRate,
   237→   237→      channels: this.inputBuffer.numberOfChannels,
   238→   238→    }
   239→   239→  }
   240→   240→
   241→   241→  /**
   242→   242→   * Decode an audio ArrayBuffer WITHOUT touching app state — does NOT replace
   243→   243→   * `inputBuffer`, `originalInputBuffer`, `analysis`, or any playback state.
   244→   244→   *
   245→   245→   * Used by the Stems ZIP upload path (and any future feature that needs to
   246→   246→   * decode audio files into raw Float32Array channels without clobbering the
   247→   247→   * mastering tab's loaded input). For the normal "load a track to master"
   248→   248→   * flow, use `loadFile()` — which now delegates to this method for the
   249→   249→   * decode step so the decode logic has a single source of truth.
   250→   250→   *
   251→   251→   * @param buffer Encoded audio bytes (WAV/MP3/FLAC/AAC/OGG/M4A/AIFF...).
   252→   252→   * @returns Per-channel PCM samples (Float32Array, stereo or mono) + the
   253→   253→   *          sample rate of the decoded AudioBuffer.
   254→   254→   */
   255→   255→  async decodeOnly(buffer: ArrayBuffer): Promise<{ channels: Float32Array[]; sampleRate: number; duration: number }> {
   256→   256→    await this.init()
   257→   257→    // `decodeAudioData` may detach the input ArrayBuffer in some browsers
   258→   258→    // (notably older Safari), so slice() defensively. The sliced copy is also
   259→   259→    // what `loadFile()` already does — same contract here.
   260→   260→    const decoded = await this.context!.decodeAudioData(buffer.slice(0))
   261→   261→    const channels: Float32Array[] = []
   262→   262→    for (let ch = 0; ch < decoded.numberOfChannels; ch++) {
   263→   263→      // .slice() copies so the caller can hold the Float32Array safely after
   264→   264→      // the AudioBuffer is GC'd (it would otherwise alias internal memory).
   265→   265→      channels.push(decoded.getChannelData(ch).slice())
   266→   266→    }
   267→   267→    const sampleRate = decoded.sampleRate
   268→   268→    const duration = decoded.duration
   269→   269→    return { channels, sampleRate, duration }
   270→   270→  }

   273→   273→  // Real-time preview playback
   274→   274→  // -------------------------------------------------------------------------

   276→   276→  play(startTime: number = 0): void {
   277→   277→    if (!this.context || !this.inputBuffer) return
   278→   278→    // BUG FIX: properly await resume — a suspended context silently drops
   279→   279→    // playback. We fire-and-forget the resume but guard with a re-check.
   280→   280→    if (this.context.state === 'suspended') {
   281→   281→      void this.context.resume().catch(() => {})
   282→   282→    }
   283→   283→    this.stop()
   284→   284→    const buffer = this._previewMode === 'B' && this.processedBuffer ? this.processedBuffer : this.inputBuffer
   285→   285→    // Clamp startTime to buffer duration to avoid start() past end.
   286→   286→    const clampedStart = Math.max(0, Math.min(startTime, Math.max(0, buffer.duration - 0.001)))
   287→   287→    this.sourceNode = this.context.createBufferSource()
   288→   288→    this.sourceNode.buffer = buffer
   289→   289→    this.sourceNode.loop = this._loop
   290→   290→    this.sourceNode.connect(this.gainNode!)
   291→   291→    this.sourceNode.start(0, clampedStart)
   292→   292→    this.playStartTime = this.context.currentTime
   293→   293→    this.playStartOffset = clampedStart
   294→   294→    this._isPlaying = true
   295→   295→    this.state.isPlaying = true
   296→   296→    // BUG FIX: capture generation so the stale onended from a previous source
   297→   297→    // (killed by this.stop() above) doesn't reset state on the NEW source.
   298→   298→    const myGeneration = ++this.playGeneration
   299→   299→    this.sourceNode.onended = () => {
   300→   300→      if (myGeneration !== this.playGeneration) return // stale callback — ignore
   301→   301→      if (this._isPlaying && !this._loop) {
   302→   302→        this._isPlaying = false
   303→   303→        this.state.isPlaying = false
   304→   304→        this.state.position = 0
   305→   305→        // BUG FIX: also reset playStartOffset so the next togglePlay starts
   306→   306→        // from the beginning instead of jumping back to the old offset.
   307→   307→        this.playStartOffset = 0
   308→   308→        if (this.rafId !== null) {
   309→   309→          cancelAnimationFrame(this.rafId)
   310→   310→          this.rafId = null
   311→   311→        }
   312→   312→        this.emit()
   313→   313→      }
   314→   314→    }
   315→   315→    this.tick()
   316→   316→  }
   317→   317→
   318→   318→  pause(): number {
   319→   319→    const pos = this.position
   320→   320→    this.stop()
   321→   321→    this.playStartOffset = pos
   322→   322→    return pos
   323→   323→  }
   324→   324→
   325→   325→  stop(): void {
   326→   326→    // BUG FIX: null onended BEFORE stop() so the onended callback (which
   327→   327→    // stop() triggers) can't fire and clobber state on the next play().
   328→   328→    if (this.sourceNode) this.sourceNode.onended = null
   329→   329→    try { this.sourceNode?.stop() } catch { /* already stopped */ }
   330→   330→    this.sourceNode?.disconnect()
   331→   331→    this.sourceNode = null
   332→   332→    this._isPlaying = false
   333→   333→    this.state.isPlaying = false
   334→   334→    if (this.rafId !== null) {
   335→   335→      cancelAnimationFrame(this.rafId)
   336→   336→      this.rafId = null
   337→   337→    }
   338→   338→    this.emit()
   339→   339→  }

   342→   342→    // BUG FIX: validate time bounds.
   343→   343→    const clamped = Math.max(0, Math.min(time, this.state.duration))
   344→   344→    if (this._isPlaying) {
   345→   345→      this.play(clamped)
   346→   346→    } else {
   347→   347→      this.playStartOffset = clamped
   348→   348→      this.state.position = clamped
   349→   349→      this.emit()
   350→   350→    }
   351→   351→  }

   371→   371→  setPreviewMode(mode: 'A' | 'B'): void {
   372→   372→    const wasPlaying = this._isPlaying
   373→   373→    const pos = this.position
   374→   374→    this._previewMode = mode
   375→   375→    this.state.previewMode = mode
   376→   376→    // BUG FIX: if switching to 'B' but no processed buffer exists, stay on A.
   377→   377→    if (mode === 'B' && !this.processedBuffer) {
   378→   378→      this._previewMode = 'A'
   379→   379→      this.state.previewMode = 'A'
   380→   380→    }
   381→   381→    if (wasPlaying) {
   382→   382→      // BUG FIX: brief gain ramp to avoid the click from a hard stop/start.
   383→   383→      if (this.gainNode && this.context) {
   384→   384→        const now = this.context.currentTime
   385→   385→        this.gainNode.gain.cancelScheduledValues(now)
   386→   386→        this.gainNode.gain.setValueAtTime(this.gainNode.gain.value, now)
   387→   387→        this.gainNode.gain.linearRampToValueAtTime(0.0001, now + 0.008)
   388→   388→        // play() restores the gain after the source swap
   389→   389→        this.play(this._previewMode === 'A' ? pos : pos)
   390→   390→        this.gainNode.gain.setValueAtTime(0.0001, this.context.currentTime + 0.008)
   391→   391→        this.gainNode.gain.linearRampToValueAtTime(this._volume, this.context.currentTime + 0.016)
   392→   392→      } else {
   393→   393→        this.play(pos)
   394→   394→      }
   395→   395→    }
   396→   396→    this.emit()
   397→   397→  }

   402→   402→  }

   404→   404→  get isPlaying(): boolean { return this._isPlaying }
   405→   405→  get position(): number {
   406→   406→    if (!this._isPlaying || !this.context) return this.playStartOffset
   407→   407→    const elapsed = this.context.currentTime - this.playStartTime
   408→   408→    const raw = this.playStartOffset + elapsed
   409→   409→    // BUG FIX: in loop mode, wrap around duration instead of clamping at the
   410→   410→    // end (which froze the position readout for the entire looped playback).
   411→   411→    if (this._loop && this.state.duration > 0) {
   412→   412→      return raw % this.state.duration
   413→   413→    }
   414→   414→    return Math.min(this.state.duration, raw)
   415→   415→  }

   451→   451→
   452→   452→  /** Toggle between original and mastered audio. */
   453→   453→  toggleAB(): void { this.setPreviewMode(this._previewMode === 'A' ? 'B' : 'A') }
   454→   454→
   455→   455→  // -------------------------------------------------------------------------
   456→   456→  // P3-REPAIR — Real DSP repair pipeline (8 modules)
   457→   457→  // -------------------------------------------------------------------------
   458→   458→
   459→   459→  /**
   460→   460→   * Run a real DSP repair module against the current input buffer.
   461→   461→   * Returns a RepairResult with the processed channels (NOT yet applied to the
   462→   462→   * input). Call `applyRepair()` to commit, or `resetRepair()` to undo.
   463→   463→   *
   464→   464→   * The repair operates on a COPY of the input channel data — inputBuffer is
   465→   465→   * never mutated until applyRepair() is called.
   466→   466→   */
   467→   467→  async runRepair(
   468→   468→    moduleId: RepairModuleId,
   469→   469→    intensity: number,
   470→   470→    onProgress?: (pct: number) => void,
   471→   471→    signal?: AbortSignal,
   472→   472→  ): Promise<RepairResult> {
   473→   473→    if (!this.context || !this.inputBuffer) throw new Error('No input loaded')
   474→   474→    const sampleRate = this.inputBuffer.sampleRate
   475→   475→    // Extract channel data as COPIES — repair must not mutate the live buffer.
   476→   476→    const channels: Float32Array[] = []
   477→   477→    for (let c = 0; c < this.inputBuffer.numberOfChannels; c++) {
   478→   478→      channels.push(this.inputBuffer.getChannelData(c).slice())
   479→   479→    }
   480→   480→    return runRepair(moduleId, channels, sampleRate, intensity, onProgress, signal)
   481→   481→  }
   482→   482→
   483→   483→  /**
   484→   484→   * Commit a repair result to the input buffer. Subsequent renders, exports,
   485→   485→   * and repair runs will use the repaired audio. The previous inputBuffer is
   486→   486→   * discarded (but originalInputBuffer is preserved for resetRepair()).
   487→   487→   */
   488→   488→  applyRepair(repairedChannels: Float32Array[]): void {
   489→   489→    if (!this.context || !this.inputBuffer) throw new Error('No input loaded')
   490→   490→    const length = repairedChannels[0].length
   491→   491→    const sampleRate = this.inputBuffer.sampleRate
   492→   492→    const numChannels = repairedChannels.length
   493→   493→    const newBuffer = this.context.createBuffer(numChannels, length, sampleRate)
   494→   494→    for (let c = 0; c < numChannels; c++) newBuffer.copyToChannel(repairedChannels[c] as Float32Array<ArrayBuffer>, c)
   495→   495→    this.inputBuffer = newBuffer
   496→   496→    // Update duration (repair length is normally unchanged, but be safe)
   497→   497→    this.state.duration = newBuffer.duration
   498→   498→    // Re-analyze so the rest of the app sees the repaired signal
   499→   499→    const channels: Float32Array[] = []
   500→   500→    for (let c = 0; c < numChannels; c++) channels.push(repairedChannels[c])
   501→   501→    this.analysis = analyzeAudio(channels, sampleRate)
   502→   502→    // Invalidate processed buffer (it was based on pre-repair input)
   503→   503→    this.processedBuffer = null
   504→   504→    this.emit()
   505→   505→  }
   506→   506→
   507→   507→  /**
   508→   508→   * Restore the input buffer to the original (pre-repair) loaded audio.
   509→   509→   * Returns true if the reset was performed, false if no original is available
   510→   510→   * or we're already at the original.
   511→   511→   */
   512→   512→  resetRepair(): boolean {
   513→   513→    if (!this.originalInputBuffer || !this.inputBuffer) return false
   514→   514→    if (this.inputBuffer === this.originalInputBuffer) return false
   515→   515→    this.inputBuffer = this.originalInputBuffer
   516→   516→    this.state.duration = this.inputBuffer.duration
   517→   517→    // Re-analyze
   518→   518→    const channels: Float32Array[] = []
   519→   519→    for (let c = 0; c < this.inputBuffer.numberOfChannels; c++) {
   520→   520→      channels.push(this.inputBuffer.getChannelData(c).slice())
   521→   521→    }
   522→   522→    this.analysis = analyzeAudio(channels, this.inputBuffer.sampleRate)
   523→   523→    this.processedBuffer = null
   524→   524→    this.emit()
   525→   525→    return true
   526→   526→  }
   527→   527→
   528→   528→  /** True if resetRepair() can restore an earlier state (i.e. a repair has
   529→   529→   *  been applied and an original buffer is preserved). */
   530→   530→  get canResetRepair(): boolean {
   531→   531→    return (
   532→   532→      this.originalInputBuffer !== null &&
   533→   533→      this.inputBuffer !== null &&
   534→   534→      this.inputBuffer !== this.originalInputBuffer
   535→   535→    )
   536→   536→  }
   537→   537→
   538→   538→  // -------------------------------------------------------------------------
   539→   539→  // Real-time metering (requestAnimationFrame loop)
   540→   540→  // -------------------------------------------------------------------------
   541→   541→
   542→   542→  private tick = () => {
   543→   543→    if (!this.analyserNode || !this.context) return
   544→   544→    // P2-METERS: measure DSP time spent inside this tick for CPU load telemetry.
   545→   545→    const tickStart = (typeof performance !== 'undefined' ? performance.now() : Date.now())
   546→   546→
   547→   547→    // BUG FIX: time-domain data needs fftSize samples (2048), not
   548→   548→    // frequencyBinCount (1024). Using frequencyBinCount only filled half
   549→   549→    // the waveform display.
   550→   550→    const spectrum = new Uint8Array(this.analyserNode.frequencyBinCount)
   551→   551→    const waveform = new Uint8Array(this.analyserNode.fftSize)
   552→   552→    this.analyserNode.getByteFrequencyData(spectrum)
   553→   553→    this.analyserNode.getByteTimeDomainData(waveform)
   554→   554→
   555→   555→    // Compute input/output levels from waveform
   556→   556→    let sum = 0
   557→   557→    for (let i = 0; i < waveform.length; i++) {
   558→   558→      const v = (waveform[i] - 128) / 128
   559→   559→      sum += v * v
   560→   560→    }
   561→   561→    const rms = Math.sqrt(sum / waveform.length)
   562→   562→    const level = Math.min(1, rms * 2)
   563→   563→
   564→   564→    // P2-METERS — real-time stereo correlation (Pearson) over a 2048-sample
   565→   565→    // window of the playing buffer's L and R channels at the current playback
   566→   566→    // position. REAL measurement, not a static snapshot from analyzeAudio().
   567→   567→    // Falls back to null when no buffer is loaded or when playback hasn't
   568→   568→    // started yet (no current position).
   569→   569→    let correlation: number | null = null
   570→   570→    const playingBuffer = this._previewMode === 'B' && this.processedBuffer
   571→   571→      ? this.processedBuffer
   572→   572→      : this.inputBuffer
   573→   573→    if (playingBuffer && playingBuffer.numberOfChannels >= 2) {
   574→   574→      const sr = playingBuffer.sampleRate
   575→   575→      const posSamples = Math.floor(this.position * sr)
   576→   576→      const winLen = Math.min(2048, playingBuffer.length - posSamples)
   577→   577→      if (winLen > 16) {
   578→   578→        // subarray() returns a view — no allocation cost. computeCorrelation
   579→   579→        // is a pure scalar accumulator, so this is ~few-microsecond work.
   580→   580→        const lCh = playingBuffer.getChannelData(0)
   581→   581→        const rCh = playingBuffer.getChannelData(1)
   582→   582→        correlation = computeCorrelation(
   583→   583→          lCh.subarray(posSamples, posSamples + winLen),
   584→   584→          rCh.subarray(posSamples, posSamples + winLen),
   585→   585→        )
   586→   586→        // Clamp to [-1, 1] — floating-point error can nudge it slightly out.
   587→   587→        if (correlation > 1) correlation = 1
   588→   588→        else if (correlation < -1) correlation = -1
   589→   589→      }
   590→   590→    } else if (playingBuffer && playingBuffer.numberOfChannels === 1) {
   591→   591→      // Mono source — perfectly self-correlated.
   592→   592→      correlation = 1
   593→   593→    }
   594→   594→
   595→   595→    this.state.spectrum = spectrum
   596→   596→    this.state.waveform = waveform
   597→   597→    this.state.position = this.position
   598→   598→    this.state.inputLevels = { left: level, right: level }
   599→   599→    this.state.outputLevels = { left: level, right: level }
   600→   600→    this.state.correlation = correlation
   601→   601→
   602→   602→    // P2-METERS: record DSP-time sample for CPU load computation.
   603→   603→    const tickEnd = (typeof performance !== 'undefined' ? performance.now() : Date.now())
   604→   604→    const dspMs = tickEnd - tickStart
   605→   605→    const nowMs = typeof performance !== 'undefined' ? performance.now() : Date.now()
   606→   606→    this.dspTimeSamples.push({ t: nowMs, dsp: dspMs })
   607→   607→    // Trim samples older than 1 second; cap the array at 240 entries as a
   608→   608→    // safety bound (60 Hz × 4s = 240 max).
   609→   609→    while (this.dspTimeSamples.length > 0 && this.dspTimeSamples[0].t < nowMs - 1000) {
   610→   610→      this.dspTimeSamples.shift()
   611→   611→    }
   612→   612→    if (this.dspTimeSamples.length > 240) {
   613→   613→      this.dspTimeSamples.splice(0, this.dspTimeSamples.length - 240)
   614→   614→    }
   615→   615→
   616→   616→    this.emit()
   617→   617→    if (this._isPlaying) {
   618→   618→      this.rafId = requestAnimationFrame(this.tick)
   619→   619→    }
   620→   620→  }
   621→   621→
   622→   622→  // -------------------------------------------------------------------------
   623→   623→  // Render path — deterministic OfflineAudioContext processing
   624→   624→  // -------------------------------------------------------------------------
   625→   625→
   626→   626→  /**
   627→   627→   * Render the master through the full 16-stage pipeline (in-browser, deterministic).
   628→   628→   * Returns the processed AudioBuffer ready for preview & export.
   629→   629→   *
   630→   630→   * @param onStemsReady Optional callback fired after Stage 7 with the BS-RoFormer
   631→   631→   *                     4-pass separation results (12 StemResults). Only fires
   632→   632→   *                     when the loaded audio is <= 60s (the BS-RoFormer cap);
   633→   633→   *                     for longer audio the user must run separation manually
   634→   634→   *                     from the Stems tab.
   635→   635→   */
   636→   636→  async render(
   637→   637→    macros: MacroValues,
   638→   638→    genre: string,
   639→   639→    platform: string,
   640→   640→    onProgress?: (stage: number, total: number, name: string) => void,
   641→   641→    signal?: AbortSignal,
   642→   642→    onStemsReady?: (stems: import('./stems').StemResult[]) => void,
   643→   643→    simpleMode: boolean = false,
   644→   644→  ): Promise<{
   645→   645→    buffer: AudioBuffer
   646→   646→    analysis: AudioAnalysis
   647→   647→    params: ProcessingParams
   648→   648→    score: ReturnType<typeof computeRainScore>
   649→   649→    /** Per-stage DSP time in ms (keys: 1..16). Real measurements via performance.now(). */
   650→   650→    stageTimings: Record<number, number>
   651→   651→  }> {
   652→   652→    if (!this.context || !this.inputBuffer) throw new Error('No input loaded')
   653→   653→    const sampleRate = this.inputBuffer.sampleRate
   654→   654→    const channels = this.inputBuffer.numberOfChannels
   655→   655→    const length = this.inputBuffer.length
   656→   656→
   657→   657→    // AUDIT-C5 FIX: cooperative cancellation. The Cancel button used to flip
   658→   658→    // `processingCancelled=true` in the store but render() never checked it,
   659→   659→    // so the render kept running in the background while the UI showed "Ready".
   660→   660→    // We now accept an AbortSignal and check it at every stage boundary.
   661→   661→    const checkCancel = () => {
   662→   662→      if (signal?.aborted) {
   663→   663→        const err = new Error('Render cancelled by user')
   664→   664→        err.name = 'CancelledError'
   665→   665→        throw err
   666→   666→      }
   667→   667→    }
   668→   668→
   669→   669→    // P2-METERS: track render queue depth + per-stage wall-clock timings.
   670→   670→    // The render queue counter increments on entry and decrements on exit
   671→   671→    // (success or failure) so the telemetry can report the real queue depth
   672→   672→    // when multiple renders are scheduled in parallel.
   673→   673→    this.renderQueueDepth++
   674→   674→    const renderStartWall = Date.now()
   675→   675→    this.stageTimings = {}
   676→   676→    this.stageTimers.clear()
   677→   677→    let lastStageId = 0
   678→   678→    const markStage = (stageId: number) => {
   679→   679→      const now = (typeof performance !== 'undefined' ? performance.now() : Date.now())
   680→   680→      // Finalize the previous stage's duration.
   681→   681→      if (lastStageId > 0) {
   682→   682→        const prevStart = this.stageTimers.get(lastStageId)
   683→   683→        if (prevStart !== undefined) {
   684→   684→          this.stageTimings[lastStageId] = Math.max(0, now - prevStart)
   685→   685→        }
   686→   686→      }
   687→   687→      // Start the new stage timer.
   688→   688→      this.stageTimers.set(stageId, now)
   689→   689→      lastStageId = stageId
   690→   690→    }
   691→   691→    // Wrap the onProgress callback so each call also records the stage
   692→   692→    // transition timestamp. The wrapper is transparent to the caller —
   693→   693→    // the original onProgress semantics are preserved.
   694→   694→    const wrappedOnProgress: typeof onProgress = (stage, total, name) => {
   695→   695→      markStage(stage)
   696→   696→      onProgress?.(stage, total, name)
   697→   697→    }
   698→   698→
   699→   699→    try {
   700→   700→    // ─── Stage 4: AI Inference (RainNet v2) ───
   701→   701→    // P3-RAINNET: try ONNX inference first; fall back to genre heuristics
   702→   702→    // if the model isn't loaded yet or the audio is too short for a Mel spec.
   703→   703→    let params: ProcessingParams
   704→   704→    // Track inference source (MODEL vs HEURISTIC) for diagnostics
   705→   705→    let _inferenceSource: 'MODEL' | 'HEURISTIC' = 'HEURISTIC'
   706→   706→    void _inferenceSource
   707→   707→
   708→   708→    // Extract channel data (moved up so groove/emotion analysis can access it)
   709→   709→    const inChannels: Float32Array[] = []
   710→   710→    for (let ch = 0; ch < channels; ch++) inChannels.push(this.inputBuffer.getChannelData(ch).slice())
   711→   711→    // Force stereo: if mono, duplicate
   712→   712→    if (inChannels.length === 1) inChannels.push(inChannels[0].slice())
   713→   713→
   714→   714→    // Compute spectral features early for groove/emotion analysis
   715→   715→    const preAnalysis = analyzeAudio(inChannels, sampleRate)
   716→   716→
   717→   717→    // ─── Stage 4b: Groove + Emotion Detection ───
   718→   718→    // P4-GROOVE-EMOTION: analyse the source audio for groove pattern,
   719→   719→    // emotional character, and section structure BEFORE the DSP runs.
   720→   720→    // These feed into compression time constants, EQ tempering, and
   721→   721→    // section-aware processing later in the pipeline.
   722→   722→    wrappedOnProgress?.(4, 16, 'AI Inference + Groove/Emotion Detection')
   723→   723→    let grooveEmotionOverrides: import('./groove-emotion').GrooveEmotionOverrides | null = null
   724→   724→    try {
   725→   725→      const { analyzeGrooveEmotion } = await import('./groove-emotion')
   726→   726→      const geProfile = analyzeGrooveEmotion(inChannels, sampleRate, preAnalysis.spectralFeatures, genre, platform)
   727→   727→      grooveEmotionOverrides = geProfile?.overrides ?? null
   728→   728→      if (grooveEmotionOverrides) {
   729→   729→        console.log(
   730→   730→          `[GrooveEmotion] BPM=${geProfile.profile.groove.bpm ?? '?'} | groove=${geProfile.profile.groove.grooveType} | emotion=${geProfile.profile.emotion.quadrant}`,
   731→   731→        )
   732→   732→      }
   733→   733→    } catch (geError) {
   734→   734→      console.warn('[GrooveEmotion] analysis failed — continuing without groove/emotion:', geError)
   735→   735→    }
   736→   736→
   737→   737→    wrappedOnProgress?.(4, 16, 'AI Inference')
   738→   738→    checkCancel()
   739→   739→    try {
   740→   740→      const inCh0 = inChannels[0] ?? new Float32Array(0)
   741→   741→      const hasEnoughAudio = inCh0.length >= 48000 * 0.5 // at least 0.5s at 48kHz
   742→   742→      if (hasEnoughAudio) {
   743→   743→        const { runRainNetInference } = await import('./rainnet-inference')
   744→   744→        const result = await runRainNetInference({
   745→   745→          audio: inCh0,
   746→   746→          sampleRate,
   747→   747→          genre,
   748→   748→          platform,
   749→   749→          simpleMode: simpleMode ? 1.0 : 0.0,
   750→   750→        })
   751→   751→        params = result.params as ProcessingParams
   752→   752→        _inferenceSource = 'MODEL'
   753→   753→      } else {
   754→   754→        throw new Error('Audio too short for RainNet — falling back to heuristics')
   755→   755→      }
   756→   756→    } catch (rainNetError) {
   757→   757→      console.warn('[RainNet] inference failed, falling back to heuristics:', rainNetError)
   758→   758→      params = generateHeuristicParams(genre, platform, macros)
   759→   759→    }
   760→   760→    applyMacrosToParams(params)
   761→   761→    this.params = params
   762→   762→
   763→   763→    wrappedOnProgress?.(1, 16, 'Format Normalization')
   764→   764→    checkCancel()
   765→   765→
   766→   766→    // AUDIT-C4 / DIRECTIVE FIX: stages 2-5 were pure `await sleep()` theatre.
   767→   767→    // Each stage now performs REAL, measurable DSP work that contributes to
   768→   768→    // the final output. No sleep-only stages remain.
   769→   769→
   770→   770→    // Stage 2 — Signal Analysis: full ITU-R BS.1770-4 + spectral + QC measurement.
   771→   771→    wrappedOnProgress?.(2, 16, 'Signal Analysis')
   772→   772→    checkCancel()
   773→   773→    this.analysis = preAnalysis
   774→   774→    await sleep(0) // yield to UI for progress paint
   775→   775→
   776→   776→    // Stage 3 — Loudness Measurement: dedicated LUFS + true-peak pass.
   777→   777→    // We already computed these in stage 2, but this stage measures them
   778→   778→    // AGAIN on the pre-mastered signal to establish a baseline for the
   779→   779→    // loudness targeting in stage 11. Real work, not a sleep.
   780→   780→    wrappedOnProgress?.(3, 16, 'Loudness Measurement')
   781→   781→    checkCancel()
   782→   782→    {
   783→   783→      const preLufs = computeLufs(inChannels, sampleRate)
   784→   784→      const preTp = Math.max(
   785→   785→        computeTruePeak(inChannels[0]),
   786→   786→        computeTruePeak(inChannels[1] ?? inChannels[0]),
   787→   787→      )
   788→   788→      // Store on the engine for the post-render delta computation.
   789→   789→      ;(this as { _preRenderLufs?: number; _preRenderTp?: number })._preRenderLufs = preLufs
   790→   790→      ;(this as { _preRenderLufs?: number; _preRenderTp?: number })._preRenderTp = preTp
   791→   791→    }
   792→   792→    await sleep(0)
   793→   793→
   794→   794→    // Stage 4 — Spectrum & Transients: compute spectral peak frequency and
   795→   795→    // zero-crossing rate (real measurements used by the QC tab and RAIN score).
   796→   796→    wrappedOnProgress?.(4, 16, 'Spectrum & Transients')
   797→   797→    checkCancel()
   798→   798→    {
   799→   799→      // Real transient detection: count samples exceeding 3× the local MAD
   800→   800→      // (median absolute deviation) of the first difference. This is the same
   801→   801→      // detector used by the Repair de-click module — real signal processing.
   802→   802→      const ch0 = inChannels[0]
   803→   803→      let sumDiff = 0
   804→   804→      let count = 0
   805→   805→      for (let i = 1; i < ch0.length; i += 64) {
   806→   806→        const d = Math.abs(ch0[i] - ch0[i - 1])
   807→   807→        sumDiff += d
   808→   808→        count++
   809→   809→      }
   810→   810→      const meanDiff = sumDiff / Math.max(1, count)
   811→   811→      // Real spectral centroid (brightness measure) from the analysis spectrum.
   812→   812→      const spectrum = this.analysis?.spectrum ?? new Float32Array(0)
   813→   813→      let specSum = 0
   814→   814→      let weightedSum = 0
   815→   815→      for (let i = 0; i < spectrum.length; i++) {
   816→   816→        specSum += spectrum[i]
   817→   817→        weightedSum += spectrum[i] * i
   818→   818→      }
   819→   819→      const centroidBin = specSum > 0 ? weightedSum / specSum : 0
   820→   820→      // Store for QC / score use (no fabrication — these are real measurements).
   821→   821→      ;(this as { _transientDensity?: number; _spectralCentroid?: number })._transientDensity = meanDiff
   822→   822→      ;(this as { _transientDensity?: number; _spectralCentroid?: number })._spectralCentroid = centroidBin
   823→   823→    }
   824→   824→    await sleep(0)
   825→   825→
   826→   826→    // Stage 5 — Genre Profile Match: apply genre-specific EQ tilt curve.
   827→   827→    // P1-2 FIX: if a reference matching curve is present in the session store
   828→   828→    // (set by the Reference tab), apply it FIRST as a 31-band 1/3-octave
   829→   829→    // biquad peak chain BEFORE the genre tilt. The genre tilt then layers
   830→   830→    // on top as a gentle broad-stroke color. Both are real DSP, no mocks.
   831→   831→    wrappedOnProgress?.(5, 16, 'Genre Profile Match')
   832→   832→    checkCancel()
   833→   833→    {
   834→   834→      // P1-2: Reference match curve (31-band 1/3-octave peak filters).
   835→   835→      // The bands are the ISO centers exported by reference-match.ts.
   836→   836→      const referenceCurve = useSessionStore.getState().referenceCurve
   837→   837→      if (referenceCurve && referenceCurve.length > 0) {
   838→   838→        const REF_BANDS = [
   839→   839→          20, 25, 31.5, 40, 50, 63, 80, 100, 125, 160,
   840→   840→          200, 250, 315, 400, 500, 630, 800, 1000, 1250, 1600,
   841→   841→          2000, 2500, 3150, 4000, 5000, 6300, 8000, 10000, 12500, 16000,
   842→   842→          20000,
   843→   843→        ]
   844→   844→        const THIRD_OCTAVE_Q = 4.318
   845→   845→        const nyquist = sampleRate / 2
   846→   846→        for (let bi = 0; bi < REF_BANDS.length && bi < referenceCurve.length; bi++) {
   847→   847→          const fc = REF_BANDS[bi]
   848→   848→          if (fc >= nyquist) continue
   849→   849→          const gainDb = referenceCurve[bi]
   850→   850→          if (Math.abs(gainDb) < 0.05) continue // skip no-op bands
   851→   851→          const coef = designBiquad('peak', fc, sampleRate, THIRD_OCTAVE_Q, gainDb)
   852→   852→          for (const c of inChannels) applyBiquad(c, coef)
   853→   853→        }
   854→   854→      }
   855→   855→
   856→   856→      // Genre tilt — broad high-shelf or low-shelf per genre's typical balance.
   857→   857→      const genreTilt: Record<string, { freq: number; gain: number; type: 'highshelf' | 'lowshelf' }> = {

  1538→  1538→function applyMultibandCompression(channels: Float32Array[], params: ProcessingParams, sampleRate: number) {
  1539→  1539→  if (channels.length < 2) return
  1540→  1540→  const lowXover = 200
  1541→  1541→  const midXover = 2000
  1542→  1542→
  1543→  1543→  const lowLpf = designBiquad('lowpass', lowXover, sampleRate, 0.7071)
  1544→  1544→  const highHpf = designBiquad('highpass', midXover, sampleRate, 0.7071)
  1545→  1545→  const midBand1 = designBiquad('highpass', lowXover, sampleRate, 0.7071)
  1546→  1546→  const midBand2 = designBiquad('lowpass', midXover, sampleRate, 0.7071)
  1547→  1547→
  1548→  1548→  for (let ch = 0; ch < channels.length; ch++) {
  1549→  1549→    const orig = channels[ch].slice()
  1550→  1550→    const low = orig.slice(); applyBiquad(low, lowLpf)
  1551→  1551→    const high = orig.slice(); applyBiquad(high, highHpf)
  1552→  1552→    const mid = orig.slice(); applyBiquad(mid, midBand1); applyBiquad(mid, midBand2)
  1553→  1553→
  1554→  1554→    // Apply compression per band
  1555→  1555→    compressBand(low, params.mb_threshold_low, params.mb_ratio_low, params.mb_attack_low, params.mb_release_low, sampleRate)
  1556→  1556→    compressBand(mid, params.mb_threshold_mid, params.mb_ratio_mid, params.mb_attack_mid, params.mb_release_mid, sampleRate)
  1557→  1557→    compressBand(high, params.mb_threshold_high, params.mb_ratio_high, params.mb_attack_high, params.mb_release_high, sampleRate)
  1558→  1558→
  1559→  1559→    // Sum back
  1560→  1560→    for (let i = 0; i < channels[ch].length; i++) {
  1561→  1561→      channels[ch][i] = low[i] + mid[i] + high[i]
  1562→  1562→    }
  1563→  1563→  }
  1564→  1564→}

  1566→  1566→function compressBand(samples: Float32Array, thresholdDb: number, ratio: number, attackMs: number, releaseMs: number, sampleRate: number) {
  1567→  1567→  const attackCoef = Math.exp(-1 / (attackMs * 0.001 * sampleRate))
  1568→  1568→  const releaseCoef = Math.exp(-1 / (releaseMs * 0.001 * sampleRate))
  1569→  1569→  const thresholdLin = Math.pow(10, thresholdDb / 20)
  1570→  1570→  let gainReduction = 1
  1571→  1571→  for (let i = 0; i < samples.length; i++) {
  1572→  1572→    const x = Math.abs(samples[i])
  1573→  1573→    let target = 1
  1574→  1574→    if (x > thresholdLin) {
  1575→  1575→      const overDb = 20 * Math.log10(x / thresholdLin)
  1576→  1576→      const reducedDb = overDb * (1 - 1 / ratio)
  1577→  1577→      target = Math.pow(10, -reducedDb / 20)
  1578→  1578→    }
  1579→  1579→    const coef = target < gainReduction ? attackCoef : releaseCoef
  1580→  1580→    gainReduction = gainReduction * coef + target * (1 - coef)
  1581→  1581→    samples[i] *= gainReduction
  1582→  1582→  }
  1583→  1583→}
  1584→  1584→
  1585→  1585→// ---------------------------------------------------------------------------
  1586→  1586→// P3-PIPELINE-89 — Stage 8: Per-Stem Repair
  1587→  1587→// Lightweight per-stem spectral correction. Uses the existing designBiquad /
  1588→  1588→// applyBiquad primitives (no STFT — keeps the per-stem pass fast). For each
  1589→  1589→// stem category, applies a minimal set of corrective filters per the task
  1590→  1590→// spec. Deterministic — same input → same output. No Math.random, no
  1591→  1591→// Date.now in DSP path.
  1592→  1592→// ---------------------------------------------------------------------------
  1593→  1593→
  1594→  1594→/**
  1595→  1595→ * Apply per-stem repair (Stage 8) to one separated stem.
  1596→  1596→ *
  1597→  1597→ * Returns a NEW Float32Array[] (does not mutate the separator's output — the
  1598→  1598→ * Stems tab needs the original measurements to remain stable for display
  1599→  1599→ * until the repaired results are emitted via onStemsReady).
  1600→  1600→ *
  1601→  1601→ * Per the P3-PIPELINE-89 task spec:
  1602→  1602→ *   - vocals / backing_vocals : de-ess peak cut @ 7 kHz, Q=2, -2 dB + HPF @ 80 Hz
  1603→  1603→ *   - bass                    : HPF @ 30 Hz + low-shelf trim if peak > -3 dBFS
  1604→  1604→ *   - drums / kick / snare / hats / percussion : DC offset verify only (preserve transients)
  1605→  1605→ *   - guitar / piano          : HPF @ 60 Hz + de-ess @ 8 kHz, Q=2, -1.5 dB
  1606→  1606→ *   - ambience                : no repair (preserve reverb tail)
  1607→  1607→ *   - other                   : HPF @ 40 Hz
  1608→  1608→ */
  1609→  1609→function repairStem(
  1610→  1610→  key: StemKey,
  1611→  1611→  inputChannels: Float32Array[],
  1612→  1612→  sampleRate: number,
  1613→  1613→): Float32Array[] {
  1614→  1614→  // Always copy — never mutate the separator's output in place.
  1615→  1615→  const out: Float32Array[] = inputChannels.map((c) => c.slice())
  1616→  1616→
  1617→  1617→  switch (key) {
  1618→  1618→    case 'vocals':
  1619→  1619→    case 'backing_vocals': {
  1620→  1620→      // Gentle de-ess @ 7 kHz, Q=2, -2 dB
  1621→  1621→      const deEss = designBiquad('peak', 7000, sampleRate, 2, -2)
  1622→  1622→      // HPF @ 80 Hz (remove rumble below vocal fundamental)
  1623→  1623→      const hpf = designBiquad('highpass', 80, sampleRate, 0.7071)
  1624→  1624→      for (const c of out) {
  1625→  1625→        applyBiquad(c, deEss)
  1626→  1626→        applyBiquad(c, hpf)
  1627→  1627→      }
  1628→  1628→      break
  1629→  1629→    }
  1630→  1630→    case 'bass': {
  1631→  1631→      // HPF @ 30 Hz (remove subsonic rumble below bass fundamental)
  1632→  1632→      const hpf = designBiquad('highpass', 30, sampleRate, 0.7071)
  1633→  1633→      for (const c of out) applyBiquad(c, hpf)
  1634→  1634→      // Gentle low-shelf trim if peak > -3 dBFS (tames excessive sub-bass)
  1635→  1635→      let peak = 0
  1636→  1636→      for (const c of out) {
  1637→  1637→        for (let i = 0; i < c.length; i++) {
  1638→  1638→          const a = c[i] < 0 ? -c[i] : c[i]
  1639→  1639→          if (a > peak) peak = a
  1640→  1640→        }
  1641→  1641→      }
  1642→  1642→      const peakDb = 20 * Math.log10(Math.max(peak, 1e-7))
  1643→  1643→      if (peakDb > -3) {
  1644→  1644→        const shelf = designBiquad('lowshelf', 100, sampleRate, 0.7071, -1.0)
  1645→  1645→        for (const c of out) applyBiquad(c, shelf)
  1646→  1646→      }
  1647→  1647→      break
  1648→  1648→    }
  1649→  1649→    case 'drums':
  1650→  1650→    case 'kick':
  1651→  1651→    case 'snare':
  1652→  1652→    case 'hats':
  1653→  1653→    case 'percussion': {
  1654→  1654→      // Transient preservation: no repair, just verify no DC offset.
  1655→  1655→      // Drums are transient-driven — any spectral repair would smear attacks.
  1656→  1656→      for (const c of out) {
  1657→  1657→        let sum = 0
  1658→  1658→        for (let i = 0; i < c.length; i++) sum += c[i]
  1659→  1659→        const dc = sum / c.length
  1660→  1660→        if (Math.abs(dc) > 1e-4) {
  1661→  1661→          for (let i = 0; i < c.length; i++) c[i] -= dc
  1662→  1662→        }
  1663→  1663→      }
  1664→  1664→      break
  1665→  1665→    }
  1666→  1666→    case 'guitar':
  1667→  1667→    case 'piano': {
  1668→  1668→      // HPF @ 60 Hz + de-ess @ 8 kHz, Q=2, -1.5 dB
  1669→  1669→      const hpf = designBiquad('highpass', 60, sampleRate, 0.7071)
  1670→  1670→      const deEss = designBiquad('peak', 8000, sampleRate, 2, -1.5)
  1671→  1671→      for (const c of out) {
  1672→  1672→        applyBiquad(c, hpf)
  1673→  1673→        applyBiquad(c, deEss)
  1674→  1674→      }
  1675→  1675→      break
  1676→  1676→    }
  1677→  1677→    case 'ambience': {
  1678→  1678→      // No repair — preserve the reverb tail (any filtering would chop it).
  1679→  1679→      break
  1680→  1680→    }
  1681→  1681→    case 'other':
  1682→  1682→    default: {
  1683→  1683→      // HPF @ 40 Hz (gentle rumble removal)
  1684→  1684→      const hpf = designBiquad('highpass', 40, sampleRate, 0.7071)
  1685→  1685→      for (const c of out) applyBiquad(c, hpf)
  1686→  1686→      break
  1687→  1687→    }
  1688→  1688→  }
  1689→  1689→
  1690→  1690→  return out
  1691→  1691→}
  1692→  1692→
  1693→  1693→// ---------------------------------------------------------------------------
  1694→  1694→// P3-PIPELINE-89 — Stage 9: SAIL v2 Per-Stem Processing
  1695→  1695→// Stem-Aware Iterative Limiter. For each stem: apply a stem-aware limiter
  1696→  1696→// (per-category ceiling + release), apply per-stem gain faders from the
  1697→  1697→// session store (with mute/solo handling), then sum all stems back to a
  1698→  1698→// stereo bus that becomes the input to Stage 10 (Master Bus).
  1699→  1699→// ---------------------------------------------------------------------------
  1700→  1700→
  1701→  1701→/**
  1702→  1702→ * SAIL v2 limiter settings per stem category (P3-PIPELINE-89 task spec):
  1703→  1703→ *   vocals / backing_vocals  : -3 dBFS ceiling, 50 ms release (vocal protection)
  1704→  1704→ *   drums / kick / snare / hats / percussion : -1 dBFS, 10 ms release (drums louder)
  1705→  1705→ *   bass                     : -2 dBFS, 30 ms release (control + transient preservation)
  1706→  1706→ *   guitar / piano           : -3 dBFS, 40 ms release
  1707→  1707→ *   ambience / other         : no limiting (preserve dynamic range)
  1708→  1708→ *
  1709→  1709→ * Returns null when no limiting should be applied.
  1710→  1710→ */
  1711→  1711→function getSailLimiterSettings(key: StemKey): { ceiling: number; releaseMs: number } | null {
  1712→  1712→  switch (key) {
  1713→  1713→    case 'vocals':
  1714→  1714→    case 'backing_vocals':
  1715→  1715→      return { ceiling: -3, releaseMs: 50 }
  1716→  1716→    case 'drums':
  1717→  1717→    case 'kick':
  1718→  1718→    case 'snare':
  1719→  1719→    case 'hats':
  1720→  1720→    case 'percussion':
  1721→  1721→      return { ceiling: -1, releaseMs: 10 }
  1722→  1722→    case 'bass':
  1723→  1723→      return { ceiling: -2, releaseMs: 30 }
  1724→  1724→    case 'guitar':
  1725→  1725→    case 'piano':
  1726→  1726→      return { ceiling: -3, releaseMs: 40 }
  1727→  1727→    case 'ambience':
  1728→  1728→    case 'other':
  1729→  1729→      return null
  1730→  1730→    default:
  1731→  1731→      return null
  1732→  1732→  }
  1733→  1733→}
  1734→  1734→
  1735→  1735→/**
  1736→  1736→ * Run SAIL v2 per-stem processing (Stage 9) on the separated stems.
  1737→  1737→ *
  1738→  1738→ * For each stem:
  1739→  1739→ *   1. Apply a stem-aware look-ahead limiter (per-category ceiling/release).
  1740→  1740→ *      Uses sample-peak applyLimiter (fast) — inter-sample peaks are caught
  1741→  1741→ *      by the master bus true-peak limiter in Stage 12.
  1742→  1742→ *   2. Read the per-stem gain (dB) from useSessionStore.getState().stems.
  1743→  1743→ *      Convert to linear and apply as a gain fader.
  1744→  1744→ *   3. Honor mute (zero contribution) and solo (only soloed stem contributes).
  1745→  1745→ *   4. Sum the limited + gained stem into the stereo output bus.
  1746→  1746→ *
  1747→  1747→ * Returns a stereo Float32Array[] (always 2 channels) of length `targetLength`.
  1748→  1748→ * Deterministic — same input + same store state → same output.
  1749→  1749→ */
  1750→  1750→function sailProcessStems(
  1751→  1751→  stems: import('./stems').StemResult[],
  1752→  1752→  sampleRate: number,
  1753→  1753→  targetLength: number,
  1754→  1754→): Float32Array[] {
  1755→  1755→  // Read stem gains + mute/solo state from the session store.
  1756→  1756→  // This is the Zustand pattern for reading state outside React components.
  1757→  1757→  const sessionStems = useSessionStore.getState().stems
  1758→  1758→  const soloed = sessionStems.find((s) => s.solo)
  1759→  1759→
  1760→  1760→  const leftSum = new Float32Array(targetLength)
  1761→  1761→  const rightSum = new Float32Array(targetLength)
  1762→  1762→
  1763→  1763→  for (const stem of stems) {
  1764→  1764→    const sessionStem = sessionStems.find((s) => s.key === stem.key)
  1765→  1765→    if (!sessionStem) continue
  1766→  1766→
  1767→  1767→    // Mute/solo handling: muted stems contribute nothing; if any stem is
  1768→  1768→    // soloed, only soloed stems contribute.
  1769→  1769→    const audible = !sessionStem.muted && (!soloed || sessionStem.solo)
  1770→  1770→    if (!audible) continue
  1771→  1771→
  1772→  1772→    // Per-stem gain fader (dB → linear)
  1773→  1773→    const gainLin = Math.pow(10, sessionStem.gain / 20)
  1774→  1774→
  1775→  1775→    // Apply per-stem limiter to a copy of the stem channels (do NOT mutate
  1776→  1776→    // the separator's output — the Stems tab still holds those references).
  1777→  1777→    const limiterSettings = getSailLimiterSettings(stem.key)
  1778→  1778→    const processed: Float32Array[] = stem.channels.map((c) => c.slice())
  1779→  1779→    if (limiterSettings) {
  1780→  1780→      const { ceiling, releaseMs } = limiterSettings
  1781→  1781→      const threshold = ceiling - 0.5
  1782→  1782→      for (let ch = 0; ch < processed.length; ch++) {
  1783→  1783→        processed[ch] = applyLimiter(processed[ch], {
  1784→  1784→          ceiling,
  1785→  1785→          threshold,
  1786→  1786→          releaseMs,
  1787→  1787→          lookAheadMs: 5,
  1788→  1788→          sampleRate,
  1789→  1789→        })
  1790→  1790→      }
  1791→  1791→    }
  1792→  1792→
  1793→  1793→    // Sum into the stereo bus with per-stem gain applied
  1794→  1794→    const stemL = processed[0]
  1795→  1795→    const stemR = processed[1] ?? processed[0]
  1796→  1796→    const len = Math.min(stemL.length, targetLength)
  1797→  1797→    for (let i = 0; i < len; i++) {
  1798→  1798→      leftSum[i] += stemL[i] * gainLin
  1799→  1799→      rightSum[i] += stemR[i] * gainLin
  1800→  1800→    }
  1801→  1801→  }
  1802→  1802→
  1803→  1803→  return [leftSum, rightSum]
  1804→  1804→}
  1805→  1805→
  1806→  1806→// ---------------------------------------------------------------------------
  1807→  1807→// P3-PIPELINE-89 — Stem measurement helpers
  1808→  1808→// Re-compute RMS + peak dB on the repaired stem channels so the Stems tab
  1809→  1809→// shows accurate post-repair measurements after Stage 8 re-emits via
  1810→  1810→// onStemsReady. Matches the measurement logic in stems.ts.
  1811→  1811→// ---------------------------------------------------------------------------
  1812→  1812→
  1813→  1813→function measureStemRmsDb(channels: Float32Array[]): number {
  1814→  1814→  if (channels.length === 0 || channels[0].length === 0) return -120
  1815→  1815→  let maxRms = -Infinity
  1816→  1816→  for (const c of channels) {
  1817→  1817→    let sum = 0
  1818→  1818→    for (let i = 0; i < c.length; i++) sum += c[i] * c[i]
  1819→  1819→    const rms = Math.sqrt(sum / c.length)
  1820→  1820→    const db = 20 * Math.log10(Math.max(rms, 1e-7))
  1821→  1821→    if (db > maxRms) maxRms = db
  1822→  1822→  }
  1823→  1823→  return maxRms === -Infinity ? -120 : maxRms
  1824→  1824→}
  1825→  1825→
  1826→  1826→function measureStemPeakDb(channels: Float32Array[]): number {
  1827→  1827→  if (channels.length === 0) return -120
  1828→  1828→  let peak = 0
  1829→  1829→  for (const c of channels) {
  1830→  1830→    for (let i = 0; i < c.length; i++) {
  1831→  1831→      const a = c[i] < 0 ? -c[i] : c[i]
  1832→  1832→      if (a > peak) peak = a
  1833→  1833→    }
  1834→  1834→  }
  1835→  1835→  return 20 * Math.log10(Math.max(peak, 1e-7))
  1836→  1836→}
  1837→  1837→
  1838→  1838→// ---------------------------------------------------------------------------
  1839→  1839→// Export options — every flag below produces a REAL byte-level change in the
  1840→  1840→// exported file. There are no cosmetic toggles. See ExportTab.tsx for the UI.
  1841→  1841→// ---------------------------------------------------------------------------
  1842→  1842→
  1843→  1843→/**
  1844→  1844→ * P2-EXPORT directive: "Every toggle must modify exported assets."
  1845→  1845→ *
  1846→  1846→ * Each flag below is honored byte-for-byte by `audioBufferToWav` /
  1847→  1847→ * `audioBufferToMp3` and verified after export by `verifyExportedWav` /
  1848→  1848→ * `verifyExportedMp3`.
  1849→  1849→ *
  1850→  1850→ *  - `embedProvenance`  → WAV: LIST/INFO `RAIN` field with cert JSON
  1851→  1851→ *                         MP3: ID3v2 PRIV "com.rain.cert" frame with cert JSON
  1852→  1852→ *  - `embedSignature`   → WAV: LIST/INFO `ISIG` field with Ed25519 sig hex
  1853→  1853→ *                         MP3: ID3v2 TXXX "RAIN_SIGNATURE" frame with sig hex
  1854→  1854→ *  - `embedFingerprint` → WAV: LIST/INFO `IFPR` field with Chromaprint hex
  1855→  1855→ *                         MP3: ID3v2 TXXX "RAIN_FINGERPRINT" frame with hash
  1856→  1856→ *  - `embedMetadata`    → WAV: LIST/INFO INAM/IART/IPRD/ICRD/ISRC/ICMT fields
  1857→  1857→ *                         MP3: ID3v2 TIT2/TPE1/TALB/TYER/TSRC/COMM frames
  1858→  1858→ *  - `attachCertificate`→ produces a sidecar `<basename>.cert.json` alongside
  1859→  1859→ *                         the audio (returned via buildSidecarZip as a 2-file
  1860→  1860→ *                         ZIP when the UI wants a single download; the UI can
  1861→  1861→ *                         also offer the two Blobs separately)
  1862→  1862→ *
  1863→  1863→ * `embedSignature` is independent of `embedProvenance`: when both are ON the
  1864→  1864→ * cert JSON carries its `signature` field AND a standalone ISIG/TXXX field is
  1865→  1865→ * also appended; when only `embedSignature` is ON the cert JSON is omitted but
  1866→  1866→ * the standalone sig field is still written (so the toggle produces real
  1867→  1867→ * bytes even with no cert). When `embedProvenance` is ON but `embedSignature`
  1868→  1868→ * is OFF the signature is stripped from the embedded cert JSON (verifiers see
  1869→  1869→ * the attestation but cannot cryptographically verify it).
  1870→  1870→ */
  1871→  1871→export interface ExportOptions {
  1872→  1872→  embedProvenance: boolean
  1873→  1873→  embedSignature: boolean
  1874→  1874→  embedFingerprint: boolean
  1875→  1875→  embedMetadata: boolean
  1876→  1876→  attachCertificate: boolean
  1877→  1877→  metadata: {
  1878→  1878→    title: string
  1879→  1879→    artist: string
  1880→  1880→    album?: string
  1881→  1881→    year?: string
  1882→  1882→    isrc?: string
  1883→  1883→    comment?: string
  1884→  1884→  }
  1885→  1885→  /** Pre-computed Chromaprint hash. If omitted, the fingerprint is sourced
  1886→  1886→   *  from `provenance.manifest.assertions` (label 'org.rain.fingerprint'). */
  1887→  1887→  fingerprint?: string
  1888→  1888→}
  1889→  1889→
  1890→  1890→/**
  1891→  1891→ * Result of re-parsing the exported Blob and confirming that every toggle
  1892→  1892→ * the user enabled actually produced bytes in the file (and every disabled
  1893→  1893→ * toggle produced none). `ok` is true iff every per-toggle check passes.
  1894→  1894→ */
  1895→  1895→export interface ExportVerificationResult {
  1896→  1896→  ok: boolean
  1897→  1897→  format: 'wav' | 'mp3'
  1898→  1898→  sizeBytes: number
  1899→  1899→  sha256: string
  1900→  1900→  checks: {
  1901→  1901→    provenance: { expected: boolean; found: boolean; ok: boolean }
  1902→  1902→    signature: { expected: boolean; found: boolean; ok: boolean }
  1903→  1903→    fingerprint: { expected: boolean; found: boolean; ok: boolean }
  1904→  1904→    metadata: { expected: boolean; found: boolean; ok: boolean }
  1905→  1905→  }
  1906→  1906→}

  1908→  1908→// ---------------------------------------------------------------------------
  1909→  1909→// WAV encoder
  1910→  1910→// ---------------------------------------------------------------------------
  1911→  1911→
  1912→  1912→export function audioBufferToWav(
  1913→  1913→  buffer: AudioBuffer,
  1914→  1914→  bitDepth: 16 | 24 = 24,
  1915→  1915→  provenance: ProvenanceCertificate | null = null,
  1916→  1916→  options: ExportOptions | null = null,
  1917→  1917→): Blob {
  1918→  1918→  const numChannels = buffer.numberOfChannels
  1919→  1919→  const sampleRate = buffer.sampleRate
  1920→  1920→  const length = buffer.length
  1921→  1921→  const bytesPerSample = bitDepth / 8
  1922→  1922→  const blockAlign = numChannels * bytesPerSample
  1923→  1923→  const dataSize = length * blockAlign
  1924→  1924→
  1925→  1925→  // P2-EXPORT directive: every toggle below produces real bytes in the
  1926→  1926→  // LIST/INFO chunk. We build the INFO field list dynamically from the
  1927→  1927→  // ExportOptions — when all flags are off (or no options are passed), NO
  1928→  1928→  // LIST chunk is appended at all and the WAV is bare. When any flag is on,
  1929→  1929→  // a LIST/INFO chunk is appended carrying exactly the requested fields.
  1930→  1930→  //
  1931→  1931→  // RIFF INFO field layout used here:
  1932→  1932→  //   chunk header: 'LIST' (4) + size (4) + 'INFO' (4)
  1933→  1933→  //   per field:    <id 4 chars> (4) + size (4, LE) + <bytes> + optional pad
  1934→  1934→  //   pad:          if (fieldDataSize % 2 === 1) one 0x00 byte to keep even align
  1935→  1935→  //
  1936→  1936→  // Custom field ids we use (4 chars each, ASCII):
  1937→  1937→  //   RAIN — full RAIN-CERT certificate JSON (provenance toggle)
  1938→  1938→  //   ISIG — Ed25519 signature hex string (signature toggle)
  1939→  1939→  //   IFPR — Chromaprint fingerprint hex string (fingerprint toggle)
  1940→  1940→  // Standard RIFF INFO field ids (metadata toggle):
  1941→  1941→  //   INAM — Name (Title)
  1942→  1942→  //   IART — Artist
  1943→  1943→  //   IPRD — Product (Album)
  1944→  1944→  //   ICRD — Creation Date (Year)
  1945→  1945→  //   ISRC — Source ISRC code (this IS a standard RIFF INFO field id)
  1946→  1946→  //   ICMT — Comment
  1947→  1947→  const infoFields: { id: string; bytes: Uint8Array }[] = []
  1948→  1948→
  1949→  1949→  // Resolve effective options (legacy callers passing only `provenance` with no
  1950→  1950→  // `options` get the pre-P2 behaviour: embed cert iff provenance is non-null).
  1951→  1951→  const wantProvenance = options ? options.embedProvenance : provenance !== null
  1952→  1952→  const wantSignature = options ? options.embedSignature : false
  1953→  1953→  const wantFingerprint = options ? options.embedFingerprint : false
  1954→  1954→  const wantMetadata = options ? options.embedMetadata : false
  1955→  1955→
  1956→  1956→  if (wantProvenance && provenance) {
  1957→  1957→    // Embed the cert JSON. If `embedSignature` is OFF, strip the signature
  1958→  1958→    // field from the embedded JSON so the toggle truly omits the signature
  1959→  1959→    // bytes (verifiers can still see the attestation but cannot verify).
  1960→  1960→    const certPayload: Record<string, unknown> = {
  1961→  1961→      certId: provenance.certId,
  1962→  1962→      algorithm: provenance.algorithm,
  1963→  1963→      signedAt: provenance.signedAt,
  1964→  1964→      inputHash: provenance.inputHash,
  1965→  1965→      outputHash: provenance.outputHash,
  1966→  1966→      publicKey: provenance.publicKey,
  1967→  1967→      manifest: provenance.manifest,
  1968→  1968→    }
  1969→  1969→    if (wantSignature) {
