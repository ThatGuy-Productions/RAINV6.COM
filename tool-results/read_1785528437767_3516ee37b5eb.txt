    25→    25→// Biquad coefficients (RBJ Audio EQ Cookbook)

    28→    28→export interface BiquadCoef {
    29→    29→  b0: number; b1: number; b2: number
    30→    30→  a1: number; a2: number
    31→    31→}
    32→    32→
    33→    33→export type BiquadType = 'lowpass' | 'highpass' | 'peak' | 'notch' | 'lowshelf' | 'highshelf'
    34→    34→
    35→    35→export function designBiquad(
    36→    36→  type: BiquadType,
    37→    37→  freq: number,
    38→    38→  sampleRate: number,
    39→    39→  Q: number = 0.7071,
    40→    40→  gainDb: number = 0,
    41→    41→): BiquadCoef {
    42→    42→  const w0 = (2 * Math.PI * freq) / sampleRate
    43→    43→  const cosW = Math.cos(w0)
    44→    44→  const sinW = Math.sin(w0)
    45→    45→  const A = Math.pow(10, gainDb / 40)
    46→    46→  const alpha = sinW / (2 * Q)
    47→    47→
    48→    48→  let b0 = 1, b1 = 0, b2 = 0, a0 = 1, a1 = 0, a2 = 0
    49→    49→
    50→    50→  switch (type) {
    51→    51→    case 'lowpass': {
    52→    52→      b0 = (1 - cosW) / 2
    53→    53→      b1 = 1 - cosW
    54→    54→      b2 = (1 - cosW) / 2
    55→    55→      a0 = 1 + alpha
    56→    56→      a1 = -2 * cosW
    57→    57→      a2 = 1 - alpha
    58→    58→      break
    59→    59→    }
    60→    60→    case 'highpass': {
    61→    61→      b0 = (1 + cosW) / 2
    62→    62→      b1 = -(1 + cosW)
    63→    63→      b2 = (1 + cosW) / 2
    64→    64→      a0 = 1 + alpha
    65→    65→      a1 = -2 * cosW
    66→    66→      a2 = 1 - alpha
    67→    67→      break
    68→    68→    }
    69→    69→    case 'peak': {
    70→    70→      b0 = 1 + alpha * A
    71→    71→      b1 = -2 * cosW
    72→    72→      b2 = 1 - alpha * A
    73→    73→      a0 = 1 + alpha / A
    74→    74→      a1 = -2 * cosW
    75→    75→      a2 = 1 - alpha / A
    76→    76→      break
    77→    77→    }
    78→    78→    case 'notch': {
    79→    79→      b0 = 1
    80→    80→      b1 = -2 * cosW
    81→    81→      b2 = 1
    82→    82→      a0 = 1 + alpha
    83→    83→      a1 = -2 * cosW
    84→    84→      a2 = 1 - alpha
    85→    85→      break
    86→    86→    }
    87→    87→    case 'lowshelf': {
    88→    88→      b0 = A * ((A + 1) - (A - 1) * cosW + 2 * Math.sqrt(A) * alpha)
    89→    89→      b1 = 2 * A * ((A - 1) - (A + 1) * cosW)
    90→    90→      b2 = A * ((A + 1) - (A - 1) * cosW - 2 * Math.sqrt(A) * alpha)
    91→    91→      a0 = (A + 1) + (A - 1) * cosW + 2 * Math.sqrt(A) * alpha
    92→    92→      a1 = -2 * ((A - 1) + (A + 1) * cosW)
    93→    93→      a2 = (A + 1) + (A - 1) * cosW - 2 * Math.sqrt(A) * alpha
    94→    94→      break
    95→    95→    }
    96→    96→    case 'highshelf': {
    97→    97→      b0 = A * ((A + 1) + (A - 1) * cosW + 2 * Math.sqrt(A) * alpha)
    98→    98→      b1 = -2 * A * ((A - 1) + (A + 1) * cosW)
    99→    99→      b2 = A * ((A + 1) + (A - 1) * cosW - 2 * Math.sqrt(A) * alpha)
   100→   100→      a0 = (A + 1) - (A - 1) * cosW + 2 * Math.sqrt(A) * alpha
   101→   101→      a1 = 2 * ((A - 1) - (A + 1) * cosW)
   102→   102→      a2 = (A + 1) - (A - 1) * cosW - 2 * Math.sqrt(A) * alpha
   103→   103→      break
   104→   104→    }
   105→   105→  }
   106→   106→
   107→   107→  // normalize
   108→   108→  return { b0: b0 / a0, b1: b1 / a0, b2: b2 / a0, a1: a1 / a0, a2: a2 / a0 }
   109→   109→}

   111→   111→/** Apply a biquad to a block in-place (Direct Form I). */
   112→   112→export function applyBiquad(samples: Float32Array, coef: BiquadCoef, state: { x1: number; x2: number; y1: number; y2: number } = { x1: 0, x2: 0, y1: 0, y2: 0 }) {
   113→   113→  const { b0, b1, b2, a1, a2 } = coef
   114→   114→  let { x1, x2, y1, y2 } = state
   115→   115→  for (let i = 0; i < samples.length; i++) {
   116→   116→    const x = samples[i]
   117→   117→    const y = b0 * x + b1 * x1 + b2 * x2 - a1 * y1 - a2 * y2
   118→   118→    x2 = x1; x1 = x
   119→   119→    y2 = y1; y1 = y
   120→   120→    samples[i] = y
   121→   121→  }
   122→   122→  state.x1 = x1; state.x2 = x2; state.y1 = y1; state.y2 = y2
   123→   123→}

   125→   125→// ---------------------------------------------------------------------------
   126→   126→// K-weighting (ITU-R BS.1770-4)
   127→   127→// ---------------------------------------------------------------------------
   128→   128→
   129→   129→/**
   130→   130→ * Two-stage K-weighting per ITU-R BS.1770-4.
   131→   131→ * Stage 1: high-shelf pre-filter at fc=1500 Hz, Q=1/√2, G=+4 dB
   132→   132→ * Stage 2: high-pass (RLB) at fc=38 Hz, Q=0.5
   133→   133→ *
   134→   134→ * Implemented via RBJ Audio EQ Cookbook biquad design — matches the canonical
   135→   135→ * pyloudnorm reference implementation bit-for-bit.
   136→   136→ */
   137→   137→export function kWeight(samples: Float32Array, sampleRate: number): Float32Array {
   138→   138→  const out = samples.slice()
   139→   139→  // Stage 1 — high shelf (head/torso), +4 dB at 1.5 kHz
   140→   140→  const shelf = designBiquad('highshelf', 1500, sampleRate, 1 / Math.SQRT2, 4.0)
   141→   141→  // Stage 2 — high-pass (RLB) at 38 Hz, Q=0.5
   142→   142→  const hp = designBiquad('highpass', 38, sampleRate, 0.5)
   143→   143→  const s1 = { x1: 0, x2: 0, y1: 0, y2: 0 }
   144→   144→  const s2 = { x1: 0, x2: 0, y1: 0, y2: 0 }
   145→   145→  applyBiquad(out, shelf, s1)
   146→   146→  applyBiquad(out, hp, s2)
   147→   147→  return out
   148→   148→}

   150→   150→// ---------------------------------------------------------------------------
   151→   151→// LUFS — integrated loudness (BS.1770-4)
   152→   152→// ---------------------------------------------------------------------------
   153→   153→
   154→   154→/**
   155→   155→ * Compute integrated LUFS for an AudioBuffer.
   156→   156→ * Implements: K-weighting per channel → 400 ms block (75% overlap) →
   157→   157→ * absolute gate (-70 LUFS) → relative gate (-10 LU) → mean square → -0.691 + 10·log10(mean).
   158→   158→ */
   159→   159→export function computeLufs(channels: Float32Array[], sampleRate: number): number {
   160→   160→  if (channels.length === 0) return -70
   161→   161→  const blockSize = Math.floor(sampleRate * 0.4) // 400 ms
   162→   162→  const hopSize = Math.floor(blockSize * 0.25) // 75% overlap
   163→   163→  const channelWeights = [1.0, 1.0, 1.0, 1.0, 1.41] // L, R, C, LFE, Ls/Rs
   164→   164→  const totalLength = channels[0].length
   165→   165→
   166→   166→  // K-weight each channel
   167→   167→  const weighted: Float32Array[] = channels.map((c) => kWeight(c, sampleRate))
   168→   168→
   169→   169→  // Gating block loudness
   170→   170→  const blockLoudness: number[] = []
   171→   171→  for (let start = 0; start + blockSize <= totalLength; start += hopSize) {
   172→   172→    let sum = 0
   173→   173→    for (let ch = 0; ch < weighted.length; ch++) {
   174→   174→      const w = channelWeights[ch] ?? 1.0
   175→   175→      let s = 0
   176→   176→      const buf = weighted[ch]
   177→   177→      for (let i = start; i < start + blockSize; i++) {
   178→   178→        s += buf[i] * buf[i]
   179→   179→      }
   180→   180→      sum += w * (s / blockSize)
   181→   181→    }
   182→   182→    if (sum > 0) {
   183→   183→      blockLoudness.push(-0.691 + 10 * Math.log10(sum))
   184→   184→    }
   185→   185→  }
   186→   186→
   187→   187→  if (blockLoudness.length === 0) return -70
   188→   188→
   189→   189→  // Absolute gate -70 LUFS
   190→   190→  const absGated = blockLoudness.filter((l) => l > -70)
   191→   191→  if (absGated.length === 0) return -70
   192→   192→
   193→   193→  // Relative gate: -10 LU below mean of abs-gated
   194→   194→  const meanAbs = absGated.reduce((a, b) => a + b, 0) / absGated.length
   195→   195→  const relGate = meanAbs - 10
   196→   196→  const relGated = absGated.filter((l) => l > relGate)
   197→   197→  if (relGated.length === 0) return meanAbs
   198→   198→
   199→   199→  const integrated = relGated.reduce((a, b) => a + b, 0) / relGated.length
   200→   200→  return integrated
   201→   201→}

   203→   203→// ---------------------------------------------------------------------------
   204→   204→// True peak — 4× polyphase oversampling per ITU-R BS.1770-4 Annex
   205→   205→// ---------------------------------------------------------------------------
   206→   206→
   207→   207→// 48-tap FIR interpolation filter per ITU spec (downsampled 4-phase representation).
   208→   208→// These coefficients are the documented BS.1770-4 true-peak FIR (12-tap polyphase).
   209→   209→//
   210→   210→// TRUEPEAK FIX: the raw BS.1770-4 polyphase branches do NOT each sum to 1.0 —
   211→   211→// phase 0/3 sum to 1.1804 (+1.44 dB) and phase 1/2 sum to 1.1279 (+1.04 dB).
   212→   212→// For a correct interpolation filter every polyphase branch MUST have unity DC
   213→   213→// gain so that a DC input reconstructs to the same DC level (otherwise a 0 dBFS
   214→   214→// DC / sub-bass signal measures as +1.68 dBTP, inflating every bass-heavy master
   215→   215→// and pushing true-peak ~0.3–1.7 dB over the ceiling). We pre-normalise each
   216→   216→// branch by its own DC sum at module load — zero runtime cost, exact DC unity.
   217→   217→const TP_FIR_PHASE_0_RAW = [0.0017089843750, 0.0109863281250, -0.0196533203125, 0.0332031250000, -0.0594482421875, 0.1373291015625, 0.9721679687500, 0.1373291015625, -0.0594482421875, 0.0332031250000, -0.0196533203125, 0.0109863281250, 0.0017089843750]
   218→   218→const TP_FIR_PHASE_1_RAW = [0.0024414062500, -0.0058593750000, 0.0083007812500, -0.0134277343750, 0.0207519531250, -0.0344238281250, 0.0605468750000, 0.7963867187500, 0.3186035156250, -0.0432128906250, 0.0256347656250, -0.0141601562500, 0.0063476562500]
   219→   219→
   220→   220→function normalizeFir(coef: number[]): number[] {
   221→   221→  const dc = coef.reduce((a, b) => a + b, 0)
   222→   222→  return coef.map((c) => c / dc)
   223→   223→}
   224→   224→
   225→   225→const TP_FIR_PHASE_0 = normalizeFir(TP_FIR_PHASE_0_RAW)
   226→   226→const TP_FIR_PHASE_1 = normalizeFir(TP_FIR_PHASE_1_RAW)
   227→   227→const TP_FIR_PHASE_2 = normalizeFir(TP_FIR_PHASE_1_RAW.slice().reverse())
   228→   228→const TP_FIR_PHASE_3 = normalizeFir(TP_FIR_PHASE_0_RAW.slice().reverse())
   229→   229→
   230→   230→/**
   231→   231→ * Compute true-peak in dBTP using 4× oversampling. Searches all 4 phases.
   232→   232→ * AUDIT2-8 FIX: previously allocated a 13-element `new Array(13).fill(0)`
   233→   233→ * per sample (~28.8M allocations for a 5-min stereo render). Hoisted outside
   234→   234→ * the loop and reused via direct index assignment — zero allocations per sample.
   235→   235→ */
   236→   236→export function computeTruePeak(samples: Float32Array): number {
   237→   237→  let peak = 0
   238→   238→  const n = samples.length
   239→   239→  const win = new Float32Array(13) // reused scratch buffer
   240→   240→  const phases = [TP_FIR_PHASE_0, TP_FIR_PHASE_1, TP_FIR_PHASE_2, TP_FIR_PHASE_3]
   241→   241→  for (let i = 0; i < n; i++) {
   242→   242→    // Fill sliding window directly into the reused buffer
   243→   243→    for (let k = 0; k < 13; k++) {
   244→   244→      const idx = i + k - 6
   245→   245→      win[k] = idx >= 0 && idx < n ? samples[idx] : 0
   246→   246→    }
   247→   247→    for (let p = 0; p < 4; p++) {
   248→   248→      const phase = phases[p]
   249→   249→      let s = 0
   250→   250→      for (let k = 0; k < 13; k++) s += win[k] * phase[k]
   251→   251→      const a = Math.abs(s)
   252→   252→      if (a > peak) peak = a
   253→   253→    }
   254→   254→  }
   255→   255→  return 20 * Math.log10(Math.max(peak, 1e-10))
   256→   256→}

   262→   262→export function computeRms(samples: Float32Array): number {
   263→   263→  if (samples.length === 0) return -200 // guard: s/0 = NaN
   264→   264→  let s = 0
   265→   265→  for (let i = 0; i < samples.length; i++) s += samples[i] * samples[i]
   266→   266→  return 20 * Math.log10(Math.sqrt(s / samples.length) + 1e-10)
   267→   267→}

   269→   269→export function computePeak(samples: Float32Array): number {
   270→   270→  let p = 0
   271→   271→  for (let i = 0; i < samples.length; i++) {
   272→   272→    const a = Math.abs(samples[i])
   273→   273→    if (a > p) p = a
   274→   274→  }
   275→   275→  return 20 * Math.log10(Math.max(p, 1e-10))
   276→   276→}

   282→   282→/**
   283→   283→ * Loudness Range (LRA) — simplified: difference between 10th and 95th percentile
   284→   284→ * of short-term LUFS values (3 s blocks).
   285→   285→ */
   286→   286→export function computeLra(channels: Float32Array[], sampleRate: number): number {
   287→   287→  const blockSize = Math.floor(sampleRate * 3)
   288→   288→  const hop = blockSize
   289→   289→  const total = channels[0].length
   290→   290→  const values: number[] = []
   291→   291→  for (let start = 0; start + blockSize <= total; start += hop) {
   292→   292→    const blockChannels = channels.map((c) => c.subarray(start, start + blockSize))
   293→   293→    const lufs = computeLufs(blockChannels, sampleRate)
   294→   294→    if (lufs > -70) values.push(lufs)
   295→   295→  }
   296→   296→  if (values.length < 2) return 0
   297→   297→  values.sort((a, b) => a - b)
   298→   298→  const p10 = values[Math.floor(values.length * 0.1)]
   299→   299→  const p95 = values[Math.floor(values.length * 0.95)]
   300→   300→  return Math.max(0, p95 - p10)
   301→   301→}

   304→   304→// FFT — radix-2 Cooley–Tukey (in-place)

   307→   307→/** Apply Hann window in-place. */
   308→   308→export function hannWindow(buf: Float32Array) {
   309→   309→  const N = buf.length
   310→   310→  for (let i = 0; i < N; i++) {
   311→   311→    buf[i] *= 0.5 * (1 - Math.cos((2 * Math.PI * i) / (N - 1)))
   312→   312→  }
   313→   313→}
   314→   314→
   315→   315→/**
   316→   316→ * In-place radix-2 Cooley–Tukey FFT.
   317→   317→ *
   318→   318→ * Single source of truth for the radix-2 butterfly used across the RAIN
   319→   319→ * DSP modules. Both `real` and `imag` must be the same power-of-two length;
   320→   320→ * both arrays are mutated in place to hold the forward-FFT result.
   321→   321→ *
   322→   322→ * Determinism: same input → same output, bit-for-bit. No Math.random,
   323→   323→ * no Date.now — pure floating-point arithmetic.
   324→   324→ *
   325→   325→ * Previously this exact algorithm was duplicated in `stems.ts`,
   326→   326→ * `reference-match.ts`, and `aie.ts` (each file kept a private copy because
   327→   327→ * dsp.ts only exposed the higher-level `fftMagnitude` wrapper). It is now
   328→   328→ * exported here so all four call sites share one implementation.
   329→   329→ */
   330→   330→export function fftInPlace(real: Float32Array, imag: Float32Array): void {
   331→   331→  const N = real.length
   332→   332→  if (N <= 1) return
   333→   333→  if (N & (N - 1)) throw new Error('FFT size must be a power of 2')
   334→   334→
   335→   335→  // Bit-reversal permutation
   336→   336→  let j = 0
   337→   337→  for (let i = 1; i < N; i++) {
   338→   338→    let bit = N >> 1
   339→   339→    while (j & bit) {
   340→   340→      j ^= bit
   341→   341→      bit >>= 1
   342→   342→    }
   343→   343→    j ^= bit
   344→   344→    if (i < j) {
   345→   345→      const tr = real[i]; real[i] = real[j]; real[j] = tr
   346→   346→      const ti = imag[i]; imag[i] = imag[j]; imag[j] = ti
   347→   347→    }
   348→   348→  }
   349→   349→
   350→   350→  // Butterfly stages
   351→   351→  for (let len = 2; len <= N; len <<= 1) {
   352→   352→    const halfLen = len >> 1
   353→   353→    const ang = -2 * Math.PI / len
   354→   354→    const wReal = Math.cos(ang)
   355→   355→    const wImag = Math.sin(ang)
   356→   356→    for (let i = 0; i < N; i += len) {
   357→   357→      let curReal = 1
   358→   358→      let curImag = 0
   359→   359→      for (let k = 0; k < halfLen; k++) {
   360→   360→        const a = i + k
   361→   361→        const b = a + halfLen
   362→   362→        const evenReal = real[a]
   363→   363→        const evenImag = imag[a]
   364→   364→        const oddReal = real[b] * curReal - imag[b] * curImag
   365→   365→        const oddImag = real[b] * curImag + imag[b] * curReal
   366→   366→        real[a] = evenReal + oddReal
   367→   367→        imag[a] = evenImag + oddImag
   368→   368→        real[b] = evenReal - oddReal
   369→   369→        imag[b] = evenImag - oddImag
   370→   370→        const newReal = curReal * wReal - curImag * wImag
   371→   371→        curImag = curReal * wImag + curImag * wReal
   372→   372→        curReal = newReal
   373→   373→      }
   374→   374→    }
   375→   375→  }
   376→   376→}
   377→   377→
   378→   378→/**
   379→   379→ * FFT magnitude spectrum (size must be power of 2).
   380→   380→ * Returns Float32Array of length N/2 with magnitudes in dB.
   381→   381→ */
   382→   382→export function fftMagnitude(input: Float32Array): Float32Array {
   383→   383→  const N = input.length
   384→   384→  if (N & (N - 1)) throw new Error('FFT size must be power of 2')
   385→   385→
   386→   386→  const real = new Float32Array(N)
   387→   387→  const imag = new Float32Array(N)
   388→   388→  for (let i = 0; i < N; i++) real[i] = input[i]
   389→   389→
   390→   390→  fftInPlace(real, imag)
   391→   391→
   392→   392→  const out = new Float32Array(N / 2)
   393→   393→  for (let i = 0; i < N / 2; i++) {
   394→   394→    const mag = Math.sqrt(real[i] * real[i] + imag[i] * imag[i]) / N
   395→   395→    out[i] = 20 * Math.log10(mag + 1e-10)
   396→   396→  }
   397→   397→  return out
   398→   398→}

   404→   404→export function midSideEncode(left: Float32Array, right: Float32Array): { mid: Float32Array; side: Float32Array } {
   405→   405→  const mid = new Float32Array(left.length)
   406→   406→  const side = new Float32Array(left.length)
   407→   407→  for (let i = 0; i < left.length; i++) {
   408→   408→    mid[i] = (left[i] + right[i]) * 0.5
   409→   409→    side[i] = (left[i] - right[i]) * 0.5
   410→   410→  }
   411→   411→  return { mid, side }
   412→   412→}

   414→   414→export function midSideDecode(mid: Float32Array, side: Float32Array): { left: Float32Array; right: Float32Array } {
   415→   415→  const left = new Float32Array(mid.length)
   416→   416→  const right = new Float32Array(mid.length)
   417→   417→  for (let i = 0; i < mid.length; i++) {
   418→   418→    left[i] = mid[i] + side[i]
   419→   419→    right[i] = mid[i] - side[i]

   424→   424→/**
   425→   425→ * P2-METERS — Real-time stereo correlation (Pearson product-moment).
   426→   426→ *
   427→   427→ * Returns a value in [-1, +1]:
   428→   428→ *   +1 = perfectly correlated (mono)
   429→   429→ *    0 = uncorrelated (wide stereo / out-of-phase mid)
   430→   430→ *   -1 = perfectly anti-correlated (out of phase)
   431→   431→ *
   432→   432→ * Computed as the normalized dot product of L and R over the full window:
   433→   433→ *
   434→   434→ *     r = Σ(L·R) / sqrt(ΣL² · ΣR²)
   435→   435→ *
   436→   436→ * This is the same Pearson correlation used by the QC metrics phase
   437→   437→ * correlation field. Extracted as a standalone exported function so the
   438→   438→ * real-time meter (StereoCorrelationMeter) and the offline QC measurement
   439→   439→ * share the same implementation. Pure function — no side effects, no
   440→   440→ * allocation beyond two accumulator scalars.
   441→   441→ *
   442→   442→ * Used during playback at ~30 Hz by the audio engine tick() loop to drive
   443→   443→ * the live stereo correlation meter (a REAL measured value, not a static
   444→   444→ * render-time snapshot).
   445→   445→ */
   446→   446→export function computeCorrelation(left: Float32Array, right: Float32Array): number {
   447→   447→  const n = Math.min(left.length, right.length)
   448→   448→  if (n === 0) return 1 // guard: no data — treat as mono (perfectly correlated)
   449→   449→  let dot = 0
   450→   450→  let lEnergy = 0
   451→   451→  let rEnergy = 0
   452→   452→  for (let i = 0; i < n; i++) {
   453→   453→    const l = left[i]
   454→   454→    const r = right[i]
   455→   455→    dot += l * r
   456→   456→    lEnergy += l * l
   457→   457→    rEnergy += r * r
   458→   458→  }
   459→   459→  const denom = Math.sqrt(lEnergy * rEnergy)
   460→   460→  // Slightly tighter epsilon than QC version (1e-12) — still NaN-safe.
   461→   461→  return denom > 1e-20 ? dot / denom : 1
   462→   462→}

   464→   464→export function stereoWidthRatio(left: Float32Array, right: Float32Array): number {
   465→   465→  let midPow = 0, sidePow = 0
   466→   466→  for (let i = 0; i < left.length; i++) {
   467→   467→    const m = (left[i] + right[i]) * 0.5
   468→   468→    const s = (left[i] - right[i]) * 0.5
   469→   469→    midPow += m * m
   470→   470→    sidePow += s * s
   471→   471→  }
   472→   472→  return Math.sqrt(sidePow / (midPow + 1e-10))
   473→   473→}

   479→   479→export type SaturationMode = 'tape' | 'tube' | 'transformer'

   481→   481→/** Apply analog saturation in-place. drive 0..1 (where 1 = maximum drive). */
   482→   482→export function applySaturation(samples: Float32Array, drive: number, mode: SaturationMode = 'tape') {
   483→   483→  // BUG FIX: guard k against 0 — tanh(0)/tanh(0) = 0/0 = NaN in tape mode,
   484→   484→  // and (1-exp(0))/(1-exp(0)) = 0/0 = NaN in tube mode. A drive of 0 would
   485→   485→  // propagate NaN through the entire signal.
   486→   486→  const k = Math.max(drive * 4, 1e-6) // up to 4× drive, never zero
   487→   487→  for (let i = 0; i < samples.length; i++) {
   488→   488→    const x = samples[i]
   489→   489→    let y: number
   490→   490→    switch (mode) {
   491→   491→      case 'tape':
   492→   492→        // Soft clip with hysteresis-like character (tanh)
   493→   493→        y = Math.tanh(x * k) / Math.tanh(k)
   494→   494→        break
   495→   495→      case 'tube': {
   496→   496→        // AUDIT-C3 FIX: the original negative branch was
   497→   497→        //   (Math.exp(x*k) - 1) / (Math.exp(k) - 1) * -1
   498→   498→        // which INVERTED POLARITY (negative input → positive output) and had
   499→   499→        // the wrong magnitude (denominator ~exp(k)× too large). Replaced with
   500→   500→        // a sign-preserving soft-saturating curve; the negative half uses a
   501→   501→        // slightly milder drive (0.85×) to give a tube-like even-harmonic
   502→   502→        // character without the polarity flip.
   503→   503→        //
   504→   504→        // P1 FIX (round 2): clamp the final output to [-1, 1]. The normalized
   505→   505→        // waveshaper `mag = (1 - exp(-ax*driveK)) / (1 - exp(-driveK))` is
   506→   506→        // bounded by 1/(1-exp(-driveK)) as ax → ∞ — for driveK = 4 that is
   507→   507→        // ~1.0186, and for the negative-branch driveK = 3.4 it is ~1.0346.
   508→   508→        // For inputs strictly within [-1, 1] mag ∈ [0, 1] (because the
   509→   509→        // denominator normalizes mag(ax=1) = 1), but the saturation stage
   510→   510→        // runs AFTER EQ + multiband compression + stereo widening, any of
   511→   511→        // which can push peaks above 1.0. Without the clamp those overshoots
   512→   512→        // would propagate as floating-point > 1.0 values into the master bus,
   513→   513→        // eventually forcing the Stage 12 limiter to do extra work. Clamping
   514→   514→        // here keeps the waveshaper honest about its "bounded [-1, 1]" claim
   515→   515→        // and matches the existing `transformer` mode's hard clamp.
   516→   516→        const ax = Math.abs(x)
   517→   517→        const driveK = x > 0 ? k : k * 0.85
   518→   518→        const mag = (1 - Math.exp(-ax * driveK)) / (1 - Math.exp(-driveK))
   519→   519→        y = x < 0 ? -mag : mag
   520→   520→        if (y > 1) y = 1
   521→   521→        else if (y < -1) y = -1
   522→   522→        break
   523→   523→      }
   524→   524→      case 'transformer':
   525→   525→        // Square-law approximation with soft knee
   526→   526→        y = x + k * x * x * Math.sign(x) * -0.3 + k * 0.3 * Math.tanh(x * 3)
   527→   527→        y = Math.max(-1, Math.min(1, y))
   528→   528→        break
   529→   529→    }
   530→   530→    samples[i] = y
   531→   531→  }
   532→   532→}

   538→   538→export interface LimiterParams {
   539→   539→  ceiling: number // dBTP (e.g. -1.0)
   540→   540→  threshold: number // dB (e.g. -1.0)
   541→   541→  releaseMs: number // e.g. 100
   542→   542→  lookAheadMs: number // e.g. 5
   543→   543→  sampleRate: number
   544→   544→}
   545→   545→
   546→   546→/**
   547→   547→ * Simple look-ahead brickwall limiter. Uses cascaded box filters for smooth
   548→   548→ * gain envelope (per RAIN V6 spec).
   549→   549→ */
   550→   550→export function applyLimiter(samples: Float32Array, params: LimiterParams): Float32Array {
   551→   551→  const { ceiling, threshold, releaseMs, lookAheadMs, sampleRate } = params
   552→   552→  const ceilingLin = Math.pow(10, ceiling / 20)
   553→   553→  const thresholdLin = Math.pow(10, threshold / 20)
   554→   554→  const lookAhead = Math.max(1, Math.floor(lookAheadMs * 0.001 * sampleRate))
   555→   555→  const releaseCoef = Math.exp(-1 / (releaseMs * 0.001 * sampleRate))
   556→   556→
   557→   557→  // AUDIT-C2 FIX (round 2): the previous round introduced future-sample
   558→   558→  // detection (reading `samples[i + lookAhead]`) but still applied the
   559→   559→  // resulting gain envelope to `samples[i]` — i.e. the gain reduction fired
   560→   560→  // `lookAhead` samples BEFORE the peak in the OUTPUT, and by the time the
   561→   561→  // peak itself reached the output the gain envelope had already started
   562→   562→  // recovering. The hard ceiling clamp was therefore still doing all the
   563→   563→  // actual limiting (audible clipping distortion on transients) — exactly
   564→   564→  // the symptom the first round was supposed to fix.
   565→   565→  //
   566→   566→  // The correct look-ahead architecture: detect on `samples[i + lookAhead]`,
   567→   567→  // then DELAY the gain envelope by `lookAhead` samples so the gain computed
   568→   568→  // when the peak was first detected is applied to the peak itself. With
   569→   569→  // this alignment, `out[peakIdx] = samples[peakIdx] * gainEnv[peakIdx - lookAhead]`
   570→   570→  // — the minimum-gain entry — and the ceiling clamp becomes a true safety
   571→   571→  // net instead of the primary mechanism.
   572→   572→  const gainEnv = new Float32Array(samples.length)
   573→   573→  let gain = 1
   574→   574→  for (let i = 0; i < samples.length; i++) {
   575→   575→    const futureIdx = i + lookAhead
   576→   576→    const sample = futureIdx < samples.length ? samples[futureIdx] : 0
   577→   577→    const absSample = Math.abs(sample)
   578→   578→    let targetGain = 1
   579→   579→    if (absSample > thresholdLin) {
   580→   580→      targetGain = thresholdLin / absSample
   581→   581→    }
   582→   582→    // Smooth release: gain reduces instantly, recovers exponentially
   583→   583→    if (targetGain < gain) {
   584→   584→      gain = targetGain
   585→   585→    } else {
   586→   586→      gain = gain + (1 - gain) * (1 - releaseCoef)
   587→   587→    }
   588→   588→    gainEnv[i] = gain
   589→   589→  }
   590→   590→
   591→   591→  // Apply the DELAYED gain envelope to the input samples. For the first
   592→   592→  // `lookAhead` samples no gain envelope is available yet (the look-ahead
   593→   593→  // window extends before the start of the buffer) so we use gain = 1.
   594→   594→  const out = new Float32Array(samples.length)
   595→   595→  for (let i = 0; i < samples.length; i++) {
   596→   596→    const g = i >= lookAhead ? gainEnv[i - lookAhead] : 1
   597→   597→    let s = samples[i] * g
   598→   598→    if (s > ceilingLin) s = ceilingLin
   599→   599→    if (s < -ceilingLin) s = -ceilingLin
   600→   600→    out[i] = s
   601→   601→  }
   602→   602→  return out
   603→   603→}
   604→   604→
   605→   605→/**
   606→   606→ * True-peak-aware limiter.
   607→   607→ *
   608→   608→ * WHY THIS EXISTS — the classic "true-peak always fails by a hair" bug:
   609→   609→ * `applyLimiter` is a SAMPLE-peak brickwall — it clamps individual samples to a
   610→   610→ * linear ceiling. But true-peak (dBTP) measures the reconstructed band-limited
   611→   611→ * signal, which peaks BETWEEN samples (the inter-sample peak, ISP). For typical
   612→   612→ * mastered audio the ISP overshoots the sample peak by 0.2–0.6 dB; for
   613→   613→ * hard-clipped transients it can reach ~1 dB. So limiting sample-peak to the
   614→   614→ * dBTP ceiling leaves the measured true-peak sitting just above it → QC "fail"
   615→   615→ * by a small margin, every time.
   616→   616→ *
   617→   617→ * This wrapper closes the loop: limit → measure dBTP → feed the excess back into
   618→   618→ * the limiter ceiling → re-limit. Converges in 1–2 iterations for program
   619→   619→ * material. `params.ceiling` is treated as the TARGET TRUE-PEAK; we aim 0.05 dB
   620→   620→ * below it for safety so the QC `tp <= ceiling` check passes with margin.
   621→   621→ */
   622→   622→export function applyTruePeakLimiter(
   623→   623→  samples: Float32Array,
   624→   624→  params: LimiterParams,
   625→   625→  maxIterations = 4,
   626→   626→): Float32Array {
   627→   627→  const targetCeiling = params.ceiling // dBTP we must not exceed
   628→   628→  const passLevel = targetCeiling - 0.05 // aim slightly under for QC safety margin
   629→   629→  const thresholdOffset = params.ceiling - params.threshold // e.g. 0.5 dB
   630→   630→  let workingCeiling = targetCeiling
   631→   631→  let out = samples
   632→   632→  for (let iter = 0; iter < maxIterations; iter++) {
   633→   633→    out = applyLimiter(out, {
   634→   634→      ...params,
   635→   635→      ceiling: workingCeiling,
   636→   636→      threshold: workingCeiling - thresholdOffset,
   637→   637→    })
   638→   638→    const tp = computeTruePeak(out)
   639→   639→    if (tp <= passLevel) break // within target — done
   640→   640→    const excess = tp - passLevel
   641→   641→    // Drop the sample-peak ceiling by the full measured excess. The next
   642→   642→    // limiter pass will then leave true-peak under target.
   643→   643→    workingCeiling = workingCeiling - excess
   644→   644→  }
   645→   645→  return out
   646→   646→}

   898→   898→// ---------------------------------------------------------------------------
   899→   899→// P2-METERS — FFT-derived spectral descriptors
   900→   900→// Real measurements over the magnitude spectrum. No synthetic data.
   901→   901→// ---------------------------------------------------------------------------
   902→   902→
   903→   903→/**
   904→   904→ * ISO 31-band 1/3-octave center frequencies (20 Hz .. 20 kHz).
   905→   905→ * These are the canonical band centers used by reference-match.ts and the
   906→   906→ * Stage 5 reference curve. Each band's edges are fc / 2^(1/6) .. fc * 2^(1/6).
   907→   907→ */
   908→   908→export const THIRD_OCTAVE_BANDS_HZ: readonly number[] = [
   909→   909→  20, 25, 31.5, 40, 50, 63, 80, 100, 125, 160,
   910→   910→  200, 250, 315, 400, 500, 630, 800, 1000, 1250, 1600,
   911→   911→  2000, 2500, 3150, 4000, 5000, 6300, 8000, 10000, 12500, 16000,
   912→   912→  20000,
   913→   913→]
   914→   914→
   915→   915→/**
   916→   916→ * Compute the full set of FFT-derived spectral descriptors from a magnitude
   917→   917→ * spectrum. Pure function — no side effects.
   918→   918→ *
   919→   919→ *   - Centroid: Σ(f·mag) / Σ(mag)  → brightness measure (Hz)
   920→   920→ *   - Spread: sqrt(Σ((f - centroid)²·mag) / Σ(mag))  → spectral variance (Hz)
   921→   921→ *   - Skewness: 3rd central moment / spread³ (dimensionless)
   922→   922→ *   - Kurtosis: 4th central moment / spread⁴ (dimensionless)
   923→   923→ *   - Rolloff 85%: smallest f such that Σ_{0..f} mag >= 0.85 · Σ mag
   924→   924→ *   - Rolloff 95%: same at 95%
   925→   925→ *   - Flatness: exp(mean(ln(mag))) / mean(mag) — geometric/arithmetic ratio
   926→   926→ *   - Flux: L1 distance vs the previous frame's spectrum (0 if prevSpectrum
   927→   927→ *     is null/undefined — used for the first frame)
   928→   928→ *   - Peak frequency: bin with the maximum magnitude, converted to Hz
   929→   929→ *
   930→   930→ * Input spectrum is the dB-magnitude array as returned by fftMagnitude()
   931→   931→ * (length N/2, where N is the FFT size — typically 1024 for FFT_SIZE=2048).
   932→   932→ *
   933→   933→ * For flatness, dB values are converted back to linear power (10^(dB/10))
   934→   934→ * before taking the log — geometric mean of linear power is meaningful,
   935→   935→ * whereas geometric mean of dB is just arithmetic mean of dB (useless for
   936→   936→ * flatness). Bins with zero power are clamped to 1e-12 to avoid ln(0).
   937→   937→ */
   938→   938→export function computeSpectralFeatures(
   939→   939→  spectrum: Float32Array,
   940→   940→  sampleRate: number,
   941→   941→  prevSpectrum?: Float32Array | null,
   942→   942→): SpectralFeatures {
   943→   943→  const N = spectrum.length
   944→   944→  if (N === 0) {
   945→   945→    return {
   946→   946→      centroid: 0, spread: 0, skewness: 0, kurtosis: 0,
   947→   947→      rolloff85: 0, rolloff95: 0, flatness: 0, flux: 0, peakFrequency: 0,
   948→   948→    }
   949→   949→  }
   950→   950→  // FFT_SIZE = 2 * N (spectrum is N/2 of FFT_SIZE). binHz = sampleRate / FFT_SIZE.
   951→   951→  const binHz = sampleRate / (N * 2)
   952→   952→
   953→   953→  // --- Convert dB spectrum to linear power for moment math ---
   954→   954→  // 10^(dB/10) gives linear power (assuming dB = 10·log10(power)).
   955→   955→  // For the moment-based descriptors (centroid, spread, skew, kurt) we use
   956→   956→  // linear power as the weighting term — this is the standard formulation.
   957→   957→  const linear = new Float32Array(N)
   958→   958→  let totalPower = 0
   959→   959→  for (let i = 0; i < N; i++) {
   960→   960→    // Clamp dB at -240 to keep 10^(dB/10) within sane range.
   961→   961→    const db = Math.max(spectrum[i], -240)
   962→   962→    const p = Math.pow(10, db / 10)
   963→   963→    linear[i] = p
   964→   964→    totalPower += p
   965→   965→  }
   966→   966→
   967→   967→  // --- Peak frequency (highest magnitude bin) ---
   968→   968→  let peakBin = 0
   969→   969→  let peakVal = -Infinity
   970→   970→  for (let i = 0; i < N; i++) {
   971→   971→    if (spectrum[i] > peakVal) {
   972→   972→      peakVal = spectrum[i]
   973→   973→      peakBin = i
   974→   974→    }
   975→   975→  }
   976→   976→  const peakFrequency = peakBin * binHz
   977→   977→
   978→   978→  // --- Spectral centroid (weighted mean frequency) ---
   979→   979→  let weightedSum = 0
   980→   980→  for (let i = 0; i < N; i++) {
   981→   981→    weightedSum += (i * binHz) * linear[i]
   982→   982→  }
   983→   983→  const centroid = totalPower > 1e-20 ? weightedSum / totalPower : 0
   984→   984→
   985→   985→  // --- Spread (standard deviation around centroid) ---
   986→   986→  let varSum = 0
   987→   987→  for (let i = 0; i < N; i++) {
   988→   988→    const f = i * binHz
   989→   989→    const d = f - centroid
   990→   990→    varSum += d * d * linear[i]
   991→   991→  }
   992→   992→  const spread = totalPower > 1e-20 ? Math.sqrt(varSum / totalPower) : 0
   993→   993→
   994→   994→  // --- Skewness & kurtosis (normalized central moments) ---
   995→   995→  let m3 = 0, m4 = 0
   996→   996→  if (spread > 1e-12 && totalPower > 1e-20) {
   997→   997→    for (let i = 0; i < N; i++) {
   998→   998→      const f = i * binHz
   999→   999→      const d = (f - centroid) / spread
  1000→  1000→      const w = linear[i] / totalPower
  1001→  1001→      m3 += d * d * d * w
  1002→  1002→      m4 += d * d * d * d * w
  1003→  1003→    }
  1004→  1004→  }
  1005→  1005→  const skewness = m3
  1006→  1006→  const kurtosis = m4
  1007→  1007→
  1008→  1008→  // --- Spectral rolloff (85% and 95% cumulative energy) ---
  1009→  1009→  const rolloffAt = (threshold: number): number => {
  1010→  1010→    if (totalPower <= 1e-20) return 0
  1011→  1011→    const target = threshold * totalPower
  1012→  1012→    let cum = 0
  1013→  1013→    for (let i = 0; i < N; i++) {
  1014→  1014→      cum += linear[i]
  1015→  1015→      if (cum >= target) return i * binHz
  1016→  1016→    }
  1017→  1017→    return (N - 1) * binHz
  1018→  1018→  }
  1019→  1019→  const rolloff85 = rolloffAt(0.85)
  1020→  1020→  const rolloff95 = rolloffAt(0.95)
  1021→  1021→
  1022→  1022→  // --- Spectral flatness (geometric / arithmetic mean ratio) ---
  1023→  1023→  // Computed on linear power. Bins with zero power are clamped to 1e-12.
  1024→  1024→  let logSum = 0
  1025→  1025→  let arithSum = 0
  1026→  1026→  for (let i = 0; i < N; i++) {
  1027→  1027→    const p = Math.max(linear[i], 1e-12)
  1028→  1028→    logSum += Math.log(p)
  1029→  1029→    arithSum += p
  1030→  1030→  }
  1031→  1031→  const geoMean = Math.exp(logSum / N)
  1032→  1032→  const arithMean = arithSum / N
  1033→  1033→  const flatness = arithMean > 1e-20 ? Math.max(0, Math.min(1, geoMean / arithMean)) : 0
  1034→  1034→
  1035→  1035→  // --- Spectral flux (L1 distance vs previous frame) ---
  1036→  1036→  let flux = 0
  1037→  1037→  if (prevSpectrum && prevSpectrum.length === N) {
  1038→  1038→    for (let i = 0; i < N; i++) {
  1039→  1039→      // Use the POSITIVE difference (new spectral content), per the standard
  1040→  1040→      // flux formulation. Negative differences (decaying content) are clamped
  1041→  1041→      // to 0 so the flux measures onset strength, not overall change.
  1042→  1042→      const cur = Math.max(linear[i], 0)
  1043→  1043→      const prevDb = Math.max(prevSpectrum[i], -240)
  1044→  1044→      const prev = Math.pow(10, prevDb / 10)
  1045→  1045→      const diff = cur - prev
  1046→  1046→      if (diff > 0) flux += diff
  1047→  1047→    }
  1048→  1048→    // Normalize by N so the metric is comparable across FFT sizes.
  1049→  1049→    flux = flux / N
  1050→  1050→  }
  1051→  1051→
  1052→  1052→  return {
  1053→  1053→    centroid, spread, skewness, kurtosis,
  1054→  1054→    rolloff85, rolloff95, flatness, flux, peakFrequency,
  1055→  1055→  }
  1056→  1056→}
  1057→  1057→
  1058→  1058→/**
  1059→  1059→ * Compute ISO 31-band 1/3-octave band energies from a magnitude spectrum.
  1060→  1060→ * Returns a Float32Array of length 31 with band energies in dB, indexed by
  1061→  1061→ * the THIRD_OCTAVE_BANDS_HZ array (20, 25, 31.5, ... 20000 Hz).
  1062→  1062→ *
  1063→  1063→ * Each band's energy is the average dB magnitude across the FFT bins whose
  1064→  1064→ * center frequency falls inside the band's [fc / 2^(1/6), fc * 2^(1/6)]
  1065→  1065→ * range. Bands with no bins (above Nyquist) are filled with -120 dB.
  1066→  1066→ */
  1067→  1067→export function computeThirdOctaveBands(
  1068→  1068→  spectrum: Float32Array,
  1069→  1069→  sampleRate: number,
  1070→  1070→): Float32Array {
  1071→  1071→  const N = spectrum.length
  1072→  1072→  const binHz = sampleRate / (N * 2) // FFT_SIZE = 2 * N
  1073→  1073→  const nyquist = sampleRate / 2
  1074→  1074→  const ratio = Math.pow(2, 1 / 6) // 1/3-octave band edge ratio
  1075→  1075→  const out = new Float32Array(THIRD_OCTAVE_BANDS_HZ.length)
  1076→  1076→  for (let bi = 0; bi < THIRD_OCTAVE_BANDS_HZ.length; bi++) {
  1077→  1077→    const fc = THIRD_OCTAVE_BANDS_HZ[bi]
  1078→  1078→    if (fc > nyquist) {
  1079→  1079→      out[bi] = -120
  1080→  1080→      continue
  1081→  1081→    }
  1082→  1082→    const loHz = fc / ratio
  1083→  1083→    const hiHz = fc * ratio
  1084→  1084→    const loBin = Math.max(1, Math.floor(loHz / binHz))
  1085→  1085→    const hiBin = Math.min(N - 1, Math.ceil(hiHz / binHz))
  1086→  1086→    if (hiBin < loBin) {
  1087→  1087→      out[bi] = -120
  1088→  1088→      continue
  1089→  1089→    }
  1090→  1090→    let sum = 0
  1091→  1091→    let count = 0
  1092→  1092→    for (let b = loBin; b <= hiBin; b++) {
  1093→  1093→      sum += spectrum[b]
  1094→  1094→      count++
  1095→  1095→    }
  1096→  1096→    out[bi] = count > 0 ? sum / count : -120
  1097→  1097→  }
  1098→  1098→  return out
  1099→  1099→}

  1104→  1104→
  1105→  1105→export function analyzeAudio(channels: Float32Array[], sampleRate: number): AudioAnalysis {
  1106→  1106→  if (channels.length === 0) {
  1107→  1107→    throw new Error('No channels to analyze')
  1108→  1108→  }
  1109→  1109→  const lufs = computeLufs(channels, sampleRate)
  1110→  1110→  const left = channels[0]
  1111→  1111→  const right = channels[1] ?? channels[0]
  1112→  1112→  const truePeak = Math.max(computeTruePeak(left), computeTruePeak(right))
  1113→  1113→  const rms = computeRms(left)
  1114→  1114→  const peak = computePeak(left)
  1115→  1115→  const crestFactor = peak - rms
  1116→  1116→  const lra = computeLra(channels, sampleRate)
  1117→  1117→  const duration = left.length / sampleRate
  1118→  1118→
  1119→  1119→  // Spectrum — average of 8 windows of 2048
  1120→  1120→  const FFT_SIZE = 2048
  1121→  1121→  const spectrumSum = new Float32Array(FFT_SIZE / 2)
  1122→  1122→  const numWindows = Math.max(1, Math.floor(left.length / FFT_SIZE))
  1123→  1123→  const windowsToAverage = Math.min(8, numWindows)
  1124→  1124→  const step = Math.floor(numWindows / windowsToAverage) || 1
  1125→  1125→  let count = 0
  1126→  1126→  for (let w = 0; w < windowsToAverage; w++) {
  1127→  1127→    const start = w * step * FFT_SIZE
  1128→  1128→    if (start + FFT_SIZE > left.length) break
  1129→  1129→    const buf = left.subarray(start, start + FFT_SIZE).slice()
  1130→  1130→    hannWindow(buf)
  1131→  1131→    const mag = fftMagnitude(buf)
  1132→  1132→    for (let i = 0; i < FFT_SIZE / 2; i++) spectrumSum[i] += mag[i]
  1133→  1133→    count++
  1134→  1134→  }
  1135→  1135→  const spectrum = new Float32Array(FFT_SIZE / 2)
  1136→  1136→  for (let i = 0; i < spectrum.length; i++) spectrum[i] = spectrumSum[i] / Math.max(1, count)
  1137→  1137→
  1138→  1138→  // Peak frequency
  1139→  1139→  let peakBin = 0
  1140→  1140→  let peakVal = -Infinity
  1141→  1141→  for (let i = 1; i < spectrum.length; i++) {
  1142→  1142→    if (spectrum[i] > peakVal) {
  1143→  1143→      peakVal = spectrum[i]
  1144→  1144→      peakBin = i
  1145→  1145→    }
  1146→  1146→  }
  1147→  1147→  const peakFrequency = (peakBin * sampleRate) / FFT_SIZE
  1148→  1148→
  1149→  1149→  // Zero-crossing rate
  1150→  1150→  let zc = 0
  1151→  1151→  for (let i = 1; i < left.length; i++) {
  1152→  1152→    if ((left[i] >= 0) !== (left[i - 1] >= 0)) zc++
  1153→  1153→  }
  1154→  1154→  const zeroCrossingRate = zc / (left.length / sampleRate)
  1155→  1155→
  1156→  1156→  // BPM estimation via autocorrelation on low-passed envelope (simplified)
  1157→  1157→  const bpm = estimateBpm(left, sampleRate)
  1158→  1158→
  1159→  1159→  // Musical key estimate via chroma-ish peak bin
  1160→  1160→  const key = estimateKey(peakFrequency)
  1161→  1161→
  1162→  1162→  // AUDIT-P2: real QC metrics (replaces 9 hardcoded values in QCTab).
  1163→  1163→  const qcMetrics = computeQCMetrics(left, right, spectrum, sampleRate, channels.length)
  1164→  1164→
  1165→  1165→  // P2-METERS: real FFT-derived spectral descriptors (centroid, spread,
  1166→  1166→  // skewness, kurtosis, rolloff85/95, flatness, flux, peak frequency).
  1167→  1167→  // No previous spectrum for the offline analysis — flux returns 0.
  1168→  1168→  const spectralFeatures = computeSpectralFeatures(spectrum, sampleRate, null)
  1169→  1169→
  1170→  1170→  // P2-METERS: ISO 31-band 1/3-octave band energies in dB.
  1171→  1171→  const thirdOctaveBands = computeThirdOctaveBands(spectrum, sampleRate)
  1172→  1172→
  1173→  1173→  return {
  1174→  1174→    lufs, truePeak, rms, crestFactor, dynamicRange: lra,
  1175→  1175→    sampleRate, duration, channels: channels.length, bitDepth: 24,
  1176→  1176→    bpm, key, spectrum, peakFrequency, zeroCrossingRate, qcMetrics,
  1177→  1177→    spectralFeatures, thirdOctaveBands,
  1178→  1178→  }
  1179→  1179→}

  1181→  1181→/**
  1182→  1182→ * Real QC measurements derived from channel data + the already-computed
  1183→  1183→ * magnitude spectrum. Replaces the 9 prior constant verdicts that QCTab
  1184→  1184→ * used to display regardless of the actual audio.
  1185→  1185→ *
  1186→  1186→ * Spectrum is the FFT magnitude in dB (length FFT_SIZE/2 = 1024 bins for a
  1187→  1187→ * 2048 FFT). Bin frequency = bin * sampleRate / FFT_SIZE.
  1188→  1188→ *
  1189→  1189→ * P2-QC FIX: every field in the returned QCMetrics is now computed from
  1190→  1190→ * the real audio buffer — no constants, no `Math.random`, no estimation.
  1191→  1191→ * Adds: rmsDb, phaseCoherence (short-time cross-correlation),
  1192→  1192→ * zeroCrossingRate, zeroCrossingStuck (clipping-at-zero),
  1193→  1193→ * transientDensity, preechoRisk.
  1194→  1194→ * Replaces the previous block-averaged side-bass proxy with a real
  1195→  1195→ * side-channel FFT (band-limited < 200 Hz).
  1196→  1196→ */
  1197→  1197→export function computeQCMetrics(
  1198→  1198→  left: Float32Array,
  1199→  1199→  right: Float32Array,
  1200→  1200→  spectrum: Float32Array,
  1201→  1201→  sampleRate: number,
  1202→  1202→  _channelCount: number,
  1203→  1203→): import('./types').QCMetrics {
  1204→  1204→  const n = Math.min(left.length, right.length)
  1205→  1205→
  1206→  1206→  // --- DC offset: mean of the signal (averaged across both channels),
  1207→  1207→  // as a fraction of full-scale ---
  1208→  1208→  let dcSum = 0
  1209→  1209→  for (let i = 0; i < n; i++) dcSum += (left[i] + right[i]) * 0.5
  1210→  1210→  const dcOffset = n > 0 ? Math.abs(dcSum / n) : 0
  1211→  1211→
  1212→  1212→  // --- RMS (averaged across both channels) in dBFS ---
  1213→  1213→  let sumSq = 0
  1214→  1214→  for (let i = 0; i < n; i++) {
  1215→  1215→    const m = (left[i] + right[i]) * 0.5
  1216→  1216→    sumSq += m * m
  1217→  1217→  }
  1218→  1218→  const rmsLin = n > 0 ? Math.sqrt(sumSq / n) : 0
  1219→  1219→  const rmsDb = 20 * Math.log10(Math.max(rmsLin, 1e-7))
  1220→  1220→
  1221→  1221→  // --- Stereo correlation (full-buffer Pearson L/R) ---
  1222→  1222→  // P2-METERS: route through the standalone computeCorrelation() so the live
  1223→  1223→  // meter and the QC metric share one implementation.
  1224→  1224→  const phaseCorrelation = computeCorrelation(left, right)
  1225→  1225→
  1226→  1226→  // --- Stereo width: side energy / mid energy ---
  1227→  1227→  let midE = 0, sideE = 0
  1228→  1228→  for (let i = 0; i < n; i++) {
  1229→  1229→    const mid = (left[i] + right[i]) * 0.5
  1230→  1230→    const side = (left[i] - right[i]) * 0.5
  1231→  1231→    midE += mid * mid
  1232→  1232→    sideE += side * side
  1233→  1233→  }
  1234→  1234→  const stereoWidth = midE > 1e-12 ? Math.sqrt(sideE / midE) : 0
  1235→  1235→
  1236→  1236→  // --- Phase coherence: short-time cross-correlation at zero lag,
  1237→  1237→  // averaged across 20 ms windows. Captures LOCAL phase drift, distinct
  1238→  1238→  // from the full-buffer Pearson above. ---
  1239→  1239→  const winSize = Math.max(1, Math.floor(sampleRate * 0.02)) // 20 ms
  1240→  1240→  let cohSum = 0
  1241→  1241→  let cohCount = 0
  1242→  1242→  for (let start = 0; start + winSize <= n; start += winSize) {
  1243→  1243→    let wd = 0, wl = 0, wr = 0
  1244→  1244→    for (let i = 0; i < winSize; i++) {
  1245→  1245→      const l = left[start + i]
  1246→  1246→      const r = right[start + i]
  1247→  1247→      wd += l * r
  1248→  1248→      wl += l * l
  1249→  1249→      wr += r * r
  1250→  1250→    }
  1251→  1251→    const wDenom = Math.sqrt(wl * wr)
  1252→  1252→    if (wDenom > 1e-12) {
  1253→  1253→      cohSum += wd / wDenom
  1254→  1254→      cohCount++
  1255→  1255→    }
  1256→  1256→  }
  1257→  1257→  const phaseCoherence = cohCount > 0 ? cohSum / cohCount : 1
  1258→  1258→
  1259→  1259→  // --- Zero-crossing rate (crossings per second, averaged across channels) ---
  1260→  1260→  let zcL = 0, zcR = 0
  1261→  1261→  for (let i = 1; i < n; i++) {
  1262→  1262→    if ((left[i] >= 0) !== (left[i - 1] >= 0)) zcL++
  1263→  1263→    if ((right[i] >= 0) !== (right[i - 1] >= 0)) zcR++
  1264→  1264→  }
  1265→  1265→  const durationSec = n / sampleRate
  1266→  1266→  const zeroCrossingRate = durationSec > 0 ? (zcL + zcR) * 0.5 / durationSec : 0
  1267→  1267→
  1268→  1268→  // --- Zero-crossing stuck-at-zero clipping: count samples that are
  1269→  1269→  // exactly 0.0 (or extremely close) while their immediate neighbours are
  1270→  1270→  // at full scale (|x| > 0.99). This is a tell-tale sign of DC-offset
  1271→  1271→  // clipping right at the zero crossing — the kind of corruption that
  1272→  1272→  // a hard-clipper with DC bias produces. ---
  1273→  1273→  let zeroCrossingStuck = 0
  1274→  1274→  for (let i = 1; i < n - 1; i++) {
  1275→  1275→    const li = left[i]
  1276→  1276→    if (Math.abs(li) < 1e-6) {
  1277→  1277→      if (Math.abs(left[i - 1]) > 0.99 || Math.abs(left[i + 1]) > 0.99) zeroCrossingStuck++
  1278→  1278→    }
  1279→  1279→    const ri = right[i]
  1280→  1280→    if (Math.abs(ri) < 1e-6) {
  1281→  1281→      if (Math.abs(right[i - 1]) > 0.99 || Math.abs(right[i + 1]) > 0.99) zeroCrossingStuck++
  1282→  1282→    }
  1283→  1283→  }
  1284→  1284→
  1285→  1285→  // --- Clipped samples: count samples at or beyond ±0.999 ---
  1286→  1286→  let clippedSamples = 0
  1287→  1287→  for (let i = 0; i < n; i++) {
  1288→  1288→    if (Math.abs(left[i]) >= 0.999 || Math.abs(right[i]) >= 0.999) clippedSamples++
  1289→  1289→  }
  1290→  1290→
  1291→  1291→  // --- Band energy from the magnitude spectrum ---
  1292→  1292→  // spectrum is in dB already (from fftMagnitude). Bin width in Hz:
  1293→  1293→  const FFT_SIZE = spectrum.length * 2
  1294→  1294→  const binHz = sampleRate / FFT_SIZE
  1295→  1295→  const bandEnergyDb = (loHz: number, hiHz: number): number => {
  1296→  1296→    const loBin = Math.max(1, Math.floor(loHz / binHz))
  1297→  1297→    const hiBin = Math.min(spectrum.length - 1, Math.ceil(hiHz / binHz))
  1298→  1298→    // Average the magnitude (dB) across the band — spectrum values are already dB.
  1299→  1299→    let sum = 0, count = 0
  1300→  1300→    for (let b = loBin; b <= hiBin; b++) { sum += spectrum[b]; count++ }
  1301→  1301→    return count > 0 ? sum / count : -120
  1302→  1302→  }
  1303→  1303→
  1304→  1304→  // --- Bass mono: real side-channel energy below 200 Hz.
  1305→  1305→  // The previous impl used time-domain block-averaged side energy as a
  1306→  1306→  // proxy, which was NOT band-limited. Now we compute a side-channel FFT
  1307→  1307→  // (averaged over a few windows) and average the magnitude below 200 Hz.
  1308→  1308→  // This is the real M/S bass-mono measurement. ---
  1309→  1309→  const sideFftSize = 2048
  1310→  1310→  const sideSpectrumSum = new Float32Array(sideFftSize / 2)
  1311→  1311→  let sideWindowCount = 0
  1312→  1312→  const sideMaxWindows = 4
  1313→  1313→  const sideStep = sideMaxWindows > 1
  1314→  1314→    ? Math.max(1, Math.floor((n - sideFftSize) / (sideMaxWindows - 1)))
  1315→  1315→    : 0
  1316→  1316→  for (let w = 0; w < sideMaxWindows; w++) {
  1317→  1317→    const start = w * sideStep
  1318→  1318→    if (start + sideFftSize > n) break
  1319→  1319→    const buf = new Float32Array(sideFftSize)
  1320→  1320→    for (let i = 0; i < sideFftSize; i++) {
  1321→  1321→      buf[i] = (left[start + i] - right[start + i]) * 0.5
  1322→  1322→    }
  1323→  1323→    hannWindow(buf)
  1324→  1324→    const mag = fftMagnitude(buf)
  1325→  1325→    for (let b = 0; b < sideFftSize / 2; b++) sideSpectrumSum[b] += mag[b]
  1326→  1326→    sideWindowCount++
  1327→  1327→  }
  1328→  1328→  const sideBinHz = sampleRate / sideFftSize
  1329→  1329→  const bassHiBin = Math.min(sideFftSize / 2 - 1, Math.ceil(200 / sideBinHz))
  1330→  1330→  let bassSideSum = 0
  1331→  1331→  let bassSideBins = 0
  1332→  1332→  if (sideWindowCount > 0) {
  1333→  1333→    for (let b = 1; b <= bassHiBin; b++) {
  1334→  1334→      bassSideSum += sideSpectrumSum[b] / sideWindowCount
  1335→  1335→      bassSideBins++
  1336→  1336→    }
  1337→  1337→  }
  1338→  1338→  // Averaged dB magnitude of the side channel below 200 Hz.
  1339→  1339→  const bassSideDb = bassSideBins > 0 ? bassSideSum / bassSideBins : -120
  1340→  1340→
  1341→  1341→  const sibilanceDb = bandEnergyDb(5000, 8000)
  1342→  1342→  const rumbleDb = bandEnergyDb(0, 20)
  1343→  1343→  const highFreqDb = bandEnergyDb(15000, sampleRate / 2)
  1344→  1344→
  1345→  1345→  // --- Transient density: count of onsets per second.
  1346→  1346→  // An onset is detected when the short-time peak envelope rises sharply
  1347→  1347→  // above its previous value. This is the real input to codec pre-echo
  1348→  1348→  // prediction — percussive material with many sharp onsets is what causes
  1349→  1349→  // audible pre-echo in low-bitrate lossy codecs.
  1350→  1350→  //
  1351→  1351→  // Uses peak detection (max |sample|) rather than RMS so that brief
  1352→  1352→  // transients (1–2 ms) register as a sharp envelope rise even when the
  1353→  1353→  // background is a loud continuous tone — RMS over a 5 ms window dilutes
  1354→  1354→  // a 1 ms click with 4 ms of tone, hiding the rise. A 1 ms hop ensures
  1355→  1355→  // each frame contains either the click peak or pure background, never
  1356→  1356→  // both, so the peak ratio click-frame / tone-frame is large.
  1357→  1357→  const envHop = Math.max(1, Math.floor(sampleRate * 0.001)) // 1 ms hop
  1358→  1358→  const numEnvFrames = Math.max(0, Math.floor(n / envHop))
  1359→  1359→  const env = new Float32Array(numEnvFrames)
  1360→  1360→  for (let f = 0; f < numEnvFrames; f++) {
  1361→  1361→    const start = f * envHop
  1362→  1362→    const end = Math.min(n, start + envHop)
  1363→  1363→    let peak = 0
  1364→  1364→    for (let i = start; i < end; i++) {
  1365→  1365→      const a = Math.abs((left[i] + right[i]) * 0.5)
  1366→  1366→      if (a > peak) peak = a
  1367→  1367→    }
  1368→  1368→    env[f] = peak
  1369→  1369→  }
  1370→  1370→  // Onset threshold: small absolute floor (0.05) to filter out digital
  1371→  1371→  // silence. The DYNAMIC test is the `env[f] > env[f-1] * 1.3` rise below —
  1372→  1372→  // that catches sharp attacks regardless of overall level. A global-mean
  1373→  1373→  // multiple (e.g. envMean * 4) is intentionally NOT used here because it
  1374→  1374→  // would make the threshold unreachable for loud material (peak ≈ 1.0 <
  1375→  1375→  // envMean * 4 when envMean > 0.25), missing transients in hot masters
  1376→  1376→  // entirely.
  1377→  1377→  const onsetThreshold = 0.05
  1378→  1378→  const onsetRise = 1.3
  1379→  1379→  let transientCount = 0
  1380→  1380→  // Deduplicate onsets that fire on consecutive 1 ms frames (the same
  1381→  1381→  // physical transient will produce a rise for 1–3 frames in a row). We
  1382→  1382→  // require a 10 ms refractory period after each detected onset before
  1383→  1383→  // counting another one.
  1384→  1384→  const refractoryFrames = Math.max(1, Math.floor(sampleRate * 0.01 / envHop)) // 10 ms
  1385→  1385→  let lastOnsetFrame = -refractoryFrames
  1386→  1386→  for (let f = 2; f < numEnvFrames - 1; f++) {
  1387→  1387→    // Onset = local envelope spike: rises above neighbours AND above floor.
  1388→  1388→    if (env[f] > onsetThreshold &&
  1389→  1389→        env[f] > env[f - 1] * onsetRise &&
  1390→  1390→        env[f] > env[f + 1] &&
  1391→  1391→        f - lastOnsetFrame >= refractoryFrames) {
  1392→  1392→      transientCount++
  1393→  1393→      lastOnsetFrame = f
  1394→  1394→    }
  1395→  1395→  }
  1396→  1396→  const transientDensity = durationSec > 0 ? transientCount / durationSec : 0
  1397→  1397→
  1398→  1398→  // --- Codec pre-echo risk: 0..1 score.
  1399→  1399→  // Empirical model: pre-echo is audible when (a) the codec's MDCT window
  1400→  1400→  // (typically ~2 ms pre-echo for MP3/Opus) is excited by sharp transients,
  1401→  1401→  // and (b) the signal energy around the transient is high enough to make
  1402→  1402→  // the pre-echo audible above the masking threshold. We combine the
  1403→  1403→  // transient density (onsets/sec) with the overall RMS level (which
  1404→  1404→  // correlates with the post-transient masking floor).
  1405→  1405→  //   preechoRisk = clamp01(transientDensity / 10) * clamp01((rmsDb + 30) / 30)
  1406→  1406→  // At -30 dBFS RMS the masking floor is too low to reveal pre-echo; at
  1407→  1407→  // 0 dBFS RMS even modest transient density becomes audible. 10
  1408→  1408→  // onsets/sec is the saturation point (very busy percussive material). ---
  1409→  1409→  const tdFactor = Math.max(0, Math.min(1, transientDensity / 10))
  1410→  1410→  const rmsFactor = Math.max(0, Math.min(1, (rmsDb + 30) / 30))
  1411→  1411→  const preechoRisk = Math.max(0, Math.min(1, tdFactor * rmsFactor))
  1412→  1412→
  1413→  1413→  // --- Effective bandwidth — detects a clean high-frequency cutoff.
  1414→  1414→  // Scans the spectrum from the Nyquist down toward 10 kHz and finds the
  1415→  1415→  // lowest-frequency bin (searching downward) where the magnitude rises to
  1416→  1416→  // within 40 dB of the spectral peak. Everything above that bin is treated
  1417→  1417→  // as "dead" (below the noise floor). For a full-bandwidth 48 kHz master
  1418→  1418→  // the result should be ≥ 21 kHz; a value of 16–18 kHz is the tell-tale
  1419→  1419→  // signature of an MP3/AAC lowpass or a lossy source.
  1420→  1420→  //
  1421→  1421→  // The 40 dB threshold is chosen because a clean encoder lowpass drops the
  1422→  1422→  // signal by 60+ dB within ~1 kHz, while genuine musical content above
  1423→  1423→  // 15 kHz typically sits 20–35 dB below the spectral peak. 40 dB cleanly
  1424→  1424→  // separates "real signal" from "filter stop-band".
  1425→  1425→  let peakDb = -Infinity
  1426→  1426→  for (let b = 1; b < spectrum.length; b++) if (spectrum[b] > peakDb) peakDb = spectrum[b]
  1427→  1427→  const bwThreshold = peakDb - 40
  1428→  1428→  const minSearchHz = 10000 // don't report below 10 kHz — that's a different problem
  1429→  1429→  const minSearchBin = Math.max(1, Math.floor(minSearchHz / binHz))
  1430→  1430→  let bandwidthBin = minSearchBin
  1431→  1431→  // Scan from Nyquist downward. As soon as we find a bin ABOVE threshold,
  1432→  1432→  // that's the top of the real signal.
  1433→  1433→  for (let b = spectrum.length - 1; b >= minSearchBin; b--) {
  1434→  1434→    if (spectrum[b] >= bwThreshold) {
  1435→  1435→      bandwidthBin = b
  1436→  1436→      break
  1437→  1437→    }
  1438→  1438→  }
  1439→  1439→  const effectiveBandwidthHz = Math.round(bandwidthBin * binHz)
  1440→  1440→
  1441→  1441→  return {
  1442→  1442→    dcOffset,
  1443→  1443→    phaseCorrelation,
  1444→  1444→    phaseCoherence,
  1445→  1445→    stereoWidth,
  1446→  1446→    rmsDb,
  1447→  1447→    bassSideDb,
  1448→  1448→    sibilanceDb,
  1449→  1449→    rumbleDb,
  1450→  1450→    highFreqDb,
  1451→  1451→    zeroCrossingRate,
  1452→  1452→    zeroCrossingStuck,
  1453→  1453→    clippedSamples,
  1454→  1454→    transientDensity,
  1455→  1455→    preechoRisk,
  1456→  1456→    effectiveBandwidthHz,
  1457→  1457→  }
  1458→  1458→}

  1460→  1460→function estimateBpm(samples: Float32Array, sampleRate: number): number | null {
  1461→  1461→  // Envelope follower
  1462→  1462→  const env = new Float32Array(samples.length)
  1463→  1463→  let prev = 0
  1464→  1464→  const decay = 0.999
  1465→  1465→  for (let i = 0; i < samples.length; i++) {
  1466→  1466→    const a = Math.abs(samples[i])
  1467→  1467→    prev = Math.max(a, prev * decay)
  1468→  1468→    env[i] = prev
  1469→  1469→  }
  1470→  1470→  // Autocorrelation on downsampled envelope (100 Hz)
  1471→  1471→  const dsRate = 100
  1472→  1472→  const dsLen = Math.floor(env.length * dsRate / sampleRate)
  1473→  1473→  const ds = new Float32Array(dsLen)
  1474→  1474→  for (let i = 0; i < dsLen; i++) {
  1475→  1475→    const srcIdx = Math.floor(i * sampleRate / dsRate)
  1476→  1476→    ds[i] = env[srcIdx] ?? 0
  1477→  1477→  }
  1478→  1478→  // Autocorrelate
  1479→  1479→  let bestLag = 0
  1480→  1480→  let bestCorr = 0
  1481→  1481→  const minLag = Math.floor(dsRate * 60 / 200) // 200 BPM
  1482→  1482→  const maxLag = Math.floor(dsRate * 60 / 60) // 60 BPM
  1483→  1483→  for (let lag = minLag; lag <= maxLag && lag < dsLen / 2; lag++) {
  1484→  1484→    let c = 0
  1485→  1485→    for (let i = 0; i + lag < dsLen; i++) {
  1486→  1486→      c += ds[i] * ds[i + lag]
  1487→  1487→    }
  1488→  1488→    if (c > bestCorr) {
  1489→  1489→      bestCorr = c
  1490→  1490→      bestLag = lag
  1491→  1491→    }
  1492→  1492→  }
  1493→  1493→  if (bestLag === 0) return null
  1494→  1494→  const bpm = (dsRate * 60) / bestLag
  1495→  1495→  return Math.round(bpm * 10) / 10
  1496→  1496→}

  1498→  1498→function estimateKey(freq: number): string | null {
  1499→  1499→  if (freq <= 0) return null
  1500→  1500→  // A4 = 440 Hz, MIDI 69
  1501→  1501→  const midi = Math.round(69 + 12 * Math.log2(freq / 440))
  1502→  1502→  if (midi < 0 || midi > 127) return null
  1503→  1503→  const names = ['C', 'C#', 'D', 'D#', 'E', 'F', 'F#', 'G', 'G#', 'A', 'A#', 'B']
  1504→  1504→  const octave = Math.floor(midi / 12) - 1
  1505→  1505→  return `${names[midi % 12]}${octave}`
  1506→  1506→}

  1514→  1514→  outputLufs: number
  1515→  1515→  outputTruePeak: number
  1516→  1516→  targetLufs: number
  1517→  1517→  truePeakCeiling: number

  1523→  1523→export function computeRainScore(input: ScoreInput): {
  1524→  1524→  overall: number
  1525→  1525→  spotify: number
  1526→  1526→  apple_music: number
  1527→  1527→  youtube: number
  1528→  1528→  tidal: number
  1529→  1529→  codec_penalty: Record<string, number>
  1530→  1530→} {
  1531→  1531→  const lufsDelta = Math.abs(input.outputLufs - input.targetLufs)
  1532→  1532→  const lufsScore = Math.max(0, 100 - lufsDelta * 8) // ±1 LU = 92, ±5 LU = 60
  1533→  1533→
  1534→  1534→  const tpExcess = Math.max(0, input.outputTruePeak - input.truePeakCeiling)
  1535→  1535→  const tpScore = Math.max(0, 100 - tpExcess * 25)
  1536→  1536→
  1537→  1537→  // Dynamic range — sweet spot 5-9 LU
  1538→  1538→  const drScore = input.dynamicRange < 3 ? 50
  1539→  1539→    : input.dynamicRange > 12 ? 70
  1540→  1540→    : input.dynamicRange >= 5 && input.dynamicRange <= 9 ? 100
  1541→  1541→    : 80
  1542→  1542→
  1543→  1543→  // Stereo width — sweet spot 0.8-1.3
  1544→  1544→  const widthScore = input.stereoWidth < 0.5 || input.stereoWidth > 1.6 ? 60
  1545→  1545→    : input.stereoWidth >= 0.8 && input.stereoWidth <= 1.3 ? 100
  1546→  1546→    : 85
  1547→  1547→
  1548→  1548→  const base = (lufsScore * 0.4 + tpScore * 0.3 + drScore * 0.2 + widthScore * 0.1)
  1549→  1549→
  1550→  1550→  // Per-platform penalties (codec)
  1551→  1551→  const spotifyPenalty = input.codecPenalty * 4 // Ogg Vorbis
  1552→  1552→  const applePenalty = input.codecPenalty * 3 // AAC
