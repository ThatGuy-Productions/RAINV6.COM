    20→// BETA-ANALYTICS: fires the server-side export_completed event (source of
    21→// truth for activation/retention — see server-analytics.ts). Separate from
    22→// recordExportDetails above, which writes local IndexedDB telemetry for
    23→// this browser's own Analytics tab. Best-effort; never blocks the download.
    24→async function reportBetaExportEvent(format: string, outputFileHash: string | undefined) {
    25→  try {
    26→    await fetch('/api/rain/render', {
    27→      method: 'POST',
    28→      headers: { 'Content-Type': 'application/json' },
    29→      body: JSON.stringify({ kind: 'export', format, outputFileHash }),
    30→    })
    31→  } catch (e) {
    32→    console.warn('[analytics] reportBetaExportEvent failed:', e)
    33→  }
    34→}

    58→/** Extract the Chromaprint hash from a RAIN-CERT manifest (if present). */
    59→function fingerprintFromCert(cert: ProvenanceCertificate | null): string | undefined {
    60→  if (!cert) return undefined
    61→  const a = cert.manifest.assertions.find((x) => x.label === 'org.rain.fingerprint')
    62→  const h = a?.data?.hash
    63→  return typeof h === 'string' ? h : undefined
    64→}

    66→export function ExportTab() {
    67→  const fileName = useSessionStore((s) => s.fileName)
    68→  const hasProcessed = useSessionStore((s) => s.hasProcessed)
    69→  const metadata = useSessionStore((s) => s.metadata)
    70→  const setMetadata = useSessionStore((s) => s.setMetadata)
    71→  // P2-EXPORT: read the real cert from the store so the toggles can embed it.
    72→  const rainCert = useSessionStore((s) => s.rainCert)
    73→
    74→  const [selectedFormat, setSelectedFormat] = useState('wav24')
    75→  // P2-EXPORT directive: five real toggles. Each one is honored byte-for-byte
    76→  // by the encoder AND re-verified after export via verifyExported*().

    78→  const [embedSignature, setEmbedSignature] = useState(true)
    79→  const [embedFingerprint, setEmbedFingerprint] = useState(true)
    80→  const [embedMetadata, setEmbedMetadata] = useState(true)
    81→  const [attachCertificate, setAttachCertificate] = useState(false)
    82→  const [isExporting, setIsExporting] = useState(false)
    83→  const [verification, setVerification] = useState<ExportVerificationResult | null>(null)
    84→
    85→  const fingerprint = useMemo(() => fingerprintFromCert(rainCert), [rainCert])
    86→
    87→  // Build the ExportOptions object that gets passed to the encoder. This is
    88→  // the SINGLE source of truth for what gets embedded — the encoder reads no
    89→  // other state.
    90→  const buildOptions = (): ExportOptions => ({
    91→    embedProvenance,
    92→    embedSignature,
    93→    embedFingerprint,
    94→    embedMetadata,
    95→    attachCertificate,
    96→    metadata: {
    97→      title: metadata.title,
    98→      artist: metadata.artist,
    99→      album: metadata.album || undefined,
   100→      year: metadata.year || undefined,
   101→      isrc: metadata.isrc || undefined,
   102→      comment: metadata.comment || undefined,

   104→    fingerprint,
   105→  })
   106→
   107→  const isAtmos = selectedFormat === 'atmos'
   108→  const isMp3 = selectedFormat === 'mp3_320'
   109→  const isWav = selectedFormat === 'wav24' || selectedFormat === 'wav16'
   110→
   111→  // Toggles are only meaningful for WAV/MP3. Atmos package uses its own
   112→  // embedding path (axml chunk + sidecar XML), so we show a notice rather
   113→  // than fake-enabled toggles.
   114→  const togglesRelevant = isWav || isMp3
   115→
   116→  const handleExport = async () => {
   117→    if (!hasProcessed) return
   118→    setIsExporting(true)
   119→    setVerification(null)
   120→    // P2-ANALYTICS: measure real export wall-clock + actual blob size for
   121→    // the Analytics tab's "Export Format Distribution" chart. Each branch
   122→    // (atmos / mp3 / wav) records its own format string + size + duration.
   123→    const exportStart = Date.now()
   124→    const fileState = useSessionStore.getState()
   125→    const recordExport = (format: string, blob: Blob, bitDepthOverride?: number) => {
   126→      void recordExportDetails({
   127→        timestamp: Date.now(),
   128→        format,
   129→        bytes: blob.size,
   130→        durationMs: Date.now() - exportStart,
   131→        sampleRate: fileState.fileSampleRate || 48000,
   132→        bitDepth: bitDepthOverride ?? fileState.fileBitDepth ?? 24,
   133→        channels: fileState.fileChannels || 2,
   134→        fileName: fileState.fileName ?? '(unknown)',
   135→      }).catch((e) => console.warn('[analytics] recordExportDetails failed:', e))
   136→      void reportBetaExportEvent(selectedFormat, rainCert?.outputHash)
   137→    }
   138→    try {
   139→      const format = FORMATS.find((f) => f.id === selectedFormat)!
   140→      const baseName = (metadata.title || fileName || 'rain-master').replace(/\s+/g, '_')
   141→
   142→      // --- Atmos path: separate code path, no ID3/LIST toggles apply ---
   143→      if (format.id === 'atmos') {
   144→        // EXPORT CAP: the spatial pipeline processes the full track (no silent
   145→        // 60s preview truncation). 360s (6 min) is the memory-safe ceiling for
   146→        // a 7.1.4 bed (12ch × 360s × 48k × 4B ≈ 830 MB persistent). Tracks
   147→        // beyond this get an EXPLICIT error — never a silently truncated file.
   148→        const ATMOS_EXPORT_MAX_SEC = 360
   149→        const sourceBuffer = audioEngine.getProcessedBuffer() ?? null
   150→        const sourceDuration = sourceBuffer?.duration ?? 0
   151→        if (sourceDuration > ATMOS_EXPORT_MAX_SEC) {
   152→          const mm = Math.floor(sourceDuration / 60)
   153→          const ss = Math.floor(sourceDuration % 60)
   154→          throw new Error(
   155→            `Atmos export supports tracks up to ${Math.floor(ATMOS_EXPORT_MAX_SEC / 60)} minutes. ` +
   156→            `Your track is ${mm}:${ss.toString().padStart(2, '0')}. ` +
   157→            `Please trim the track or export as WAV / MP3 instead.`,
   158→          )
   159→        }
   160→        const config: SpatialConfig = {
   161→          bedFormat: '7.1.4',
   162→          outputMode: 'MULTICHANNEL',
   163→          hrtf: 'SPHERICAL',
   164→          objects: 16,
   165→          width: 100,
   166→          centerFocus: 50,
   167→        }
   168→        // Pass the export cap so the full track (up to 6 min) is processed.

   172→          throw new Error(

   176→          )
   177→        }

   181→        triggerDownload(blob, `${baseName}_Atmos_7.1.4.zip`)
   182→        recordExport('ATMOS-7.1.4', blob, 24)
   183→        notifySuccess(
   184→          'Atmos full-source package complete',
   185→          `${result.processedSeconds.toFixed(1)}s · 5 files · ${(blob.size / 1024 / 1024).toFixed(1)} MB`,

   188→        return
   189→      }
   190→

   198→        // and reports per-toggle ✓/✗.
   199→        const result = await verifyExportedMp3(blob, options)
   200→        setVerification(result)
   201→
   202→        if (options.attachCertificate && rainCert) {
   203→          // Sidecar path: ZIP the MP3 + the cert.json together. The cert.json
   204→          // is the FULL cert (sig + manifest + fingerprint intact) — the
   205→          // sidecar IS the authoritative cert.
   206→          const audioBytes = new Uint8Array(await blob.arrayBuffer())
   207→          const certJson = JSON.stringify(rainCert, null, 2)
   208→          const zipBlob = buildSidecarZip(
   209→            audioBytes,
   210→            `${baseName}_RAIN.mp3`,
   211→            certJson,
   212→            `${baseName}.cert.json`,
   213→          )
   214→          triggerDownload(zipBlob, `${baseName}_RAIN.zip`)
   215→          recordExport('MP3-320+SIDECAR', zipBlob, 16)
   216→        } else {
   217→          triggerDownload(blob, `${baseName}_RAIN.mp3`)
   218→          recordExport('MP3-320', blob, 16)
   219→        }
   220→        reportExportResult(result, 'MP3')
   221→        setIsExporting(false)
   222→        return
   223→      }
   224→
   225→      // --- WAV path (24-bit / 16-bit) ---
   226→      const bitDepth = (format.bitDepth === 24 ? 24 : 16) as 16 | 24
   227→      const blob = audioEngine.exportWav(bitDepth, rainCert, options)
   228→      const result = await verifyExportedWav(blob, options)
   229→      setVerification(result)
   230→
   231→      if (options.attachCertificate && rainCert) {
   232→        const audioBytes = new Uint8Array(await blob.arrayBuffer())
   233→        const certJson = JSON.stringify(rainCert, null, 2)
   234→        const zipBlob = buildSidecarZip(
   235→          audioBytes,
   236→          `${baseName}_RAIN.wav`,
   237→          certJson,
   238→          `${baseName}.cert.json`,
   239→        )
   240→        triggerDownload(zipBlob, `${baseName}_RAIN.zip`)
   241→        recordExport(bitDepth === 24 ? 'WAV-24+SIDECAR' : 'WAV-16+SIDECAR', zipBlob, bitDepth)
   242→      } else {
   243→        triggerDownload(blob, `${baseName}_RAIN.wav`)
   244→        recordExport(bitDepth === 24 ? 'WAV-24' : 'WAV-16', blob, bitDepth)
   245→      }
   246→      reportExportResult(result, 'WAV')
   247→    } catch (e) {
   248→      console.error('[RAIN export] error:', e)
   249→      notifyError('Export failed', e instanceof Error ? e.message : 'Unknown error')
   250→    } finally {
   251→      setIsExporting(false)
   252→    }
   253→  }

   257→      <div className="lg:col-span-2 space-y-4">

   259→          <div className="text-[10px] font-mono uppercase tracking-wider text-muted-foreground mb-1">
   260→            Export Format
   261→          </div>
   262→          <div className="text-sm font-semibold mb-3">Choose authoritative master</div>
   263→          <div className="grid sm:grid-cols-2 gap-2">
   264→            {FORMATS.map((f) => (
   265→              <button
   266→                key={f.id}
   267→                onClick={() => setSelectedFormat(f.id)}
   268→                disabled={!hasProcessed}
   269→                className={`text-left p-3 rounded-md border transition-all disabled:opacity-50 disabled:cursor-not-allowed ${
   270→                  selectedFormat === f.id
   271→                    ? 'border-rain-accent bg-rain-accent/10'
   272→                    : 'border-rain-border bg-rain-surface-2/60 hover:border-rain-accent/50'
   273→                }`}
   274→              >
   275→                <div className="flex items-start gap-2">
   276→                  <div className={selectedFormat === f.id ? 'text-rain-accent' : 'text-muted-foreground'}>
   277→                    {f.icon}
   278→                  </div>
   279→                  <div className="flex-1 min-w-0">
   280→                    <div className="text-sm font-semibold mb-0.5">{f.label}</div>
   281→                    <div className="text-[10px] text-muted-foreground leading-tight mb-1">{f.desc}</div>
   282→                    <div className="text-[9px] font-mono text-muted-foreground">{f.size}</div>
   283→                  </div>
   284→                </div>
   285→              </button>
   286→            ))}
   287→          </div>
   288→          {/* Full-bandwidth guarantee banner — addresses the "clean cutoff
   289→              above 16-18 kHz" issue. WAV is lossless PCM (Nyquist = 24 kHz
   290→              @ 48 k). MP3 LAME's default lowpass was disabled (patched in
   291→              node_modules) so 320 kbps exports preserve the full top octave. */}
   292→          <div className="mt-3 flex items-start gap-2 p-2.5 rounded-md border border-rain-accent/30 bg-rain-accent/5">
   293→            <CheckCircle2 className="w-4 h-4 text-rain-accent flex-shrink-0 mt-0.5" />
   294→            <div className="text-[10px] leading-tight">
   295→              <span className="font-semibold text-rain-accent">Full Bandwidth Guarantee</span>
   296→              <span className="text-muted-foreground"> — WAV exports are lossless PCM (Nyquist = sample-rate / 2). MP3 320 kbps exports have LAME's default lowpass <em>disabled</em>, preserving the full 20–24 kHz top octave. No 16–18 kHz encoder cliff.</span>
   297→            </div>
   298→          </div>
   299→        </div>
   300→
   301→        <div className="rain-panel rounded-lg p-4">
   302→          <div className="text-[10px] font-mono uppercase tracking-wider text-muted-foreground mb-1">
   303→            Metadata (ID3 / RIFF INFO — used by Metadata toggle)
   304→          </div>
   305→          <div className="text-sm font-semibold mb-3">Tag the master</div>
   306→          <div className="grid sm:grid-cols-2 gap-3">
   307→            <MetadataField label="Title" value={metadata.title} onChange={(v) => setMetadata({ title: v })} />
   308→            <MetadataField label="Artist" value={metadata.artist} onChange={(v) => setMetadata({ artist: v })} />
   309→            <MetadataField label="Album" value={metadata.album} onChange={(v) => setMetadata({ album: v })} />
   310→            <MetadataField label="Track #" value={metadata.trackNumber} onChange={(v) => setMetadata({ trackNumber: v })} />
   311→            <MetadataField label="Year" value={metadata.year} onChange={(v) => setMetadata({ year: v })} />

   314→          </div>
   315→        </div>
   316→
   317→        {/* P2-EXPORT: 5 real toggles, grouped logically per the directive. */}
   318→        <div className="rain-panel rounded-lg p-4">
   319→          <div className="text-[10px] font-mono uppercase tracking-wider text-muted-foreground mb-1">
   320→            Provenance
   321→          </div>
   322→          <div className="text-sm font-semibold mb-3">What gets embedded in the audio bytes</div>

   324→            <div className="text-[10px] text-muted-foreground/80 italic mb-2">

   330→          <div className={`space-y-2 ${!togglesRelevant ? 'opacity-50 pointer-events-none' : ''}`}>

   384→        {/* P2-EXPORT Step 3: verification report — shows what was ACTUALLY

   386→        {verification && (

   390→              Verification Report
   391→            </div>
   392→            <div className="text-sm font-semibold mb-3 flex items-center gap-2">
   393→              {verification.ok ? (
   394→                <><CheckCircle2 className="w-4 h-4 text-rain-accent" /> Verified ✓</>

   396→                <><XCircle className="w-4 h-4 text-red-400" /> Verification failed ✗</>

   504→/** Toast the right message based on verification result. */
   505→function reportExportResult(result: ExportVerificationResult, formatLabel: string): void {
   506→  if (result.ok) {
   507→    notifySuccess(
   508→      `${formatLabel} export verified`,
   509→      `All toggles honored · ${result.sizeBytes.toLocaleString()} bytes`,
   510→    )
   511→  } else {
   512→    const failed = Object.entries(result.checks)
   513→      .filter(([, v]) => !v.ok)
   514→      .map(([k]) => k)
   515→    notifyError(
   516→      `${formatLabel} export verification failed`,
   517→      `Mismatch on: ${failed.join(', ')}. The encoder did not honor one or more toggles.`,
   518→    )
   519→  }
   520→}

   522→/** Trigger a browser download for the given Blob. */
   523→function triggerDownload(blob: Blob, filename: string): void {
   524→  const url = URL.createObjectURL(blob)
   525→  const a = document.createElement('a')
   526→  a.href = url
   527→  a.download = filename
   528→  document.body.appendChild(a)
   529→  a.click()
   530→  document.body.removeChild(a)
   531→  URL.revokeObjectURL(url)
   532→}
